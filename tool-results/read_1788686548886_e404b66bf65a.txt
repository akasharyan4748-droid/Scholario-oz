   190→/**
   191→ * FINANCIAL CATEGORY of a payment/transaction (Core vs Additional vs Exam).
   192→ *
   193→ *   CORE        — money against the standard annual fee structure
   194→ *                 (recurring heads: Tuition, Transport, Library…).
   195→ *   EXAMINATION — money against per-examination charges (the exam fee
   196→ *                 schedule + board/exam heads).
   197→ *   ADDITIONAL  — money against an event-based Additional Charge
   198→ *                 (Educational Tour, Workshop, Competition…) — tracked
   199→ *                 SEPARATELY from core fees so the two are never mixed.
   200→ *
   201→ * Optional on FeeTransaction for backward compatibility: legacy
   202→ * transactions (recorded before this field existed) derive their category
   203→ * via `txnCategory()` (feeHead containing "exam" → EXAMINATION, else CORE).
   204→ */
   205→export type TransactionCategory = 'CORE' | 'EXAMINATION' | 'ADDITIONAL'
   206→
   207→/** Resolve the financial category of any transaction (new or legacy). */
   208→export function txnCategory(
   209→  t: Pick<FeeTransaction, 'category' | 'additionalChargeId' | 'feeHead'>,
   210→): TransactionCategory {
   211→  if (t.additionalChargeId) return 'ADDITIONAL'
   212→  if (t.category) return t.category
   213→  if (/exam/i.test(t.feeHead ?? '')) return 'EXAMINATION'
   214→  return 'CORE'
   215→}

   223→/**
   224→ * Collection lifecycle (FIN-COLLECTION → APPS-IA-1 generalization):
   225→ *   Draft    — being composed; NOT an obligation yet (excluded from student
   226→ *              accounts, the collect-payment wizard and student fees).
   227→ *              Editable in full and DELETABLE (no money can exist).
   228→ *   Active   — open for collection (linked applications publish → Active)

   231→ *              closed charge stops being an obligation for students.
   232→ *   Archived — historical read-only record; payment history stays

   235→ *              stay on record (audit trail).
   236→ */

   960→/**
   961→ * Returns the reason a structure cannot be edited in place, or null when
   962→ * editing is allowed. Rules (PART 7/8/9):
   963→ *   • draft / not-configured structures → editable freely
   964→ *   • historical (non-current) session → read-only, always
   965→ *   • published CURRENT-session structure → locked unless a temporary
   966→ *     editing window is open for it
   967→ */
   968→export function structureLockReason(
   969→  state: Pick<FeeState, 'versions' | 'structureEditWindow'>,
   970→  struct: Pick<FeeStructureConfig, 'id' | 'academicYear' | 'className'>,
   971→): string | null {
   972→  const isPublished = state.versions.some((v) => v.structureId === struct.id && v.status === 'current')
   973→  if (!isPublished) return null // draft — editable
   974→  const year = struct.academicYear ?? CURRENT_ACADEMIC_YEAR
   975→  if (year !== CURRENT_ACADEMIC_YEAR) {
   976→    return 'Historical session structures are read-only — the record stays historically intact.'
   977→  }
   978→  const w = state.structureEditWindow
   979→  const live = w.structureId === struct.id && !!w.expiresAt && Date.now() < w.expiresAt
   980→  if (live) return null
   981→  return `"${struct.className}" is locked — it is active for students. Request a temporary edit window to make exceptional changes.`
   982→}

   994→// ─── Helper: find the FeeStructureConfig that applies to a student ──
   995→// FEE-PER-CLASS (Phase 5): tries an EXACT classId match first (using the
   996→// student's classId when available, derived from className via the
   997→// ACADEMIC_CLASSES catalogue). Falls back to className exact match,
   998→// then classLevel substring matching. This priority order ensures:
   999→//   1. A student in Class 9 (C12) finds FS04 (classId='C12') directly —
  1000→//      no substring walk.
  1001→//   2. A student in Class 11-PCB (C14-PCB) finds the future Class 11
  1002→//      structure via applicableClassIds=['C14-PCM','C14-PCB'] — handles
  1003→//      the stream-class ambiguity that className-only matching couldn't.
  1004→//   3. A student whose className has no structure (e.g. "Class 4") falls
  1005→//      back to the level (Primary → FS02).
  1006→//
  1007→// The signature accepts an optional classId (Phase 5 addition). Callers
  1008→// that pass only className continue to work — backward compatible.
  1009→export function findStructureForStudent(className: string, classId?: string): FeeStructureConfig | undefined {
  1010→  // STRUCT-SESSION/REV — resolve against the LIVE store state (the pool the
  1011→  // publish pipeline updates), falling back to the seed constant only when
  1012→  // the store is not yet hydrated. Previously this read the static
  1013→  // FEE_STRUCTURES seed directly, so a newly published version never
  1014→  // reached student accounts (PART 21 violation).
  1015→  let pool: FeeStructureConfig[] = FEE_STRUCTURES
  1016→  try {
  1017→    const live = useFeeStore.getState().feeStructures
  1018→    if (Array.isArray(live) && live.length > 0) pool = live
  1019→  } catch {
  1020→    /* store not initialised yet — seed fallback is correct at init time */
  1021→  }
  1022→  // 1. classId exact match (preferred path — Phase 5)
  1023→  if (classId) {
  1024→    const byApplicable = pool.find((f) => f.applicableClassIds?.includes(classId))
  1025→    if (byApplicable) return byApplicable
  1026→    const byClassId = pool.find((f) => f.classId === classId)
  1027→    if (byClassId) return byClassId
  1028→  }
  1029→  // 2. className exact match (legacy path — pre-Phase-5 primary lookup)
  1030→  const byName = pool.find((f) => f.className === className)
  1031→  if (byName) return byName
  1032→  // 3. classLevel fallback (last resort — keeps backward compat with
  1033→  //    pre-FEE-PER-CLASS seed which used range names like "Class 9–10")
  1034→  return pool.find((f) => f.classLevel === studentClassLevel(className))
  1035→}
  1036→
  1037→// ─── Helper: count affected students for a structure ───────────────
  1038→// FEE-PER-CLASS (Phase 5): tries an EXACT classId/applicableClassIds
  1039→// match first (so a Class 9 structure reports only Class 9 students,
  1040→// and a Class 12 structure reports BOTH PCM and PCB stream students).
  1041→// Falls back to className exact match, then classLevel substring
  1042→// matching when no student has an exact className match (e.g. a custom
  1043→// structure with className="Custom").
  1044→//
  1045→// Mirrors the matching logic in `findStructureForStudent` so the
  1046→// changeLog's `affectedStudents` field reflects the same students
  1047→// whose fee accounts would be re-derived when the structure changes.
  1048→function countStudentsForStructure(struct: {
  1049→  className: string
  1050→  classLevel: string
  1051→  classId?: string
  1052→  applicableClassIds?: string[]
  1053→}): number {
  1054→  const students = useStudentsStore.getState().students.filter((s) => s.status === 'Active')
  1055→  // 1. classId / applicableClassIds exact match (Phase 5 — preferred
  1056→  //    path when the structure is bound to academic class ids).
  1057→  if (struct.applicableClassIds && struct.applicableClassIds.length > 0) {
  1058→    const matched = students.filter((s) => {
  1059→      const sid = deriveStudentClassId(s.className)
  1060→      return sid != null && struct.applicableClassIds!.includes(sid)
  1061→    })
  1062→    if (matched.length > 0) return matched.length
  1063→  }
  1064→  if (struct.classId) {
  1065→    const matched = students.filter((s) => deriveStudentClassId(s.className) === struct.classId)
  1066→    if (matched.length > 0) return matched.length
  1067→  }
  1068→  // 2. className exact match (legacy path)
  1069→  if (struct.className) {
  1070→    const exact = students.filter((s) => s.className === struct.className)
  1071→    if (exact.length > 0) return exact.length
  1072→  }
  1073→  // 3. classLevel fallback (last resort)
  1074→  return students.filter((s) => studentClassLevel(s.className) === struct.classLevel).length
  1075→}

  1077→// PHASE 5 — derive a student's AcademicClassDef.id from their className.
  1078→// Looks up the canonical class catalogue to find the matching id (e.g.
  1079→// "Class 9" → "C12", "Pre-Nursery" → "C01"). Returns undefined if no
  1080→// academic class matches (e.g. custom draft className "Class 9 — Copy").
  1081→// Used by countStudentsForStructure's classId path. Cached at module
  1082→// scope so the lookup happens once per className.
  1083→const _studentClassNameToClassId = new Map<string, string | undefined>()
  1084→export function deriveStudentClassId(className: string): string | undefined {
  1085→  if (_studentClassNameToClassId.has(className)) {
  1086→    return _studentClassNameToClassId.get(className)
  1087→  }
  1088→  const def = ACADEMIC_CLASSES.find((c) => c.name === className)
  1089→  _studentClassNameToClassId.set(className, def?.id)
  1090→  return def?.id
  1091→}

  1458→// Compute the diff between two head arrays for the changeLog `changes` field.
  1459→function diffHeads(oldHeads: FeeHead[], newHeads: FeeHead[]): { headName: string; oldValue: number; newValue: number }[] {
  1460→  const changes: { headName: string; oldValue: number; newValue: number }[] = []
  1461→  const seen = new Set<string>()
  1462→  for (const h of newHeads) {
  1463→    seen.add(h.name)
  1464→    const old = oldHeads.find((o) => o.name === h.name)
  1465→    if (!old) {
  1466→      changes.push({ headName: h.name, oldValue: 0, newValue: h.amount })
  1467→    } else if (old.amount !== h.amount) {
  1468→      changes.push({ headName: h.name, oldValue: old.amount, newValue: h.amount })
  1469→    }
  1470→  }
  1471→  for (const h of oldHeads) {
  1472→    if (!seen.has(h.name)) {
  1473→      changes.push({ headName: h.name, oldValue: h.amount, newValue: 0 })
  1474→    }
  1475→  }
  1476→  return changes
  1477→}
