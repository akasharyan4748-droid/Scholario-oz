   129→function initialsOf(name: string): string {
   130→  return name
   131→    .split(' ')
   132→    .map((p) => p[0] ?? '')
   133→    .slice(0, 2)
   134→    .join('')
   135→    .toUpperCase()
   136→}

   138→function classLabels(classIds: string[]): string {
   139→  const names = classIds.map((id) => ACADEMIC_CLASSES.find((c) => c.id === id)?.name ?? id)
   140→  if (names.length === 0) return 'No classes'
   141→  if (names.length <= 3) return names.join(', ')
   142→  return `${names.slice(0, 3).join(', ')} +${names.length - 3}`
   143→}

   145→/** Students scoped by a charge — explicit studentIds win, else the roster of
   146→ *  the applicable classes (Active students only). */
   147→function scopedStudentsFor(charge: AdditionalCharge, activeStudents: StudentRecord[]): StudentRecord[] {
   148→  const ids = charge.studentIds
   149→  if (ids && ids.length > 0) return activeStudents.filter((s) => ids.includes(s.id))
   150→  return activeStudents.filter((s) => charge.applicableClassIds.includes(s.classId))
   151→}

   153→/** Per-charge roll-up used by rows and the detail drawer. */
   154→interface ChargeFacts {

   160→  expected: number
   161→  /** null when there is no denominator (open-amount drive). */
   162→  pct: number | null
   163→  /** ALL bound transactions (Payments tab). */
   164→  txns: FeeTransaction[]
   165→}

   171→export function FeesAdditionalCharges({ data }: { data: ReturnType<typeof useFeeData> }) {
   172→  const { additionalCharges, transactions } = data
   173→  const students = useStudentsStore((s) => s.students)
   174→  const applications = useApplicationsStore((s) => s.applications)
   175→
   176→  const [detailId, setDetailId] = useState<string | null>(null)
   177→
   178→  const activeStudents = useMemo(() => students.filter((s) => s.status === 'Active'), [students])
   179→
   180→  // Reverse lookup — application-linked source (no drift-prone field):
   181→  // the SchoolApplication whose payment.chargeId === charge.id.
   182→  const appByChargeId = useMemo(() => {
   183→    const map = new Map<string, SchoolApplication>()
   184→    for (const app of applications) {
   185→      const cid = app.payment.chargeId
   186→      if (cid && !map.has(cid)) map.set(cid, app)
   187→    }
   188→    return map
   189→  }, [applications])
   190→
   191→  // Live roll-up per charge from bound transactions.
   192→  const facts = useMemo(() => {
   193→    const map = new Map<string, ChargeFacts>()
   194→    for (const c of additionalCharges) {
   195→      const scoped = scopedStudentsFor(c, activeStudents)
   196→      const txns = transactions.filter((t) => t.additionalChargeId === c.id)
   197→      const collected = txns
   198→        .filter((t) => t.status === 'Success' || t.status === 'Under Verification')
   199→        .reduce((sum, t) => sum + t.amount, 0)
   200→      const custom = c.allowCustomAmount === true
   201→      const expected = custom ? (c.targetAmount ?? 0) : scoped.length * c.amount
   202→      const pct = expected > 0 ? Math.min(100, Math.round((collected / expected) * 100)) : null
   203→      map.set(c.id, { students: scoped.length, collected, expected, pct, txns })
   204→    }
   205→    return map
   206→  }, [additionalCharges, transactions, activeStudents])
   207→
   208→  // Status-only list — open collections first (the actionable surface),
   209→  // then closed/cancelled ones with their status chip (permanent record).
   210→  // "Application payments" filter (spec §57): only charges generated by an
   211→  // Applications & Forms form (reverse lookup app.payment.chargeId) — the
   212→  // Educational Tour view, never donations/standalone collections.
   213→  const [scopeFilter, setScopeFilter] = useState<'all' | 'application'>('all')
   214→  const orderedCharges = useMemo(() => {
   215→    const rank = (c: AdditionalCharge) => (c.status === 'Active' ? 0 : 1)
   216→    return [...additionalCharges]
   217→      .filter((c) => (scopeFilter === 'application' ? appByChargeId.has(c.id) : true))
   218→      .sort((a, b) => rank(a) - rank(b) || a.dueDate.localeCompare(b.dueDate))
   219→  }, [additionalCharges, scopeFilter, appByChargeId])
   220→
   221→  const detailCharge = detailId ? additionalCharges.find((c) => c.id === detailId) ?? null : null
   222→
   223→  return (
   224→    <>
   225→      <Panel
   226→        title={
   227→          <span className="inline-flex items-center gap-2">
   228→            Additional Collections
   229→            <Badge variant="outline" className="text-[9px] h-4 px-1.5">
   230→              {orderedCharges.filter((c) => c.status === 'Active').length}
   231→            </Badge>
   232→          </span>
   233→        }
   234→        subtitle="Event and other non-annual collections · created from Applications & Forms."
   235→        action={
   236→          <div className="inline-flex h-7 items-center rounded-full bg-muted/70 p-0.5" role="group" aria-label="Collection scope">
   237→            {([['all', 'All collections'], ['application', 'Application payments']] as const).map(([v, label]) => (
   238→              <button
   239→                key={v}
   240→                type="button"
   241→                onClick={() => setScopeFilter(v)}
   242→                aria-pressed={scopeFilter === v}
   243→                className={cn(
   244→                  'px-2.5 h-6 rounded-full text-[10.5px] font-medium whitespace-nowrap transition-all',
   245→                  scopeFilter === v ? 'bg-white dark:bg-white/10 shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   246→                )}
   247→              >
   248→                {label}
   249→              </button>
   250→            ))}
   251→          </div>
   252→        }
   253→        bodyClassName="p-0"
   254→      >
   255→        {orderedCharges.length > 0 ? (
   256→          <div className="divide-y divide-border">
   257→            {orderedCharges.map((c) => {
   258→              const f = facts.get(c.id) ?? EMPTY_FACTS
   259→              const app = appByChargeId.get(c.id)
   260→              return (
   261→                <motion.div key={c.id} initial={{ opacity: 0, y: 4 }} animate={{ opacity: 1, y: 0 }}>
   262→                  <button
   263→                    type="button"
   264→                    onClick={() => setDetailId(c.id)}
   265→                    className="w-full flex items-start justify-between gap-3 px-4 py-2.5 text-left hover:bg-muted/30 transition-colors"
   266→                    aria-label={`Open ${c.name} payment status`}
   267→                  >
   268→                    {/* Left — category icon chip + name + chips + meta line */}
   269→                    <div className="flex items-start gap-3 min-w-0 flex-1">
   270→                      <span className={cn('flex h-9 w-9 shrink-0 items-center justify-center rounded-lg', categoryTone(c.category))}>
   271→                        <CategoryIcon category={c.category} className="h-4 w-4" />
   272→                      </span>
   273→                      <div className="min-w-0">
   274→                        <div className="flex items-center gap-1.5 min-w-0">
   275→                          <p className={cn('text-xs font-semibold truncate', c.status !== 'Active' && 'text-muted-foreground')}>{c.name}</p>
   276→                          <MiniChip className={c.mandatory ? 'bg-slate-500/10 text-slate-600 dark:text-slate-300' : 'bg-muted text-muted-foreground'}>
   277→                            {c.mandatory ? 'Mandatory' : 'Optional'}
   278→                          </MiniChip>
   279→                          {c.allowCustomAmount && (
   280→                            <MiniChip className="bg-violet-500/10 text-violet-700 dark:text-violet-300">Custom amount</MiniChip>
   281→                          )}
   282→                        </div>
   283→                        <p className="text-[10px] text-muted-foreground mt-0.5 truncate">
   284→                          <span className="font-medium">{c.category}</span>
   285→                          {' · '}{app ? `via Application · ${app.title}` : 'Standalone'}
   286→                          {' · '}{classLabels(c.applicableClassIds)}
   287→                          {' · '}due {formatDate(c.dueDate)}
   288→                          {' · '}{f.students} student{f.students === 1 ? '' : 's'}
   289→                        </p>
   290→                      </div>
   291→                    </div>
   292→
   293→                    {/* Right — labelled figures, progress, status */}
   294→                    <div className="shrink-0 w-32 sm:w-40 text-right">
   295→                      <p className={cn('text-sm font-bold tabular-nums leading-tight truncate', f.collected > 0 && 'text-emerald-600 dark:text-emerald-400')}>
   296→                        {formatINR(f.collected)}{' '}
   297→                        <span className="text-[9px] font-semibold text-muted-foreground">collected</span>
   298→                      </p>
   299→                      <p className="text-[9px] text-muted-foreground truncate">
   300→                        {c.allowCustomAmount
   301→                          ? c.targetAmount
   302→                            ? <>open amount · target {formatINR(c.targetAmount)}</>
   303→                            : 'open amount'
   304→                          : <>of {formatINR(f.expected)} expected</>}
   305→                      </p>
   306→                      {f.pct != null && (
   307→                        <div className="mt-1.5 flex items-center gap-1.5">
   308→                          <div aria-hidden className="h-1.5 flex-1 rounded-full bg-muted overflow-hidden">
   309→                            <motion.div
   310→                              initial={{ width: 0 }}
   311→                              animate={{ width: `${f.pct}%` }}
   312→                              transition={{ duration: 0.5, ease: 'easeOut' }}
   313→                              className="h-full rounded-full bg-emerald-500"
   314→                            />
   315→                          </div>
   316→                          <span className="w-7 text-right text-[9px] font-semibold text-muted-foreground tabular-nums">{f.pct}%</span>
   317→                        </div>
   318→                      )}
   319→                      <div className="mt-1.5 flex items-center justify-end gap-1.5">
   320→                        <StatusChip status={c.status} />
   321→                        <span className="text-[10px] text-muted-foreground tabular-nums">
   322→                          {f.txns.length} payment{f.txns.length === 1 ? '' : 's'}
   323→                        </span>
   324→                      </div>
   325→                    </div>
   326→                  </button>
   327→                </motion.div>
   328→              )
   329→            })}
   330→          </div>
   331→        ) : (
   332→          <FeeEmptyState
   333→            icon={<CalendarDays className="h-6 w-6" />}
   334→            title={scopeFilter === 'application' ? 'No application payments' : 'No additional collections'}
   335→            description={scopeFilter === 'application'
   336→              ? 'No published Applications & Forms form carries a charge yet — tour collections appear here once a paid form goes live.'
   337→              : 'Collections are generated from Applications & Forms when a paid form is published — they appear here with their live payment status.'}
   338→          />
   339→        )}
   340→      </Panel>
   341→
   342→      {/* Collection status drawer (read-only — spec §2) */}
   343→      {detailCharge && (
   344→        <CollectionStatusSheet
   345→          charge={detailCharge}
   346→          facts={facts.get(detailCharge.id) ?? EMPTY_FACTS}
   347→          app={appByChargeId.get(detailCharge.id)}
   348→          activeStudents={activeStudents}
   349→          onClose={() => setDetailId(null)}
   350→        />
   351→      )}
   352→    </>
   353→  )
   354→}

   374→function CollectionStatusSheet({
   375→  charge,
   376→  facts,
   377→  app,
   378→  activeStudents,
   379→  onClose,
   380→}: {
   381→  charge: AdditionalCharge
   382→  facts: ChargeFacts
   383→  app?: SchoolApplication
   384→  activeStudents: StudentRecord[]
   385→  onClose: () => void
   386→}) {
   387→  const [tab, setTab] = useState<DetailTab>('students')
   388→
   389→  const isCustom = charge.allowCustomAmount === true
   390→  const scoped = useMemo(() => scopedStudentsFor(charge, activeStudents), [charge, activeStudents])
   391→
   392→  // Per-student payment state derived from the bound transactions.
   393→  const studentStates = useMemo(() => {
   394→    const map = new Map<string, { paid: number; verifying: boolean }>()
   395→    for (const t of facts.txns) {
   396→      const cur = map.get(t.studentId) ?? { paid: 0, verifying: false }
   397→      if (t.status === 'Success' || t.status === 'Under Verification') cur.paid += t.amount
   398→      if (t.status === 'Under Verification') cur.verifying = true
   399→      map.set(t.studentId, cur)
   400→    }
   401→    return map
   402→  }, [facts.txns])
   403→
   404→  const paidCount = scoped.filter((s) => {
   405→    const st = studentStates.get(s.id)
   406→    const paid = st?.paid ?? 0
   407→    return isCustom ? paid > 0 : paid >= charge.amount
   408→  }).length
   409→
   410→  // Bound payments, newest first (Payments tab).
   411→  const payments = useMemo(
   412→    () => [...facts.txns].sort((a, b) => b.date.localeCompare(a.date) || b.receiptNo.localeCompare(a.receiptNo)),
   413→    [facts.txns],
   414→  )
   415→
   416→  // Compact meta rows — record context without a separate management tab.
   417→  const metaRows: Array<{ label: string; value: string }> = []
   418→  if (charge.description) metaRows.push({ label: 'Description', value: charge.description })
   419→  if (charge.reference) metaRows.push({ label: 'Reference', value: charge.reference })
   420→  metaRows.push({
   421→    label: 'Created by',
   422→    value: `${charge.createdBy}${charge.createdAt ? ` · ${formatDate(charge.createdAt)}` : ''}`,
   423→  })
   424→  metaRows.push({ label: 'Session', value: charge.academicYear })
   425→  metaRows.push({ label: 'Due date', value: formatDate(charge.dueDate) })
   426→  if (charge.status === 'Closed') {
   427→    metaRows.push({
   428→      label: 'Closed',
   429→      value: `${charge.closedAt ? formatDate(charge.closedAt) : '—'}${charge.closeNote ? ` — ${charge.closeNote}` : ''}`,
   430→    })
   431→  }
   432→  if (charge.status === 'Cancelled') {
   433→    metaRows.push({ label: 'Cancelled', value: charge.cancelReason ?? 'No reason recorded' })
   434→  }
   435→
   436→  const expectedDisplay = isCustom
   437→    ? charge.targetAmount && charge.targetAmount > 0 ? formatINR(charge.targetAmount) : 'Open'
   438→    : formatINR(facts.expected)
   439→  const pendingDisplay = facts.expected > 0
   440→    ? formatINR(Math.max(0, facts.expected - facts.collected))
   441→    : isCustom ? '—' : formatINR(0)
   442→
   443→  return (
   444→    <Sheet open onOpenChange={(o) => { if (!o) onClose() }}>
   445→      <SheetContent side="right" className="w-full sm:max-w-md p-0 gap-0 flex flex-col">
   446→        {/* Header — icon chip + name + status + source line */}
   447→        <SheetHeader className="space-y-0 border-b border-border/60 px-4 pb-3 pt-4 text-left">
   448→          <div className="flex items-start gap-3 pr-8">
   449→            <span className={cn('flex h-9 w-9 shrink-0 items-center justify-center rounded-lg', categoryTone(charge.category))}>
   450→              <CategoryIcon category={charge.category} className="h-4 w-4" />
   451→            </span>
   452→            <div className="min-w-0">
   453→              <div className="flex items-center gap-1.5 min-w-0">
   454→                <SheetTitle className="truncate text-sm font-bold leading-tight">{charge.name}</SheetTitle>
   455→                <StatusChip status={charge.status} />
   456→              </div>
   457→              <SheetDescription className="mt-0.5 truncate text-[10px] text-muted-foreground">
   458→                {app ? `via Application · ${app.title}` : 'Standalone'} · {charge.category}
   459→              </SheetDescription>
   460→            </div>
   461→          </div>
   462→        </SheetHeader>
   463→
   464→        {/* Summary strip — 4 labelled mini tiles + progress */}
   465→        <div className="border-b border-border/60 px-4 pb-2.5 pt-3">
   466→          <div className="grid grid-cols-4 gap-1.5">
   467→            <MiniTile label="Expected" value={expectedDisplay} />
   468→            <MiniTile label="Collected" value={formatINR(facts.collected)} accent={facts.collected > 0} />
   469→            <MiniTile label="Pending" value={pendingDisplay} />
   470→            <MiniTile label="Students" value={String(facts.students)} />
   471→          </div>
   472→          {facts.pct != null && (
   473→            <div className="mt-2 flex items-center gap-1.5">
   474→              <div aria-hidden className="h-1.5 flex-1 rounded-full bg-muted overflow-hidden">
   475→                <motion.div
   476→                  initial={{ width: 0 }}
   477→                  animate={{ width: `${facts.pct}%` }}
   478→                  transition={{ duration: 0.5, ease: 'easeOut' }}
