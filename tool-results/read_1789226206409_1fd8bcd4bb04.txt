    69→// Queue chip mapping (spec): Pending Principal Acceptance → amber 'Pending',
    70→// Collected by Teacher → sky, Clarification Requested → violet.
    71→function queueChip(status: CashRequest['status']): { label: string; tone: string } {
    72→  switch (status) {
    73→    case 'Pending Principal Acceptance':
    74→      return { label: 'Pending', tone: 'bg-amber-500/10 text-amber-700 dark:text-amber-300' }
    75→    case 'Collected by Teacher':
    76→      return { label: 'Collected', tone: 'bg-sky-500/10 text-sky-700 dark:text-sky-300' }
    77→    case 'Clarification Requested':
    78→      return { label: 'Clarification', tone: 'bg-violet-500/10 text-violet-700 dark:text-violet-300' }
    79→    default:
    80→      return { label: status, tone: '' }
    81→  }
    82→}

    97→/** Who recorded the payment awaiting verification — SaaS-STAGE-1 keeps ONE

   100→function collectorLabel(role: FeeTransaction['collectorRole'], collectedBy: string): string {
   101→  if (role === 'teacher') return `Teacher · ${collectedBy}`
   102→  if (role === 'class_teacher') return `Class Teacher · ${collectedBy}`
   103→  if (role === 'self') return 'Student self-service'
   104→  return `Office · ${collectedBy}`
   105→}

   426→            </div>
   427→          </details>
   428→        )}
   429→      </Panel>
   430→
   431→      {/* ── Confirmation modals (business logic unchanged) ───────────── */}
   432→      <AnimatePresence>
   433→        {approvingReq && (
   434→          <ApproveModal
   435→            req={approvingReq}
   436→            currentOutstanding={getStudentOutstanding(approvingReq.studentId)}
   437→            loading={actionLoading}
   438→            onClose={() => setApprovingReq(null)}
   439→            onConfirm={handleApprove}
   440→          />
   441→        )}
   442→      </AnimatePresence>
   443→
   444→      <AnimatePresence>
   445→        {rejectingReq && (
   446→          <RejectModal
   447→            req={rejectingReq}
   448→            reason={rejectReason}
   449→            setReason={setRejectReason}
   450→            note={rejectNote}
   451→            setNote={setRejectNote}
   452→            loading={actionLoading}
   453→            onClose={() => setRejectingReq(null)}
   454→            onConfirm={handleReject}
   455→          />
   456→        )}
   457→      </AnimatePresence>
   458→
   459→      <AnimatePresence>
   460→        {clarifyReq && (

   467→            onConfirm={handleClarify}
   468→          />
   469→        )}
   470→      </AnimatePresence>
   471→
   472→      {/* ── Direct cash reject (mandatory reason, no money recorded) ── */}
   473→      <Dialog open={!!rejectingDirect} onOpenChange={(o) => !o && setRejectingDirect(null)}>

   481→          <Textarea
   482→            className="min-h-[64px] text-xs"
   483→            placeholder="Reason (required) — shown in the audit trail"
   484→            value={rejectingDirectReason}
   485→            onChange={(e) => setRejectingDirectReason(e.target.value)}
   486→          />
   487→          <DialogFooter>
   488→            <Button variant="outline" size="sm" className="h-8 text-xs" onClick={() => setRejectingDirect(null)}>Cancel</Button>
   489→            <Button
   490→              size="sm"
   491→              className="h-8 text-xs"
   492→              disabled={!rejectingDirectReason.trim()}
   493→              onClick={() => {
   494→                if (!rejectingDirect) return
   495→                const r = rejectDirectCashTxn(rejectingDirect.id, 'Principal', rejectingDirectReason.trim())
   496→                if (r.success) toast.error('Direct cash rejected', { description: `${rejectingDirect.receiptNo} — ${rejectingDirectReason.trim()}` })
   497→                else toast.error('Could not reject', { description: r.error })
   498→                setRejectingDirect(null)
   499→              }}
   500→            >
   501→              Reject entry
   502→            </Button>
   503→          </DialogFooter>
   504→        </DialogContent>
   505→      </Dialog>
   506→    </>
   507→  )
   508→}
