   234→/** Compact display name for a canonical class ("Class 11 PCM" only when
   235→ *  the selection covers SOME of that grade's streams). */
   236→function classChipLabel(id: string): string {
   237→  const c = ACADEMIC_CLASSES.find((x) => x.id === id)
   238→  if (!c) return id
   239→  return c.stream ? `${c.name} ${c.stream}` : c.name
   240→}
   241→
   242→/** Human one-line summary of an applicability rule:
   243→ *  "All students" · "Boys" · "Class 9 & Class 11" · "Boys · Class 6 & Class 8". */
   244→function describeAudience(a: EntryFeeAudience): string {
   245→  if (a.scope === 'all') return 'All students'
   246→  const classIds = a.classIds ?? []
   247→  const list = summarizeClassIds(classIds)
   248→  if (a.scope === 'gender') return a.gender === 'girls' ? 'Girls' : 'Boys'
   249→  if (a.scope === 'classes') return list || 'No classes selected'
   250→  return `${a.gender === 'girls' ? 'Girls' : 'Boys'} · ${list || 'No classes selected'}`
   251→}
   252→
   253→/** Join selected canonical classes into a short human list, collapsing a
   254→ *  grade to its plain name when every stream of that grade is selected. */
   255→function summarizeClassIds(ids: string[]): string {
   256→  if (ids.length === 0) return ''
   257→  const picked = new Set(ids)
   258→  const byName = new Map<string, { total: number; picked: number; first: string; stream: string | undefined }>()
   259→  for (const c of ACADEMIC_CLASSES) {
   260→    const e = byName.get(c.name) ?? { total: 0, picked: 0, first: c.id, stream: c.stream }
   261→    e.total += 1
   262→    if (picked.has(c.id)) e.picked += 1
   263→    byName.set(c.name, e)
   264→  }
   265→  const tokens: string[] = []
   266→  const seen = new Set<string>()
   267→  for (const c of ACADEMIC_CLASSES) {
   268→    if (!picked.has(c.id) || seen.has(c.name)) continue
   269→    seen.add(c.name)
   270→    const e = byName.get(c.name)!
   271→    tokens.push(e.picked === e.total ? c.name : `${c.name} ${c.stream ?? ''}`.trim())
   272→  }
   273→  if (tokens.length === 0) return ''
   274→  if (tokens.length === 1) return tokens[0]
   275→  return `${tokens.slice(0, -1).join(', ')} & ${tokens[tokens.length - 1]}`
   276→}

   280→function EntryFeePolicyCard() {
   281→  const policy = useFeeStore((s) => s.entryFeePolicy)
   282→  const updateEntryFeePolicy = useFeeStore((s) => s.updateEntryFeePolicy)
   283→  const feeStructures = useFeeStore((s) => s.feeStructures)
   284→  const perms = useEffectiveFeeCapabilities()
   285→  const canManage = perms.fee_entry_policy_manage
   286→  const [editingRule, setEditingRule] = useState<string | null>(null)
   287→
   288→  // Registration Fee amount is DERIVED from the live published fee
   289→  // structures (canonical source) — draft copies are excluded because a
   290→  // draft charges nobody.
   291→  const registration = useMemo(() => {
   292→    const entries = feeStructures
   293→      .filter((st) => !/draft/i.test(st.className))
   294→      .flatMap((st) => st.components
   295→        .filter((c) => c.active && (c.catalogueId === 'fh-2' || c.name === 'Registration Fee'))
   296→        .map((c) => ({ amount: c.amount })))
   297→    const amounts = Array.from(new Set(entries.map((e) => e.amount)))
   298→    return {
   299→      found: entries.length > 0,
   300→      label: amounts.length === 1 ? formatINR(amounts[0]) : 'per structure',
   301→    }
   302→  }, [feeStructures])
   303→
   304→  const ruleLabel = (r: EntryFeeRule) => (r.id === 'admission' ? 'Admission Fee' : r.id === 'registration' ? 'Registration Fee' : r.id)
   305→  const ruleAmount = (r: EntryFeeRule) =>
   306→    r.id === 'admission'
   307→      ? `${formatINR(policy.admissionAmount)} · one-time`
   308→      : `${registration.found ? registration.label : '—'} · one-time`
   309→
   310→  const saveRule = (ruleId: string, applies: EntryFeeAudience) => {
   311→    updateEntryFeePolicy({
   312→      rules: policy.rules.map((r) => (r.id === ruleId ? { ...r, applies } : r)),
   313→    })
   314→    toast.success('Applicability updated', {
   315→      description: `${ruleLabel(policy.rules.find((r) => r.id === ruleId)!)}, now: ${describeAudience(applies)}`,
   316→    })
   317→  }
   318→
   319→  const editing = editingRule ? policy.rules.find((r) => r.id === editingRule) ?? null : null
   320→
   321→  return (
   322→    <SettingsCard
   323→      label="One-Time Entry Fees"
   324→      icon={<UserPlus />}
   325→      summary={policy.enabled ? 'enabled' : 'off'}
   326→      action={
   327→        canManage ? (
   328→          <span className="text-[10px] text-muted-foreground">Tap a fee to edit who pays it</span>
   329→        ) : (
   330→          <ViewOnlyChip label="View only" />
   331→        )
   332→      }
   333→    >
   334→      <div className="space-y-3">
   335→        {/* Status · who manages — one compact line */}
   336→        <div className="flex items-center flex-wrap gap-x-4 gap-y-1 text-[11px]">
   337→          <span className="text-muted-foreground">Status
   338→            <span className={cn('ml-1.5 font-semibold', policy.enabled ? 'text-emerald-600' : 'text-muted-foreground')}>
   339→              {policy.enabled ? 'Enabled' : 'Disabled'}
   340→            </span>
   341→          </span>
   342→          <span className="text-muted-foreground">Who can manage
   343→            <span className="ml-1.5 font-semibold text-foreground">{canManage ? 'Principal' : 'Platform (Super Admin)'}</span>
   344→          </span>
   345→        </div>
   346→
   347→        {/* Fee rows — name + amount, then the rule in one human line */}
   348→        <div className="divide-y divide-border/40 border-t border-b border-border/40">
   349→          {policy.rules.map((r) => (
   350→            <div key={r.id} className="py-2.5">
   351→              <div className="flex items-center justify-between gap-2">
   352→                <p className="text-xs font-medium shrink-0">{ruleLabel(r)}</p>

   403→function RuleBuilderDialog({ rule, label, onClose, onSave }: {
   404→  rule: EntryFeeRule
   405→  label: string
   406→  onClose: () => void
   407→  onSave: (applies: EntryFeeAudience) => void
   408→}) {
   409→  const students = useStudentsStore((s) => s.students)
   410→  const [scope, setScope] = useState<EntryFeeAudience['scope']>(rule.applies.scope)
   411→  const [gender, setGender] = useState<'boys' | 'girls'>(rule.applies.gender ?? 'boys')
   412→  const [classIds, setClassIds] = useState<string[]>(rule.applies.classIds ?? [])
   413→
   414→  const draft: EntryFeeAudience = useMemo(() => ({
   415→    scope,
   416→    ...(scope === 'gender' || scope === 'classes_gender' ? { gender } : {}),
   417→    ...(scope === 'classes' || scope === 'classes_gender' ? { classIds } : {}),
   418→  }), [scope, gender, classIds])
   419→
   420→  const affected = useMemo(
   421→    () => students.filter((s) => s.status === 'Active' && entryFeeApplies(draft, s)).length,
   422→    [students, draft],
   423→  )
   424→
   425→  const needsClasses = scope === 'classes' || scope === 'classes_gender'
   426→  const needsGender = scope === 'gender' || scope === 'classes_gender'
   427→
   428→  const toggleClass = (id: string) =>
   429→    setClassIds((prev) => (prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]))
   430→
   431→  const save = () => {
   432→    if (needsClasses && classIds.length === 0) {
   433→      toast.error('Select at least one class')
   434→      return
   435→    }
   436→    onSave(draft)
   437→  }
   438→
   439→  return (
   440→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   441→      <DialogContent className="sm:max-w-md">
   442→        <DialogHeader>
   443→          <DialogTitle className="text-sm font-semibold">{label} — who pays it?</DialogTitle>
   444→          <DialogDescription className="text-[11px]">
   445→            Applies to future admissions only. Recorded payments never change.
   446→          </DialogDescription>
   447→        </DialogHeader>
   448→
