     3→/**
     4→ * Class Attendance — the daily attendance workspace.
     5→ *
     6→ * WORKFLOW (server-backed, /api/teacher/attendance):
     7→ *   · Classes listed here are the ones the session teacher is authorized
     8→ *     for — class teacher of the class OR subject teacher with an active
     9→ *     ClassSubjectAssignment. Nothing is hardcoded per teacher.
    10→ *   · SAVE DRAFT persists the working state (editable, re-saveable).
    11→ *   · SUBMIT ATTENDANCE finalizes the teacher's own submission — after
    12→ *     which it is read-only. One teacher's submission never overwrites
    13→ *     another's: class teacher and each subject teacher own separate,
    14→ *     auditable submissions for the same class/date.
    15→ *   · Subject teachers are never blocked: with no class-teacher record
    16→ *     they get a blank workspace; once the class teacher SUBMITS, they
    17→ *     see the pre-filled roster ("Pre-filled from Class Teacher") which
    18→ *     they can review, adjust, save and submit as their own record.
    19→ *   · If the class teacher submits while a subject teacher already has a
    20→ *     draft, the draft is preserved and a subtle notice with a
    21→ *     "Review latest class attendance" action appears.
    22→ *
    23→ * DATA DISCIPLINE (same as the Teacher Hub modules):
    24→ *   · { cache: 'no-store', credentials: 'same-origin' } — the httpOnly
    25→ *     erp_session cookie is the ONLY auth source; the server resolves
    26→ *     teacher / school / authorized classes on every call.
    27→ *   · { ok, data } envelope; 401 → signOut() once.
    28→ *   · Honest states everywhere: first-load skeleton, quiet reloads,
    29→ *     section-level errors with retry, purposeful empty states.

    31→ * VISUAL LANGUAGE: the four summary cards + roster rows preserve the
    32→ * established attendance recipe verbatim (compact icon chip, label,

    92→function todayISO(): string {
    93→  const d = new Date()
    94→  const pad = (n: number) => (n < 10 ? `0${n}` : `${n}`)
    95→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`

    98→function longDate(iso: string): string {
    99→  const d = new Date(`${iso}T00:00:00`)
   100→  return d.toLocaleDateString('en-IN', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })
   101→}

   103→function shortTime(iso: string | null): string {
   104→  if (!iso) return ''

   131→async function request<T>(url: string, init?: RequestInit): Promise<T> {
   132→  const r = await fetch(url, {
   133→    ...init,
   134→    cache: 'no-store',
   135→    credentials: 'same-origin',
   136→    headers: init?.body ? { 'Content-Type': 'application/json', ...init?.headers } : init?.headers,
   137→  })
   138→  if (!r.ok) {
   139→    if (r.status === 401) {
   140→      handleExpiredSession()
   141→      throw new Error('Your session has expired. Please sign in again.')
   142→    }
   143→    let message = `Request failed (${r.status})`
   144→    try {
   145→      const j: unknown = await r.json()
   146→      if (j && typeof j === 'object' && typeof (j as { error?: unknown }).error === 'string') {
   147→        message = (j as { error: string }).error
   148→      }
   149→    } catch {
   150→      /* non-JSON error body — keep the concise fallback */
   151→    }
   152→    throw new Error(message)
   153→  }
   154→  const j: unknown = await r.json()
   155→  if (!j || typeof j !== 'object' || (j as { ok?: unknown }).ok !== true || !('data' in j)) {
   156→    throw new Error('Unexpected response from the server.')
   157→  }
   158→  return (j as { data: T }).data
   159→}

   163→/** Build the working status map for a detail: own submission → prefill → default. */
   164→function initialWorking(detail: AttendanceDetail | null): Record<string, Status> {
   165→  const w: Record<string, Status> = {}
   166→  if (!detail) return w
   167→  const source = detail.you?.entries ?? detail.prefill?.entries ?? null
   168→  const map = source ? new Map(source.map((e) => [e.studentId, e.status.toLowerCase() as Status])) : null
   169→  for (const s of detail.roster) {
   170→    w[s.id] = map?.get(s.id) ?? 'present'
   171→  }
   172→  return w
   173→}
   174→
   175→function sameWorking(a: Record<string, Status>, b: Record<string, Status>): boolean {
   176→  const ka = Object.keys(a)
   177→  if (ka.length !== Object.keys(b).length) return false
   178→  return ka.every((k) => a[k] === b[k])
   179→}

   213→  const [working, setWorking] = useState<Record<string, Status>>({})
   214→  const [baseline, setBaseline] = useState<Record<string, Status>>({})
   215→  const [search, setSearch] = useState('')

   217→  const [savedFlash, setSavedFlash] = useState(false)
   218→  const [submitting, setSubmitting] = useState(false)
   219→  const [detailError, setDetailError] = useState<string | null>(null)
   220→
   221→  const reload = useCallback(() => setTick((t) => t + 1), [])
   222→
   223→  // ── aggregate fetch: authorized classes + working detail ──
   224→  useEffect(() => {
   225→    let cancelled = false
   226→    setError(null)
   227→    setDetailError(null)
   228→    if (!hasData.current) setLoading(true)
   229→    const qs = new URLSearchParams({ date })
   230→    if (classId) qs.set('classId', classId)
   231→    request<AttendancePayload>(`/api/teacher/attendance?${qs.toString()}`)
   232→      .then((d) => {
   233→        hasData.current = true
   234→        if (cancelled) return

   236→        // auto-select the first authorized class on first load
   237→        if (!classId && d.classes.length > 0) setClassId(d.classes[0].id)
   238→      })
   239→      .catch((e: unknown) => {
   240→        if (cancelled) return
   241→        // a class-scoped failure (e.g. stale selection) keeps the class list usable
   242→        if (classId) {
   243→          setDetailError(e instanceof Error ? e.message : 'Attendance could not load.')
   244→        } else {
   245→          setError(e instanceof Error ? e.message : 'Attendance could not load.')
   246→        }
   247→      })
   248→      .finally(() => {
   249→        if (!cancelled) setLoading(false)
   250→      })
   251→    return () => {
   252→      cancelled = true
   253→    }
   254→  }, [date, classId, tick])
   255→
   256→  const detail = payload?.detail ?? null
   257→  const selectedClass = payload?.classes.find((c) => c.id === classId) ?? null
   258→
   259→  // ── (re)initialize the working state whenever the detail changes ──
   260→  useEffect(() => {
   261→    const w = initialWorking(detail)
   262→    setWorking(w)
   263→    setBaseline(w)
   264→    setSearch('')
   265→  }, [detail])
   266→
   267→  // ── honest sidebar badge: classes not yet submitted TODAY ──
   268→  useEffect(() => {
   269→    if (!payload || payload.date !== todayISO()) return

   271→    useTeacherHubStore.getState().setCounts({ attendancePending: pending })
   272→  }, [payload])
   273→
   274→  const readOnly = detail?.you?.status === 'SUBMITTED'
   275→  const persistedStatus: PersistedStatus = readOnly
   276→    ? 'submitted'
   277→    : detail?.you?.status === 'DRAFT'
   278→      ? 'draft'
   279→      : detail?.prefill
   280→        ? 'prefilled'
   281→        : 'not-started'
   282→  const dirty = !sameWorking(working, baseline)
   283→
   284→  // ── unsaved-changes guards (Part 19) ──

   288→      e.preventDefault()

   290→    }

   298→        action()

   303→    [dirty, readOnly],
