    67→    67→function drawFieldRows(doc: jsPDF, y: number, rows: FieldRow[]): number {
    68→    68→  let cursor = y
    69→    69→  let i = 0
    70→    70→  while (i < rows.length) {
    71→    71→    const left = rows[i]
    72→    72→    if (left.wide) {
    73→    73→      // Full-width row (address, emergency contact, medical note…).
    74→    74→      const lines = renderCell(doc, M.left, left, CW, cursor)
    75→    75→      cursor += 4.2 + (lines - 1) * 3.1
    76→    76→      i += 1
    77→    77→      continue
    78→    78→    }
    79→    79→    const right = (i + 1 < rows.length && !rows[i + 1].wide) ? rows[i + 1] : undefined
    80→    80→    const leftLines = renderCell(doc, M.left, left, COL_W, cursor)
    81→    81→    const rightLines = right
    82→    82→      ? renderCell(doc, M.left + COL_W + COL_GAP, right, COL_W, cursor)
    83→    83→      : 1
    84→    84→    cursor += 4.2 + (Math.max(leftLines, rightLines) - 1) * 3.1
    85→    85→    i += right ? 2 : 1
    86→    86→  }
    87→    87→  return cursor
    88→    88→}

    90→    90→/** Draws one label + value-on-rule cell; returns the number of value lines. */
    91→    91→function renderCell(doc: jsPDF, x: number, row: FieldRow, w: number, y: number): number {
    92→    92→  const label = row.label.toUpperCase()
    93→    93→  doc.setFont('times', 'bold')
    94→    94→  doc.setFontSize(6)
    95→    95→  doc.setTextColor(INK.soft)
    96→    96→  const labelW = doc.getTextWidth(label) + label.length * 0.22
    97→    97→  doc.text(label, x, y + 2.7, { charSpace: 0.22 })
    98→    98→
    99→    99→  const ruleX = x + labelW + 2.2
   100→   100→  const ruleW = Math.max(8, x + w - ruleX)
   101→   101→
   102→   102→  doc.setDrawColor(INK.dotted)
   103→   103→  if (row.value !== undefined && row.value !== '') {
   104→   104→    const valueFont = row.mono ? 'courier' : 'times'
   105→   105→    doc.setFont(valueFont, 'normal')
   106→   106→    doc.setFontSize(7.5)
   107→   107→    doc.setTextColor(INK.main)
   108→   108→    // Wrap DOWNWARD: the first value line sits on the label baseline, wrapped
   109→   109→    // lines follow below, and the fill rule runs under the LAST line.
   110→   110→    const lines = doc.splitTextToSize(row.value, ruleW - 1) as string[]
   111→   111→    const list = Array.isArray(lines) ? lines : [String(lines)]
   112→   112→    for (let k = 0; k < list.length; k++) {
   113→   113→      doc.text(list[k], ruleX, y + 3.1 + k * 3.1)
   114→   114→    }
   115→   115→    const lastRuleY = y + 3.4 + (list.length - 1) * 3.1
   116→   116→    doc.setDrawColor(INK.rule)
   117→   117→    doc.setLineWidth(0.2)
   118→   118→    doc.setLineDashPattern([], 0)
   119→   119→    doc.line(ruleX, lastRuleY, x + w, lastRuleY)
   120→   120→    return list.length
   121→   121→  }
   122→   122→  // Blank copy — the classic dotted fill-in rule.
   123→   123→  const ruleY = y + 3.4
   124→   124→  doc.setLineWidth(0.35)
   125→   125→  doc.setLineDashPattern([0.45, 0.6], 0)
   126→   126→  doc.line(ruleX, ruleY, x + w, ruleY)
   127→   127→  doc.setLineDashPattern([], 0)
   128→   128→  return 1
   129→   129→}

   384→   384→function drawDeclaration(
   385→   385→  doc: jsPDF,
   386→   386→  y: number,
   387→   387→  app: SchoolApplication,
   388→   388→  sub: ApplicationSubmission | undefined,
   389→   389→  destination: string,
   390→   390→  datesLabel: string,
   391→   391→): number {
   392→   392→  const pad = 1.8
   393→   393→  const inner = CW - 2 * pad
   394→   394→
   395→   395→  doc.setFont('times', 'normal')
   396→   396→  doc.setFontSize(7)
   397→   397→  doc.setTextColor(INK.main)
   398→   398→  const intro = doc.splitTextToSize(
   399→   399→    `I/We, ${sub?.guardianName ?? '________________'}, parent/guardian of ${sub?.studentName ?? '________________'} of ${sub ? `${sub.className} — ${sub.section}` : '____________'}, hereby declare and undertake as follows:`,
   400→   400→    inner,
   401→   401→  ) as string[]
   402→   402→
   403→   403→  const items = [
   404→   404→    `I/We give full consent for my/our ward to participate in the Educational Tour to ${destination}${datesLabel ? ` (${datesLabel})` : ''} organised by the school.`,
   405→   405→    'The particulars furnished above are true and correct to the best of my/our knowledge.',
   406→   406→    'I/We have noted the tour dates, the fee payable and the conditions stated in the school circular.',
   407→   407→    'In case of illness or emergency during the tour, I/We authorise the school and the escorting staff to secure necessary medical assistance and treatment for my/our ward.',
   408→   408→    'My/our ward shall abide by the school\u2019s rules and the instructions of the escorting staff throughout the tour.',
   409→   409→  ].map((t) => doc.splitTextToSize(t, inner - 4) as string[])
   410→   410→
   411→   411→  const introH = intro.length * 3.1
   412→   412→  const itemsH = items.reduce((sum, lines) => sum + lines.length * 3.1 + 0.9, 0)
   413→   413→  const boxH = pad * 2 + introH + itemsH + 4.5
   414→   414→

   416→   416→  doc.setDrawColor(INK.soft)

   418→   418→  doc.setLineDashPattern([], 0)

   420→   420→
   421→   421→  let cy = y + pad + 2.6

   424→   424→    cy += 3.1
   425→   425→  }
   426→   426→  cy += 0.9
   427→   427→  let num = 1
   428→   428→  for (const lines of items) {

   447→   447→async function drawSignatureArea(
   448→   448→  doc: jsPDF,
   449→   449→  y: number,
   450→   450→  app: SchoolApplication,
   451→   451→  sub: ApplicationSubmission | undefined,
   452→   452→): Promise<number> {
   453→   453→  const colW = (CW - 12) / 3
   454→   454→  const cols = [
   455→   455→    { who: 'Student\u2019s Signature', name: sub?.studentName },
   456→   456→    { who: 'Parent / Guardian\u2019s Signature', name: sub?.guardianName, sig: sub?.signature },
   457→   457→    { who: 'Class Teacher / Tour In-charge', name: app.inChargeName },
   458→   458→  ]
   459→   459→  y += 4
   460→   460→  for (let c = 0; c < cols.length; c++) {
   461→   461→    const x = M.left + c * (colW + 6)
   462→   462→    const ruleY = y + 10
   463→   463→    const sig = cols[c].sig
   464→   464→    if (sig && sig.mode === 'drawn' && sig.data.startsWith('data:image/png')) {
   465→   465→      const dims = await loadImageSize(sig.data)
   466→   466→      if (dims && dims.w > 0 && dims.h > 0) {
   467→   467→        const maxW = colW - 6
   468→   468→        const maxH = 9
   469→   469→        const scale = Math.min(maxW / dims.w, maxH / dims.h)
   470→   470→        const w = dims.w * scale
   471→   471→        const h = dims.h * scale
   472→   472→        try {
   473→   473→          doc.addImage(sig.data, 'PNG', x + (colW - w) / 2, ruleY - h - 0.6, w, h)
   474→   474→        } catch { /* unembeddable image — leave the blank rule */ }
   475→   475→      }
   476→   476→    } else if (sig && sig.mode === 'typed') {
   477→   477→      doc.setFont('times', 'italic')
   478→   478→      doc.setFontSize(10)
   479→   479→      doc.setTextColor(INK.main)
   480→   480→      doc.text(sig.data, x + colW / 2, ruleY - 0.8, { align: 'center' })
   481→   481→    }
   482→   482→    // Dotted signature rule.
   483→   483→    doc.setDrawColor(INK.mute)
   484→   484→    doc.setLineWidth(0.35)
   485→   485→    doc.setLineDashPattern([0.45, 0.6], 0)
   486→   486→    doc.line(x, ruleY, x + colW, ruleY)
   487→   487→    doc.setLineDashPattern([], 0)
   488→   488→    // Caption.
   489→   489→    doc.setFont('times', 'bold')
   490→   490→    doc.setFontSize(5.5)
   491→   491→    doc.setTextColor(INK.main)
   492→   492→    doc.text(cols[c].who, x + colW / 2, ruleY + 2.6, { align: 'center' })
   493→   493→    doc.setFont('times', 'normal')
   494→   494→    doc.setFontSize(5.25)
   495→   495→    doc.setTextColor(INK.mute)
   496→   496→    const caption = sig
   497→   497→      ? `Recorded online · ${formatDate(sig.signedAt)}`
   498→   498→      : cols[c].name ?? 'Name: ______________'
   499→   499→    doc.text(caption, x + colW / 2, ruleY + 5.2, { align: 'center' })
   500→   500→  }
   501→   501→  return y + 16.5
   502→   502→}

   548→   548→function paymentOf(app: SchoolApplication, pay: SubmissionPaymentInfo | undefined): string | undefined {
   549→   549→  if (!pay || app.payment.mode === 'None') return undefined
   550→   550→  if (pay.status === 'Paid') return `PAID · Receipt ${pay.receiptNos.join(', ') || '—'}`
   551→   551→  if (pay.status === 'Awaiting Verification') return `PAYMENT PENDING · Receipt ${pay.pendingReceiptNo ?? '—'} (cash — under verification)`
   552→   552→  return 'NOT PAID'
   553→   553→}
