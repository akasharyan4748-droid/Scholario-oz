     1→'use client'
     2→
     3→/**
     4→ * ApplyDialog — the student's Educational Tour application: fill → submit →
     5→ * pay, inside ONE dialog.
     6→ *
     7→ * Structure mirrors the printed official form so what the student fills is
     8→ * exactly what the school office files:
     9→ *   1. Official document header (school, form no., session)
    10→ *   2. Tour details (read-only — destination, dates, fee, in-charge)
    11→ *   3. Student particulars (READ-ONLY, from the school record — applicants
    12→ *      never edit their master record here; corrections go through the office)
    13→ *   4. Guardian details (read-only)
    14→ *   5. Tour-specific details (the template's editable questions — saved to
    15→ *      the submission snapshot only, never the canonical student record)
    16→ *   6. Declaration & guardian consent
    17→ *   → review step: compact REVIEW APPLICATION summary before final submit
    18→ *   → payment step: Online (instant receipt) or Cash / Pay at School
    19→ *     (recorded Under Verification until the Principal confirms).
    20→ *
    21→ * MONEY RULE: payments go through fee-store recordPayment() only, bound to
    22→ * the application's Additional Charge + applicationId — tour money is
    23→ * separate from the student's annual fees and lands in the normal ledger.
    24→ *
    25→ * Idempotency: re-submitting an already-submitted form returns
    26→ * existingSubmissionId from the store — we surface "You already applied"
    27→ * and close instead of creating a duplicate.
    28→ */
    29→

    52→} from '@/lib/store/applications-store'

    60→type PayMode = 'UPI' | 'Cash'
    61→
    62→interface ApplyDialogProps {
    63→  open: boolean
    64→  onOpenChange: (open: boolean) => void
    65→  app: SchoolApplication | null
    66→  /** Display + canonical identity pair resolved by the module. */
    67→  identity: StudentIdentityPair | null
    68→  /** Present when fixing a 'Correction Required' submission. */
    69→  existingSubmission?: ApplicationSubmission | null
    70→  /** 'payment' skips straight to the payment step (awaiting-payment rows). */
    71→  initialStep?: 'form' | 'payment'
    72→}
    73→
    74→type Answers = Record<string, string | string[] | boolean>
    75→type Attachments = Record<string, { name: string; size: number }>
    76→type Errors = Record<string, string>
    77→
    78→const EM_DASH = ' — '
    79→
    80→function splitEmergencyContact(value: Answers[string]): { name: string; phone: string } {
    81→  if (typeof value === 'string' && value.includes(EM_DASH)) {
    82→    const [name = '', phone = ''] = value.split(EM_DASH)
    83→    return { name, phone }
    84→  }
    85→  return { name: '', phone: '' }
    86→}
    87→
    88→export function ApplyDialog({ open, onOpenChange, app, identity, existingSubmission = null, initialStep = 'form' }: ApplyDialogProps) {
    89→  const [step, setStep] = useState<'form' | 'review' | 'payment'>(initialStep === 'payment' ? 'payment' : 'form')
    90→  const [answers, setAnswers] = useState<Answers>({})
    91→  const [attachments, setAttachments] = useState<Attachments>({})
    92→  const [emergency, setEmergency] = useState<Record<string, { name: string; phone: string }>>({})
    93→  const [consentAccepted, setConsentAccepted] = useState(false)
    94→  const [errors, setErrors] = useState<Errors>({})
    95→  const [submitting, setSubmitting] = useState(false)
    96→  const [paying, setPaying] = useState<PayMode | null>(null)
    97→  const [activeSubmissionId, setActiveSubmissionId] = useState<string | null>(null)
    98→
    99→  const isFixMode = !!existingSubmission
   100→  // Active school's own branding (tenant-scoped School Settings).
   101→  const schGeneral = useSchoolSettingsStore((s) => s.general)
   102→
   103→  // ── Reset / prefill whenever the dialog opens for an application ──
   104→  useEffect(() => {
   105→    if (!open || !app) return
   106→    setStep(initialStep === 'payment' ? 'payment' : 'form')
   107→    setErrors({})
   108→    setSubmitting(false)
   109→    setPaying(null)
   110→    if (existingSubmission) {
   111→      setAnswers({ ...existingSubmission.answers })
   112→      setAttachments(existingSubmission.attachments ? { ...existingSubmission.attachments } : {})
   113→      setEmergency(Object.fromEntries(
   114→        app.formFields
   115→          .filter((f) => f.type === 'emergency-contact')
   116→          .map((f) => [f.id, splitEmergencyContact(existingSubmission.answers[f.id])]),
   117→      ))
   118→      setConsentAccepted(!!existingSubmission.consentGivenAt)
   119→      setActiveSubmissionId(existingSubmission.id)
   120→    } else {
   121→      setAnswers({})
   122→      setAttachments({})
   123→      setEmergency(Object.fromEntries(
   124→        app.formFields
   125→          .filter((f) => f.type === 'emergency-contact')
   126→          .map((f) => [f.id, { name: '', phone: '' }]),
   127→      ))
   128→      setConsentAccepted(false)
   129→      setActiveSubmissionId(null)
   130→    }
   131→  }, [open, app, existingSubmission, initialStep])
   132→
   133→  const amount = app?.payment.amount ?? 0
   134→  const needsPayment = !!app && app.payment.mode !== 'None'
   135→  const isTour = !!app && (app.category === 'Tour' || app.category === 'Trip' || app.templateKey === 'educational_tour')
   136→
   137→  const closeDialog = () => {
   138→    onOpenChange(false)
   139→  }
   140→
   141→  // ── Validation (application-specific fields + digital guardian consent) ──
   142→  const validate = (): boolean => {
   143→    if (!app) return false
   144→    const errs: Errors = {}
   145→    for (const f of app.formFields) {
   146→      const v = answers[f.id]
   147→      if (f.type === 'emergency-contact') {
   148→        const { name, phone } = emergency[f.id] ?? { name: '', phone: '' }
   149→        if (f.required && (!name.trim() || !phone.trim())) {
   150→          errs[f.id] = 'Enter both a contact name and a phone number.'
   151→        }
   152→        continue
   153→      }
   154→      if (f.type === 'file') {
   155→        if (f.required && !attachments[f.id]) errs[f.id] = 'Please attach a file.'
   156→        continue
   157→      }
   158→      if (f.type === 'signature' || f.type === 'declaration') {
   159→        if (f.required && v !== true) errs[f.id] = 'Tick the acknowledgement to continue.'
   160→        continue
   161→      }
   162→      if (f.type === 'checkbox' || f.type === 'multiselect') {
   163→        if (f.required && (!Array.isArray(v) || v.length === 0)) errs[f.id] = 'Pick at least one option.'
   164→        continue
   165→      }
   166→      if (f.type === 'yesno') {
   167→        if (f.required && typeof v !== 'boolean') errs[f.id] = 'Choose Yes or No.'
   168→        continue
   169→      }
   170→      if (f.required && (typeof v !== 'string' || !v.trim())) {
   171→        errs[f.id] = 'This field is required.'
   172→      }
   173→    }
   174→    if (app.guardianConsent.required && app.guardianConsent.method === 'Digital' && !consentAccepted) {
   175→      errs.consent = 'Guardian consent is required before submitting.'
   176→    }
   177→    setErrors(errs)
   178→    if (Object.keys(errs).length > 0) {
   179→      toast.error('Please complete the highlighted fields.')
   180→      return false
   181→    }
   182→    return true
   183→  }
   184→
   185→  // ── Submit (or resubmit after corrections) — reached from the review step ──
   186→  const handleSubmit = () => {
   187→    if (!app || !identity || submitting) return
   188→    if (initialStep !== 'payment' && !validate()) {
   189→      setStep('form')
   190→      return
   191→    }
   192→
   193→    setSubmitting(true)
   194→
   195→    if (isFixMode && existingSubmission) {
   196→      const res = useApplicationsStore.getState().resubmitSubmission(
   197→        existingSubmission.id,
   198→        answers,
   199→        Object.keys(attachments).length ? attachments : undefined,
   200→        identity.canonical.name,
   201→      )
   202→      setSubmitting(false)
   203→      if (!res.success) {
   204→        toast.error(res.error ?? 'Could not resubmit.')
   205→        return
   206→      }
   207→      const updated = useApplicationsStore.getState().submissions.find((s) => s.id === existingSubmission.id)
   208→      const pay = updated ? deriveSubmissionPayment(app, updated) : null
   209→      if (needsPayment && (!pay || pay.status === 'Not Paid')) {
   210→        setActiveSubmissionId(existingSubmission.id)
   211→        setStep('payment')
   212→        toast.info('Corrections submitted — one more step: payment.')
   213→      } else {
   214→        toast.success('Corrections submitted for review.')
   215→        closeDialog()
   216→      }
   217→      return
   218→    }
   219→
   220→    const res = useApplicationsStore.getState().submitApplication({
   221→      applicationId: app.id,
   222→      student: identity.canonical,
   223→      answers,
   224→      attachments: Object.keys(attachments).length ? attachments : undefined,
   225→      consentAccepted,
   226→      submittedByRole: 'Student',
   227→    })
   228→
   229→    if (res.success && res.existingSubmissionId) {
   230→      // Idempotent hit — the store refused to duplicate. Be honest + close.
   231→      setSubmitting(false)
   232→      toast.info('You already applied', {
   233→        description: 'A submission for this form already exists — nothing was duplicated.',
   234→      })
   235→      closeDialog()
   236→      return
   237→    }
   238→    if (!res.success || !res.submission) {
   239→      setSubmitting(false)
   240→      toast.error(res.error ?? 'Submission failed.')
   241→      return
   242→    }
   243→
   244→    if (needsPayment) {
   245→      setActiveSubmissionId(res.submission.id)
   246→      setStep('payment')
   247→      setSubmitting(false)
   248→      toast.success('Application submitted — complete the payment to finish.', {
   249→        description: formatINR(amount),
   250→      })
   251→    } else {
   252→      setSubmitting(false)
   253→      toast.success('Application submitted.', {
   254→        description: 'Track its status under My submissions.',
   255→      })
   256→      closeDialog()
   257→    }
   258→  }
   259→
   260→  // ── Payment — the ONLY way money moves (canonical fee store) ──
   261→  const runPayment = (mode: PayMode) => {
   262→    if (!app || !identity || paying) return
   263→    if (!app.payment.chargeId) return // guarded in the UI
   264→    setPaying(mode)
   265→    const res = useFeeStore.getState().recordPayment({
   266→      studentId: identity.canonical.id,
   267→      amount: app.payment.amount,
   268→      mode,
   269→      purpose: `Application: ${app.title}`,
   270→      feeHead: app.payment.feeHeadLabel || app.title,
   271→      collectedBy: 'Student Self-Service',
   272→      additionalChargeId: app.payment.chargeId,
   273→      applicationId: app.id,
   274→      ...(mode === 'UPI' ? { referenceNo: `UPI-${Date.now()}` } : {}),
   275→    })
   276→    setPaying(null)
   277→    if (!res.success || !res.transaction) {
   278→      toast.error(res.error ?? 'Payment could not be recorded.')
   279→      return
   280→    }
   281→    const receipt = res.transaction.receiptNo
   282→    addAuditEvent({
   283→      applicationId: app.id,
   284→      submissionId: activeSubmissionId ?? undefined,
   285→      ts: new Date().toISOString(),
   286→      actor: identity.canonical.name,
   287→      actorRole: 'Student',
   288→      action: 'payment.completed',
   289→      message: mode === 'Cash'
   290→        ? `Cash ${formatINR(app.payment.amount)} recorded for "${app.title}" — receipt ${receipt}, awaiting verification by the Principal.`
   291→        : `Paid ${formatINR(app.payment.amount)} online (UPI ref ${res.transaction.referenceNo ?? '—'}) for "${app.title}" — receipt ${receipt}.`,
   292→    })
   293→    if (mode === 'Cash') {
   294→      toast.info('Recorded — awaiting cash verification by the Principal', {
   295→        description: `Receipt ${receipt} · ${formatINR(app.payment.amount)} · ${app.title}`,
   296→      })
   297→    } else {
   298→      toast.success(`Payment successful — receipt ${receipt}`, {
   299→        description: `${formatINR(app.payment.amount)} paid for ${app.title}.`,
   300→      })
   301→    }
   302→    closeDialog()
   303→  }
   304→
   305→  const skipPayment = () => {
   306→    toast.info('Payment skipped for now', {
   307→      description: 'You can pay anytime from My submissions.',
   308→    })
   309→    closeDialog()
   310→  }
   311→
   312→  return (
   313→    <Dialog open={open} onOpenChange={(o) => { if (!o) closeDialog() }}>
   314→      <DialogContent className="sm:max-w-lg max-h-[88vh] overflow-hidden flex flex-col">
   315→        {app && identity && (
   316→          <>
   317→            <DialogHeader className="text-left shrink-0">
   318→              <div className="flex items-start justify-between gap-3">
   319→                <div className="min-w-0">
   320→                  <DialogTitle className="text-base leading-snug">{app.title}</DialogTitle>
   321→                  <DialogDescription className="mt-1 flex flex-wrap items-center gap-1.5 text-[11px]">
   322→                    <Badge variant="outline" className="text-[9px] h-4 px-1.5">Educational Tour</Badge>
   323→                    <span>Deadline {formatDate(app.deadline)}</span>
   324→                    {needsPayment && <span>· {formatINR(amount)}{app.payment.mode === 'Optional' ? ' (optional)' : ''}</span>}
   325→                  </DialogDescription>
   326→                </div>
   327→                {step === 'payment' && (
   328→                  <Badge variant="outline" className="shrink-0 text-[9px] h-4 px-1.5 border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-400">
   329→                    Payment
   330→                  </Badge>
   331→                )}
   332→                {step === 'review' && (
   333→                  <Badge variant="outline" className="shrink-0 text-[9px] h-4 px-1.5 border-sky-300 bg-sky-50 text-sky-700 dark:border-sky-500/30 dark:bg-sky-500/10 dark:text-sky-400">
   334→                    Review
   335→                  </Badge>
   336→                )}
   337→              </div>
   338→            </DialogHeader>
   339→
   340→            {step === 'form' ? (
   341→              <div className="min-h-0 flex-1 overflow-y-auto -mx-1 px-1 space-y-4">
   342→                {/* ── Official document header — the form reads as a real
   343→                    school-issued document, not a SaaS card. ── */}
   344→                <div className="rounded-lg border border-border bg-card px-3.5 py-3">
   345→                  <div className="flex items-start justify-between gap-3 border-b border-dashed border-border pb-2.5">
   346→                    <div className="min-w-0">
   347→                      <p className="text-[13px] font-extrabold tracking-tight text-foreground leading-tight">{schGeneral.schoolName}</p>
   348→                      {schGeneral.affiliation && <p className="text-[9px] text-muted-foreground mt-0.5">{schGeneral.affiliation}</p>}
   349→                    </div>
   350→                    <div className="text-right shrink-0">
   351→                      <p className="text-[8px] uppercase tracking-[0.18em] text-muted-foreground">Form No.</p>
   352→                      <p className="text-[10px] font-mono font-semibold">APPF-{app.id.slice(-8).toUpperCase()}</p>
   353→                      <p className="text-[8px] uppercase tracking-[0.18em] text-muted-foreground mt-1">Session</p>
   354→                      <p className="text-[10px] font-semibold">{app.academicYear}</p>
   355→                    </div>
   356→                  </div>
   357→                  <p className="mt-2 text-[8.5px] uppercase tracking-[0.2em] text-muted-foreground">Official {formPurposeOf(app)} Form · {app.category}</p>
   358→                  <p className="text-sm font-bold leading-tight mt-0.5">{app.title}</p>
   359→                </div>
   360→
   361→                {/* ── Activity details (read-only) ── */}
   362→                <section className="rounded-lg border border-border px-3.5 py-3">
   363→                  <p className="text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">1. {isTour ? 'Tour' : 'Activity'} details</p>
   364→                  <div className="mt-2 grid grid-cols-2 gap-x-4 gap-y-1.5 text-[11px]">
   365→                    {app.destination && <Particular icon={<Bus className="h-3 w-3" />} label="Destination" value={app.destination} />}
   366→                    <Particular icon={<CalendarDays className="h-3 w-3" />} label={isTour ? 'Tour date' : 'Event date'} value={app.eventDate ? formatDate(app.eventDate) : 'To be announced'} />
   367→                    <Particular icon={<CalendarDays className="h-3 w-3" />} label="Last date to apply" value={formatDate(app.deadline)} />
   368→                    <Particular icon={<Landmark className="h-3 w-3" />} label="Fee per student" value={needsPayment ? `${formatINR(amount)}${app.payment.mode === 'Optional' ? ' (optional)' : ''}` : 'Free'} />
   369→                    <Particular icon={<Users className="h-3 w-3" />} label="Teacher in-charge" value={app.inChargeName ?? 'To be assigned'} />
   370→                  </div>
   371→                  {app.description && (
   372→                    <p className="mt-2 text-[10.5px] leading-relaxed text-muted-foreground border-l-2 border-border pl-2.5">{app.description}</p>
   373→                  )}
   374→                </section>
   375→
   376→                {/* ── Student particulars (read-only, canonical record — §20/§21
   377→                    minimal set: only what the form needs; DOB/blood group stay
   378→                    in the school record) ── */}
   379→                <section className="rounded-lg border border-border bg-muted/25 px-3.5 py-3">
   380→                  <p className="flex items-center gap-1.5 text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">
   381→                    <User className="h-3 w-3" /> 2. Student particulars
   382→                  </p>
   383→                  <div className="mt-2 grid grid-cols-2 gap-x-4 gap-y-1.5 text-[11px]">
   384→                    <Particular label="Name" value={identity.canonical.name} />
   385→                    <Particular label="Admission no." value={identity.canonical.admissionNo} mono />
   386→                    <Particular label="Class / Section" value={`${identity.canonical.className} — ${identity.canonical.section}`} />
   387→                    <Particular label="Roll no." value={identity.canonical.rollNo} />
   388→                  </div>
   389→                  <p className="mt-2 text-[9.5px] text-muted-foreground">Taken from the school record — corrections go through the office.</p>
   390→                </section>
   391→
   392→                {/* ── Guardian details (read-only) ── */}
   393→                <section className="rounded-lg border border-border bg-muted/25 px-3.5 py-3">
   394→                  <p className="flex items-center gap-1.5 text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">
   395→                    <Users className="h-3 w-3" /> 3. Parent / guardian details
   396→                  </p>
   397→                  <div className="mt-2 grid grid-cols-2 gap-x-4 gap-y-1.5 text-[11px]">
   398→                    <Particular label="Guardian name" value={identity.canonical.guardianName} />
   399→                    <Particular label="Guardian phone" value={identity.canonical.guardianPhone} />
   400→                  </div>
   401→                </section>
   402→
   403→                {/* ── Form questions (editable), grouped by section ── */}
   404→                {(() => {
   405→                  const groups = new Map<string, typeof app.formFields>()
   406→                  for (const f of app.formFields) {
   407→                    const key = f.section ?? 'Additional Questions'
   408→                    const arr = groups.get(key) ?? []
   409→                    arr.push(f)
   410→                    groups.set(key, arr)
   411→                  }
   412→                  let sectionNo = 3
   413→                  return Array.from(groups.entries()).map(([section, fields]) => {
   414→                    sectionNo += 1
   415→                    return (
   416→                      <section key={section} className="space-y-3.5">
   417→                        <p className="text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">{sectionNo}. {section}</p>
   418→                        {fields.map((f) => (
   419→                          <FieldRenderer
   420→                            key={f.id}
   421→                            field={f}
   422→                            value={answers[f.id]}
   423→                            attachment={attachments[f.id]}
   424→                            emergency={emergency[f.id] ?? { name: '', phone: '' }}
   425→                            error={errors[f.id]}
   426→                            onAnswer={(v) => {
   427→                              setAnswers((prev) => ({ ...prev, [f.id]: v }))
   428→                              setErrors((prev) => { if (!prev[f.id]) return prev; const next = { ...prev }; delete next[f.id]; return next })
   429→                            }}
   430→                            onEmergency={(v) => {
   431→                              setEmergency((prev) => ({ ...prev, [f.id]: v }))
   432→                              setAnswers((prev) => ({ ...prev, [f.id]: [v.name.trim(), v.phone.trim()].filter(Boolean).join(EM_DASH) }))
   433→                              setErrors((prev) => { if (!prev[f.id]) return prev; const next = { ...prev }; delete next[f.id]; return next })
   434→                            }}
   435→                            onAttachment={(file) => {
   436→                              setAttachments((prev) => {
   437→                                const next = { ...prev }
   438→                                if (file) next[f.id] = { name: file.name, size: file.size }
   439→                                else delete next[f.id]
   440→                                return next
   441→                              })
   442→                              setErrors((prev) => { if (!prev[f.id]) return prev; const next = { ...prev }; delete next[f.id]; return next })
   443→                            }}
   444→                          />
   445→                        ))}
   446→                      </section>
   447→                    )
   448→                  })
   449→                })()}
   450→
   451→                {/* ── Declaration & consent ── */}
   452→                <section className="rounded-lg border border-border px-3.5 py-3">
   453→                  <p className="flex items-center gap-1.5 text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">
   454→                    <ShieldCheck className="h-3 w-3" /> Declaration &amp; guardian consent
   455→                  </p>
   456→                  <p className="mt-2 text-[10.5px] leading-relaxed text-muted-foreground">
   457→                    The particulars above are taken from the school record. All tour-specific information furnished is true to the best of my knowledge.
   458→                  </p>
   459→                  {app.guardianConsent.required && (
   460→                    app.guardianConsent.method === 'Digital' ? (
   461→                      <label
   462→                        className={cn(
   463→                          'mt-2 flex items-start gap-2.5 rounded-md p-2 -m-2 cursor-pointer',
   464→                          errors.consent && 'ring-1 ring-rose-300 rounded-md',

   569→                  <p className="rounded-md border border-amber-200 bg-amber-50 px-3 py-2.5 text-[11px] leading-relaxed text-amber-800 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-400">
   570→                    This form is not linked to a fee-ledger charge yet, so self-service payment is unavailable. Please pay at the school office.
   571→                  </p>
   572→                )}
   573→              </div>
   574→            )}
   575→
   576→            <DialogFooter className="shrink-0 border-t border-border pt-3 mt-1">
   577→              {step === 'form' ? (
   578→                <>
   579→                  <Button variant="ghost" size="sm" className="h-8 text-xs" onClick={closeDialog} disabled={submitting}>
   580→                    Cancel
   581→                  </Button>
   582→                  <Button size="sm" className="h-8 text-xs" onClick={() => { if (validate()) setStep('review') }}>
   583→                    Review Application
   584→                  </Button>
   585→                </>
   586→              ) : step === 'review' ? (
   587→                <>
   588→                  <Button variant="ghost" size="sm" className="h-8 text-xs" onClick={() => setStep('form')} disabled={submitting}>
   589→                    Back to form
   590→                  </Button>
   591→                  <Button size="sm" className="h-8 text-xs" onClick={handleSubmit} disabled={submitting}>
   592→                    <CheckCircle2 className="h-3.5 w-3.5" />
   593→                    {submitting ? 'Submitting…' : 'Submit Application'}
   594→                  </Button>
   595→                </>
   596→              ) : (
   597→                <div className="flex w-full items-center justify-between gap-2">
   598→                  {app.payment.mode === 'Optional' ? (
   599→                    <button type="button" onClick={skipPayment} className="text-[11px] font-medium text-muted-foreground underline-offset-2 hover:underline">
   600→                      Skip for now
   601→                    </button>
   602→                  ) : (
   603→                    <span className="text-[10px] text-muted-foreground">You can also pay later from My submissions.</span>
   604→                  )}
   605→                  <Button variant="ghost" size="sm" className="h-8 text-xs" onClick={closeDialog}>
   606→                    Close
   607→                  </Button>
   608→                </div>
   609→              )}
   610→            </DialogFooter>
   611→          </>
   612→        )}
   613→      </DialogContent>
   614→    </Dialog>
   615→  )

   643→interface FieldRendererProps {

   654→function FieldRenderer({ field, value, attachment, emergency, error, onAnswer, onEmergency, onAttachment }: FieldRendererProps) {
   655→  const toggleOption = (opt: string) => {
   656→    const current = Array.isArray(value) ? value : []
   657→    onAnswer(current.includes(opt) ? current.filter((o) => o !== opt) : [...current, opt])
   658→  }
   659→
   660→  return (
   661→    <div>
   662→      <Label className="text-xs font-medium">
   663→        {field.label}
   664→        {field.required && <span className="text-rose-500 ml-0.5">*</span>}
   665→      </Label>
   666→      {field.helpText && <p className="mt-0.5 text-[10px] text-muted-foreground">{field.helpText}</p>}
   667→
   668→      <div className="mt-1.5">
   669→        {field.type === 'text' && (
   670→          <Input
   671→            type="text"
   672→            value={typeof value === 'string' ? value : ''}
   673→            onChange={(e) => onAnswer(e.target.value)}
   674→            className="h-8 text-xs"
   675→            aria-label={field.label}
   676→          />
   677→        )}
   678→
   679→        {field.type === 'number' && (
   680→          <Input
   681→            type="number"
   682→            value={typeof value === 'string' ? value : ''}
   683→            onChange={(e) => onAnswer(e.target.value)}
   684→            className="h-8 text-xs"
   685→            aria-label={field.label}
   686→          />
   687→        )}
   688→
   689→        {/* Compact pop-over date picker (Examination-module pattern) — never
   690→            the oversized native date input. */}
   691→        {field.type === 'date' && (
   692→          <DatePicker
   693→            compact
   694→            value={typeof value === 'string' ? value : ''}
   695→            onChange={(v) => onAnswer(v)}
   696→            placeholder="Select date"
   697→          />
   698→        )}
   699→
   700→        {field.type === 'longtext' && (
   701→          <Textarea
   702→            value={typeof value === 'string' ? value : ''}
   703→            onChange={(e) => onAnswer(e.target.value)}
   704→            rows={3}
   705→            className="text-xs min-h-[64px]"
   706→            aria-label={field.label}
   707→          />
   708→        )}
   709→
   710→        {field.type === 'dropdown' && (
   711→          <Select value={typeof value === 'string' ? value : ''} onValueChange={(v) => onAnswer(v)}>
   712→            <SelectTrigger className="h-8 text-xs" aria-label={field.label}><SelectValue placeholder="Select…" /></SelectTrigger>
   713→            <SelectContent className="z-[70]">
   714→              {(field.options ?? []).map((opt) => (
   715→                <SelectItem key={opt} value={opt} className="text-xs">{opt}</SelectItem>
   716→              ))}
   717→            </SelectContent>
   718→          </Select>
   719→        )}
   720→
   721→        {field.type === 'radio' && (
   722→          <RadioGroup
   723→            value={typeof value === 'string' ? value : ''}
   724→            onValueChange={(v) => onAnswer(v)}
   725→            className="mt-0.5 flex flex-wrap gap-x-4 gap-y-1.5"
   726→            aria-label={field.label}
   727→          >
   728→            {(field.options ?? []).map((opt) => (
   729→              <label key={opt} className="flex cursor-pointer items-center gap-1.5 text-xs">
   730→                <RadioGroupItem value={opt} className="h-3.5 w-3.5" />
   731→                {opt}
   732→              </label>
   733→            ))}
   734→          </RadioGroup>
   735→        )}
   736→
   737→        {(field.type === 'checkbox' || field.type === 'multiselect') && (
   738→          <div className="flex flex-wrap gap-1.5">
   739→            {(field.options ?? []).map((opt) => {
   740→              const active = Array.isArray(value) && value.includes(opt)
   741→              return (
   742→                <button
   743→                  key={opt}
   744→                  type="button"
   745→                  onClick={() => toggleOption(opt)}
   746→                  aria-pressed={active}
   747→                  className={cn(
   748→                    'rounded-full border px-2.5 py-1 text-[11px] transition-colors',
   749→                    active
   750→                      ? 'border-primary/40 bg-primary/10 text-primary font-medium'
   751→                      : 'border-border text-muted-foreground hover:bg-muted/60',
   752→                  )}
   753→                >
   754→                  {opt}
   755→                </button>
   756→              )
   757→            })}
   758→          </div>
   759→        )}
   760→
   761→        {field.type === 'yesno' && (
   762→          <RadioGroup
   763→            value={typeof value === 'boolean' ? (value ? 'yes' : 'no') : ''}
   764→            onValueChange={(v) => onAnswer(v === 'yes')}

   794→              aria-label={field.label}
   795→            />

   831→              I understand a physical signature is required

   840→              onCheckedChange={(c) => onAnswer(c === true)}
   841→              className="mt-0.5"

   852→    </div>
