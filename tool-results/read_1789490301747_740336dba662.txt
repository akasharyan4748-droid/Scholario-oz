   131→   131→async function request<T>(url: string, init?: RequestInit): Promise<T> {
   132→   132→  const r = await fetch(url, {
   133→   133→    ...init,
   134→   134→    cache: 'no-store',
   135→   135→    credentials: 'same-origin',
   136→   136→    headers: init?.body ? { 'Content-Type': 'application/json', ...init?.headers } : init?.headers,
   137→   137→  })
   138→   138→  if (!r.ok) {
   139→   139→    if (r.status === 401) {
   140→   140→      handleExpiredSession()
   141→   141→      throw new Error('Your session has expired. Please sign in again.')
   142→   142→    }
   143→   143→    let message = `Request failed (${r.status})`
   144→   144→    try {
   145→   145→      const j: unknown = await r.json()
   146→   146→      if (j && typeof j === 'object' && typeof (j as { error?: unknown }).error === 'string') {
   147→   147→        message = (j as { error: string }).error
   148→   148→      }
   149→   149→    } catch {
   150→   150→      /* non-JSON error body — keep the concise fallback */
   151→   151→    }
   152→   152→    throw new Error(message)
   153→   153→  }
   154→   154→  const j: unknown = await r.json()
   155→   155→  if (!j || typeof j !== 'object' || (j as { ok?: unknown }).ok !== true || !('data' in j)) {
   156→   156→    throw new Error('Unexpected response from the server.')
   157→   157→  }
   158→   158→  return (j as { data: T }).data
   159→   159→}

   164→   164→function initialWorking(detail: AttendanceDetail | null): Record<string, Status> {
   165→   165→  const w: Record<string, Status> = {}
   166→   166→  if (!detail) return w
   167→   167→  const source = detail.you?.entries ?? detail.prefill?.entries ?? null
   168→   168→  const map = source ? new Map(source.map((e) => [e.studentId, e.status.toLowerCase() as Status])) : null
   169→   169→  for (const s of detail.roster) {
   170→   170→    w[s.id] = map?.get(s.id) ?? 'present'
   171→   171→  }
   172→   172→  return w
   173→   173→}

   175→   175→function sameWorking(a: Record<string, Status>, b: Record<string, Status>): boolean {
   176→   176→  const ka = Object.keys(a)
   177→   177→  if (ka.length !== Object.keys(b).length) return false
   178→   178→  return ka.every((k) => a[k] === b[k])
   179→   179→}
