   340→   340→// Quick "in words" for date of birth (very compact).
   341→   341→function birthWords(dob: string): string {
   342→   342→  const d = new Date(dob)
   343→   343→  if (Number.isNaN(d.getTime())) return '—'
   344→   344→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December']
   345→   345→  return `${ordinal(d.getDate())} ${months[d.getMonth()]} ${d.getFullYear()}`
   346→   346→}
   347→   347→function ordinal(n: number): string {
   348→   348→  const s = ['th', 'st', 'nd', 'rd']
   349→   349→  const v = n % 100
   350→   350→  return n + (s[(v - 20) % 10] || s[v] || s[0])
   351→   351→}

   499→   499→function gradeFor(pct: number): string {
   500→   500→  if (pct >= 90) return 'A1'
   501→   501→  if (pct >= 80) return 'A2'
   502→   502→  if (pct >= 70) return 'B1'
   503→   503→  if (pct >= 60) return 'B2'
   504→   504→  if (pct >= 50) return 'C1'
   505→   505→  if (pct >= 33) return 'C2'
   506→   506→  return 'E'
   507→   507→}
   508→   508→function divisionFor(pct: number): string {
   509→   509→  if (pct >= 60) return 'First'
   510→   510→  if (pct >= 45) return 'Second'
   511→   511→  if (pct >= 33) return 'Third'
   512→   512→  return 'Fail'
   513→   513→}
