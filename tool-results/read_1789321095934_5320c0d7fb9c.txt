   211→// ─── ClassSubjectAssignment (Spec §19/§20) ─────────────────────────────
   212→// Separates the SUBJECT (canonical, school-scoped) from its ASSIGNMENT to
   213→// a specific class (+ optional stream). One canonical "Hindi" subject can
   214→// be assigned to many classes. Removing an assignment does NOT delete the
   215→// canonical subject — historical records remain valid (Spec §16/§25).
   216→model ClassSubjectAssignment {
   217→  id           String    @id @default(cuid())
   218→  schoolId     String
   219→  school       School    @relation(fields: [schoolId], references: [id], onDelete: Cascade)
   220→  classId      String
   221→  class        Class     @relation(fields: [classId], references: [id], onDelete: Cascade)
   222→  subjectId    String
   223→  subject      Subject   @relation(fields: [subjectId], references: [id], onDelete: Cascade)
   224→  // Spec §34: distinguishes core/default subjects from additional/optional ones.
   225→  isCore       Boolean   @default(true)
   226→  // Spec §18/§35: an inactive assignment is excluded from new examinations
   227→  // but historical records remain valid. Archived subjects are not selectable.
   228→  isActive     Boolean   @default(true)
   229→  // Spec §33: some subjects are examinable, others are not (e.g. PE, Music).
   230→  examinable   Boolean   @default(true)
   231→  // Spec §33: deterministic ordering for display.
   232→  displayOrder Int       @default(0)
   233→  createdAt    DateTime  @default(now())
   234→  updatedAt    DateTime? @updatedAt
   235→
   236→  // Spec §12/§19: one assignment per (class, subject) — prevents duplicates.
   237→  @@unique([classId, subjectId])
   238→}

   931→// ─── Phase 9 — Enterprise Fee Architecture (DB-backed) ───────────────
   932→//
   933→// The fee-store has been in-memory + Zustand-persist (localStorage) since
   934→// takeover. These new models make the fee system truly DB-backed and
   935→// multi-tenant-safe, and let the webhook route auto-reconcile instead of
   936→// just logging. All new models are ADDITIVE — the existing Fee + Payment
   937→// models remain untouched so the legacy /api/fees route keeps working.
   938→//
   939→// Architectural mapping (store type → DB row):
   940→//   - school-settings-store.fees.feeHeads[]  → MasterFeeHead
   941→//   - fee-store.FEE_STRUCTURES[]             → FeeStructure (+ FeeHead rows)
   942→//   - fee-store.FeeStructureVersion          → FeeStructureVersion
   943→//   - fee-store.SEED_TRANSACTIONS[]          → FeeTransaction (immutable)
   944→//   - fee-store.settlements[]                → Settlement
   945→//   - fee-store.reconciliations[]            → Reconciliation
   946→//   - fee-store.webhookEvents[]              → WebhookEvent
   947→//
   948→// Settlement status lifecycle:
   949→//   pending → settled → reconciled
   950→//                     ↘ exception → (manual review) → reconciled
   951→//   failure → reversed
   952→
   953→/// School-wide master fee-head catalogue — single source of truth for
   954→/// head names, default frequencies, default amounts, categories.
   955→/// Picked from when building per-class fee structures.
   956→model MasterFeeHead {
   957→  id          String   @id @default(cuid())

   960→  name        String
   961→  category    String   @default("Other") // Tuition, Admission, Annual, Transport, Lab, Library, Exam, Activity, Board, Other
   962→  frequency   String   @default("Monthly") // Annual, Half-Yearly, Quarterly, Monthly, Per Term, One-Time
   963→  amount      Float    @default(0)
   964→  mandatory   Boolean  @default(true)
   965→  active      Boolean  @default(true)
   966→  description String?
   967→  sortOrder   Int      @default(0)
   968→  createdAt   DateTime @default(now())
   969→  updatedAt   DateTime @updatedAt
   970→
   971→  /// Per-structure line items bound to this catalogue entry (read model).
   972→  structures FeeHead[]
   973→
   974→  @@unique([schoolId, name])
   975→}
