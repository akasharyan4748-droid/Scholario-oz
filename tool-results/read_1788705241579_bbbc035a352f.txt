    40→    40→/** Teachers the student may request a meeting with (class teacher + subject
    41→    41→ *  teachers of THEIR class — never a school-wide directory, spec §30). */
    42→    42→function allowedTeachers(classId: string, section: string): { id: string; name: string; subject: string }[] {
    43→    43→  const cls = useStudentsStore.getState().classes.find((c) => c.id === classId)
    44→    44→  if (!cls) return []
    45→    45→  const sec = cls.sections.find((s) => s.name === section)
    46→    46→  const roster: { id: string; name: string; subject: string }[] = []
    47→    47→  const seen = new Set<string>()
    48→    48→  const push = (id: string | undefined, subject: string) => {
    49→    49→    if (!id || seen.has(id)) return
    50→    50→    const t = teachers.find((x) => x.id === id)
    51→    51→    if (!t) return
    52→    52→    seen.add(id)
    53→    53→    roster.push({ id: t.id, name: t.name, subject })
    54→    54→  }
    55→    55→  push(sec?.classTeacherId ?? cls.classTeacherId, 'Class Teacher')
    56→    56→  for (const [subId, tid] of Object.entries(cls.subjectTeachers)) {
    57→    57→    const name = cls.subjects.find((n) => n === subId) ?? subId.replace('sub-', '')
    58→    58→    push(tid, name.charAt(0).toUpperCase() + name.slice(1))
    59→    59→  }
    60→    60→  return roster
    61→    61→}
