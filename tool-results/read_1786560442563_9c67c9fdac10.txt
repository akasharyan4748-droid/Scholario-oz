    69→function formatDateRange(start: string, end: string): string {
    70→  if (!start) return '—'
    71→  const [sy, sm, sd] = start.split('-').map(Number)
    72→  const startDate = new Date(sy, sm - 1, sd)
    73→  if (start === end) {
    74→    return startDate.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    75→  }
    76→  const [ey, em, ed] = end.split('-').map(Number)
    77→  const endDate = new Date(ey, em - 1, ed)
    78→  return `${startDate.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })} – ${endDate.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })}`
    79→}

   611→function KpiCard({ label, value, sub, icon, tone }: {

   618→  }[tone]
