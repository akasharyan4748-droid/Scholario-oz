     1→/**
     2→ * curriculum-execution — the AUTOMATED CURRICULUM EXECUTION engine.
     3→ *
     4→ * Lesson Planner is NOT manual planning. The teacher selects a (class,
     5→ * subject) teaching assignment; the platform resolves the MASTER CURRICULUM
     6→ * (structured CBSE data — never AI-generated), derives the day-wise teaching
     7→ * schedule from the class TIMETABLE + ACADEMIC CALENDAR (holidays), and the
     8→ * teacher simply executes: start → complete. Progress updates automatically.
     9→ *
    10→ * SECURITY MODEL (mirrors teacher-attendance.ts):
    11→ *   erp_session cookie → requireTeacher → school scope → ACTIVE teaching
    12→ *   assignments (ClassSubjectAssignment). Being class teacher grants NO
    13→ *   lesson-planner scope (§24) — only assignments do. Client-supplied
    14→ *   classId/subjectId/topicId are re-validated server-side on every call;
    15→ *   LessonExecution rows are (schoolId, classId, subjectId, teacherId)-
    16→ *   scoped, so no other school's or teacher's execution state is reachable.
    17→ *
    18→ * SCHEDULING MODEL (cursor-based projection):
    19→ *   · COMPLETED/SKIPPED topics are history — they keep their real dates and
    20→ *     are never recalculated (§33/§34).
    21→ *   · The CURSOR = first topic without a COMPLETED/SKIPPED record. It is
    22→ *     always the next lesson to teach, on the next eligible slot — a missed
    23→ *     day self-heals (the topic repeats) and a holiday simply shifts the
    24→ *     projection forward (§11/§33).
    25→ *   · Future topics are PROJECTED onto upcoming eligible slots (weekday has
    26→ *     a timetable slot for the class+subject AND the date is not a holiday).
    27→ *     A topic needing more minutes than one slot spans consecutive slots.
    28→ *   · POSTPONED marks today's slot as moved — the topic re-appears at the
    29→ *     next eligible slot (§16/§17). All overrides are persisted rows
    30→ *     (auditability §17).
    31→ */

    49→function dayISO(d: Date): string {
    50→  return d.toISOString().slice(0, 10)
    51→}

    53→function parseDay(v: unknown, field = 'Date'): Date {
    54→  if (typeof v !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(v.trim())) {
    55→    throw new Error(`${field} must be a YYYY-MM-DD date`)
    56→  }
    57→  const d = new Date(`${v.trim()}T00:00:00.000Z`)
    58→  if (Number.isNaN(d.getTime())) throw new Error(`${field} is not a valid date`)
    59→  return d
    60→}

    62→function todayUTC(): Date {
    63→  const d = new Date()
    64→  d.setUTCHours(0, 0, 0, 0)
    65→  return d
    66→}

    68→function addDays(d: Date, n: number): Date {
    69→  const x = new Date(d)
    70→  x.setUTCDate(x.getUTCDate() + n)
    71→  return x
    72→}

    74→const DAY_NAMES = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'] as const
    75→
    76→function dayIndex(name: string): number {
    77→  const i = DAY_NAMES.findIndex((d) => d.toLowerCase() === name.trim().toLowerCase())
    78→  if (i < 0) throw new Error(`Unknown weekday in timetable: ${name}`)
    79→  return i
    80→}

    82→/** minutes between "HH:MM" strings (timetable rows) */
    83→function minutesBetween(start: string | null, end: string | null): number | null {
    84→  if (!start || !end) return null
    85→  const m = (s: string): number => {
    86→    const [h, mm] = s.split(':').map((x) => Number.parseInt(x, 10))
    87→    if (Number.isNaN(h) || Number.isNaN(mm)) return NaN
    88→    return h * 60 + mm
    89→  }
    90→  const a = m(start)
    91→  const b = m(end)
    92→  if (Number.isNaN(a) || Number.isNaN(b) || b <= a) return null
    93→  return b - a
    94→}

    98→export async function teachingAssignments(ctx: TeacherHubContext): Promise<TeachingAssignmentRef[]> {
    99→  const rows = await db.classSubjectAssignment.findMany({
   100→    where: { schoolId: ctx.schoolId, teacherId: ctx.userId, isActive: true },
   101→    include: {
   102→      class: { select: { id: true, name: true, section: true, gradeLevel: true } },
   103→      subject: { select: { id: true, name: true } },
   104→    },
   105→    orderBy: [{ class: { name: 'asc' } }, { displayOrder: 'asc' }],
   106→  })
   107→  const out: TeachingAssignmentRef[] = []
   108→  for (const r of rows) {
   109→    if (!out.some((a) => a.classId === r.class.id && a.subjectId === r.subject.id)) {
   110→      out.push({
   111→        classId: r.class.id,
   112→        classLabel: classLabelOf(r.class),
   113→        subjectId: r.subject.id,
   114→        subjectName: r.subject.name,
   115→      })
   116→    }
   117→  }
   118→  return out
   119→}

   121→async function assertAssignment(
   122→  ctx: TeacherHubContext,
   123→  classId: unknown,
   124→  subjectId: unknown,
   125→): Promise<TeachingAssignmentRef> {
   126→  if (typeof classId !== 'string' || !classId.trim()) throw new Error('Class is required')
   127→  if (typeof subjectId !== 'string' || !subjectId.trim()) throw new Error('Subject is required')
   128→  const assignments = await teachingAssignments(ctx)
   129→  const match = assignments.find((a) => a.classId === classId && a.subjectId === subjectId)
   130→  if (!match) {
   131→    const cls = await db.class.findFirst({ where: { id: classId, schoolId: ctx.schoolId }, select: { id: true } })
   132→    if (!cls) throw new Error('NOT_FOUND')
   133→    throw new Error('FORBIDDEN')
   134→  }
   135→  return match
   136→}

   140→/** class level digits from gradeLevel or the class name ("Grade 9 - A" → "9") */
   141→function classLevelOf(cls: { gradeLevel: string | null; name: string }): string {
   142→  if (cls.gradeLevel) {
   143→    const digits = cls.gradeLevel.replace(/[^0-9]/g, '')
   144→    if (digits) return digits
   145→  }
   146→  const digits = cls.name.replace(/[^0-9]/g, '')
   147→  return digits || cls.name
   148→}

   150→/**
   151→ * Resolve the ACTIVE master curriculum for a (class, subject). Picks the
   152→ * highest ACTIVE version for the board/classLevel/subjectName. Returns null
   153→ * when the library has no curriculum for this pair — the UI then shows the
   154→ * honest "Curriculum unavailable" state (§32). NEVER fabricates one.
   155→ */
   156→async function resolveCurriculum(
   157→  ctx: TeacherHubContext,
   158→  cls: { gradeLevel: string | null; name: string },
   159→  subjectName: string,
   160→): Promise<CurriculumRef | null> {
   161→  const classLevel = classLevelOf(cls)
   162→  const curricula = await db.curriculum.findMany({
   163→    where: { board: 'CBSE', classLevel, subjectName, status: 'ACTIVE' },
   164→    orderBy: { version: 'desc' },
   165→    select: {
   166→      id: true, board: true, academicSession: true, classLevel: true,
   167→      subjectName: true, version: true, status: true,
   168→    },
   169→  })
   170→  // prefer the school's own academic session when the library has it
   171→  const school = await db.school.findUnique({ where: { id: ctx.schoolId }, select: { academicYear: true } })
   172→  const sessionOf = (s: string): string => s.replace('/', '-').replace(/\s/g, '')
   173→  const schoolSession = school?.academicYear ? sessionOf(school.academicYear) : null
   174→  const match =
   175→    (schoolSession && curricula.find((c) => sessionOf(c.academicSession) === schoolSession)) || curricula[0] || null
   176→  return match
   177→}

   187→async function subjectSlots(schoolId: string, classId: string, subjectId: string): Promise<Slot[]> {
   188→  const rows = await db.timetable.findMany({ where: { schoolId, classId, subjectId } })
   189→  const slots: Slot[] = []
   190→  for (const r of rows) {
   191→    const minutes = minutesBetween(r.startTime, r.endTime) ?? 45
   192→    slots.push({ weekday: dayIndex(r.day), period: r.period, minutes })
   193→  }
   194→  slots.sort((a, b) => a.weekday - b.weekday || a.period - b.period)
   195→  return slots
   196→}

   198→/** HOLIDAY calendar: ISO-date set + title map ( SchoolEvent, ranges expanded ). */
   199→async function holidayCalendar(schoolId: string): Promise<{ dates: Map<string, string>; titles: Map<string, string> }> {
   200→  const events = await db.schoolEvent.findMany({
   201→    where: { schoolId, type: 'HOLIDAY' },
   202→    select: { title: true, startDate: true, endDate: true },
   203→  })
   204→  const dates = new Map<string, string>()
   205→  const titles = new Map<string, string>()
   206→  for (const e of events) {
   207→    const start = new Date(e.startDate)
   208→    start.setUTCHours(0, 0, 0, 0)
   209→    const end = e.endDate ? new Date(e.endDate) : start
   210→    end.setUTCHours(0, 0, 0, 0)
   211→    for (let d = new Date(start); d.getTime() <= end.getTime(); d = addDays(d, 1)) {
   212→      const iso = dayISO(d)
   213→      dates.set(iso, iso)
   214→      titles.set(iso, e.title)
   215→    }
   216→  }
   217→  return { dates, titles }
   218→}

   222→// ---------- schedule projection (§9/§13/§21) ----------
   223→
   224→interface FlatTopic {
   225→  id: string
   226→  unitId: string
   227→  unitTitle: string
   228→  unitSequence: number
   229→  sequence: number
   230→  title: string
   231→  estimatedMinutes: number
   232→  learningOutcomes: string[]
   233→}
   234→
   235→interface ExecutionRow {
   236→  curriculumTopicId: string
   237→  date: Date
   238→  period: number | null
   239→  status: string
   240→  completedAt: Date | null
   241→}
   242→
   243→/** Iterate eligible lesson slots (date+period) from `from` (inclusive). */
   244→function* eligibleSlots(
   245→  from: Date,
   246→  slots: Slot[],
   247→  holidays: Map<string, string>,
   248→): Generator<{ date: Date; period: number; minutes: number }> {
   249→  const slotsByWeekday = new Map<number, Slot[]>()
   250→  for (const s of slots) {
   251→    const arr = slotsByWeekday.get(s.weekday) ?? []
   252→    arr.push(s)
   253→    slotsByWeekday.set(s.weekday, arr)
   254→  }
   255→  let d = new Date(from)
   256→  d.setUTCHours(0, 0, 0, 0)
   257→  for (let guard = 0; guard < 400; guard++) {
   258→    const iso = dayISO(d)
   259→    if (!holidays.has(iso)) {
   260→      const daySlots = (slotsByWeekday.get(d.getUTCDay()) ?? []).slice().sort((a, b) => a.period - b.period)
   261→      for (const s of daySlots) yield { date: new Date(d), period: s.period, minutes: s.minutes }
   262→    }
   263→    d = addDays(d, 1)
   264→  }
   265→}
   266→
   267→/**
   268→ * Assign upcoming topics to eligible slots. A topic spans
   269→ * ceil(estimatedMinutes / slotMinutes) consecutive slots.
   270→ */
   271→function projectTopics(
   272→  topics: FlatTopic[],
   273→  startIndex: number,
   274→  slotIterator: Generator<{ date: Date; period: number; minutes: number }>,
   275→): Map<string, { date: string; period: number }> {
   276→  const out = new Map<string, { date: string; period: number }>()
   277→  let idx = startIndex
   278→  let current = slotIterator.next()
   279→  while (!current.done && idx < topics.length) {
   280→    const topic = topics[idx]
   281→    const slotsNeeded = Math.max(1, Math.ceil(topic.estimatedMinutes / current.value.minutes))
   282→    out.set(topic.id, { date: dayISO(current.value.date), period: current.value.period })
   283→    for (let i = 0; i < slotsNeeded; i++) {
   284→      current = slotIterator.next()
   285→      if (current.done) return out
   286→    }
   287→    idx++
   288→  }
   289→  return out
   290→}

   294→async function buildSelected(
   295→  ctx: TeacherHubContext,
   296→  assignment: TeachingAssignmentRef,
   297→): Promise<CurriculumExecutionPayload['selected']> {
   298→  const cls = await db.class.findFirst({
   299→    where: { id: assignment.classId, schoolId: ctx.schoolId },
   300→    select: { gradeLevel: true, name: true },
   301→  })
   302→
   303→  const curriculum = cls
   304→    ? await resolveCurriculum(ctx, cls, assignment.subjectName)
   305→    : null
   306→
   307→  const today = todayUTC()
   308→  const todayIso = dayISO(today)
   309→
   310→  if (!curriculum) {
   311→    return {
   312→      classId: assignment.classId,
   313→      classLabel: assignment.classLabel,
   314→      subjectId: assignment.subjectId,
   315→      subjectName: assignment.subjectName,
   316→      curriculum: null,
   317→      timetable: null,
   318→      todayLesson: null,
   319→      todayNote: null,
   320→      upNext: [],
   321→      units: [],
   322→      progress: {
   323→        completedTopics: 0,
   324→        skippedTopics: 0,
   325→        totalTopics: 0,
   326→        remainingTopics: 0,
   327→        percent: 0,
   328→        completedMinutes: 0,
   329→        totalMinutes: 0,
   330→      },
   331→    }
   332→  }
   333→
   334→  // full topic tree (flat, in teaching sequence)
   335→  const unitRows = await db.curriculumUnit.findMany({
   336→    where: { curriculumId: curriculum.id },
   337→    orderBy: { sequence: 'asc' },
   338→    include: { topics: { orderBy: { sequence: 'asc' } } },
   339→  })
   340→  const topics: FlatTopic[] = []
   341→  for (const u of unitRows) {
   342→    for (const t of u.topics) {
   343→      let outcomes: string[] = []
   344→      try {
   345→        const v: unknown = t.learningOutcomes ? JSON.parse(t.learningOutcomes) : []
   346→        if (Array.isArray(v)) outcomes = v.filter((x): x is string => typeof x === 'string')
   347→      } catch {
   348→        outcomes = []
   349→      }
   350→      topics.push({
   351→        id: t.id,
   352→        unitId: u.id,
   353→        unitTitle: u.title,
   354→        unitSequence: u.sequence,
   355→        sequence: t.sequence,
   356→        title: t.title,
   357→        estimatedMinutes: t.estimatedMinutes,
   358→        learningOutcomes: outcomes,
   359→      })
   360→    }
   361→  }
   362→
   363→  // execution state for this (class, subject, teacher)
   364→  const executions = await db.lessonExecution.findMany({
   365→    where: {
   366→      schoolId: ctx.schoolId,
   367→      classId: assignment.classId,
   368→      subjectId: assignment.subjectId,
   369→      teacherId: ctx.userId,
   370→    },
   371→  })
   372→  const byTopic = new Map<string, ExecutionRow>()
   373→  for (const e of executions) byTopic.set(e.curriculumTopicId, e)
   374→
   375→  // slots + holidays (timetable is the preferred source; default fallback otherwise)
   376→  const realSlots = await subjectSlots(ctx.schoolId, assignment.classId, assignment.subjectId)
   377→  const useDefault = realSlots.length === 0
   378→  const slots = useDefault ? DEFAULT_SLOTS : realSlots
   379→  const { dates: holidays, titles: holidayTitles } = await holidayCalendar(ctx.schoolId)
   380→  const minutesPerWeek = slots.reduce((acc, s) => acc + s.minutes, 0)
   381→
   382→  // cursor = first topic not COMPLETED/SKIPPED
   383→  const isDone = (t: FlatTopic): boolean => {
   384→    const e = byTopic.get(t.id)
   385→    return !!e && (e.status === 'COMPLETED' || e.status === 'SKIPPED')
   386→  }
   387→  const cursorIdx = topics.findIndex((t) => !isDone(t))
   388→
   389→  // today's slots (empty on holidays / no-slot weekdays)
   390→  const todaySlotList = holidays.has(todayIso)
   391→    ? []
   392→    : (slots.filter((s) => s.weekday === today.getUTCDay()).sort((a, b) => a.period - b.period))
   393→
   394→  // today's execution overlay for the cursor topic
   395→  const cursorTopic = cursorIdx >= 0 ? topics[cursorIdx] : null
   396→  const cursorExec = cursorTopic ? byTopic.get(cursorTopic.id) ?? null : null
   397→  const cursorPostponedToday =
   398→    cursorExec?.status === 'POSTPONED' && dayISO(cursorExec.date) === todayIso
   399→
   400→  // slots already consumed today by completed lessons (spanning topics take
   401→  // ceil(est/minutes) slots — the honest approximation of real consumption)
   402→  const slotMinutes = todaySlotList[0]?.minutes ?? 45
   403→  const completedToday = topics.filter((t) => {
   404→    const e = byTopic.get(t.id)
   405→    return e?.status === 'COMPLETED' && dayISO(e.date) === todayIso
   406→  })
   407→  const consumedSlotsToday = completedToday.reduce(
   408→    (acc, t) => acc + Math.max(1, Math.ceil(t.estimatedMinutes / slotMinutes)),
   409→    0,
   410→  )
   411→  const slotsLeftToday = todaySlotList.length - consumedSlotsToday
   412→  const lastCompletedToday = completedToday[completedToday.length - 1] ?? null
   413→
   414→  // ── projection start: the first slot from which uncompleted topics are placed ──
   415→  // · today still has a free slot and the cursor is pending: the cursor sits
   416→  //   on today's first FREE slot; the NEXT topics start after today.
   417→  // · cursor postponed today: cursor re-appears at the next slot after today.
   418→  // · today fully taught / holiday / no slots: projection starts after today.
   419→  const todayPending =
   420→    slotsLeftToday > 0 && !!cursorTopic && !cursorPostponedToday
   421→
   422→  let projectionIterator: Generator<{ date: Date; period: number; minutes: number }>
   423→  let projectionStartIdx: number
   424→  if (todayPending && cursorTopic) {
   425→    projectionIterator = eligibleSlots(addDays(today, 1), slots, holidays)
   426→    projectionStartIdx = cursorIdx + 1
   427→  } else {
   428→    projectionIterator = eligibleSlots(addDays(today, 1), slots, holidays)
   429→    projectionStartIdx = cursorIdx
   430→  }
   431→  const projection = projectTopics(topics, projectionStartIdx, projectionIterator)
   432→
   433→  // the cursor topic's own slot when it is today's pending lesson
   434→  if (todayPending && cursorTopic) {
   435→    projection.set(cursorTopic.id, {
   436→      date: todayIso,
   437→      period: todaySlotList[Math.min(consumedSlotsToday, todaySlotList.length - 1)].period,
   438→    })
   439→  }
   440→
   441→  // where a postponed cursor moves to (next eligible slot)
   442→  let movedTo: { date: string; period: number | null } | null = null
   443→  if (cursorPostponedToday && cursorTopic) {
   444→    const next = eligibleSlots(addDays(today, 1), slots, holidays).next()
   445→    if (!next.done) movedTo = { date: dayISO(next.value.date), period: next.value.period }
   446→  }
   447→
   448→  // ── topic DTOs ──
   449→  const scheduled: ScheduledTopicDTO[] = topics.map((t) => {
   450→    const e = byTopic.get(t.id)
   451→    let status: ScheduledTopicDTO['status'] = 'PLANNED'
   452→    let completedAt: string | null = null
   453→    let scheduledDate: string | null = null
   454→    let period: number | null = null
   455→    if (e) {
   456→      if (e.status === 'COMPLETED') {
   457→        status = 'COMPLETED'
   458→        completedAt = (e.completedAt ?? e.date).toISOString()
   459→        scheduledDate = dayISO(e.date)
   460→        period = e.period
   461→      } else if (e.status === 'SKIPPED') {
   462→        status = 'SKIPPED'
   463→        scheduledDate = dayISO(e.date)
   464→        period = e.period
   465→      } else if (e.status === 'IN_PROGRESS') {
   466→        status = 'IN_PROGRESS'
   467→        scheduledDate = dayISO(e.date)
   468→        period = e.period
   469→      } else if (e.status === 'POSTPONED') {
   470→        status = 'POSTPONED'
   471→        scheduledDate = dayISO(e.date)
   472→        period = e.period
   473→      }
   474→    }
   475→    if (status === 'PLANNED') {
   476→      const p = projection.get(t.id)
   477→      if (p) {
   478→        scheduledDate = p.date
   479→        period = p.period
   480→      }
   481→    }
   482→    return {
   483→      id: t.id,
   484→      unitId: t.unitId,
   485→      unitTitle: t.unitTitle,
   486→      unitSequence: t.unitSequence,
   487→      sequence: t.sequence,
   488→      title: t.title,
   489→      estimatedMinutes: t.estimatedMinutes,
   490→      learningOutcomes: t.learningOutcomes,
   491→      status,
   492→      completedAt,
   493→      scheduledDate,
   494→      period,
   495→      isToday: scheduledDate === todayIso,
   496→      postponedTo: null,
   497→    }
   498→  })
   499→  const topicById = new Map(scheduled.map((t) => [t.id, t]))
   500→  if (cursorTopic && movedTo) {
   501→    const dto = topicById.get(cursorTopic.id)
   502→    if (dto) {
   503→      dto.postponedTo = movedTo
   504→      // display target date instead of the postponed-from date
   505→      dto.scheduledDate = movedTo.date
   506→      dto.period = movedTo.period
   507→      dto.isToday = false
   508→    }
   509→  }
   510→
   511→  // ── today's lesson ──
   512→  let todayLesson: TodayLessonDTO | null = null
   513→  let todayNote: string | null = null
   514→  if (holidays.has(todayIso)) {
   515→    todayNote = `School holiday — ${holidayTitles.get(todayIso) ?? 'Holiday'}`
   516→  } else if (todaySlotList.length === 0) {
   517→    todayNote = `No ${assignment.subjectName} period scheduled today`
   518→  } else if (!cursorTopic) {
   519→    todayNote = 'All curriculum topics are completed'
   520→  } else if (slotsLeftToday <= 0 && lastCompletedToday) {
   521→    // today's teaching slots are fully consumed — show the last completion
   522→    const lastExec = byTopic.get(lastCompletedToday.id)
   523→    todayLesson = {
   524→      topic: topicById.get(lastCompletedToday.id)!,
   525→      period: lastExec?.period ?? todaySlotList[0].period,
   526→      slotMinutes: todaySlotList[0].minutes,
   527→      state: 'COMPLETED',
   528→      movedTo: null,
   529→      nextLessonToday: null,
   530→    }
   531→  } else {
   532→    const cursorState: TodayLessonDTO['state'] = cursorPostponedToday
   533→      ? 'POSTPONED'
   534→      : cursorExec?.status === 'IN_PROGRESS'
   535→        ? 'IN_PROGRESS'
   536→        : 'PLANNED'
   537→    todayLesson = {
   538→      topic: topicById.get(cursorTopic.id)!,
   539→      period: todaySlotList[Math.min(consumedSlotsToday, todaySlotList.length - 1)].period,
   540→      slotMinutes: todaySlotList[0].minutes,
   541→      state: cursorState,
   542→      movedTo,
   543→      nextLessonToday: null,
   544→    }
   545→  }
   546→
   547→  // ── up next: the next ~8 projected lessons, with skipped holidays in between ──
   548→  const upNext: UpNextItemDTO[] = []
   549→  const projectedDates = [...projection.entries()]
   550→    .filter(([topicId, v]) => {
   551→      const t = topics.find((x) => x.id === topicId)
   552→      return t && !isDone(t) && v.date > todayIso
   553→    })
   554→    .map(([, v]) => v)
   555→    .sort((a, b) => a.date.localeCompare(b.date) || (a.period ?? 0) - (b.period ?? 0))
   556→  for (const p of projectedDates.slice(0, 8)) {
   557→    const topic = topics.find((t) => projection.get(t.id) === p)
   558→    if (!topic) continue
   559→    upNext.push({
   560→      date: p.date,
   561→      kind: 'lesson',
   562→      topic: topicById.get(topic.id)!,
   563→      period: p.period,
   564→      slotMinutes: todaySlotList[0]?.minutes ?? slots[0]?.minutes ?? 45,
   565→    })
   566→  }
   567→  // holidays that fall inside the visible horizon → shown as skipped days
   568→  const horizon = upNext.length > 0 ? upNext[upNext.length - 1].date : dayISO(addDays(today, 7))
   569→  for (let d = addDays(today, 1); dayISO(d) <= horizon; d = addDays(d, 1)) {
   570→    const iso = dayISO(d)
   571→    if (holidays.has(iso)) {
   572→      upNext.push({ date: iso, kind: 'holiday', holidayTitle: holidayTitles.get(iso) ?? 'Holiday' })
   573→    }
   574→  }
   575→  upNext.sort((a, b) => a.date.localeCompare(b.date) || (a.period ?? 0) - (b.period ?? 0))
   576→  const trimmed = upNext.slice(0, 12)
   577→
   578→  // ── units + progress ──
   579→  const units: CurriculumUnitDTO[] = unitRows.map((u) => {
   580→    const unitTopics = scheduled.filter((t) => t.unitId === u.id)
   581→    return {
   582→      id: u.id,
   583→      sequence: u.sequence,
   584→      title: u.title,
   585→      estimatedMinutes: u.estimatedMinutes,
   586→      topics: unitTopics,
   587→      completedTopics: unitTopics.filter((t) => t.status === 'COMPLETED').length,
   588→    }
   589→  })
   590→  const completed = scheduled.filter((t) => t.status === 'COMPLETED')
   591→  const skipped = scheduled.filter((t) => t.status === 'SKIPPED')
   592→  const totalMinutes = topics.reduce((acc, t) => acc + t.estimatedMinutes, 0)
   593→  const completedMinutes = completed.reduce((acc, t) => acc + t.estimatedMinutes, 0)
   594→
   595→  return {
   596→    classId: assignment.classId,
   597→    classLabel: assignment.classLabel,
   598→    subjectId: assignment.subjectId,
   599→    subjectName: assignment.subjectName,
   600→    curriculum: {
   601→      id: curriculum.id,
   602→      board: curriculum.board,
   603→      academicSession: curriculum.academicSession,
   604→      classLevel: curriculum.classLevel,
   605→      subjectName: curriculum.subjectName,
   606→      version: curriculum.version,
   607→    },
   608→    timetable: {
   609→      periodsPerWeek: slots.length,
   610→      minutesPerWeek,
   611→      source: useDefault ? 'default' : 'timetable',
   612→    },
   613→    todayLesson,
   614→    todayNote,
   615→    upNext: trimmed,
   616→    units,
   617→    progress: {
   618→      completedTopics: completed.length,
   619→      skippedTopics: skipped.length,
   620→      totalTopics: topics.length,
   621→      remainingTopics: topics.length - completed.length - skipped.length,
   622→      percent: topics.length > 0 ? Math.round((completed.length / topics.length) * 100) : 0,
   623→      completedMinutes,
   624→      totalMinutes,
   625→    },
   626→  }
   627→}

   629→/** count of assignments with a lesson due today (sidebar badge) */
   630→async function dueTodayCount(ctx: TeacherHubContext, assignments: TeachingAssignmentRef[]): Promise<number> {
   631→  const today = todayUTC()
   632→  const todayIso = dayISO(today)
   633→  let due = 0
   634→  for (const a of assignments) {
   635→    const selected = await buildSelected(ctx, a)
   636→    if (selected?.todayLesson && selected.todayLesson.state !== 'COMPLETED' && selected.todayLesson.state !== 'POSTPONED') {
   637→      // only count when today's slot is genuinely still pending
   638→      if (selected.todayLesson.topic.scheduledDate === todayIso || selected.todayLesson.state === 'IN_PROGRESS') due++
   639→    }
   640→  }
   641→  return due
   642→}

   674→async function resolveTopicInCurriculum(
   675→  ctx: TeacherHubContext,
   676→  assignment: TeachingAssignmentRef,
   677→  topicId: unknown,
   678→): Promise<string> {
   679→  if (typeof topicId !== 'string' || !topicId.trim()) throw new Error('Topic is required')
   680→  const cls = await db.class.findFirst({
   681→    where: { id: assignment.classId, schoolId: ctx.schoolId },
   682→    select: { gradeLevel: true, name: true },
   683→  })
   684→  if (!cls) throw new Error('NOT_FOUND')
   685→  const curriculum = await resolveCurriculum(ctx, cls, assignment.subjectName)
   686→  if (!curriculum) throw new Error('No curriculum is available for this class and subject')
   687→  const unitRows = await db.curriculumUnit.findMany({
   688→    where: { curriculumId: curriculum.id },
   689→    select: { topics: { select: { id: true } } },
   690→  })
   691→  const topicIds = new Set(unitRows.flatMap((u) => u.topics.map((t) => t.id)))
   692→  if (!topicIds.has(topicId)) throw new Error('Topic does not belong to this curriculum')
   693→  return topicId
   694→}

   696→/**
   697→ * Apply a lesson action to an owned execution row (one per topic — the
   698→ * school curriculum instance state for this teacher/class/subject).
   699→ */
   700→export async function applyLessonAction(opts: {
   701→  user: AuthUser
   702→  ctx: TeacherHubContext
   703→  assignment: TeachingAssignmentRef
   704→  topicId: string
   705→  action: LessonAction
   706→}): Promise<void> {
   707→  const { user, ctx, assignment, topicId, action } = opts
   708→  const today = todayUTC()
   709→  const todayIso = dayISO(today)
   710→
   711→  // the topic's current slot (for honest date/period anchoring)
   712→  const slots = await subjectSlots(ctx.schoolId, assignment.classId, assignment.subjectId)
   713→  const todaySlots = slots
   714→    .filter((s) => s.weekday === today.getUTCDay())
   715→    .sort((a, b) => a.period - b.period)
   716→  const { dates: holidays } = await holidayCalendar(ctx.schoolId)
   717→  const eligibleToday = !holidays.has(todayIso) && todaySlots.length > 0
   718→  const period = eligibleToday ? todaySlots[0].period : null
   719→
   720→  const unique = {
   721→    schoolId: ctx.schoolId,
   722→    classId: assignment.classId,
   723→    subjectId: assignment.subjectId,
   724→    teacherId: ctx.userId,
   725→    curriculumTopicId: topicId,
   726→  }
   727→
   728→  const existing = await db.lessonExecution.findUnique({
   729→    where: { schoolId_classId_subjectId_teacherId_curriculumTopicId: unique },
   730→  })
   731→
   732→  if (action === 'start') {
   733→    if (existing?.status === 'COMPLETED') throw new Error('This lesson is already completed')
   734→    if (existing?.status === 'SKIPPED') throw new Error('This lesson was skipped')
   735→    await db.lessonExecution.upsert({
   736→      where: { schoolId_classId_subjectId_teacherId_curriculumTopicId: unique },
   737→      update: { status: 'IN_PROGRESS', date: today, period },
   738→      create: { ...unique, date: today, period, status: 'IN_PROGRESS' },
   739→    })
   740→  } else if (action === 'complete') {
   741→    if (existing?.status === 'COMPLETED') throw new Error('This lesson is already completed')
   742→    if (existing?.status === 'SKIPPED') throw new Error('This lesson was skipped')
   743→    await db.lessonExecution.upsert({
   744→      where: { schoolId_classId_subjectId_teacherId_curriculumTopicId: unique },
   745→      update: { status: 'COMPLETED', date: today, period, completedAt: new Date() },
   746→      create: { ...unique, date: today, period, status: 'COMPLETED', completedAt: new Date() },
   747→    })
   748→  } else if (action === 'postpone') {
   749→    if (existing?.status === 'COMPLETED') throw new Error('A completed lesson cannot be rescheduled')
   750→    await db.lessonExecution.upsert({
   751→      where: { schoolId_classId_subjectId_teacherId_curriculumTopicId: unique },
   752→      update: { status: 'POSTPONED', date: today, period, completedAt: null },
   753→      create: { ...unique, date: today, period, status: 'POSTPONED' },
   754→    })
   755→  } else if (action === 'skip') {
   756→    if (existing?.status === 'COMPLETED') throw new Error('A completed lesson cannot be skipped')
   757→    await db.lessonExecution.upsert({
   758→      where: { schoolId_classId_subjectId_teacherId_curriculumTopicId: unique },
   759→      update: { status: 'SKIPPED', date: today, period, completedAt: null },
   760→      create: { ...unique, date: today, period, status: 'SKIPPED' },
   761→    })
   762→  }
   763→
   764→  const topicRow = await db.curriculumTopic.findUnique({ where: { id: topicId }, select: { title: true } })
   765→  const actionLabel: Record<LessonAction, string> = {
   766→    start: 'started',
   767→    complete: 'completed',
   768→    postpone: 'postponed to the next teaching slot',
   769→    skip: 'skipped',
   770→  }
   771→  await auditTeacherAction(
   772→    user,
   773→    ctx.schoolId,
   774→    action === 'complete' ? 'LESSON_COMPLETED' : 'LESSON_EXECUTION',
   775→    `${assignment.subjectName} · ${assignment.classLabel} · ${topicRow?.title ?? 'Topic'} · ${actionLabel[action]}`,
   776→  )
   777→}
