    40→/** Teachers the student may request a meeting with (class teacher + subject
    41→ *  teachers of THEIR class — never a school-wide directory, spec §30). */
    42→function allowedTeachers(classId: string, section: string): { id: string; name: string; subject: string }[] {
    43→  const cls = useStudentsStore.getState().classes.find((c) => c.id === classId)
    44→  if (!cls) return []
    45→  const sec = cls.sections.find((s) => s.name === section)
    46→  const roster: { id: string; name: string; subject: string }[] = []
    47→  const seen = new Set<string>()
    48→  const push = (id: string | undefined, subject: string) => {
    49→    if (!id || seen.has(id)) return
    50→    const t = teachers.find((x) => x.id === id)
    51→    if (!t) return
    52→    seen.add(id)
    53→    roster.push({ id: t.id, name: t.name, subject })
    54→  }
    55→  push(sec?.classTeacherId ?? cls.classTeacherId, 'Class Teacher')
    56→  for (const [subId, tid] of Object.entries(cls.subjectTeachers)) {
    57→    const name = cls.subjects.find((n) => n === subId) ?? subId.replace('sub-', '')
    58→    push(tid, name.charAt(0).toUpperCase() + name.slice(1))
    59→  }
    60→  return roster
    61→}

   327→function TaskRow({ task }: { task: ResponsibilityTask }) {

   337→    )}>

   341→      )}>

   344→      <div className="flex-1 min-w-0">

   400→function PostUpdateDialog({ positions, onClose }: { positions: StudentPosition[]; onClose: () => void }) {
   401→  const postClassUpdate = useClassResponsibilityStore((s) => s.postClassUpdate)
   402→  const [title, setTitle] = useState('')
   403→  const [body, setBody] = useState('')
   404→  const [category, setCategory] = useState<ClassUpdateCategory>('notice')
   405→
   406→  const submit = () => {
   407→    if (!title.trim() || !body.trim()) return
   408→    const result = postClassUpdate({ actorStudentId: DEMO_STUDENT_ID, title: title.trim(), body: body.trim(), category })
   409→    if (result.ok) {
   410→      toast.success('Update submitted for review', { description: 'Your class teacher will review it before it appears on the class notice board.' })
   411→      onClose()
   412→    } else {
   413→      toast.error('Could not post update', { description: result.error })
   414→    }
   415→  }
   416→
   417→  return (
   418→    <DialogShell
   419→      title="Post a class update"
   420→      subtitle={`${POSITION_DEFS[positions[0]?.key]?.title} · reviewed by your teacher`}
   421→      onClose={onClose}
   422→      footer={
   423→        <>
