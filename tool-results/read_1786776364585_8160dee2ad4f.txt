    49→export function SeatingSection({ examId, exam, onReload }: SectionProps) {
    50→  const [classId, setClassId] = useState(exam?.classes[0]?.classId ?? '')
    51→  const { seats, loading, reload } = useSeatingPlan(examId, classId)
    52→  const { generate, loading: generating } = useGenerateSeating()
    53→  const gate = useRoleGate()
    54→  const [rooms, setRooms] = useState<Array<{ name: string; capacity: number }>>([{ name: 'Room A', capacity: 30 }])
    55→
    56→  const handleGenerate = async () => {
    57→    if (!classId) { toast.error('Select a class'); return }
    58→    if (rooms.length === 0 || rooms.some((r) => !r.name || r.capacity <= 0)) {
    59→      toast.error('Add at least one valid room')
    60→      return
    61→    }
    62→    try {
    63→      const r = await generate(examId, classId, rooms)
    64→      toast.success(`Seating plan generated: ${r.generated} students assigned`)
    65→      reload()
    66→      onReload()
    67→    } catch (e: any) {
    68→      toast.error('Failed to generate seating', { description: e.message })
    69→    }
    70→  }
    71→
    72→  const handleExport = () => {
    73→    if (seats.length === 0) { toast.error('No seating plan to export'); return }
    74→    const className = exam?.classes.find((c: any) => c.classId === classId)?.className ?? 'Class'
    75→    try {
    76→      const { filename } = generateSeatingPlanPDF(exam, className, seats)
    77→      toast.success('Seating plan exported', { description: filename })
    78→    } catch (e: any) {
    79→      toast.error('Export failed', { description: e.message })
    80→    }
    81→  }
    82→
    83→  const handleBatchAdmitCards = async () => {
    84→    if (!classId) { toast.error('Select a class'); return }
    85→    try {
    86→      const data = await fetchAdmitCardsBatch(examId, classId)
    87→      if (data.students.length === 0) { toast.error('No students in this class'); return }
    88→      const className = exam?.classes.find((c: any) => c.classId === classId)?.className ?? 'Class'
    89→      const { filename } = generateBatchAdmitCardPDF(data.exam, className, data.students)
    90→      toast.success(`${data.students.length} admit cards exported`, { description: filename })
    91→    } catch (e: any) {
    92→      toast.error('Export failed', { description: e.message })
    93→    }
    94→  }
    95→
    96→  const addRoom = () => setRooms((r) => [...r, { name: `Room ${String.fromCharCode(65 + r.length)}`, capacity: 30 }])
    97→  const removeRoom = (i: number) => setRooms((r) => r.filter((_, idx) => idx !== i))
    98→  const updateRoom = (i: number, field: 'name' | 'capacity', value: string | number) => {
    99→    setRooms((r) => r.map((room, idx) => idx === i ? { ...room, [field]: value } : room))
   100→  }
   101→
   102→  return (
   103→    <div className="space-y-3">
   104→      {gate.canGenerateSeating && (
   105→        <div className="rounded-xl border border-border bg-card p-3">
   106→          <div className="flex items-center gap-2 mb-2">
   107→            <MapPin className="h-3.5 w-3.5 text-primary" />
   108→            <p className="text-[10px] uppercase font-semibold text-muted-foreground">Generate Seating Plan</p>
   109→          </div>
   110→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mb-2">
   111→            <Select value={classId} onValueChange={setClassId}>
   112→              <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Select class" /></SelectTrigger>
   113→              <SelectContent>
   114→                {exam?.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   115→              </SelectContent>
   116→            </Select>
   117→          </div>
   118→          <div className="space-y-1.5 mb-2">
   119→            <div className="flex items-center justify-between">
   120→              <Label className="text-[10px] uppercase font-semibold text-muted-foreground">Rooms</Label>
   121→              <Button size="sm" variant="ghost" className="h-6 text-xs gap-1" onClick={addRoom}><Plus className="h-3 w-3" /> Add Room</Button>
   122→            </div>
   123→            {rooms.map((r, i) => (
   124→              <div key={i} className="flex items-center gap-2">
   125→                <Input value={r.name} onChange={(e) => updateRoom(i, 'name', e.target.value)} placeholder="Room name" className="h-7 text-xs flex-1" />
   126→                <Input type="number" value={r.capacity} onChange={(e) => updateRoom(i, 'capacity', Number(e.target.value))} className="h-7 text-xs w-20" />
   127→                <button onClick={() => removeRoom(i)} className="text-muted-foreground hover:text-rose-500"><Trash2 className="h-3.5 w-3.5" /></button>
   128→              </div>
   129→            ))}
   130→          </div>
   131→          <div className="flex flex-wrap gap-2">
   132→            <Button size="sm" variant="default" className="h-7 text-xs gap-1.5" onClick={handleGenerate} disabled={generating}>
   133→              <Sparkles className="h-3 w-3" /> {generating ? 'Generating…' : 'Generate Seating'}
   134→            </Button>
   135→            <Button size="sm" variant="outline" className="h-7 text-xs" onClick={handleExport} disabled={seats.length === 0}>
   136→              Export Seating PDF
   137→            </Button>
   138→            <Button size="sm" variant="outline" className="h-7 text-xs" onClick={handleBatchAdmitCards}>
   139→              Generate All Admit Cards (PDF)
   140→            </Button>
   141→          </div>
   142→        </div>
   143→      )}
   144→
   145→      {loading ? <InlineLoading label="Loading seating plan…" /> : seats.length === 0 ? (
   146→        <div className="rounded-xl border border-border bg-card p-6 text-center text-xs text-muted-foreground">
   147→          No seating assignments yet.
   148→        </div>
   149→      ) : (
   150→        <div className="rounded-xl border border-border bg-card overflow-hidden">
   151→          <div className="px-3 py-2 border-b border-border flex items-center justify-between">
   152→            <p className="text-xs font-semibold">Seating Assignments ({seats.length})</p>
   153→            <Button size="sm" variant="ghost" className="h-6 text-xs" onClick={reload}><RefreshCw className="h-3 w-3" /></Button>
   154→          </div>
   155→          <Table>
   156→            <TableHeader>
   157→              <TableRow>
   158→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Room</TableHead>
   159→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Seat #</TableHead>
   160→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Row</TableHead>
   161→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Roll No</TableHead>
   162→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Student</TableHead>
   163→              </TableRow>
   164→            </TableHeader>
   165→            <TableBody>
   166→              {seats.map((s) => (
   167→                <TableRow key={s.id}>
   168→                  <TableCell className="text-xs font-medium">{s.room}</TableCell>
   169→                  <TableCell className="text-xs tabular-nums">{s.seatNumber}</TableCell>
   170→                  <TableCell className="text-xs tabular-nums">{s.row ?? '—'}</TableCell>
   171→                  <TableCell className="text-xs font-mono">{s.studentRollNo ?? '—'}</TableCell>
   172→                  <TableCell className="text-xs">{s.studentName}</TableCell>
   173→                </TableRow>
   174→              ))}
   175→            </TableBody>
   176→          </Table>
   177→        </div>
   178→      )}
   179→    </div>
   180→  )
   181→}

   185→export function AttendanceSection({ examId, exam, onReload }: SectionProps) {
   186→  const [classId, setClassId] = useState(exam?.classes[0]?.classId ?? '')
   187→  const { attendance, loading, reload } = useExamAttendance(examId, classId)
   188→  const { autoMark, loading: autoLoading } = useAutoMarkAttendance()
   189→
   190→  const handleAutoMark = async () => {
   191→    if (!classId) return
   192→    try {
   193→      const r = await autoMark(examId, classId)
   194→      toast.success(`Auto-marked ${r.marked} attendance entries from marks`, { description: 'All ABSENT/MEDICAL/EXEMPTED statuses synced.' })
   195→      reload()
   196→    } catch (e: any) {
   197→      toast.error('Failed to auto-mark', { description: e.message })
   198→    }
   199→  }
   200→
   201→  // Group by date+subject
   202→  const grouped = attendance.reduce((acc, a) => {
   203→    const key = `${a.date}|${a.subjectName ?? a.subjectId ?? 'general'}`
   204→    if (!acc.has(key)) acc.set(key, { date: a.date, subject: a.subjectName, items: [] })
   205→    acc.get(key)!.items.push(a)
   206→    return acc
   207→  }, new Map<string, { date: string; subject: string | null; items: typeof attendance }>())
   208→
   209→  return (
   210→    <div className="space-y-3">
   211→      <div className="rounded-xl border border-border bg-card p-3 flex flex-wrap items-center gap-2">
   212→        <Select value={classId} onValueChange={setClassId}>
   213→          <SelectTrigger size="sm" className="text-xs w-[180px]"><SelectValue placeholder="Select class" /></SelectTrigger>
   214→          <SelectContent>
   215→            {exam?.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   216→          </SelectContent>
   217→        </Select>
   218→        <Button size="sm" variant="outline" className="h-7 text-xs gap-1.5" onClick={handleAutoMark} disabled={autoLoading || !classId}>
   219→          <Sparkles className="h-3 w-3" /> {autoLoading ? 'Marking…' : 'Auto-mark from Marks'}
   220→        </Button>
   221→        <span className="text-[10px] text-muted-foreground ml-auto">{attendance.length} entries</span>
   222→      </div>
   223→
   224→      {loading ? <InlineLoading label="Loading exam attendance…" /> : attendance.length === 0 ? (
   225→        <div className="rounded-xl border border-border bg-card p-6 text-center text-xs text-muted-foreground">
   226→          No exam attendance records. Use "Auto-mark from Marks" to sync from existing marks, or attendance will be tracked as marks are entered.
   227→        </div>
   228→      ) : (
   229→        <div className="space-y-2">
   230→          {Array.from(grouped.values()).map((group, idx) => (
   231→            <div key={idx} className="rounded-xl border border-border bg-card overflow-hidden">
   232→              <div className="px-3 py-2 border-b border-border bg-muted/30 flex items-center gap-2">
   233→                <p className="text-xs font-semibold">{group.subject ?? 'General Session'}</p>
   234→                <span className="text-[10px] text-muted-foreground">{group.date}</span>
   235→                <span className="text-[10px] text-muted-foreground ml-auto">{group.items.length} students</span>
   236→              </div>
   237→              <Table>
   238→                <TableHeader>
   239→                  <TableRow>
   240→                    <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Roll No</TableHead>
   241→                    <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Student</TableHead>
   242→                    <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Status</TableHead>
   243→                    <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Remarks</TableHead>
   244→                  </TableRow>
   245→                </TableHeader>
   246→                <TableBody>
   247→                  {group.items.map((a) => (
   248→                    <TableRow key={a.id}>
   249→                      <TableCell className="text-xs font-mono">{a.studentRollNo ?? '—'}</TableCell>
   250→                      <TableCell className="text-xs">{a.studentName}</TableCell>
   251→                      <TableCell>
   252→                        <StatusBadge status={a.status} />
   253→                      </TableCell>
   254→                      <TableCell className="text-xs text-muted-foreground">{a.remarks ?? '—'}</TableCell>
   255→                    </TableRow>
   256→                  ))}
   257→                </TableBody>
   258→              </Table>
   259→            </div>
   260→          ))}
   261→        </div>
   262→      )}
   263→    </div>
   264→  )
   265→}

   267→function StatusBadge({ status }: { status: string }) {
   268→  const cls = {
   269→    PRESENT: 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/20',
   270→    ABSENT: 'bg-rose-500/10 text-rose-700 dark:text-rose-300 border-rose-500/20',
   271→    MEDICAL: 'bg-amber-500/10 text-amber-700 dark:text-amber-300 border-amber-500/20',
   272→    EXEMPTED: 'bg-violet-500/10 text-violet-700 dark:text-violet-300 border-violet-500/20',
   273→  }[status] || 'bg-muted text-muted-foreground border-border'
   274→  return <span className={cn('inline-flex items-center rounded-full border px-1.5 py-0.5 text-[9px] font-semibold', cls)}>{status}</span>
   275→}

   279→export function GraceSection({ examId, exam }: SectionProps) {
   280→  const [classId, setClassId] = useState(exam?.classes[0]?.classId ?? '')
   281→  const [subjectId, setSubjectId] = useState('')
   282→  const { students, marks } = useMarks(examId, classId, subjectId)
   283→  const [selectedMarkId, setSelectedMarkId] = useState<string | null>(null)
   284→  const [graceMarks, setGraceMarks] = useState(0)
   285→  const [reason, setReason] = useState('')
   286→  const { apply, loading } = useApplyGrace()
   287→  const gate = useRoleGate()
   288→
   289→  const subjectsForClass = exam?.subjects.filter((s: any) => !classId || s.classId === classId) ?? []
   290→  const markRows = students.map((s) => {
   291→    const m = marks.find((mk) => mk.studentId === s.id)
   292→    return { student: s, mark: m }
   293→  }).filter((r) => r.mark)
   294→
   295→  const handleApply = async () => {
   296→    if (!selectedMarkId) { toast.error('Select a student'); return }
   297→    if (graceMarks === 0) { toast.error('Grace marks cannot be 0'); return }
   298→    if (!reason.trim()) { toast.error('Reason is required'); return }
   299→    try {
   300→      await apply(examId, selectedMarkId, graceMarks, reason.trim())
   301→      toast.success(`+${graceMarks} grace marks applied`, { description: 'Original marks preserved in audit log.' })
   302→      setSelectedMarkId(null)
   303→      setGraceMarks(0)
   304→      setReason('')
   305→    } catch (e: any) {
   306→      toast.error('Failed to apply grace', { description: e.message })
   307→    }
   308→  }
   309→
   310→  return (
   311→    <div className="space-y-3">
   312→      <div className="rounded-xl border border-amber-500/30 bg-amber-500/5 p-3 flex items-start gap-2">
   313→        <AlertTriangle className="h-4 w-4 text-amber-600 shrink-0 mt-0.5" />
   314→        <div>
   315→          <p className="text-xs font-semibold text-amber-700 dark:text-amber-300">Grace / Moderation</p>
   316→          <p className="text-[10px] text-amber-700/80 dark:text-amber-300/80 mt-0.5">
   317→            Original marks are preserved. All grace applications are recorded in the audit log with the reason and authorizing user. After results are declared, only Principal can apply grace.
   318→          </p>
   319→        </div>
   320→      </div>
   321→
   322→      <div className="rounded-xl border border-border bg-card p-3">
   323→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mb-3">
   324→          <Select value={classId} onValueChange={(v) => { setClassId(v); setSubjectId('') }}>
   325→            <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Class" /></SelectTrigger>
   326→            <SelectContent>
   327→              {exam?.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   328→            </SelectContent>
   329→          </Select>
   330→          <Select value={subjectId} onValueChange={setSubjectId}>
   331→            <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Subject" /></SelectTrigger>
   332→            <SelectContent>
   333→              {subjectsForClass.map((s: any) => <SelectItem key={s.subjectId} value={s.subjectId}>{s.subjectName}</SelectItem>)}
   334→            </SelectContent>
   335→          </Select>
   336→        </div>
   337→
   338→        {!subjectId ? (
   339→          <p className="text-xs text-muted-foreground py-4 text-center">Select a subject to view marks.</p>
   340→        ) : markRows.length === 0 ? (
   341→          <p className="text-xs text-muted-foreground py-4 text-center">No marks entered for this subject yet.</p>
   342→        ) : (
   343→          <Table>
   344→            <TableHeader>
   345→              <TableRow>
   346→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Select</TableHead>
   347→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Roll No</TableHead>
   348→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Student</TableHead>
   349→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Marks</TableHead>
   350→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Grace</TableHead>
   351→              </TableRow>
   352→            </TableHeader>
   353→            <TableBody>
   354→              {markRows.map(({ student, mark }) => (
   355→                <TableRow key={student.id} className={cn(selectedMarkId === mark?.id && 'bg-amber-500/5')}>
   356→                  <TableCell>
   357→                    {gate.canApplyGrace && (
   358→                      <input type="radio" checked={selectedMarkId === mark?.id} onChange={() => { setSelectedMarkId(mark?.id ?? null); setGraceMarks(0); setReason('') }} />
   359→                    )}
   360→                  </TableCell>
   361→                  <TableCell className="text-xs font-mono">{student.rollNo ?? '—'}</TableCell>
   362→                  <TableCell className="text-xs">{student.name}</TableCell>
   363→                  <TableCell className="text-xs tabular-nums">{mark?.status === 'PRESENT' ? (mark?.marksObtained ?? '—') : mark?.status}</TableCell>
   364→                  <TableCell className="text-xs tabular-nums text-amber-600">+{mark?.graceMarks ?? 0}</TableCell>
   365→                </TableRow>
   366→              ))}
   367→            </TableBody>
   368→          </Table>
   369→        )}
   370→
   371→        {selectedMarkId && gate.canApplyGrace && (
   372→          <div className="mt-3 rounded-lg border border-amber-500/30 bg-amber-500/5 p-3 space-y-2">
   373→            <p className="text-[10px] uppercase font-bold text-amber-700 dark:text-amber-300">Apply Grace Marks</p>
   374→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   375→              <div>
   376→                <Label className="text-[10px]">Grace Marks (±)</Label>
   377→                <Input type="number" value={graceMarks} onChange={(e) => setGraceMarks(Number(e.target.value))} className="h-7 text-xs" />
   378→              </div>
   379→              <div>
   380→                <Label className="text-[10px]">Reason (required)</Label>
   381→                <Input value={reason} onChange={(e) => setReason(e.target.value)} placeholder="e.g. Question paper error Q3" className="h-7 text-xs" />
   382→              </div>
   383→            </div>
   384→            <Button size="sm" className="h-7 text-xs gap-1.5 bg-amber-600 hover:bg-amber-700 text-white" onClick={handleApply} disabled={loading}>
   385→              <Send className="h-3 w-3" /> {loading ? 'Applying…' : 'Apply Grace'}
   386→            </Button>
   387→          </div>
   388→        )}
   389→      </div>
   390→    </div>
   391→  )
   392→}

   394→// ─── Outcomes Section (Promotion/Compartment/Retest) ─────────────────
   395→
   396→export function OutcomesSection({ examId, exam }: SectionProps) {
   397→  const [classId, setClassId] = useState(exam?.classes[0]?.classId ?? '')
   398→  const { outcomes, loading, reload } = useOutcomes(examId, classId)
   399→  const { compute, loading: computing } = useComputeOutcomes()
   400→  const { override, loading: overriding } = useOverrideOutcome()
   401→  const gate = useRoleGate()
   402→  void overriding
   403→
   404→  const handleCompute = async () => {
   405→    if (!classId) return
   406→    try {
   407→      const r = await compute(examId, classId)
   408→      toast.success(`Outcomes computed for ${r.autoCount} students`, { description: 'PROMOTED if passed, COMPARTMENT if 1 fail, RETEST if 2 fails, NOT_PROMOTED otherwise.' })
   409→      reload()
   410→    } catch (e: any) {
   411→      toast.error('Failed to compute outcomes', { description: e.message })
   412→    }
   413→  }
   414→
   415→  const handleOverride = async (studentId: string, outcome: 'PROMOTED' | 'COMPARTMENT' | 'RETEST' | 'NOT_PROMOTED') => {
   416→    try {
   417→      await override(examId, studentId, outcome, 'Manual override by Principal')
   418→      toast.success(`Outcome overridden to ${outcome}`)
   419→      reload()
   420→    } catch (e: any) {
   421→      toast.error('Failed to override', { description: e.message })
   422→    }
   423→  }
   424→
   425→  const summary = {
   426→    total: outcomes.length,
   427→    promoted: outcomes.filter((o) => o.outcome === 'PROMOTED').length,
   428→    compartment: outcomes.filter((o) => o.outcome === 'COMPARTMENT').length,
   429→    retest: outcomes.filter((o) => o.outcome === 'RETEST').length,
   430→    notPromoted: outcomes.filter((o) => o.outcome === 'NOT_PROMOTED').length,
   431→  }
   432→
   433→  return (
   434→    <div className="space-y-3">
   435→      <div className="rounded-xl border border-border bg-card p-3 flex flex-wrap items-center gap-2">
   436→        <Select value={classId} onValueChange={setClassId}>
   437→          <SelectTrigger size="sm" className="text-xs w-[180px]"><SelectValue placeholder="Select class" /></SelectTrigger>
   438→          <SelectContent>
   439→            {exam?.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   440→          </SelectContent>
   441→        </Select>
   442→        {gate.canOverrideOutcome && (
   443→          <Button size="sm" variant="default" className="h-7 text-xs gap-1.5" onClick={handleCompute} disabled={computing || !classId}>
   444→            <Sparkles className="h-3 w-3" /> {computing ? 'Computing…' : 'Compute Auto Outcomes'}
   445→          </Button>
   446→        )}
   447→        {outcomes.length > 0 && (
   448→          <div className="ml-auto flex items-center gap-2 text-[10px]">
   449→            <OutcomeBadge outcome="PROMOTED" count={summary.promoted} />
   450→            <OutcomeBadge outcome="COMPARTMENT" count={summary.compartment} />
   451→            <OutcomeBadge outcome="RETEST" count={summary.retest} />
   452→            <OutcomeBadge outcome="NOT_PROMOTED" count={summary.notPromoted} />
   453→          </div>
   454→        )}
   455→      </div>
   456→
   457→      {loading ? <InlineLoading label="Loading outcomes…" /> : outcomes.length === 0 ? (
   458→        <div className="rounded-xl border border-border bg-card p-6 text-center text-xs text-muted-foreground">
   459→          No outcomes computed yet. Click "Compute Auto Outcomes" to derive from marks.
   460→        </div>
   461→      ) : (
   462→        <div className="rounded-xl border border-border bg-card overflow-hidden">
   463→          <Table>
   464→            <TableHeader>
   465→              <TableRow>
   466→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Roll No</TableHead>
   467→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Student</TableHead>
   468→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground text-center">%</TableHead>
   469→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground text-center">Grade</TableHead>
   470→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground text-center">Failed</TableHead>
   471→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Outcome</TableHead>
   472→                <TableHead className="text-[9px] uppercase font-semibold text-muted-foreground">Override</TableHead>
   473→              </TableRow>
   474→            </TableHeader>
   475→            <TableBody>
   476→              {outcomes.map((o) => (
   477→                <TableRow key={o.id}>
   478→                  <TableCell className="text-xs font-mono">{o.studentRollNo ?? '—'}</TableCell>
   479→                  <TableCell className="text-xs">{o.studentName}</TableCell>
   480→                  <TableCell className="text-xs tabular-nums text-center">{o.percentage}%</TableCell>
   481→                  <TableCell className="text-xs font-bold text-center">{o.grade}</TableCell>
   482→                  <TableCell className="text-xs text-center tabular-nums">{o.subjectsFailed}</TableCell>
   483→                  <TableCell><OutcomePill outcome={o.outcome} /></TableCell>
   484→                  <TableCell>
   485→                    {gate.canOverrideOutcome ? (
   486→                      <Select value={o.outcome} onValueChange={(v) => handleOverride(o.studentId, v as any)}>
   487→                        <SelectTrigger size="sm" className="h-6 text-[10px] w-[140px]"><SelectValue /></SelectTrigger>
   488→                        <SelectContent>
   489→                          <SelectItem value="PROMOTED">Promoted</SelectItem>
   490→                          <SelectItem value="COMPARTMENT">Compartment</SelectItem>
   491→                          <SelectItem value="RETEST">Retest</SelectItem>
   492→                          <SelectItem value="NOT_PROMOTED">Not Promoted</SelectItem>
   493→                        </SelectContent>
   494→                      </Select>
   495→                    ) : (
   496→                      <span className="text-[9px] text-muted-foreground/40">—</span>
   497→                    )}
   498→                  </TableCell>
   499→                </TableRow>
   500→              ))}
   501→            </TableBody>
   502→          </Table>
   503→        </div>
   504→      )}
   505→    </div>
   506→  )
   507→}
   508→
   509→function OutcomeBadge({ outcome, count }: { outcome: string; count: number }) {
   510→  const cls = outcomeStyle(outcome)
   511→  return <span className={cn('inline-flex items-center rounded-full border px-1.5 py-0.5 text-[9px] font-semibold', cls)}>{outcome} · {count}</span>
   512→}
   513→
   514→function OutcomePill({ outcome }: { outcome: string }) {
   515→  return <span className={cn('inline-flex items-center rounded-full border px-2 py-0.5 text-[10px] font-semibold', outcomeStyle(outcome))}>{outcome.replace('_', ' ')}</span>
   516→}
   517→
   518→function outcomeStyle(outcome: string): string {
   519→  switch (outcome) {
   520→    case 'PROMOTED': return 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border-emerald-500/20'
   521→    case 'COMPARTMENT': return 'bg-amber-500/10 text-amber-700 dark:text-amber-300 border-amber-500/20'
   522→    case 'RETEST': return 'bg-orange-500/10 text-orange-700 dark:text-orange-300 border-orange-500/20'
   523→    case 'NOT_PROMOTED': return 'bg-rose-500/10 text-rose-700 dark:text-rose-300 border-rose-500/20'
   524→    default: return 'bg-muted text-muted-foreground border-border'
   525→  }
   526→}

   530→export function CsvImportSection({ examId, exam }: SectionProps) {
   531→  const [classId, setClassId] = useState(exam?.classes[0]?.classId ?? '')
   532→  const [subjectId, setSubjectId] = useState('')
   533→  const [pasteText, setPasteText] = useState('')
   534→  const [preview, setPreview] = useState<CsvImportRow[] | null>(null)
   535→  const [errors, setErrors] = useState<string[]>([])
   536→  const { importCsv, loading } = useImportMarksCsv()
   537→  const gate = useRoleGate()
   538→
   539→  const subjectsForClass = exam?.subjects.filter((s: any) => !classId || s.classId === classId) ?? []
   540→
   541→  const handleParse = () => {
   542→    const lines = pasteText.trim().split('\n')
   543→    const rows: CsvImportRow[] = []
   544→    const errs: string[] = []
   545→    // Skip header if present
   546→    const startIdx = lines[0]?.toLowerCase().includes('roll') ? 1 : 0
   547→    for (let i = startIdx; i < lines.length; i++) {
   548→      const cols = lines[i].split(',').map((c) => c.trim().replace(/^"|"$/g, ''))
   549→      if (cols.length < 2) { errs.push(`Row ${i + 1}: not enough columns`); continue }
   550→      const [rollNo, name, marks, status, remarks] = cols
   551→      const statusU = (status || 'PRESENT').toUpperCase() as MarkStatus
   552→      if (!['PRESENT', 'ABSENT', 'MEDICAL', 'EXEMPTED'].includes(statusU)) {
   553→        errs.push(`Row ${i + 1}: invalid status "${status}"`)
   554→        continue
   555→      }
   556→      const marksNum = marks === '' ? null : Number(marks)
   557→      if (statusU === 'PRESENT' && marksNum !== null && (isNaN(marksNum) || marksNum < 0)) {
   558→        errs.push(`Row ${i + 1}: invalid marks "${marks}"`)
   559→        continue
   560→      }
   561→      rows.push({ rollNo, studentName: name, marksObtained: marksNum, status: statusU, remarks })
   562→    }
   563→    setPreview(rows)
   564→    setErrors(errs)
   565→  }
   566→
   567→  const handleDownloadTemplate = async () => {
   568→    if (!classId || !subjectId) { toast.error('Select class and subject first'); return }
   569→    try {
   570→      await downloadCsvTemplate(examId, classId, subjectId)
   571→      toast.success('Template downloaded')
   572→    } catch (e: any) {
   573→      toast.error('Failed to download template', { description: e.message })
   574→    }
   575→  }
   576→
   577→  const handleImport = async () => {
   578→    if (!preview || preview.length === 0) { toast.error('Nothing to import'); return }
   579→    try {
   580→      const result = await importCsv(examId, classId, subjectId, preview)
   581→      toast.success(`Imported ${result.accepted} marks`, {
   582→        description: result.rejected > 0 ? `${result.rejected} rejected: ${result.errors[0]?.message ?? 'see audit log'}` : 'All rows imported successfully',
   583→      })
   584→      setPasteText('')
   585→      setPreview(null)
   586→      setErrors([])
   587→    } catch (e: any) {
   588→      toast.error('Import failed', { description: e.message })
   589→    }
   590→  }
   591→
   592→  return (
   593→    <div className="space-y-3">
   594→      <div className="rounded-xl border border-border bg-card p-3">
   595→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 mb-3">
   596→          <Select value={classId} onValueChange={(v) => { setClassId(v); setSubjectId('') }}>
   597→            <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Class" /></SelectTrigger>
   598→            <SelectContent>
   599→              {exam?.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   600→            </SelectContent>
   601→          </Select>
   602→          <Select value={subjectId} onValueChange={setSubjectId}>
   603→            <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Subject" /></SelectTrigger>
   604→            <SelectContent>
   605→              {subjectsForClass.map((s: any) => <SelectItem key={s.subjectId} value={s.subjectId}>{s.subjectName}</SelectItem>)}
   606→            </SelectContent>
   607→          </Select>
   608→        </div>
   609→        <Button size="sm" variant="outline" className="h-7 text-xs mb-2" onClick={handleDownloadTemplate} disabled={!classId || !subjectId}>
   610→          Download CSV Template
   611→        </Button>
   612→        <Textarea
   613→          value={pasteText}
   614→          onChange={(e) => setPasteText(e.target.value)}
   615→          placeholder={'Roll Number,Student Name,Marks Obtained,Status,Remarks\n01,Aarav Sharma,35,PRESENT,\n02,Diya Patel,40,PRESENT,\n03,Vivaan Reddy,,ABSENT,'}
   616→          className="text-xs font-mono h-32"
   617→        />
   618→        <div className="flex flex-wrap gap-2 mt-2">
   619→          <Button size="sm" variant="outline" className="h-7 text-xs" onClick={handleParse} disabled={!pasteText.trim() || !gate.canImportCsv}>
   620→            Parse & Preview
   621→          </Button>
   622→          {preview && gate.canImportCsv && (
   623→            <Button size="sm" variant="default" className="h-7 text-xs gap-1.5 bg-emerald-600 hover:bg-emerald-700 text-white" onClick={handleImport} disabled={loading || preview.length === 0}>
   624→              {loading ? 'Importing…' : `Import ${preview.length} rows`}
   625→            </Button>
   626→          )}
   627→          {!gate.canImportCsv && (
   628→            <p className="text-[10px] text-muted-foreground self-center">Read-only — you don't have permission to import marks.</p>
   629→          )}
   630→        </div>
   631→        {errors.length > 0 && (
   632→          <div className="mt-2 rounded-lg border border-rose-500/30 bg-rose-500/5 p-2">
   633→            <p className="text-[10px] font-semibold text-rose-700 dark:text-rose-300 mb-1">{errors.length} parse errors:</p>
   634→            <ul className="text-[10px] text-rose-700/80 dark:text-rose-300/80 list-disc list-inside space-y-0.5">
   635→              {errors.slice(0, 5).map((e, i) => <li key={i}>{e}</li>)}
   636→            </ul>
   637→          </div>
   638→        )}
   639→        {preview && preview.length > 0 && (
   640→          <div className="mt-2">
   641→            <p className="text-[10px] font-semibold text-muted-foreground mb-1">Preview ({preview.length} rows)</p>
   642→            <div className="max-h-40 overflow-y-auto rounded-lg border border-border/60">
   643→              <Table>
   644→                <TableHeader>
   645→                  <TableRow>
   646→                    <TableHead className="text-[9px] uppercase">Roll</TableHead>
   647→                    <TableHead className="text-[9px] uppercase">Student</TableHead>
   648→                    <TableHead className="text-[9px] uppercase text-center">Marks</TableHead>
   649→                    <TableHead className="text-[9px] uppercase">Status</TableHead>
   650→                  </TableRow>
   651→                </TableHeader>
   652→                <TableBody>
   653→                  {preview.slice(0, 10).map((r, i) => (
   654→                    <TableRow key={i}>
   655→                      <TableCell className="text-[10px] font-mono">{r.rollNo}</TableCell>
   656→                      <TableCell className="text-[10px]">{r.studentName}</TableCell>
   657→                      <TableCell className="text-[10px] text-center tabular-nums">{r.marksObtained ?? '—'}</TableCell>
   658→                      <TableCell className="text-[10px]">{r.status}</TableCell>
   659→                    </TableRow>
   660→                  ))}
   661→                </TableBody>
   662→              </Table>
   663→              {preview.length > 10 && <p className="text-[10px] text-muted-foreground p-2 text-center">+ {preview.length - 10} more rows</p>}
   664→            </div>
   665→          </div>
   666→        )}
   667→      </div>
   668→    </div>
   669→  )
   670→}
