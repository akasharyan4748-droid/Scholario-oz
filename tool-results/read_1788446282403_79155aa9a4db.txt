     3→     3→/**
     4→     4→ * ApplyDialog — the student's Educational Tour application: fill → submit →
     5→     5→ * pay, inside ONE dialog.
     6→     6→ *
     7→     7→ * Structure mirrors the printed official form so what the student fills is
     8→     8→ * exactly what the school office files:
     9→     9→ *   1. Official document header (school, form no., session)
    10→    10→ *   2. Tour details (read-only — destination, dates, fee, in-charge)
    11→    11→ *   3. Student particulars (READ-ONLY, from the school record — applicants
    12→    12→ *      never edit their master record here; corrections go through the office)
    13→    13→ *   4. Guardian details (read-only)
    14→    14→ *   5. Tour-specific details (the template's editable questions — saved to
    15→    15→ *      the submission snapshot only, never the canonical student record)
    16→    16→ *   6. Declaration & guardian consent
    17→    17→ *   → review step: compact REVIEW APPLICATION summary before final submit
    18→    18→ *   → payment step: Online (instant receipt) or Cash / Pay at School
    19→    19→ *     (recorded Under Verification until the Principal confirms).
    20→    20→ *
    21→    21→ * MONEY RULE: payments go through fee-store recordPayment() only, bound to
    22→    22→ * the application's Additional Charge + applicationId — tour money is
    23→    23→ * separate from the student's annual fees and lands in the normal ledger.
    24→    24→ *
    25→    25→ * Idempotency: re-submitting an already-submitted form returns
    26→    26→ * existingSubmissionId from the store — we surface "You already applied"
    27→    27→ * and close instead of creating a duplicate.
    28→    28→ */

    75→    75→type Answers = Record<string, string | string[] | boolean>

    79→    79→const EM_DASH = ' — '
    80→    80→
    81→    81→function splitEmergencyContact(value: Answers[string]): { name: string; phone: string } {
    82→    82→  if (typeof value === 'string' && value.includes(EM_DASH)) {
    83→    83→    const [name = '', phone = ''] = value.split(EM_DASH)
    84→    84→    return { name, phone }
    85→    85→  }
    86→    86→  return { name: '', phone: '' }
    87→    87→}

    89→    89→export function ApplyDialog({ open, onOpenChange, app, identity, existingSubmission = null, initialStep = 'form' }: ApplyDialogProps) {
    90→    90→  const [step, setStep] = useState<'form' | 'review' | 'payment'>(initialStep === 'payment' ? 'payment' : 'form')
    91→    91→  const [answers, setAnswers] = useState<Answers>({})
    92→    92→  const [attachments, setAttachments] = useState<Attachments>({})
    93→    93→  const [emergency, setEmergency] = useState<Record<string, { name: string; phone: string }>>({})
    94→    94→  const [consentAccepted, setConsentAccepted] = useState(false)
    95→    95→  const [signature, setSignature] = useState<SignatureValue | null>(null)
    96→    96→  const [errors, setErrors] = useState<Errors>({})
    97→    97→  const [submitting, setSubmitting] = useState(false)
    98→    98→  const [paying, setPaying] = useState<PayMode | null>(null)
    99→    99→  const [activeSubmissionId, setActiveSubmissionId] = useState<string | null>(null)
   100→   100→
   101→   101→  const isFixMode = !!existingSubmission
   102→   102→  // Active school's own branding (tenant-scoped School Settings).
   103→   103→  const schGeneral = useSchoolSettingsStore((s) => s.general)
   104→   104→
   105→   105→  // ── Reset / prefill whenever the dialog opens for an application ──
   106→   106→  useEffect(() => {
   107→   107→    if (!open || !app) return
   108→   108→    setStep(initialStep === 'payment' ? 'payment' : 'form')
   109→   109→    setErrors({})
   110→   110→    setSubmitting(false)
   111→   111→    setPaying(null)
   112→   112→    if (existingSubmission) {
   113→   113→      setAnswers({ ...existingSubmission.answers })
   114→   114→      setAttachments(existingSubmission.attachments ? { ...existingSubmission.attachments } : {})
   115→   115→      setEmergency(Object.fromEntries(
   116→   116→        app.formFields
   117→   117→          .filter((f) => f.type === 'emergency-contact')
   118→   118→          .map((f) => [f.id, splitEmergencyContact(existingSubmission.answers[f.id])]),
   119→   119→      ))
   120→   120→      setConsentAccepted(!!existingSubmission.consentGivenAt)
   121→   121→      setActiveSubmissionId(existingSubmission.id)
   122→   122→    } else {
   123→   123→      setAnswers({})
   124→   124→      setAttachments({})
   125→   125→      setEmergency(Object.fromEntries(
   126→   126→        app.formFields
   127→   127→          .filter((f) => f.type === 'emergency-contact')
   128→   128→          .map((f) => [f.id, { name: '', phone: '' }]),
   129→   129→      ))
   130→   130→      setConsentAccepted(false)
   131→   131→      setSignature(null)
   132→   132→      setActiveSubmissionId(null)
   133→   133→    }
   134→   134→  }, [open, app, existingSubmission, initialStep])
   135→   135→
   136→   136→  const amount = app?.payment.amount ?? 0
   137→   137→  const needsPayment = !!app && app.payment.mode !== 'None'
   138→   138→  const isTour = !!app && (app.category === 'Tour' || app.category === 'Trip' || app.templateKey === 'educational_tour')
   139→   139→
   140→   140→  const closeDialog = () => {
   141→   141→    onOpenChange(false)
   142→   142→  }
   143→   143→
   144→   144→  // ── Validation (application-specific fields + digital guardian consent) ──
   145→   145→  const validate = (): boolean => {
   146→   146→    if (!app) return false
   147→   147→    const errs: Errors = {}
   148→   148→    for (const f of app.formFields) {
   149→   149→      const v = answers[f.id]
   150→   150→      if (f.type === 'emergency-contact') {
   151→   151→        const { name, phone } = emergency[f.id] ?? { name: '', phone: '' }
   152→   152→        if (f.required && (!name.trim() || !phone.trim())) {
   153→   153→          errs[f.id] = 'Enter both a contact name and a phone number.'
   154→   154→        }
   155→   155→        continue
   156→   156→      }
   157→   157→      if (f.type === 'file') {
   158→   158→        if (f.required && !attachments[f.id]) errs[f.id] = 'Please attach a file.'
   159→   159→        continue
   160→   160→      }
   161→   161→      if (f.type === 'signature' || f.type === 'declaration') {
   162→   162→        if (f.required && v !== true) errs[f.id] = 'Tick the acknowledgement to continue.'
   163→   163→        continue
   164→   164→      }
   165→   165→      if (f.type === 'checkbox' || f.type === 'multiselect') {
   166→   166→        if (f.required && (!Array.isArray(v) || v.length === 0)) errs[f.id] = 'Pick at least one option.'
   167→   167→        continue
   168→   168→      }
   169→   169→      if (f.type === 'yesno') {
   170→   170→        if (f.required && typeof v !== 'boolean') errs[f.id] = 'Choose Yes or No.'
   171→   171→        continue
   172→   172→      }
   173→   173→      if (f.required && (typeof v !== 'string' || !v.trim())) {
   174→   174→        errs[f.id] = 'This field is required.'
   175→   175→      }
   176→   176→    }
   177→   177→    if (app.guardianConsent.required && app.guardianConsent.method === 'Digital' && !consentAccepted) {
   178→   178→      errs.consent = 'Guardian consent is required before submitting.'
   179→   179→    }
   180→   180→    // PART 19 — a required digital signature is a real captured signature,
   181→   181→    // never fabricated. Only enforced on the initial submit (fix-mode keeps
   182→   182→    // the original signature on record).
   183→   183→    if (
   184→   184→      !isFixMode && app.guardianConsent.required && app.guardianConsent.method === 'Digital'
   185→   185→      && app.guardianConsent.signatureRequired !== false
   186→   186→      && !signatureComplete(signature)
   187→   187→    ) {
   188→   188→      errs.signature = 'Capture the guardian’s signature (draw or type) to continue.'
   189→   189→    }
   190→   190→    setErrors(errs)
   191→   191→    if (Object.keys(errs).length > 0) {
   192→   192→      toast.error('Please complete the highlighted fields.')
   193→   193→      return false
   194→   194→    }
   195→   195→    return true
   196→   196→  }
   197→   197→
   198→   198→  // ── Submit (or resubmit after corrections) — reached from the review step ──
   199→   199→  const handleSubmit = () => {
   200→   200→    if (!app || !identity || submitting) return
   201→   201→    if (initialStep !== 'payment' && !validate()) {
   202→   202→      setStep('form')
   203→   203→      return
   204→   204→    }

   212→   212→        Object.keys(attachments).length ? attachments : undefined,
   213→   213→        identity.canonical.name,
   214→   214→      )
   215→   215→      setSubmitting(false)
   216→   216→      if (!res.success) {
   217→   217→        toast.error(res.error ?? 'Could not resubmit.')
   218→   218→        return
   219→   219→      }
   220→   220→      const updated = useApplicationsStore.getState().submissions.find((s) => s.id === existingSubmission.id)
   221→   221→      const pay = updated ? deriveSubmissionPayment(app, updated) : null
   222→   222→      if (needsPayment && (!pay || pay.status === 'Not Paid')) {
   223→   223→        setActiveSubmissionId(existingSubmission.id)
   224→   224→        setStep('payment')
   225→   225→        toast.info('Corrections submitted — one more step: payment.')
   226→   226→      } else {
   227→   227→        toast.success('Corrections submitted for review.')
   228→   228→        closeDialog()
   229→   229→      }
   230→   230→      return
   231→   231→    }
   232→   232→
   233→   233→    const res = useApplicationsStore.getState().submitApplication({
   234→   234→      applicationId: app.id,
   235→   235→      student: identity.canonical,
   236→   236→      answers,
   237→   237→      attachments: Object.keys(attachments).length ? attachments : undefined,
   238→   238→      consentAccepted,
   239→   239→      signature: signatureComplete(signature) ? signature ?? undefined : undefined,
   240→   240→      submittedByRole: 'Student',
   241→   241→    })
   242→   242→
   243→   243→    if (res.success && res.existingSubmissionId) {
   244→   244→      // Idempotent hit — the store refused to duplicate. Be honest + close.
   245→   245→      setSubmitting(false)
   246→   246→      toast.info('You already applied', {
   247→   247→        description: 'A submission for this form already exists — nothing was duplicated.',
   248→   248→      })
   249→   249→      closeDialog()
   250→   250→      return
   251→   251→    }
   252→   252→    if (!res.success || !res.submission) {
   253→   253→      setSubmitting(false)
   254→   254→      toast.error(res.error ?? 'Submission failed.')
   255→   255→      return
   256→   256→    }
   257→   257→
   258→   258→    if (needsPayment) {
   259→   259→      setActiveSubmissionId(res.submission.id)
   260→   260→      setStep('payment')
   261→   261→      setSubmitting(false)
   262→   262→      toast.success('Application submitted — complete the payment to finish.', {
   263→   263→        description: formatINR(amount),
   264→   264→      })
   265→   265→    } else {
   266→   266→      setSubmitting(false)
   267→   267→      toast.success('Application submitted.', {
   268→   268→        description: 'Track its status under My submissions.',
   269→   269→      })
   270→   270→      closeDialog()
   271→   271→    }
   272→   272→  }
   273→   273→
   274→   274→  // ── Payment — the ONLY way money moves (canonical fee store) ──
   275→   275→  const runPayment = (mode: PayMode) => {
   276→   276→    if (!app || !identity || paying) return
   277→   277→    if (!app.payment.chargeId) return // guarded in the UI
   278→   278→    setPaying(mode)
   279→   279→    const res = useFeeStore.getState().recordPayment({
   280→   280→      studentId: identity.canonical.id,
   281→   281→      amount: app.payment.amount,
   282→   282→      mode,
   283→   283→      purpose: `Application: ${app.title}`,
   284→   284→      feeHead: app.payment.feeHeadLabel || app.title,
   285→   285→      collectedBy: 'Student Self-Service',
   286→   286→      additionalChargeId: app.payment.chargeId,
   287→   287→      applicationId: app.id,
   288→   288→      ...(mode === 'UPI' ? { referenceNo: `UPI-${Date.now()}` } : {}),
   289→   289→    })
   290→   290→    setPaying(null)
   291→   291→    if (!res.success || !res.transaction) {
   292→   292→      toast.error(res.error ?? 'Payment could not be recorded.')
   293→   293→      return
   294→   294→    }
   295→   295→    const receipt = res.transaction.receiptNo
   296→   296→    addAuditEvent({
   297→   297→      applicationId: app.id,
   298→   298→      submissionId: activeSubmissionId ?? undefined,
   299→   299→      ts: new Date().toISOString(),
   300→   300→      actor: identity.canonical.name,
   301→   301→      actorRole: 'Student',
   302→   302→      action: 'payment.completed',
   303→   303→      message: mode === 'Cash'
   304→   304→        ? `Cash ${formatINR(app.payment.amount)} recorded for "${app.title}" — receipt ${receipt}, awaiting verification by the Principal.`
   305→   305→        : `Paid ${formatINR(app.payment.amount)} online (UPI ref ${res.transaction.referenceNo ?? '—'}) for "${app.title}" — receipt ${receipt}.`,
   306→   306→    })
   307→   307→    if (mode === 'Cash') {
   308→   308→      toast.info('Recorded — awaiting cash verification by the Principal', {
   309→   309→        description: `Receipt ${receipt} · ${formatINR(app.payment.amount)} · ${app.title}`,
   310→   310→      })
   311→   311→    } else {
   312→   312→      toast.success(`Payment successful — receipt ${receipt}`, {
   313→   313→        description: `${formatINR(app.payment.amount)} paid for ${app.title}.`,
   314→   314→      })
   315→   315→    }
   316→   316→    closeDialog()
   317→   317→  }
   318→   318→
   319→   319→  const skipPayment = () => {
   320→   320→    toast.info('Payment skipped for now', {
   321→   321→      description: 'You can pay anytime from My submissions.',
   322→   322→    })
   323→   323→    closeDialog()
   324→   324→  }
   325→   325→
   326→   326→  return (

   328→   328→      <DialogContent className="sm:max-w-lg max-h-[88vh] overflow-hidden flex flex-col">
   329→   329→        {app && identity && (
   330→   330→          <>
   331→   331→            <DialogHeader className="text-left shrink-0">
   332→   332→              <div className="flex items-start justify-between gap-3">
   333→   333→                <div className="min-w-0">
   334→   334→                  <DialogTitle className="text-base leading-snug">{app.title}</DialogTitle>
   335→   335→                  <DialogDescription className="mt-1 flex flex-wrap items-center gap-1.5 text-[11px]">
   336→   336→                    <Badge variant="outline" className="text-[9px] h-4 px-1.5">Educational Tour</Badge>
   337→   337→                    <span>Deadline {formatDate(app.deadline)}</span>
   338→   338→                    {needsPayment && <span>· {formatINR(amount)}{app.payment.mode === 'Optional' ? ' (optional)' : ''}</span>}
   339→   339→                  </DialogDescription>
   340→   340→                </div>
   341→   341→                {step === 'payment' && (
   342→   342→                  <Badge variant="outline" className="shrink-0 text-[9px] h-4 px-1.5 border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-400">
   343→   343→                    Payment
   344→   344→                  </Badge>
   345→   345→                )}
   346→   346→                {step === 'review' && (
   347→   347→                  <Badge variant="outline" className="shrink-0 text-[9px] h-4 px-1.5 border-sky-300 bg-sky-50 text-sky-700 dark:border-sky-500/30 dark:bg-sky-500/10 dark:text-sky-400">
   348→   348→                    Review
   349→   349→                  </Badge>
   350→   350→                )}
   351→   351→              </div>
   352→   352→            </DialogHeader>
   353→   353→
   354→   354→            {step === 'form' ? (
   355→   355→              <div className="min-h-0 flex-1 overflow-y-auto -mx-1 px-1 space-y-4">
   356→   356→                {/* ── Official document header — the form reads as a real
   357→   357→                    school-issued document, not a SaaS card. ── */}
   358→   358→                <div className="rounded-lg border border-border bg-card px-3.5 py-3">
   359→   359→                  <div className="flex items-start justify-between gap-3 border-b border-dashed border-border pb-2.5">
   360→   360→                    <div className="min-w-0">
   361→   361→                      <p className="text-[13px] font-extrabold tracking-tight text-foreground leading-tight">{schGeneral.schoolName}</p>
   362→   362→                      {schGeneral.affiliation && <p className="text-[9px] text-muted-foreground mt-0.5">{schGeneral.affiliation}</p>}
   363→   363→                    </div>
   364→   364→                    <div className="text-right shrink-0">
   365→   365→                      <p className="text-[8px] uppercase tracking-[0.18em] text-muted-foreground">Form No.</p>
   366→   366→                      <p className="text-[10px] font-mono font-semibold">APPF-{app.id.slice(-8).toUpperCase()}</p>
   367→   367→                      <p className="text-[8px] uppercase tracking-[0.18em] text-muted-foreground mt-1">Session</p>

   369→   369→                    </div>

   376→   376→                <section className="rounded-lg border border-border px-3.5 py-3">

   390→   390→                {/* ── Student particulars (read-only, canonical record — §20/§21
