    26→function GradeDonut({ distribution, gradeScale, totalStudents, selectedGrade, onSelectGrade }: {
    27→  distribution: Record<string, number>
    28→  gradeScale: readonly { grade: string; minPct: number; color: string }[]
    29→  totalStudents: number
    30→  selectedGrade: string | null
    31→  onSelectGrade: (grade: string | null) => void
    32→}) {
    33→  const colorHex: Record<string, string> = {
    34→    A1: '#10b981', A2: '#34d399', B1: '#0ea5e9', B2: '#f59e0b',
    35→    C1: '#f97316', C2: '#f43f5e', E: '#e11d48',
    36→  }
    37→  const size = 180
    38→  const stroke = 28
    39→  const radius = (size - stroke) / 2
    40→  const circumference = 2 * Math.PI * radius
    41→  const cx = size / 2
    42→  const cy = size / 2
    43→
    44→  // Build segments without mutation — use reduce to accumulate offsets.
    45→  const { segments } = gradeScale.reduce<{ segments: Array<{ grade: string; count: number; fraction: number; dash: number; offset: number; color: string }>; cumulative: number }>(
    46→    (acc, g) => {
    47→      const count = distribution[g.grade] ?? 0
    48→      const fraction = totalStudents > 0 ? count / totalStudents : 0
    49→      const dash = fraction * circumference
    50→      acc.segments.push({
    51→        grade: g.grade,
    52→        count,
    53→        fraction,
    54→        dash,
    55→        offset: -acc.cumulative * circumference,
    56→        color: colorHex[g.grade] ?? '#94a3b8',
    57→      })
    58→      acc.cumulative += fraction
    59→      return acc
    60→    },
    61→    { segments: [], cumulative: 0 },
    62→  )
    63→
    64→  return (
    65→    <div className="flex items-center gap-4">
    66→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="shrink-0">
    67→        {/* Background ring */}
    68→        <circle cx={cx} cy={cy} r={radius} fill="none" stroke="hsl(var(--muted))" strokeWidth={stroke} opacity={0.3} />
    69→        {/* Segments — clickable */}
    70→        {segments.map((s) => s.count > 0 && (
    71→          <circle
    72→            key={s.grade}
    73→            cx={cx}
    74→            cy={cy}
    75→            r={radius}
    76→            fill="none"
    77→            stroke={s.color}
    78→            strokeWidth={selectedGrade === s.grade ? stroke + 6 : stroke}
    79→            strokeDasharray={`${s.dash} ${circumference - s.dash}`}
    80→            strokeDashoffset={s.offset}
    81→            transform={`rotate(-90 ${cx} ${cy})`}
    82→            strokeLinecap="butt"
    83→            className="transition-all duration-300 cursor-pointer hover:opacity-80"
    84→            opacity={selectedGrade === null || selectedGrade === s.grade ? 1 : 0.3}
    85→            onClick={() => onSelectGrade(selectedGrade === s.grade ? null : s.grade)}
    86→          >
    87→            <title>{s.grade}: {s.count} students ({Math.round(s.fraction * 100)}%) — click to filter</title>
    88→          </circle>
    89→        ))}
    90→        {/* Center text */}
    91→        <text x={cx} y={cy - 8} textAnchor="middle" className="fill-foreground font-bold pointer-events-none" style={{ fontSize: 28, fontWeight: 700 }}>
    92→          {selectedGrade ? (distribution[selectedGrade] ?? 0) : totalStudents}
    93→        </text>
    94→        <text x={cx} y={cy + 12} textAnchor="middle" className="fill-muted-foreground pointer-events-none" style={{ fontSize: 10, fontWeight: 600 }}>
    95→          {selectedGrade ? `Grade ${selectedGrade}` : 'Students'}
    96→        </text>
    97→      </svg>
    98→      {/* Legend — clickable */}
    99→      <div className="space-y-1">
   100→        {segments.filter((s) => s.count > 0).map((s) => (
   101→          <button
   102→            key={s.grade}
   103→            onClick={() => onSelectGrade(selectedGrade === s.grade ? null : s.grade)}
   104→            className={cn(
   105→              'flex items-center gap-2 text-[10px] rounded-md px-1.5 py-0.5 transition-colors w-full text-left',
   106→              selectedGrade === s.grade ? 'bg-primary/10 ring-1 ring-primary/30' : 'hover:bg-muted/40',
   107→            )}
   108→          >
   109→            <span className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ backgroundColor: s.color }} />
   110→            <span className="font-semibold w-6">{s.grade}</span>
   111→            <span className="text-muted-foreground tabular-nums">{s.count}</span>
   112→            <span className="text-muted-foreground/60 tabular-nums">({Math.round(s.fraction * 100)}%)</span>
   113→          </button>
   114→        ))}
   115→        {segments.every((s) => s.count === 0) && (
   116→          <p className="text-[10px] text-muted-foreground">No data</p>
   117→        )}
   118→        {selectedGrade && (
   119→          <button
   120→            onClick={() => onSelectGrade(null)}
   121→            className="flex items-center gap-1 text-[9px] text-primary hover:underline mt-1"
   122→          >
   123→            <RotateCcw className="h-2.5 w-2.5" /> Clear filter
   124→          </button>
   125→        )}
   126→      </div>
   127→    </div>
   128→  )
   129→}

   342→/** Compute a student's rank from the marks data. */
   343→function studentPerformance_rank(exam: ExamDTO, allMarks: any[], studentId: string): number {
   344→  const studentIds = new Set(allMarks.map((m) => m.studentId))
   345→  const pcts: Array<{ id: string; pct: number }> = []
   346→  for (const sid of studentIds) {
   347→    const marks = allMarks.filter((m) => m.studentId === sid)
   348→    let totalObtained = 0
   349→    let totalMax = 0
   350→    for (const subj of exam.subjects.filter((s: any) => s.classId === marks[0]?.classId)) {
   351→      const mark = marks.find((m) => m.subjectId === subj.subjectId)
   352→      if (!mark) continue
   353→      if (mark.status === 'ABSENT' || mark.marksObtained === null) {
   354→        totalMax += subj.maxMarks
   355→        continue
   356→      }
   357→      totalObtained += mark.marksObtained
   358→      totalMax += subj.maxMarks
   359→    }
   360→    const pct = totalMax > 0 ? (totalObtained / totalMax) * 100 : 0
   361→    pcts.push({ id: sid, pct })
   362→  }
   363→  pcts.sort((a, b) => b.pct - a.pct)
   364→  const idx = pcts.findIndex((p) => p.id === studentId)
   365→  return idx + 1
   366→}

   368→// ─── Subject Drill-Down Modal — student-wise marks for one subject ─────
   369→
   370→function SubjectDrillDownModal({ exam, classId, subjectId, subjectName, className, allMarks, onClose }: {
   371→  exam: ExamDTO
   372→  classId: string
   373→  subjectId: string
   374→  subjectName: string
   375→  className: string
   376→  allMarks: any[]
   377→  onClose: () => void
   378→}) {
   379→  const subjectConfig = exam.subjects.find((s: any) => s.classId === classId && s.subjectId === subjectId)
   380→  const maxMarks = subjectConfig?.maxMarks ?? 100
   381→  const passMarks = subjectConfig?.passMarks ?? 33
   382→
   383→  const studentMarks = useMemo(() => {
   384→    return allMarks
   385→      .filter((m) => m.classId === classId && m.subjectId === subjectId)
   386→      .sort((a, b) => (b.marksObtained ?? -1) - (a.marksObtained ?? -1))
   387→  }, [allMarks, classId, subjectId])
   388→
   389→  const stats = useMemo(() => {
   390→    const entered = studentMarks.filter((m) => m.marksObtained !== null && m.status !== 'ABSENT')
   391→    const values = entered.map((m) => m.marksObtained!)
   392→    const total = studentMarks.length
   393→    const present = entered.length
   394→    const absent = studentMarks.filter((m) => m.status === 'ABSENT').length
   395→    const avg = values.length > 0 ? Math.round((values.reduce((a, b) => a + b, 0) / values.length) * 10) / 10 : 0
   396→    const highest = values.length > 0 ? Math.max(...values) : 0
   397→    const lowest = values.length > 0 ? Math.min(...values) : 0
   398→    const passed = values.filter((v) => v >= passMarks).length
   399→    const failed = values.filter((v) => v < passMarks).length
   400→    const passRate = present > 0 ? Math.round((passed / present) * 100) : 0
   401→    return { total, present, absent, avg, highest, lowest, passed, failed, passRate }
   402→  }, [studentMarks, passMarks])
   403→
   404→  return (
   405→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm p-4" onClick={onClose}>
   406→      <div
   407→        className="bg-card border border-border rounded-xl shadow-2xl max-w-2xl w-full max-h-[85vh] overflow-hidden flex flex-col"
   408→        onClick={(e) => e.stopPropagation()}
   409→      >
   410→        {/* Header */}
   411→        <div className="flex items-center justify-between gap-3 px-5 py-4 border-b border-border bg-gradient-to-r from-sky-500/5 to-transparent">
   412→          <div className="flex items-center gap-3 min-w-0">
   413→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-sky-500/10 text-sky-600 dark:text-sky-400">
   414→              <BookOpen className="h-5 w-5" />
   415→            </div>
   416→            <div className="min-w-0">
   417→              <h3 className="text-sm font-bold truncate">{subjectName}</h3>
   418→              <p className="text-[10px] text-muted-foreground">
   419→                {className} · Max {maxMarks} · Pass {passMarks}
   420→              </p>
   421→            </div>
   422→          </div>
   423→          <div className="flex items-center gap-2 shrink-0">
   424→            <div className="text-right">
   425→              <p className="text-lg font-bold tabular-nums">{stats.avg}</p>
   426→              <p className="text-[9px] text-muted-foreground">Average</p>
   427→            </div>
   428→            <button
   429→              onClick={onClose}
   430→              className="flex h-7 w-7 items-center justify-center rounded-md text-muted-foreground hover:bg-muted hover:text-foreground transition-colors"
   431→            >
   432→              <X className="h-4 w-4" />

   526→  const [filterClass, setFilterClass] = useState('all')
   527→  const [filterSubject, setFilterSubject] = useState('all')
   528→  const [selectedStudentId, setSelectedStudentId] = useState<string | null>(null)
   529→  const [selectedGrade, setSelectedGrade] = useState<string | null>(null)
   530→  const [selectedSubjectPaper, setSelectedSubjectPaper] = useState<{ classId: string; subjectId: string; subjectName: string; className: string } | null>(null)
   531→  const storeMarks = useMockMarksStore((s) => s.marks)
   532→  const allMarks = useMemo(() => storeMarks.filter((m) => m.examId === exam.id), [storeMarks, exam.id])
   533→
   534→  // Use the central grading configuration (DEFAULT_GRADE_BOUNDARIES from types.ts).
   535→  // In production this would come from the school's GradeScale table.
   536→  const gradeScale = DEFAULT_GRADE_BOUNDARIES
   537→
   538→  // Compute per-student subject percentages → grades.
   539→  const gradeData = useMemo(() => {
   540→    const distribution: Record<string, number> = {}
   541→    for (const g of gradeScale) distribution[g.grade] = 0
   542→    let totalStudents = 0
   543→    let passedCount = 0
   544→    let failedCount = 0
   545→    let absentCount = 0
   546→    let highestPct = 0
   547→    let lowestPct = 100
   548→    let pctSum = 0
   549→    let pctCount = 0
   550→
   551→    // Group marks by student (within filter scope).
   552→    const studentIds = new Set(allMarks.map((m) => m.studentId))
   553→    for (const studentId of studentIds) {
   554→      const studentMarks = allMarks.filter((m) => m.studentId === studentId)
   555→      // Apply class filter.
   556→      if (filterClass !== 'all') {
   557→        const studentClassId = studentMarks[0]?.classId
   558→        if (studentClassId !== filterClass) continue
   559→      }
   560→      let totalObtained = 0
   561→      let totalMax = 0
   562→      let isAbsentInAll = true
   563→      for (const subj of exam.subjects.filter((s: any) => s.classId === studentMarks[0]?.classId)) {
   564→        // Apply subject filter for per-subject distribution.
   565→        if (filterSubject !== 'all' && subj.subjectId !== filterSubject) continue
   566→        const mark = studentMarks.find((m) => m.subjectId === subj.subjectId)
   567→        if (mark?.status === 'ABSENT' || mark?.marksObtained === null) {
   568→          // Absent or missing — don't add to totals.
   569→          continue
   570→        }
   571→        isAbsentInAll = false
   572→        totalObtained += mark!.marksObtained ?? 0
   573→        totalMax += subj.maxMarks
   574→      }
   575→      if (totalMax === 0) continue
   576→      totalStudents++
   577→      const pct = Math.round((totalObtained / totalMax) * 100 * 100) / 100
   578→      if (isAbsentInAll) {
   579→        absentCount++
   580→        continue
   581→      }
   582→      highestPct = Math.max(highestPct, pct)
   583→      lowestPct = Math.min(lowestPct, pct)
   584→      pctSum += pct
   585→      pctCount++
   586→      const { grade } = getGradeForPercentage(pct, [])
   587→      distribution[grade] = (distribution[grade] ?? 0) + 1
   588→      if (pct >= 33) passedCount++
   589→      else failedCount++
   590→    }
   591→    const avgPct = pctCount > 0 ? Math.round((pctSum / pctCount) * 10) / 10 : 0
   592→    return { distribution, totalStudents, passedCount, failedCount, absentCount, highestPct, lowestPct, avgPct }
   593→  }, [allMarks, exam, gradeScale, filterClass, filterSubject])
   594→

   599→      if (filterClass !== 'all' && c.classId !== filterClass) continue

   601→        if (filterSubject !== 'all' && subj.subjectId !== filterSubject) continue
   602→        const marks = allMarks.filter((m) => m.classId === c.classId && m.subjectId === subj.subjectId)
   603→        const dist: Record<string, number> = {}
   604→        for (const g of gradeScale) dist[g.grade] = 0
   605→        let total = 0
   606→        for (const m of marks) {
   607→          if (m.status === 'ABSENT' || m.marksObtained === null) continue
   608→          const pct = subj.maxMarks > 0 ? (m.marksObtained / subj.maxMarks) * 100 : 0
   609→          const { grade } = getGradeForPercentage(pct, [])
   610→          dist[grade] = (dist[grade] ?? 0) + 1
   611→          total++
   612→        }
   613→        rows.push({ subjectName: subj.subjectName, className: c.className, classId: c.classId, subjectId: subj.subjectId, distribution: dist, total })
   614→      }
   615→    }
   616→    return rows

   624→    const studentIds = new Set(allMarks.map((m) => m.studentId))

   652→      const { grade } = getGradeForPercentage(percentage, [])

   654→        studentId,

   656→        rollNo: studentMarks[0]?.studentRollNo ?? null,
   657→        className: exam.classes.find((c: any) => c.classId === studentMarks[0]?.classId)?.className ?? '',

   838→                return (

   840→                  key={i}
   841→                  onClick={() => setSelectedSubjectPaper({ classId: r.classId ?? '', subjectId: r.subjectId ?? '', subjectName: r.subjectName, className: r.className })}

   847→                    <ChevronRight className="h-2.5 w-2.5 text-muted-foreground/40" />
   848→                  </td>
   849→                  <td className="px-2 py-1 text-center tabular-nums">{r.total}</td>
   850→                  {gradeScale.map((g) => {
   851→                    const count = r.distribution[g.grade] ?? 0
   852→                    const intensity = count / maxCount
   853→                    return (
   854→                      <td key={g.grade} className="px-1 py-1 text-center tabular-nums text-[9px] relative">

   857→                            'inline-flex items-center justify-center w-6 h-5 rounded text-[9px] font-medium',
   858→                            count === 0 ? 'text-muted-foreground/40' : 'text-foreground',
   859→                          )}
   860→                          style={count > 0 ? {
   861→                            backgroundColor: `hsl(var(--primary) / ${0.08 + intensity * 0.25})`,
   862→                          } : undefined}
   863→                        >
   864→                          {count === 0 ? '—' : count}
   865→                        </span>
   866→                      </td>
   867→                    )
   868→                  })}
   869→                </tr>
   870→                )
   871→              })}
   872→              {subjectComparison.length === 0 && (
   873→                <tr><td colSpan={2 + gradeScale.length} className="py-4 text-center text-muted-foreground">No data available.</td></tr>
   874→              )}
   875→            </tbody>
   876→          </table>
   877→        </div>
   878→      </CollapsibleSection>
   879→
   880→      {/* Student Performance / Toppers */}
   881→      <CollapsibleSection
   882→        title="Student Performance"
   883→        subtitle={selectedGrade ? `${filteredStudentPerformance.length} of ${studentPerformance.length} · Grade ${selectedGrade}` : `${studentPerformance.length} students · ranked`}
   884→        accent="amber"
   885→        defaultOpen={false}
   886→        actions={selectedGrade ? (
   887→          <button
   888→            onClick={() => setSelectedGrade(null)}
   889→            className="inline-flex items-center gap-0.5 px-1.5 py-0.5 rounded text-[9px] font-medium text-primary hover:bg-primary/10 transition-colors"
   890→          >
   891→            <RotateCcw className="h-2.5 w-2.5" /> Clear grade filter
   892→          </button>
   893→        ) : undefined}
   894→      >
   895→        <div className="overflow-x-auto max-h-[20rem]">
   896→          <table className="w-full text-xs">
   897→            <thead className="sticky top-0 z-10 bg-muted shadow-[0_1px_0_0_hsl(var(--border))]">
   898→              <tr>
   899→                <th className="text-center px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Rank</th>
   900→                <th className="text-left px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Student</th>
   901→                <th className="text-left px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Class</th>
   902→                <th className="text-right px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Total</th>
   903→                <th className="text-right px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">%</th>
   904→                <th className="text-center px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Grade</th>
   905→                <th className="text-center px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Failed</th>
   906→                <th className="text-center px-2 py-1.5 text-[9px] uppercase font-semibold text-muted-foreground">Result</th>
   907→              </tr>
   908→            </thead>
   909→            <tbody>
   910→              {filteredStudentPerformance.map((s) => (
   911→                <tr
   912→                  key={s.studentId}
   913→                  onClick={() => setSelectedStudentId(s.studentId)}
   914→                  className="border-t border-border/30 hover:bg-primary/5 even:bg-muted/10 transition-colors cursor-pointer"
   915→                >
   916→                  <td className="px-2 py-1.5 text-center">
   917→                    {s.rank <= 3 ? (
   918→                      <span className={cn(
   919→                        'inline-flex items-center justify-center h-5 w-5 rounded-full text-[9px] font-bold',
   920→                        s.rank === 1 ? 'bg-amber-500/20 text-amber-700 dark:text-amber-300' :
   921→                        s.rank === 2 ? 'bg-slate-400/20 text-slate-600 dark:text-slate-300' :
   922→                        'bg-orange-600/20 text-orange-700 dark:text-orange-400',
   923→                      )}>
   924→                        {s.rank}
   925→                      </span>
   926→                    ) : (
   927→                      <span className="text-[9px] text-muted-foreground tabular-nums">{s.rank}</span>
   928→                    )}
   929→                  </td>
   930→                  <td className="px-2 py-1.5 font-medium flex items-center gap-1">
   931→                    {s.studentName}
   932→                    <ChevronRight className="h-2.5 w-2.5 text-muted-foreground/40" />
   933→                  </td>
   934→                  <td className="px-2 py-1.5 text-muted-foreground">{s.className}</td>
   935→                  <td className="px-2 py-1.5 text-right tabular-nums">{s.totalObtained}/{s.totalMax}</td>
   936→                  <td className="px-2 py-1.5 text-right tabular-nums font-semibold">{s.percentage}%</td>
   937→                  <td className="px-2 py-1.5 text-center font-bold">{s.grade}</td>
   938→                  <td className="px-2 py-1.5 text-center tabular-nums">{s.subjectsFailed}</td>
   939→                  <td className="px-2 py-1.5 text-center">
   940→                    {s.passed ? (
   941→                      <span className="inline-flex items-center px-1.5 py-0.5 rounded-full text-[8px] font-semibold bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 border border-emerald-500/20">PASS</span>
   942→                    ) : (
   943→                      <span className="inline-flex items-center px-1.5 py-0.5 rounded-full text-[8px] font-semibold bg-rose-500/10 text-rose-700 dark:text-rose-300 border border-rose-500/20">FAIL</span>
   944→                    )}
   945→                  </td>
   946→                </tr>
   947→              ))}
   948→              {filteredStudentPerformance.length === 0 && (
   949→                <tr><td colSpan={8} className="py-4 text-center text-muted-foreground">{selectedGrade ? `No students with grade ${selectedGrade}.` : 'No student data available.'}</td></tr>
   950→              )}
   951→            </tbody>
   952→          </table>
   953→        </div>
   954→      </CollapsibleSection>
   955→
   956→      {/* Student drill-down modal */}
   957→      {selectedStudentId && (
   958→        <StudentDrillDownModal
   959→          exam={exam}
   960→          studentId={selectedStudentId}
   961→          allMarks={allMarks}
   962→          onClose={() => setSelectedStudentId(null)}
   963→        />
   964→      )}
   965→
   966→      {/* Subject drill-down modal */}
   967→      {selectedSubjectPaper && (
   968→        <SubjectDrillDownModal
   969→          exam={exam}
   970→          classId={selectedSubjectPaper.classId}
   971→          subjectId={selectedSubjectPaper.subjectId}
   972→          subjectName={selectedSubjectPaper.subjectName}
   973→          className={selectedSubjectPaper.className}
   974→          allMarks={allMarks}
   975→          onClose={() => setSelectedSubjectPaper(null)}
   976→        />
   977→      )}
   978→
   979→      {/* Grade analysis highlights */}
   980→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   981→        <Stat label="Highest %" value={`${gradeData.highestPct}%`} />
   982→        <Stat label="Lowest %" value={`${gradeData.lowestPct}%`} />
   983→        <Stat label="Average %" value={`${gradeData.avgPct}%`} />
   984→        <Stat label="Pass %" value={`${gradeData.totalStudents > 0 ? Math.round((gradeData.passedCount / gradeData.totalStudents) * 100) : 0}%`} />
   985→      </div>
   986→    </div>
   987→  )
