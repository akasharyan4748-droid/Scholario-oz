   226→// ─── ClassSubjectAssignment (Spec §19/§20) ─────────────────────────────
   227→// Separates the SUBJECT (canonical, school-scoped) from its ASSIGNMENT to
   228→// a specific class (+ optional stream). One canonical "Hindi" subject can
   229→// be assigned to many classes. Removing an assignment does NOT delete the
   230→// canonical subject — historical records remain valid (Spec §16/§25).
   231→model ClassSubjectAssignment {
   232→  id           String    @id @default(cuid())
   233→  schoolId     String
   234→  school       School    @relation(fields: [schoolId], references: [id], onDelete: Cascade)
   235→  classId      String
   236→  class        Class     @relation(fields: [classId], references: [id], onDelete: Cascade)
   237→  subjectId    String
   238→  subject      Subject   @relation(fields: [subjectId], references: [id], onDelete: Cascade)
   239→  // Spec §34: distinguishes core/default subjects from additional/optional ones.
   240→  isCore       Boolean   @default(true)
   241→  // Spec §18/§35: an inactive assignment is excluded from new examinations
   242→  // but historical records remain valid. Archived subjects are not selectable.
   243→  isActive     Boolean   @default(true)
   244→  // Spec §33: some subjects are examinable, others are not (e.g. PE, Music).
   245→  examinable   Boolean   @default(true)
   246→  // Spec §33: deterministic ordering for display.
   247→  displayOrder Int       @default(0)
   248→  createdAt    DateTime  @default(now())
   249→  updatedAt    DateTime? @updatedAt
   250→
   251→  // Spec §12/§19: one assignment per (class, subject) — prevents duplicates.
   252→  @@unique([classId, subjectId])
   253→}

   946→// ─── Phase 9 — Enterprise Fee Architecture (DB-backed) ───────────────
   947→//
   948→// The fee-store has been in-memory + Zustand-persist (localStorage) since
   949→// takeover. These new models make the fee system truly DB-backed and
   950→// multi-tenant-safe, and let the webhook route auto-reconcile instead of
   951→// just logging. All new models are ADDITIVE — the existing Fee + Payment
   952→// models remain untouched so the legacy /api/fees route keeps working.
   953→//
   954→// Architectural mapping (store type → DB row):
   955→//   - school-settings-store.fees.feeHeads[]  → MasterFeeHead
   956→//   - fee-store.FEE_STRUCTURES[]             → FeeStructure (+ FeeHead rows)
   957→//   - fee-store.FeeStructureVersion          → FeeStructureVersion
   958→//   - fee-store.SEED_TRANSACTIONS[]          → FeeTransaction (immutable)
   959→//   - fee-store.settlements[]                → Settlement
   960→//   - fee-store.reconciliations[]            → Reconciliation
   961→//   - fee-store.webhookEvents[]              → WebhookEvent
   962→//
   963→// Settlement status lifecycle:
   964→//   pending → settled → reconciled
   965→//                     ↘ exception → (manual review) → reconciled
   966→//   failure → reversed
   967→
   968→/// School-wide master fee-head catalogue — single source of truth for
   969→/// head names, default frequencies, default amounts, categories.
   970→/// Picked from when building per-class fee structures.

   972→  id          String   @id @default(cuid())

   975→  name        String
   976→  category    String   @default("Other") // Tuition, Admission, Annual, Transport, Lab, Library, Exam, Activity, Board, Other
   977→  frequency   String   @default("Monthly") // Annual, Half-Yearly, Quarterly, Monthly, Per Term, One-Time
   978→  amount      Float    @default(0)
   979→  mandatory   Boolean  @default(true)
   980→  active      Boolean  @default(true)
   981→  description String?
   982→  sortOrder   Int      @default(0)
   983→  createdAt   DateTime @default(now())
   984→  updatedAt   DateTime @updatedAt
   985→
   986→  /// Per-structure line items bound to this catalogue entry (read model).
   987→  structures FeeHead[]
   988→
   989→  @@unique([schoolId, name])
   990→}
