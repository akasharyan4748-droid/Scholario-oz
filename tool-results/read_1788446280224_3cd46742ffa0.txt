     3→/**
     4→ * ApplyDialog — the student's Educational Tour application: fill → submit →
     5→ * pay, inside ONE dialog.
     6→ *
     7→ * Structure mirrors the printed official form so what the student fills is
     8→ * exactly what the school office files:
     9→ *   1. Official document header (school, form no., session)
    10→ *   2. Tour details (read-only — destination, dates, fee, in-charge)
    11→ *   3. Student particulars (READ-ONLY, from the school record — applicants
    12→ *      never edit their master record here; corrections go through the office)
    13→ *   4. Guardian details (read-only)
    14→ *   5. Tour-specific details (the template's editable questions — saved to
    15→ *      the submission snapshot only, never the canonical student record)
    16→ *   6. Declaration & guardian consent
    17→ *   → review step: compact REVIEW APPLICATION summary before final submit
    18→ *   → payment step: Online (instant receipt) or Cash / Pay at School
    19→ *     (recorded Under Verification until the Principal confirms).
    20→ *
    21→ * MONEY RULE: payments go through fee-store recordPayment() only, bound to
    22→ * the application's Additional Charge + applicationId — tour money is
    23→ * separate from the student's annual fees and lands in the normal ledger.
    24→ *
    25→ * Idempotency: re-submitting an already-submitted form returns
    26→ * existingSubmissionId from the store — we surface "You already applied"
    27→ * and close instead of creating a duplicate.
    28→ */
    29→

    61→type PayMode = 'UPI' | 'Cash'
    62→
    63→interface ApplyDialogProps {
    64→  open: boolean
    65→  onOpenChange: (open: boolean) => void
    66→  app: SchoolApplication | null
    67→  /** Display + canonical identity pair resolved by the module. */
    68→  identity: StudentIdentityPair | null
    69→  /** Present when fixing a 'Correction Required' submission. */
    70→  existingSubmission?: ApplicationSubmission | null
    71→  /** 'payment' skips straight to the payment step (awaiting-payment rows). */
    72→  initialStep?: 'form' | 'payment'
    73→}
    74→
    75→type Answers = Record<string, string | string[] | boolean>
    76→type Attachments = Record<string, { name: string; size: number }>
    77→type Errors = Record<string, string>
    78→
    79→const EM_DASH = ' — '
    80→
    81→function splitEmergencyContact(value: Answers[string]): { name: string; phone: string } {
    82→  if (typeof value === 'string' && value.includes(EM_DASH)) {
    83→    const [name = '', phone = ''] = value.split(EM_DASH)
    84→    return { name, phone }
    85→  }
    86→  return { name: '', phone: '' }
    87→}
    88→
    89→export function ApplyDialog({ open, onOpenChange, app, identity, existingSubmission = null, initialStep = 'form' }: ApplyDialogProps) {
    90→  const [step, setStep] = useState<'form' | 'review' | 'payment'>(initialStep === 'payment' ? 'payment' : 'form')
    91→  const [answers, setAnswers] = useState<Answers>({})
    92→  const [attachments, setAttachments] = useState<Attachments>({})
    93→  const [emergency, setEmergency] = useState<Record<string, { name: string; phone: string }>>({})
    94→  const [consentAccepted, setConsentAccepted] = useState(false)
    95→  const [signature, setSignature] = useState<SignatureValue | null>(null)
    96→  const [errors, setErrors] = useState<Errors>({})
    97→  const [submitting, setSubmitting] = useState(false)
    98→  const [paying, setPaying] = useState<PayMode | null>(null)
    99→  const [activeSubmissionId, setActiveSubmissionId] = useState<string | null>(null)
   100→
   101→  const isFixMode = !!existingSubmission
   102→  // Active school's own branding (tenant-scoped School Settings).
   103→  const schGeneral = useSchoolSettingsStore((s) => s.general)
   104→
   105→  // ── Reset / prefill whenever the dialog opens for an application ──
   106→  useEffect(() => {
   107→    if (!open || !app) return
   108→    setStep(initialStep === 'payment' ? 'payment' : 'form')
   109→    setErrors({})
   110→    setSubmitting(false)
   111→    setPaying(null)
   112→    if (existingSubmission) {
   113→      setAnswers({ ...existingSubmission.answers })
   114→      setAttachments(existingSubmission.attachments ? { ...existingSubmission.attachments } : {})
   115→      setEmergency(Object.fromEntries(
   116→        app.formFields
   117→          .filter((f) => f.type === 'emergency-contact')
   118→          .map((f) => [f.id, splitEmergencyContact(existingSubmission.answers[f.id])]),
   119→      ))
   120→      setConsentAccepted(!!existingSubmission.consentGivenAt)
   121→      setActiveSubmissionId(existingSubmission.id)
   122→    } else {
   123→      setAnswers({})
   124→      setAttachments({})
   125→      setEmergency(Object.fromEntries(
   126→        app.formFields
   127→          .filter((f) => f.type === 'emergency-contact')
   128→          .map((f) => [f.id, { name: '', phone: '' }]),
   129→      ))
   130→      setConsentAccepted(false)
   131→      setSignature(null)
   132→      setActiveSubmissionId(null)
   133→    }
   134→  }, [open, app, existingSubmission, initialStep])
   135→
   136→  const amount = app?.payment.amount ?? 0
   137→  const needsPayment = !!app && app.payment.mode !== 'None'
   138→  const isTour = !!app && (app.category === 'Tour' || app.category === 'Trip' || app.templateKey === 'educational_tour')
   139→
   140→  const closeDialog = () => {
   141→    onOpenChange(false)
   142→  }
   143→
   144→  // ── Validation (application-specific fields + digital guardian consent) ──
   145→  const validate = (): boolean => {
   146→    if (!app) return false
   147→    const errs: Errors = {}
   148→    for (const f of app.formFields) {
   149→      const v = answers[f.id]
   150→      if (f.type === 'emergency-contact') {
   151→        const { name, phone } = emergency[f.id] ?? { name: '', phone: '' }
   152→        if (f.required && (!name.trim() || !phone.trim())) {
   153→          errs[f.id] = 'Enter both a contact name and a phone number.'
   154→        }
   155→        continue
   156→      }
   157→      if (f.type === 'file') {
   158→        if (f.required && !attachments[f.id]) errs[f.id] = 'Please attach a file.'
   159→        continue
   160→      }
   161→      if (f.type === 'signature' || f.type === 'declaration') {
   162→        if (f.required && v !== true) errs[f.id] = 'Tick the acknowledgement to continue.'
   163→        continue
   164→      }
   165→      if (f.type === 'checkbox' || f.type === 'multiselect') {
   166→        if (f.required && (!Array.isArray(v) || v.length === 0)) errs[f.id] = 'Pick at least one option.'
   167→        continue
   168→      }
   169→      if (f.type === 'yesno') {
   170→        if (f.required && typeof v !== 'boolean') errs[f.id] = 'Choose Yes or No.'
   171→        continue
   172→      }
   173→      if (f.required && (typeof v !== 'string' || !v.trim())) {
   174→        errs[f.id] = 'This field is required.'
   175→      }
   176→    }
   177→    if (app.guardianConsent.required && app.guardianConsent.method === 'Digital' && !consentAccepted) {
   178→      errs.consent = 'Guardian consent is required before submitting.'
   179→    }
   180→    // PART 19 — a required digital signature is a real captured signature,
   181→    // never fabricated. Only enforced on the initial submit (fix-mode keeps
   182→    // the original signature on record).
   183→    if (
   184→      !isFixMode && app.guardianConsent.required && app.guardianConsent.method === 'Digital'
   185→      && app.guardianConsent.signatureRequired !== false
   186→      && !signatureComplete(signature)
   187→    ) {
   188→      errs.signature = 'Capture the guardian’s signature (draw or type) to continue.'
   189→    }
   190→    setErrors(errs)
   191→    if (Object.keys(errs).length > 0) {
   192→      toast.error('Please complete the highlighted fields.')
   193→      return false
   194→    }
   195→    return true
   196→  }
   197→
   198→  // ── Submit (or resubmit after corrections) — reached from the review step ──
   199→  const handleSubmit = () => {
   200→    if (!app || !identity || submitting) return
   201→    if (initialStep !== 'payment' && !validate()) {
   202→      setStep('form')
   203→      return
   204→    }
   205→
   206→    setSubmitting(true)
   207→
   208→    if (isFixMode && existingSubmission) {
   209→      const res = useApplicationsStore.getState().resubmitSubmission(
   210→        existingSubmission.id,
   211→        answers,
   212→        Object.keys(attachments).length ? attachments : undefined,
   213→        identity.canonical.name,
   214→      )
   215→      setSubmitting(false)
   216→      if (!res.success) {
   217→        toast.error(res.error ?? 'Could not resubmit.')
   218→        return
   219→      }
   220→      const updated = useApplicationsStore.getState().submissions.find((s) => s.id === existingSubmission.id)
   221→      const pay = updated ? deriveSubmissionPayment(app, updated) : null
   222→      if (needsPayment && (!pay || pay.status === 'Not Paid')) {
   223→        setActiveSubmissionId(existingSubmission.id)
   224→        setStep('payment')
   225→        toast.info('Corrections submitted — one more step: payment.')
   226→      } else {
   227→        toast.success('Corrections submitted for review.')
   228→        closeDialog()
   229→      }
   230→      return
   231→    }
   232→
   233→    const res = useApplicationsStore.getState().submitApplication({
   234→      applicationId: app.id,
   235→      student: identity.canonical,
   236→      answers,
   237→      attachments: Object.keys(attachments).length ? attachments : undefined,
   238→      consentAccepted,
   239→      signature: signatureComplete(signature) ? signature ?? undefined : undefined,
   240→      submittedByRole: 'Student',
   241→    })
   242→
   243→    if (res.success && res.existingSubmissionId) {
   244→      // Idempotent hit — the store refused to duplicate. Be honest + close.
   245→      setSubmitting(false)
   246→      toast.info('You already applied', {
   247→        description: 'A submission for this form already exists — nothing was duplicated.',
   248→      })
   249→      closeDialog()
   250→      return
   251→    }
   252→    if (!res.success || !res.submission) {
   253→      setSubmitting(false)
   254→      toast.error(res.error ?? 'Submission failed.')
   255→      return
   256→    }
   257→
   258→    if (needsPayment) {
   259→      setActiveSubmissionId(res.submission.id)
   260→      setStep('payment')
   261→      setSubmitting(false)
   262→      toast.success('Application submitted — complete the payment to finish.', {
   263→        description: formatINR(amount),
   264→      })
   265→    } else {
   266→      setSubmitting(false)
   267→      toast.success('Application submitted.', {
   268→        description: 'Track its status under My submissions.',
   269→      })
   270→      closeDialog()
   271→    }
   272→  }
   273→
   274→  // ── Payment — the ONLY way money moves (canonical fee store) ──
   275→  const runPayment = (mode: PayMode) => {
   276→    if (!app || !identity || paying) return
   277→    if (!app.payment.chargeId) return // guarded in the UI
   278→    setPaying(mode)
   279→    const res = useFeeStore.getState().recordPayment({
   280→      studentId: identity.canonical.id,
   281→      amount: app.payment.amount,
   282→      mode,
   283→      purpose: `Application: ${app.title}`,
   284→      feeHead: app.payment.feeHeadLabel || app.title,
   285→      collectedBy: 'Student Self-Service',
   286→      additionalChargeId: app.payment.chargeId,
   287→      applicationId: app.id,
   288→      ...(mode === 'UPI' ? { referenceNo: `UPI-${Date.now()}` } : {}),
   289→    })
   290→    setPaying(null)
   291→    if (!res.success || !res.transaction) {
   292→      toast.error(res.error ?? 'Payment could not be recorded.')
   293→      return
   294→    }
   295→    const receipt = res.transaction.receiptNo
   296→    addAuditEvent({
   297→      applicationId: app.id,
   298→      submissionId: activeSubmissionId ?? undefined,
   299→      ts: new Date().toISOString(),
   300→      actor: identity.canonical.name,
   301→      actorRole: 'Student',
   302→      action: 'payment.completed',
   303→      message: mode === 'Cash'
   304→        ? `Cash ${formatINR(app.payment.amount)} recorded for "${app.title}" — receipt ${receipt}, awaiting verification by the Principal.`
   305→        : `Paid ${formatINR(app.payment.amount)} online (UPI ref ${res.transaction.referenceNo ?? '—'}) for "${app.title}" — receipt ${receipt}.`,
   306→    })
   307→    if (mode === 'Cash') {
   308→      toast.info('Recorded — awaiting cash verification by the Principal', {
   309→        description: `Receipt ${receipt} · ${formatINR(app.payment.amount)} · ${app.title}`,
   310→      })
   311→    } else {
   312→      toast.success(`Payment successful — receipt ${receipt}`, {
   313→        description: `${formatINR(app.payment.amount)} paid for ${app.title}.`,
   314→      })
   315→    }
   316→    closeDialog()
   317→  }
   318→
   319→  const skipPayment = () => {
   320→    toast.info('Payment skipped for now', {
   321→      description: 'You can pay anytime from My submissions.',
   322→    })
   323→    closeDialog()
   324→  }
   325→
   326→  return (
   327→    <Dialog open={open} onOpenChange={(o) => { if (!o) closeDialog() }}>
   328→      <DialogContent className="sm:max-w-lg max-h-[88vh] overflow-hidden flex flex-col">
   329→        {app && identity && (
   330→          <>
   331→            <DialogHeader className="text-left shrink-0">
   332→              <div className="flex items-start justify-between gap-3">
   333→                <div className="min-w-0">
   334→                  <DialogTitle className="text-base leading-snug">{app.title}</DialogTitle>
   335→                  <DialogDescription className="mt-1 flex flex-wrap items-center gap-1.5 text-[11px]">
   336→                    <Badge variant="outline" className="text-[9px] h-4 px-1.5">Educational Tour</Badge>
   337→                    <span>Deadline {formatDate(app.deadline)}</span>
   338→                    {needsPayment && <span>· {formatINR(amount)}{app.payment.mode === 'Optional' ? ' (optional)' : ''}</span>}
   339→                  </DialogDescription>
   340→                </div>
   341→                {step === 'payment' && (
   342→                  <Badge variant="outline" className="shrink-0 text-[9px] h-4 px-1.5 border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-400">
   343→                    Payment
   344→                  </Badge>
   345→                )}
   346→                {step === 'review' && (
   347→                  <Badge variant="outline" className="shrink-0 text-[9px] h-4 px-1.5 border-sky-300 bg-sky-50 text-sky-700 dark:border-sky-500/30 dark:bg-sky-500/10 dark:text-sky-400">
   348→                    Review
   349→                  </Badge>
   350→                )}
   351→              </div>
   352→            </DialogHeader>
   353→
   354→            {step === 'form' ? (
   355→              <div className="min-h-0 flex-1 overflow-y-auto -mx-1 px-1 space-y-4">
   356→                {/* ── Official document header — the form reads as a real
   357→                    school-issued document, not a SaaS card. ── */}
   358→                <div className="rounded-lg border border-border bg-card px-3.5 py-3">
   359→                  <div className="flex items-start justify-between gap-3 border-b border-dashed border-border pb-2.5">
   360→                    <div className="min-w-0">
   361→                      <p className="text-[13px] font-extrabold tracking-tight text-foreground leading-tight">{schGeneral.schoolName}</p>
   362→                      {schGeneral.affiliation && <p className="text-[9px] text-muted-foreground mt-0.5">{schGeneral.affiliation}</p>}
   363→                    </div>
   364→                    <div className="text-right shrink-0">
   365→                      <p className="text-[8px] uppercase tracking-[0.18em] text-muted-foreground">Form No.</p>
   366→                      <p className="text-[10px] font-mono font-semibold">APPF-{app.id.slice(-8).toUpperCase()}</p>
   367→                      <p className="text-[8px] uppercase tracking-[0.18em] text-muted-foreground mt-1">Session</p>
   368→                      <p className="text-[10px] font-semibold">{app.academicYear}</p>
   369→                    </div>
   370→                  </div>
   371→                  <p className="mt-2 text-[8.5px] uppercase tracking-[0.2em] text-muted-foreground">Official {formPurposeOf(app)} Form · {app.category}</p>
   372→                  <p className="text-sm font-bold leading-tight mt-0.5">{app.title}</p>
   373→                </div>
   374→
   375→                {/* ── Activity details (read-only) ── */}
   376→                <section className="rounded-lg border border-border px-3.5 py-3">
   377→                  <p className="text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">1. {isTour ? 'Tour' : 'Activity'} details</p>
   378→                  <div className="mt-2 grid grid-cols-2 gap-x-4 gap-y-1.5 text-[11px]">
   379→                    {app.destination && <Particular icon={<Bus className="h-3 w-3" />} label="Destination" value={app.destination} />}
   380→                    <Particular icon={<CalendarDays className="h-3 w-3" />} label={isTour ? 'Tour date' : 'Event date'} value={app.eventDate ? formatDate(app.eventDate) : 'To be announced'} />
   381→                    <Particular icon={<CalendarDays className="h-3 w-3" />} label="Last date to apply" value={formatDate(app.deadline)} />
   382→                    <Particular icon={<Landmark className="h-3 w-3" />} label="Fee per student" value={needsPayment ? `${formatINR(amount)}${app.payment.mode === 'Optional' ? ' (optional)' : ''}` : 'Free'} />
   383→                    <Particular icon={<Users className="h-3 w-3" />} label="Teacher in-charge" value={app.inChargeName ?? 'To be assigned'} />
   384→                  </div>
   385→                  {app.description && (
   386→                    <p className="mt-2 text-[10.5px] leading-relaxed text-muted-foreground border-l-2 border-border pl-2.5">{app.description}</p>
   387→                  )}
   388→                </section>
   389→
   390→                {/* ── Student particulars (read-only, canonical record — §20/§21
   391→                    minimal set: only what the form needs; DOB/blood group stay
   392→                    in the school record) ── */}
   393→                <section className="rounded-lg border border-border bg-muted/25 px-3.5 py-3">
   394→                  <p className="flex items-center gap-1.5 text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">
   395→                    <User className="h-3 w-3" /> 2. Student particulars
   396→                  </p>
   397→                  <div className="mt-2 grid grid-cols-2 gap-x-4 gap-y-1.5 text-[11px]">
   398→                    <Particular label="Name" value={identity.canonical.name} />
   399→                    <Particular label="Admission no." value={identity.canonical.admissionNo} mono />
   400→                    <Particular label="Class / Section" value={`${identity.canonical.className} — ${identity.canonical.section}`} />
   401→                    <Particular label="Roll no." value={identity.canonical.rollNo} />
   402→                  </div>
   403→                  <p className="mt-2 text-[9.5px] text-muted-foreground">Taken from the school record — corrections go through the office.</p>
   404→                </section>
   405→
   406→                {/* ── Guardian details (read-only) ── */}
   407→                <section className="rounded-lg border border-border bg-muted/25 px-3.5 py-3">
   408→                  <p className="flex items-center gap-1.5 text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">
   409→                    <Users className="h-3 w-3" /> 3. Parent / guardian details
   410→                  </p>
   411→                  <div className="mt-2 grid grid-cols-2 gap-x-4 gap-y-1.5 text-[11px]">
   412→                    <Particular label="Guardian name" value={identity.canonical.guardianName} />
   413→                    <Particular label="Guardian phone" value={identity.canonical.guardianPhone} />
   414→                  </div>
   415→                </section>
   416→
   417→                {/* ── Form questions (editable), grouped by section ── */}
   418→                {(() => {
   419→                  const groups = new Map<string, typeof app.formFields>()
   420→                  for (const f of app.formFields) {
   421→                    const key = f.section ?? 'Additional Questions'
   422→                    const arr = groups.get(key) ?? []
   423→                    arr.push(f)
   424→                    groups.set(key, arr)
   425→                  }
   426→                  let sectionNo = 3
   427→                  return Array.from(groups.entries()).map(([section, fields]) => {
   428→                    sectionNo += 1
   429→                    return (
   430→                      <section key={section} className="space-y-3.5">
   431→                        <p className="text-[9px] font-bold uppercase tracking-[0.14em] text-muted-foreground">{sectionNo}. {section}</p>
   432→                        {fields.map((f) => (
   433→                          <FieldRenderer
   434→                            key={f.id}
   435→                            field={f}
   436→                            value={answers[f.id]}
   437→                            attachment={attachments[f.id]}
   438→                            emergency={emergency[f.id] ?? { name: '', phone: '' }}
   439→                            error={errors[f.id]}
   440→                            onAnswer={(v) => {
   441→                              setAnswers((prev) => ({ ...prev, [f.id]: v }))
   442→                              setErrors((prev) => { if (!prev[f.id]) return prev; const next = { ...prev }; delete next[f.id]; return next })
   443→                            }}
   444→                            onEmergency={(v) => {
   445→                              setEmergency((prev) => ({ ...prev, [f.id]: v }))
   446→                              setAnswers((prev) => ({ ...prev, [f.id]: [v.name.trim(), v.phone.trim()].filter(Boolean).join(EM_DASH) }))
   447→                              setErrors((prev) => { if (!prev[f.id]) return prev; const next = { ...prev }; delete next[f.id]; return next })
   448→                            }}
   449→                            onAttachment={(file) => {
   450→                              setAttachments((prev) => {
   451→                                const next = { ...prev }
   452→                                if (file) next[f.id] = { name: file.name, size: file.size }
   453→                                else delete next[f.id]
   454→                                return next
   455→                              })
   456→                              setErrors((prev) => { if (!prev[f.id]) return prev; const next = { ...prev }; delete next[f.id]; return next })
   457→                            }}
   458→                          />
   459→                        ))}
   460→                      </section>
   461→                    )
   462→                  })
   463→                })()}
   464→
   465→                {/* ── Declaration & consent ── */}
   466→                <section className="rounded-lg border border-border px-3.5 py-3">
