    40→/** Teachers the student may request a meeting with (class teacher + subject
    41→ *  teachers of THEIR class — never a school-wide directory, spec §30). */
    42→function allowedTeachers(classId: string, section: string): { id: string; name: string; subject: string }[] {
    43→  const cls = useStudentsStore.getState().classes.find((c) => c.id === classId)
    44→  if (!cls) return []
    45→  const sec = cls.sections.find((s) => s.name === section)
    46→  const roster: { id: string; name: string; subject: string }[] = []
    47→  const seen = new Set<string>()
    48→  const push = (id: string | undefined, subject: string) => {
    49→    if (!id || seen.has(id)) return
    50→    const t = teachers.find((x) => x.id === id)
    51→    if (!t) return
    52→    seen.add(id)
    53→    roster.push({ id: t.id, name: t.name, subject })
    54→  }
    55→  push(sec?.classTeacherId ?? cls.classTeacherId, 'Class Teacher')
    56→  for (const [subId, tid] of Object.entries(cls.subjectTeachers)) {
    57→    const name = cls.subjects.find((n) => n === subId) ?? subId.replace('sub-', '')
    58→    push(tid, name.charAt(0).toUpperCase() + name.slice(1))
    59→  }
    60→  return roster
    61→}

   292→function ActionCard({ icon, title, desc, stat, onClick, accent }: {

   395→function PostUpdateDialog({ positions, onClose }: { positions: StudentPosition[]; onClose: () => void }) {
   396→  const postClassUpdate = useClassResponsibilityStore((s) => s.postClassUpdate)
   397→  const [title, setTitle] = useState('')
   398→  const [body, setBody] = useState('')
   399→  const [category, setCategory] = useState<ClassUpdateCategory>('notice')
   400→
   401→  const submit = () => {
   402→    if (!title.trim() || !body.trim()) return
   403→    const result = postClassUpdate({ actorStudentId: DEMO_STUDENT_ID, title: title.trim(), body: body.trim(), category })
   404→    if (result.ok) {
   405→      toast.success('Update submitted for review', { description: 'Your class teacher will review it before it appears on the class notice board.' })
   406→      onClose()
   407→    } else {
   408→      toast.error('Could not post update', { description: result.error })
   409→    }
   410→  }
   411→
   412→  return (
   413→    <DialogShell
   414→      title="Post a class update"
   415→      subtitle={`${POSITION_DEFS[positions[0]?.key]?.title} · reviewed by your teacher`}
   416→      onClose={onClose}
   417→      footer={
   418→        <>
   419→          <Button variant="outline" size="sm" onClick={onClose}>Cancel</Button>
   420→          <Button size="sm" disabled={!title.trim() || !body.trim()} onClick={submit}>
   421→            <Send className="h-3.5 w-3.5" /> Submit for review
   422→          </Button>
   423→        </>
   424→      }
