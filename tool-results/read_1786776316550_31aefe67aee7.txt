    97→    97→function toExamDTO(e: any): ExamDTO {
    98→    98→  const marks = e.marks ?? []
    99→    99→  const total = marks.length
   100→   100→  const entered = marks.filter((m: any) => m.marksObtained !== null || m.status !== 'PRESENT').length
   101→   101→  const locked = marks.filter((m: any) => m.workflowStatus === 'LOCKED').length
   102→   102→  const submitted = marks.filter((m: any) => m.workflowStatus === 'SUBMITTED').length
   103→   103→  const verified = marks.filter((m: any) => m.workflowStatus === 'VERIFIED').length
   104→   104→  return {
   105→   105→    id: e.id,
   106→   106→    schoolId: e.schoolId,
   107→   107→    name: e.name,
   108→   108→    type: e.type ?? 'Unit Test',
   109→   109→    session: e.session ?? '2025-2026',
   110→   110→    term: e.term ?? null,
   111→   111→    status: e.status ?? 'Draft',
   112→   112→    resultStatus: e.resultStatus ?? 'Not Started',
   113→   113→    passPercentage: e.passPercentage ?? 33,
   114→   114→    startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : null,
   115→   115→    endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : null,
   116→   116→    declaredAt: e.declaredAt ? e.declaredAt.toISOString() : null,
   117→   117→    declaredBy: e.declaredBy,
   118→   118→    createdBy: e.createdBy,
   119→   119→    createdAt: e.createdAt.toISOString(),
   120→   120→    updatedAt: e.updatedAt.toISOString(),
   121→   121→    classes: (e.examClasses ?? []).map(toExamClassDTO),
   122→   122→    subjects: (e.examSubjects ?? []).map(toSubjectConfigDTO),
   123→   123→    schedule: (e.scheduleItems ?? []).map(toScheduleDTO),
   124→   124→    markSummary: {
   125→   125→      total,
   126→   126→      entered,
   127→   127→      locked,
   128→   128→      submitted,
   129→   129→      verified,
   130→   130→      pct: total > 0 ? Math.round((entered / total) * 100) : 0,
   131→   131→    },
   132→   132→  }
   133→   133→}

   280→   280→export async function createExam(
   281→   281→  schoolId: string,
   282→   282→  user: AuthUserLike | null,
   283→   283→  input: CreateExamInput
   284→   284→): Promise<ExamDTO> {
   285→   285→  if (!input.name?.trim()) throw new Error('Examination name is required')
   286→   286→  if (input.classIds.length === 0) throw new Error('At least one class is required')
   287→   287→
   288→   288→  // Validate classes belong to school
   289→   289→  const validClasses = await db.class.findMany({
   290→   290→    where: { id: { in: input.classIds }, schoolId },
   291→   291→    select: { id: true, name: true },
   292→   292→  })
   293→   293→  if (validClasses.length !== input.classIds.length) {
   294→   294→    throw new Error('One or more classes not found in this school')
   295→   295→  }
   296→   296→
   297→   297→  // Validate subjects exist
   298→   298→  for (const classId of input.classIds) {
   299→   299→    const subjects = input.subjectsByClass[classId] ?? []
   300→   300→    if (subjects.length === 0) throw new Error(`No subjects selected for class ${validClasses.find((c) => c.id === classId)?.name}`)
   301→   301→    const validSubjects = await db.subject.findMany({
   302→   302→      where: { id: { in: subjects.map((s) => s.subjectId) }, classId },
   303→   303→      select: { id: true },
   304→   304→    })
   305→   305→    if (validSubjects.length !== subjects.length) {
   306→   306→      throw new Error('One or more subjects not found for the selected class')
   307→   307→    }
   308→   308→  }
   309→   309→
   310→   310→  const exam = await db.exam.create({
   311→   311→    data: {
   312→   312→      schoolId,
   313→   313→      name: input.name.trim(),
   314→   314→      type: input.type,
   315→   315→      session: input.session ?? '2025-2026',
   316→   316→      startDate: input.startDate ? new Date(input.startDate) : null,
   317→   317→      endDate: input.endDate ? new Date(input.endDate) : null,
   318→   318→      status: 'Draft',
   319→   319→      resultStatus: 'Not Started',
   320→   320→      passPercentage: input.passPercentage ?? 33,
   321→   321→      createdBy: user?.id ?? null,
   322→   322→      examClasses: {
   323→   323→        create: input.classIds.map((classId) => ({ classId })),
   324→   324→      },
   325→   325→      examSubjects: {
   326→   326→        create: input.classIds.flatMap((classId) =>
   327→   327→          (input.subjectsByClass[classId] ?? []).map((s, idx) => ({
   328→   328→            classId,
   329→   329→            subjectId: s.subjectId,
   330→   330→            maxMarks: s.maxMarks ?? 100,
   331→   331→            passMarks: s.passMarks ?? 33,
   332→   332→            theoryMarks: s.theoryMarks ?? s.maxMarks ?? 100,
   333→   333→            practicalMarks: s.practicalMarks ?? 0,
   334→   334→            sortOrder: idx,
   335→   335→          }))
   336→   336→        ),
   337→   337→      },
   338→   338→    },
   339→   339→    include: EXAM_INCLUDE,
   340→   340→  })
   341→   341→
   342→   342→  // Create schedule items if provided
   343→   343→  if (input.schedule && input.schedule.length > 0) {
   344→   344→    for (const item of input.schedule) {
   345→   345→      await db.examScheduleItem.create({
   346→   346→        data: {
   347→   347→          examId: exam.id,
   348→   348→          classId: item.classId,
   349→   349→          subjectId: item.subjectId,
   350→   350→          date: new Date(item.date),
   351→   351→          startTime: item.startTime,
   352→   352→          endTime: item.endTime,
   353→   353→          room: item.room ?? null,
   354→   354→          invigilatorName: item.invigilatorName ?? null,
   355→   355→        },
   356→   356→      })
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  // Auto-create empty ExamMark rows for every student in every class+subject
   361→   361→  for (const classId of input.classIds) {
   362→   362→    const students = await db.student.findMany({ where: { classId, schoolId }, select: { id: true } })
   363→   363→    const subjects = input.subjectsByClass[classId] ?? []
   364→   364→    for (const student of students) {
   365→   365→      for (const subject of subjects) {
   366→   366→        await db.examMark.upsert({
   367→   367→          where: {
   368→   368→            examId_classId_subjectId_studentId: {
   369→   369→              examId: exam.id,
   370→   370→              classId,
   371→   371→              subjectId: subject.subjectId,
   372→   372→              studentId: student.id,
   373→   373→            },
   374→   374→          },
   375→   375→          create: {
   376→   376→            examId: exam.id,
   377→   377→            classId,
   378→   378→            subjectId: subject.subjectId,
   379→   379→            studentId: student.id,
   380→   380→            marksObtained: null,
   381→   381→            status: 'PRESENT',
   382→   382→            workflowStatus: 'DRAFT',
   383→   383→          },
   384→   384→          update: {},
   385→   385→        })
   386→   386→      }
   387→   387→    }
   388→   388→  }
   389→   389→
   390→   390→  await audit(exam.id, user, 'EXAM_CREATED', 'EXAM', exam.id, null, { name: exam.name, type: exam.type })
   391→   391→
   392→   392→  const fresh = await db.exam.findUnique({ where: { id: exam.id }, include: EXAM_INCLUDE })
   393→   393→  return toExamDTO(fresh!)
   394→   394→}

   398→   398→export async function updateExam(
   399→   399→  examId: string,
   400→   400→  schoolId: string,
   401→   401→  user: AuthUserLike | null,
   402→   402→  updates: { name?: string; type?: string; session?: string; startDate?: string; endDate?: string; status?: string; passPercentage?: number }
   403→   403→): Promise<ExamDTO> {
   404→   404→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   405→   405→  if (!exam) throw new Error('Exam not found')
   406→   406→  const oldSnapshot = { name: exam.name, type: exam.type, status: exam.status, resultStatus: exam.resultStatus }
   407→   407→  const updated = await db.exam.update({
   408→   408→    where: { id: examId },
   409→   409→    data: {
   410→   410→      ...(updates.name !== undefined ? { name: updates.name } : {}),
   411→   411→      ...(updates.type !== undefined ? { type: updates.type } : {}),
   412→   412→      ...(updates.session !== undefined ? { session: updates.session } : {}),
   413→   413→      ...(updates.startDate !== undefined ? { startDate: updates.startDate ? new Date(updates.startDate) : null } : {}),
   414→   414→      ...(updates.endDate !== undefined ? { endDate: updates.endDate ? new Date(updates.endDate) : null } : {}),
   415→   415→      ...(updates.status !== undefined ? { status: updates.status } : {}),
   416→   416→      ...(updates.passPercentage !== undefined ? { passPercentage: updates.passPercentage } : {}),
   417→   417→    },
   418→   418→    include: EXAM_INCLUDE,
   419→   419→  })
   420→   420→  await audit(examId, user, 'EXAM_UPDATED', 'EXAM', examId, oldSnapshot, {
   421→   421→    name: updates.name,
   422→   422→    type: updates.type,
   423→   423→    status: updates.status,
   424→   424→  })
   425→   425→  return toExamDTO(updated)
   426→   426→}

   436→   436→
   437→   437→export async function addScheduleItem(
   438→   438→  examId: string,
   439→   439→  schoolId: string,
   440→   440→  user: AuthUserLike | null,
   441→   441→  data: {
   442→   442→    classId: string
   443→   443→    subjectId: string
   444→   444→    date: string
   445→   445→    startTime: string
   446→   446→    endTime: string
   447→   447→    room?: string
   448→   448→    invigilatorName?: string
   449→   449→  }
   450→   450→): Promise<ScheduleItemDTO> {
   451→   451→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   452→   452→  if (!exam) throw new Error('Exam not found')
   453→   453→  // Conflict detection — same room+date+time OR same class+date+time OR same invigilator+date+time
   454→   454→  const conflicts = await db.examScheduleItem.findMany({
   455→   455→    where: {
   456→   456→      examId,
   457→   457→      OR: [
   458→   458→        { classId: data.classId, date: new Date(data.date), startTime: data.startTime },

   533→   533→export async function setMark(
   534→   534→  examId: string,
   535→   535→  schoolId: string,
   536→   536→  user: AuthUserLike | null,
   537→   537→  input: SetMarkInput
   538→   538→): Promise<ExamMarkDTO> {
   539→   539→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   540→   540→  if (!exam) throw new Error('Exam not found')
   541→   541→  if (exam.resultStatus === 'Result Declared') {
   542→   542→    throw new Error('Cannot modify marks after results are declared')
   543→   543→  }
   544→   544→
   545→   545→  // Validate marks against max
   546→   546→  const subjectConfig = await db.examSubjectConfig.findFirst({
   547→   547→    where: { examId, classId: input.classId, subjectId: input.subjectId },
   548→   548→  })
   549→   549→  if (!subjectConfig) throw new Error('Subject not configured for this exam/class')
   550→   550→
   551→   551→  const max = subjectConfig.maxMarks
   552→   552→  if (input.status === 'PRESENT' && input.marksObtained !== null) {
   553→   553→    if (input.marksObtained < 0) throw new Error('Marks cannot be negative')
   554→   554→    if (input.marksObtained > max) throw new Error(`Marks cannot exceed maximum (${max})`)
   555→   555→  }
   556→   556→  if (input.status !== 'PRESENT' && input.marksObtained !== null && input.marksObtained !== 0) {
   557→   557→    // For ABSENT/MEDICAL/EXEMPTED we typically store null; coerce
   558→   558→    input.marksObtained = null
   559→   559→  }
   560→   560→
   561→   561→  const existing = await db.examMark.findUnique({
   562→   562→    where: {
   563→   563→      examId_classId_subjectId_studentId: {
   564→   564→        examId,
   565→   565→        classId: input.classId,
   566→   566→        subjectId: input.subjectId,
   567→   567→        studentId: input.studentId,
   568→   568→      },
   569→   569→    },
   570→   570→  })
   571→   571→
   572→   572→  // If mark is already LOCKED, cannot modify without override flow
   573→   573→  if (existing?.workflowStatus === 'LOCKED') {
   574→   574→    throw new Error('Mark is locked. Use the unlock/reopen workflow to modify.')
   575→   575→  }
   576→   576→  if (existing?.workflowStatus === 'VERIFIED' && user?.role !== 'PRINCIPAL' && user?.role !== 'MANAGEMENT') {
   577→   577→    throw new Error('Verified marks can only be modified by Principal')
   578→   578→  }
   579→   579→
   580→   580→  const oldValue = existing ? {
   581→   581→    marksObtained: existing.marksObtained,
   582→   582→    status: existing.status,
   583→   583→  } : null
   584→   584→
   585→   585→  // Preserve workflow status if already SUBMITTED/VERIFIED — but updates after submission
   586→   586→  // remain allowed for teachers only if not yet LOCKED.
   587→   587→  const updated = await db.examMark.upsert({
   588→   588→    where: {
   589→   589→      examId_classId_subjectId_studentId: {
   590→   590→        examId,
   591→   591→        classId: input.classId,
   592→   592→        subjectId: input.subjectId,
   593→   593→        studentId: input.studentId,
   594→   594→      },
   595→   595→    },
   596→   596→    create: {
   597→   597→      examId,
   598→   598→      classId: input.classId,
   599→   599→      subjectId: input.subjectId,
   600→   600→      studentId: input.studentId,
   601→   601→      marksObtained: input.status === 'PRESENT' ? input.marksObtained : null,
   602→   602→      status: input.status,
   603→   603→      workflowStatus: 'DRAFT',
   604→   604→      originalMarks: input.status === 'PRESENT' ? input.marksObtained : null,
   605→   605→      remarks: input.remarks ?? null,
   606→   606→      enteredBy: user?.id ?? null,
   607→   607→      enteredAt: new Date(),
   608→   608→    },
   609→   609→    update: {
   610→   610→      marksObtained: input.status === 'PRESENT' ? input.marksObtained : null,
   611→   611→      status: input.status,
   612→   612→      remarks: input.remarks ?? existing?.remarks ?? null,
   613→   613→      enteredBy: user?.id ?? existing?.enteredBy ?? null,
   614→   614→      enteredAt: new Date(),
   615→   615→      // If reverting from SUBMITTED to DRAFT (any edit), allow it
   616→   616→      ...(existing?.workflowStatus === 'SUBMITTED' ? { workflowStatus: 'DRAFT' as const } : {}),
   617→   617→    },
   618→   618→    include: { student: { include: { user: { select: { name: true } } } } },
   619→   619→  })
   620→   620→
   621→   621→  // Update exam resultStatus to "Marks Entry" if it was "Not Started"
   622→   622→  if (exam.resultStatus === 'Not Started') {
   623→   623→    await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Marks Entry' } })
   624→   624→  }
   625→   625→
   626→   626→  await audit(examId, user, 'MARK_ENTERED', 'MARK', updated.id, oldValue, {
   627→   627→    marksObtained: updated.marksObtained,
   628→   628→    status: updated.status,
   629→   629→  })
   630→   630→  return toMarkDTO(updated)
   631→   631→}

   634→   634→  examId: string,
   635→   635→  schoolId: string,

   652→   652→export async function submitMarks(
   653→   653→  examId: string,
   654→   654→  schoolId: string,
   655→   655→  user: AuthUserLike | null,
   656→   656→  filter: { classId?: string; subjectId?: string }
   657→   657→): Promise<{ submitted: number }> {
   658→   658→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   659→   659→  if (!exam) throw new Error('Exam not found')
   660→   660→  if (exam.resultStatus === 'Result Declared') {
   661→   661→    throw new Error('Cannot modify marks after results are declared')
   662→   662→  }
   663→   663→  const result = await db.examMark.updateMany({
   664→   664→    where: {
   665→   665→      examId,
   666→   666→      ...(filter.classId ? { classId: filter.classId } : {}),
   667→   667→      ...(filter.subjectId ? { subjectId: filter.subjectId } : {}),
   668→   668→      workflowStatus: 'DRAFT',
   669→   669→      // Must have either marks entered OR a non-PRESENT status (absent/medical/exempted)
   670→   670→      OR: [
   671→   671→        { marksObtained: { not: null } },
   672→   672→        { status: { not: 'PRESENT' } },
   673→   673→      ],
   674→   674→    },
   675→   675→    data: {
   676→   676→      workflowStatus: 'SUBMITTED',
   677→   677→      enteredBy: user?.id ?? null,
   678→   678→    },
   679→   679→  })
   680→   680→  // Move exam to Under Verification if all marks submitted
   681→   681→  const remainingDrafts = await db.examMark.count({ where: { examId, workflowStatus: 'DRAFT' } })
   682→   682→  if (remainingDrafts === 0) {
   683→   683→    await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Under Verification' } })
   684→   684→  } else {
   685→   685→    await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Marks Entry' } })
   686→   686→  }
   687→   687→  await audit(examId, user, 'MARK_SUBMITTED', 'MARK', null, null, filter)
   688→   688→  return { submitted: result.count }
   689→   689→}

   801→   801→    {
   802→   802→      students: studentDTOs,
   803→   803→      subjects: subjectDTOs,
   804→   804→      marks: markDTOs,
   805→   805→      passPercentage: exam.passPercentage,

   807→   807→    cls.name
   808→   808→  )
   809→   809→
   810→   810→  return { students: studentDTOs, subjects: subjectDTOs, marks: markDTOs, results, analytics }
