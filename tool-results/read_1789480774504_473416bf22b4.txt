   255→// ─── ClassSubjectAssignment (Spec §19/§20) ─────────────────────────────
   256→// Separates the SUBJECT (canonical, school-scoped) from its ASSIGNMENT to
   257→// a specific class (+ optional stream). One canonical "Hindi" subject can
   258→// be assigned to many classes. Removing an assignment does NOT delete the
   259→// canonical subject — historical records remain valid (Spec §16/§25).
   260→model ClassSubjectAssignment {
   261→  id           String    @id @default(cuid())
   262→  schoolId     String
   263→  school       School    @relation(fields: [schoolId], references: [id], onDelete: Cascade)
   264→  classId      String
   265→  class        Class     @relation(fields: [classId], references: [id], onDelete: Cascade)
   266→  subjectId    String
   267→  subject      Subject   @relation(fields: [subjectId], references: [id], onDelete: Cascade)
   268→  // Spec §34: distinguishes core/default subjects from additional/optional ones.
   269→  isCore       Boolean   @default(true)
   270→  // Spec §18/§35: an inactive assignment is excluded from new examinations
   271→  // but historical records remain valid. Archived subjects are not selectable.
   272→  isActive     Boolean   @default(true)
   273→  // Spec §33: some subjects are examinable, others are not (e.g. PE, Music).
   274→  examinable   Boolean   @default(true)
   275→  // Spec §33: deterministic ordering for display.
   276→  displayOrder Int       @default(0)
   277→  createdAt    DateTime  @default(now())
   278→  updatedAt    DateTime? @updatedAt
   279→
   280→  // Spec §12/§19: one assignment per (class, subject) — prevents duplicates.
   281→  @@unique([classId, subjectId])
   282→}

   999→// ─── Phase 9 — Enterprise Fee Architecture (DB-backed) ───────────────
  1000→//
  1001→// The fee-store has been in-memory + Zustand-persist (localStorage) since
  1002→// takeover. These new models make the fee system truly DB-backed and
  1003→// multi-tenant-safe, and let the webhook route auto-reconcile instead of
  1004→// just logging. All new models are ADDITIVE — the existing Fee + Payment
  1005→// models remain untouched so the legacy /api/fees route keeps working.
  1006→//
  1007→// Architectural mapping (store type → DB row):
  1008→//   - school-settings-store.fees.feeHeads[]  → MasterFeeHead
  1009→//   - fee-store.FEE_STRUCTURES[]             → FeeStructure (+ FeeHead rows)
  1010→//   - fee-store.FeeStructureVersion          → FeeStructureVersion
  1011→//   - fee-store.SEED_TRANSACTIONS[]          → FeeTransaction (immutable)
  1012→//   - fee-store.settlements[]                → Settlement
  1013→//   - fee-store.reconciliations[]            → Reconciliation
  1014→//   - fee-store.webhookEvents[]              → WebhookEvent
  1015→//
  1016→// Settlement status lifecycle:
  1017→//   pending → settled → reconciled
  1018→//                     ↘ exception → (manual review) → reconciled
  1019→//   failure → reversed
  1020→

  1030→  frequency   String   @default("Monthly") // Annual, Half-Yearly, Quarterly, Monthly, Per Term, One-Time
  1031→  amount      Float    @default(0)
  1032→  mandatory   Boolean  @default(true)
