   274→// ─── Academic session mapping (Indian academic year: April – March) ───
   275→
   276→/** Maps a monthly period key ('2026-08') to its academic session id
   277→ *  ('2026-27'). April–December belong to the session that starts that
   278→ *  calendar year; January–March belong to the session started a year
   279→ *  earlier. This is how the archive groups real records by session. */
   280→export function sessionOfPeriod(periodKey: string): string {
   281→  const [y, m] = periodKey.split('-').map(Number)
   282→  if (!Number.isFinite(y) || !Number.isFinite(m)) return CURRENT_SESSION.id
   283→  const start = m >= 4 ? y : y - 1
   284→  return `${start}-${String((start + 1) % 100).padStart(2, '0')}`
   285→}
   286→
   287→/** '2026-27' → '2026–27' (en dash, display label). */
   288→export function sessionLabelOf(sessionId: string): string {
   289→  return sessionId.replace('-', '–')
   290→}
   291→
   292→/** All monthly period keys of an academic session, in order. `upTo`
   293→ *  (inclusive) caps the range — the live session only counts months that
   294→ *  have actually begun, so future months never inflate payroll figures. */
   295→export function sessionPeriodRange(sessionId: string, upTo?: string): string[] {
   296→  const [a, b] = sessionId.split('-').map(Number)
   297→  // '2026-27' → start 2026, end 2027. Accept 4-digit or 2-digit parts.
   298→  const startYear = a >= 100 ? a : 2000 + a
   299→  const endYear = b >= 100 ? b : startYear + 1
   300→  const endMonthCal = 3 // March of the following calendar year
   301→  const out: string[] = []
   302→  for (let y = startYear, m = 4; ; m++) {
   303→    if (m > 12) { m = 1; y++ }
   304→    const pk = `${y}-${String(m).padStart(2, '0')}`
   305→    if (upTo && pk > upTo) break
   306→    out.push(pk)
   307→    if (y === endYear && m === endMonthCal) break
   308→    if (out.length > 24) break // safety net — never loops unbounded
   309→  }
   310→  return out
   311→}

   313→/**
   314→ * Builds a session payroll snapshot from REAL store data — the single
   315→ * source used both by the in-progress session's read-only overview and by
   316→ * `archiveSession` when freezing a completed session. Per employee, every
   317→ * month of the session (up to `upTo`) contributes net payable via the same
   318→ * helpers the Payments tab uses, so the numbers always match the running
   319→ * payroll. Payment rows are the original records — netPayable was already
   320→ * snapshotted when each payment was recorded.
   321→ */
   322→export function buildSessionPayrollSnapshot(
   323→  state: Pick<SalaryState, 'employees' | 'salaries' | 'adjustments' | 'payments'>,
   324→  sessionId: string,
   325→  upTo?: string,
   326→): SessionPayrollSnapshot {
   327→  const periods = sessionPeriodRange(sessionId, upTo)
   328→  const inSession = (pk: string) => periods.includes(pk)
   329→  const sessionPayments = state.payments.filter((p) => inSession(p.periodKey))
   330→
   331→  const records: ArchivedEmployeeRecord[] = []
   332→  for (const e of state.employees) {
   333→    const state_ = state.salaries[e.id]
   334→    const empPayments = sessionPayments.filter((p) => p.employeeId === e.id)
   335→    if (!state_ && empPayments.length === 0) continue // never on payroll this session
   336→    const joinKey = e.joiningDate?.slice(0, 7) ?? ''
   337→    let totalPayable = 0
   338→    for (const pk of periods) {
   339→      if (joinKey && pk < joinKey) continue // not employed yet that month
   340→      const payable = netPayableFor(state, e.id, pk)
   341→      if (payable > 0) totalPayable += payable
   342→    }
   343→    const totalPaid = empPayments.filter((p) => p.status === 'Confirmed').reduce((s, p) => s + p.amount, 0)
   344→    const livePayments = empPayments.filter((p) => p.status !== 'Reversed')
   345→    records.push({
   346→      employeeId: e.id,
   347→      employeeCode: e.employeeId,
   348→      name: e.name,
   349→      designation: e.designation,
   350→      department: e.department,
   351→      employmentStatus: e.status,
   352→      monthlySalary: state_?.salary.netBase ?? 0,
   353→      totalPayable,
   354→      totalPaid,
   355→      outstanding: Math.max(0, totalPayable - totalPaid),
   356→      paymentsCount: livePayments.length,
   357→    })
   358→  }
   359→  records.sort((x, y) => y.totalPayable - x.totalPayable || x.name.localeCompare(y.name))
   360→
   361→  const paid = sessionPayments.filter((p) => p.status === 'Confirmed')
   362→  return {
   363→    sessionId,
   364→    sessionLabel: sessionLabelOf(sessionId),
   365→    records,
   366→    payments: [...sessionPayments].sort((a, b) => b.date.localeCompare(a.date)),
   367→    summary: {
   368→      employees: records.length,
   369→      totalPayroll: records.reduce((s, r) => s + r.totalPayable, 0),
   370→      totalPaid: paid.reduce((s, p) => s + p.amount, 0),
   371→      totalOutstanding: records.reduce((s, r) => s + r.outstanding, 0),
   372→      paymentsCount: sessionPayments.filter((p) => p.status !== 'Reversed').length,
   373→    },
   374→  }
   375→}

   554→function buildSession(template: SalaryStructureTemplate, base: number, effectiveFrom: string, exactNet?: number): SessionSalary {
   555→  const earnings: AppliedComponent[] = [
   556→    { name: 'Basic Pay', type: 'Earning', amount: base },
   557→    ...template.components
   558→      .filter((c) => c.type === 'Earning')
   559→      .map((c) => ({ name: c.name, type: 'Earning' as const, amount: c.basis === 'Percentage' ? Math.round(base * c.value / 100) : c.value })),
   560→  ]
   561→  const deductions: AppliedComponent[] = template.components
   562→    .filter((c) => c.type === 'Deduction')
   563→    .map((c) => ({ name: c.name, type: 'Deduction' as const, amount: c.basis === 'Percentage' ? Math.round(base * c.value / 100) : c.value }))
   564→  let netBase = earnings.reduce((s, c) => s + c.amount, 0) - deductions.reduce((s, c) => s + c.amount, 0)
   565→  // Absorb rupee-level rounding into Basic Pay so the net lands exactly.
   566→  if (exactNet !== undefined && netBase !== exactNet) {
   567→    earnings[0] = { ...earnings[0], amount: earnings[0].amount + (exactNet - netBase) }
   568→    netBase = exactNet
   569→  }
   570→  return { structureId: template.id, structureName: template.name, base, netBase, earnings, deductions, effectiveFrom }
   571→}

   573→/** Places an employee on a scale for a target GROSS earnings total. */
   574→export function applyStructure(template: SalaryStructureTemplate, targetGross: number, effectiveFrom: string): SessionSalary {
   575→  const pctEarnings = template.components.filter((c) => c.type === 'Earning' && c.basis === 'Percentage')
   576→    .reduce((s, c) => s + c.value, 0) / 100
   577→  const fixedEarnings = template.components.filter((c) => c.type === 'Earning' && c.basis === 'Fixed')
   578→    .reduce((s, c) => s + c.value, 0)
   579→  const base = Math.round((targetGross - fixedEarnings) / (1 + pctEarnings))
   580→  return buildSession(template, base, effectiveFrom)
   581→}

   583→/** Places an employee on a scale for an exact target NET salary. */
   584→export function applyStructureToNet(template: SalaryStructureTemplate, targetNet: number, effectiveFrom: string): SessionSalary {
   585→  const pctEarnings = template.components.filter((c) => c.type === 'Earning' && c.basis === 'Percentage')
   586→    .reduce((s, c) => s + c.value, 0) / 100
   587→  const fixedEarnings = template.components.filter((c) => c.type === 'Earning' && c.basis === 'Fixed')
   588→    .reduce((s, c) => s + c.value, 0)
   589→  const pctDeductions = template.components.filter((c) => c.type === 'Deduction' && c.basis === 'Percentage')
   590→    .reduce((s, c) => s + c.value, 0) / 100
   591→  const fixedDeductions = template.components.filter((c) => c.type === 'Deduction' && c.basis === 'Fixed')
   592→    .reduce((s, c) => s + c.value, 0)
   593→  // net = base·(1 + pctE − pctD) + fixedE − fixedD
   594→  const denom = 1 + pctEarnings - pctDeductions
   595→  const base = Math.round((targetNet - fixedEarnings + fixedDeductions) / denom)
   596→  return buildSession(template, base, effectiveFrom, targetNet)
   597→}
