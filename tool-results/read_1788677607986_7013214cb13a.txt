     1→/**
     2→ * Messaging store — Zustand store for the Messages & Inbox module.
     3→ *
     4→ * One connected messaging system:
     5→ *   Conversations → Messages → Send/Reply → Star/Archive/Draft
     6→ *   Groups → Group Members → Group Conversation → Send to Group
     7→ *
     8→ * Recipient data comes from canonical Teachers + Students (for parents).

    11→ *
    12→ * Folders: Inbox · Starred · Sent · Groups · Drafts · Archive
    13→ * Labels: Staff · Parents · Groups · Urgent (all functional filters)
    14→ *
    15→ * State mutations:
    16→ *   - sendMessage (creates/replies to conversation; group replies use a random member name)
    17→ *   - markRead (clears unread on open)
    18→ *   - starConversation / unstarConversation
    19→ *   - archiveConversation / unarchiveConversation
    20→ *   - saveDraft / deleteDraft / sendDraft
    21→ *   - markUrgent
    22→ *   - composeNew (recipient picker)
    23→ *   - createGroup / addMember / removeMember (group management)
    24→ */

   126→/** Resolve a single member ref into a display object. Returns null if not found. */
   127→export function resolveMemberRef(ref: string): MemberDisplay | null {
   128→  if (ref.startsWith('t:')) {
   129→    const id = ref.slice(2)
   130→    const t = teachers.find((x) => x.id === id)
   131→    if (!t || t.archived) return null
   132→    return {
   133→      ref,
   134→      type: 'teacher',
   135→      name: t.name,
   136→      avatar: t.avatar,
   137→      role: `${t.designation} · ${t.department}`,
   138→    }
   139→  }
   140→  if (ref.startsWith('p:')) {
   141→    const sid = ref.slice(2)
   142→    const s = useStudentsStore.getState().students.find((x) => x.id === sid)
   143→    if (!s || s.status !== 'Active') return null
   144→    const avatar = s.fatherName.split(' ').map((n) => n[0]).slice(0, 2).join('') || 'P'
   145→    return {
   146→      ref,
   147→      type: 'parent',
   148→      name: s.fatherName,
   149→      avatar,
   150→      role: `Parent · ${s.name} (${s.className}-${s.section})`,
   151→    }
   152→  }
   153→  return null
   154→}
   155→
   156→/** Resolve a list of member refs into display objects (skips missing). */
   157→export function resolveMemberRefs(refs: string[]): MemberDisplay[] {
   158→  return refs.map(resolveMemberRef).filter((x): x is MemberDisplay => x !== null)
   159→}

   186→function formatTimeAgo(iso: string): string {
   187→  const diff = Date.now() - new Date(iso).getTime()
   188→  const min = Math.floor(diff / 60000)
   189→  if (min < 1) return 'just now'
   190→  if (min < 60) return `${min} min ago`
   191→  const hr = Math.floor(min / 60)
   192→  if (hr < 24) return `${hr} hr ago`
   193→  const day = Math.floor(hr / 24)
   194→  if (day === 1) return 'yesterday'
   195→  if (day < 7) return `${day} days ago`
   196→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short' })
   197→}
   198→
   199→/** Compact relative time for conversation rows: now · 5m · 3h · Yesterday · Tue · 12 Aug */
   200→function formatListTime(iso: string): string {
   201→  const d = new Date(iso)
   202→  const now = new Date()
   203→  const diff = now.getTime() - d.getTime()
   204→  const min = Math.floor(diff / 60000)
   205→  if (min < 1) return 'now'
   206→  if (min < 60) return `${min}m`
   207→  const hr = Math.floor(min / 60)
   208→  if (hr < 24 && d.toDateString() === now.toDateString()) return `${hr}h`
   209→  const yesterday = new Date(now)
   210→  yesterday.setDate(now.getDate() - 1)
   211→  if (d.toDateString() === yesterday.toDateString()) return 'Yesterday'
   212→  if (diff < 7 * 24 * 3600 * 1000) return d.toLocaleDateString('en-IN', { weekday: 'short' })
   213→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short' })
   214→}
   215→
   216→/** Day label for chat date dividers: Today · Yesterday · Tuesday · 12 Aug 2025 */
   217→function formatDayLabel(iso: string): string {
   218→  const d = new Date(iso)
   219→  const now = new Date()
   220→  if (d.toDateString() === now.toDateString()) return 'Today'
   221→  const yesterday = new Date(now)
   222→  yesterday.setDate(now.getDate() - 1)
   223→  if (d.toDateString() === yesterday.toDateString()) return 'Yesterday'
   224→  const sameYear = d.getFullYear() === now.getFullYear()
   225→  return d.toLocaleDateString('en-IN', {
   226→    weekday: diff7(d, now) ? 'long' : undefined,
   227→    day: '2-digit',
   228→    month: 'short',
   229→    year: sameYear ? undefined : 'numeric',
   230→  })
   231→}
   232→
   233→function diff7(a: Date, b: Date): boolean {
   234→  return Math.abs(b.getTime() - a.getTime()) < 7 * 24 * 3600 * 1000
   235→}
   236→
   237→function formatMessageTime(iso: string): string {
   238→  return new Date(iso).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: true })
   239→}

   772→  getFilteredConversations: () => {
   773→    const state = get()
   774→    let result = state.conversations
   775→
   776→    // Folder filter
   777→    switch (state.activeFolder) {
   778→      case 'inbox':
   779→        result = result.filter((c) => !c.archived)
   780→        break
   781→      case 'starred':
   782→        result = result.filter((c) => c.starred && !c.archived)
   783→        break
   784→      case 'sent':
   785→        // Conversations where the last message was sent by me
   786→        result = result.filter((c) => {
   787→          const msgs = state.messages[c.id] ?? []
   788→          return msgs.length > 0 && msgs[msgs.length - 1].sender === 'me' && !c.archived
   789→        })
   790→        break
   791→      case 'groups':
   792→        result = result.filter((c) => c.type === 'group' && !c.archived)
   793→        break
   794→      case 'drafts':
   795→        // Conversations that have an associated draft
   796→        const draftConvIds = new Set(state.drafts.filter((d) => d.conversationId).map((d) => d.conversationId!))
   797→        result = result.filter((c) => draftConvIds.has(c.id) && !c.archived)
   798→        break
   799→      case 'archive':
   800→        result = result.filter((c) => c.archived)
   801→        break
   802→    }
   803→
   804→    // Label filter
   805→    if (state.activeLabel) {
   806→      switch (state.activeLabel) {
   807→        case 'Staff': result = result.filter((c) => c.type === 'staff'); break
   808→        case 'Parents': result = result.filter((c) => c.type === 'parent'); break
   809→        case 'Groups': result = result.filter((c) => c.type === 'group'); break
   810→        case 'Urgent': result = result.filter((c) => c.urgent); break
   811→      }
   812→    }
   813→
   814→    // Search filter
   815→    if (state.searchQuery.trim()) {
   816→      const q = state.searchQuery.toLowerCase()
   817→      result = result.filter((c) => {
   818→        if (c.name.toLowerCase().includes(q)) return true
   819→        if (c.lastMessage.toLowerCase().includes(q)) return true
   820→        // Search in message content
   821→        const msgs = state.messages[c.id] ?? []
   822→        return msgs.some((m) => m.text.toLowerCase().includes(q))
   823→      })
   824→    }
   825→
   826→    // Sort: starred first, then by latest activity
   827→    return result.sort((a, b) => {
   828→      if (a.starred && !b.starred) return -1
   829→      if (!a.starred && b.starred) return 1
   830→      return new Date(b.lastTimestamp).getTime() - new Date(a.lastTimestamp).getTime()
   831→    })
   832→  },
   833→
   834→  getUnreadCount: () => {
   835→    return get().conversations.filter((c) => !c.archived).reduce((sum, c) => sum + c.unread, 0)
   836→  },
   837→
   838→  getGroupById: (id) => get().groups.find((g) => g.id === id),
   839→
   840→  getGroupByConversationId: (conversationId) =>
   841→    get().groups.find((g) => g.conversationId === conversationId),
   842→}))

   846→export interface RecipientOption {
   847→  name: string
   848→  role: string
   849→  type: ConversationType
   850→  avatar: string
   851→}
   852→
   853→export function getRecipientOptions(): RecipientOption[] {
   854→  const students = useStudentsStore.getState().students
   855→  const activeStudents = students.filter((s) => s.status === 'Active')
   856→
   857→  const staff: RecipientOption[] = teachers
   858→    .filter((t) => !t.archived)
   859→    .map((t) => ({
   860→      name: t.name,
   861→      role: `${t.designation} · ${t.department}`,
   862→      type: 'staff' as ConversationType,
   863→      avatar: t.avatar,
   864→    }))
   865→
   866→  // Parents — deduped by guardian name (one entry per parent; siblings
   867→  // are collapsed into a single row so recipient keys stay unique).
   868→  const parents: RecipientOption[] = []
   869→  const seenParent = new Set<string>()
   870→  for (const s of activeStudents) {
   871→    if (seenParent.has(s.fatherName)) continue
   872→    seenParent.add(s.fatherName)
   873→    const wards = activeStudents.filter((x) => x.fatherName === s.fatherName).map((x) => x.name)
   874→    const wardLabel =
   875→      wards.length === 1
   876→        ? wards[0]
   877→        : wards.length === 2
   878→          ? wards.join(' & ')
   879→          : `${wards[0]} +${wards.length - 1} more`
   880→    parents.push({
   881→      name: s.fatherName,
   882→      role: `Parent · ${wardLabel}`,
   883→      type: 'parent' as ConversationType,
   884→      avatar: s.fatherName.split(' ').map((n) => n[0]).slice(0, 2).join(''),
   885→    })
   886→    if (parents.length >= 12) break
   887→  }
   888→
   889→  // Groups — pulled from the live store so newly-created groups appear automatically
   890→  const groups: RecipientOption[] = useMessagingStore.getState().groups.map((g) => ({
   891→    name: g.name,
   892→    role: `Group · ${g.memberRefs.length} member${g.memberRefs.length === 1 ? '' : 's'}`,
   893→    type: 'group' as ConversationType,
   894→    avatar: avatarFromName(g.name),
   895→  }))
   896→
   897→  return [...staff, ...parents, ...groups]
   898→}
