     1→/**
     2→ * Finance store — single source of truth for the Finance Dashboard.
     3→ *
     4→ * Aggregates from:
     5→ *   - Fee Management (fee-store.ts) → Fee Revenue + Receivables
     6→ *   - Salary & Payroll (salary-store.ts) → Payroll Expense + Payables
     7→ *   - P&L data (mock/finance-dashboard.ts) → Operating expenses, other income
     8→ *   - Balance sheet data → Assets / Liabilities / Net Worth
     9→ *
    10→ * All dashboard numbers derive from this store — no independent hardcoding
    11→ * in different components. Numbers reconcile mathematically:
    12→ *   Revenue - Expenses = Net Surplus
    13→ *   Assets - Liabilities = Net Worth
    14→ *   Opening Cash + Cash In - Cash Out = Closing Cash
    15→ *   Budget - Actual = Variance
    16→ */

    46→// ─── Hook: useFinanceData ────────────────────────────────────────────
    47→
    48→export function useFinanceData(periodId: string = 'fy25-26') {
    49→  const feeData = useFeeData(CURRENT_ACADEMIC_YEAR)
    50→  const salaryData = useSalaryData()
    51→
    52→  return useMemo(() => {
    53→    // ── Revenue ────────────────────────────────────────────────────
    54→    // Derive fee revenue from fee-store (collected amount).
    55→    const feeRevenue = feeData.analytics.totalCollected
    56→    const feeExpected = feeData.analytics.totalExpected
    57→    const feeOutstanding = feeData.analytics.totalOutstanding
    58→    const feeCollectionRate = feeData.analytics.collectionRate
    59→
    60→    // Other revenue (from P&L data: transport, admissions, donations, etc.)
    61→    // Excluding the tuition/fee income which is already in feeRevenue.
    62→    const otherIncomeItems = pnlData.filter((p) =>
    63→      p.type === 'income' &&
    64→      !p.category.toLowerCase().includes('tuition') &&
    65→      !p.category.toLowerCase().includes('transport fees')  // transport is in fee structure
    66→    )
    67→    const otherRevenue = otherIncomeItems.reduce((s, p) => s + p.amount, 0)
    68→
    69→    // For demo coherence, we use the canonical financeStats.monthlyRevenue
    70→    // for the trend chart and aggregate to fiscal-year totals.
    71→    const monthlyRevenue = financeStats.monthlyRevenue
    72→    const totalRevenue = monthlyRevenue.reduce((s, m) => s + m.revenue, 0)
    73→    const totalExpenses = monthlyRevenue.reduce((s, m) => s + m.expense, 0)
    74→    const netSurplus = totalRevenue - totalExpenses
    75→    const surplusMargin = totalRevenue > 0 ? Math.round((netSurplus / totalRevenue) * 1000) / 10 : 0
    76→
    77→    // ── Expenses ────────────────────────────────────────────────────
    78→    // Derive payroll expense from salary-store (current month calculated).
    79→    const monthlyPayroll = salaryData.analytics.monthlyPayroll
    80→    const annualizedPayroll = monthlyPayroll * 12
    81→
    82→    // ── LIVE PAYROLL LEDGER (Employee Accounts parity) ─────────────
    83→    // Operational payroll cash metrics mirrored from the Salary & Payroll
    84→    // module's Employee Accounts tab — the Finance Dashboard and the
    85→    // payroll workspace always agree. Balance-sheet panels keep their
    86→    // static reconciliation sources; these are the REAL numbers.
    87→    const payrollPeriods = periodOptions(6)
    88→    const payrollCurrentPeriod = currentPeriodKey()
    89→    let payrollOutstanding = 0
    90→    for (const e of salaryData.employees) {
    91→      if (e.status !== 'Active' && e.status !== 'On Leave') continue
    92→      const joinKey = e.joiningDate?.slice(0, 7) ?? ''
    93→      for (const pk of payrollPeriods) {
    94→        if (pk > payrollCurrentPeriod) continue
    95→        if (joinKey && pk < joinKey) continue // not employed yet — no accrual
    96→        const payable = netPayableFor(
    97→          { salaries: salaryData.salaries, adjustments: salaryData.adjustments }, e.id, pk,
    98→        )
    99→        if (payable <= 0) continue
   100→        const confirmed = confirmedPaidFor(salaryData.payments, e.id, pk)
   101→        payrollOutstanding += Math.max(0, payable - confirmed)
   102→      }
   103→    }
   104→    const payrollPendingReceipts = salaryData.payments
   105→      .filter((p) => p.status === 'Pending Receipt' || p.status === 'Not Received')
   106→      .reduce((s, p) => s + p.amount, 0)
   107→    const payrollPaidSession = salaryData.payments
   108→      .filter((p) => p.status === 'Confirmed')
   109→      .reduce((s, p) => s + p.amount, 0)
   110→
   111→    // ── FEE-PLAN SESSION HEALTH (Fee Structures parity) ────────────
   112→    // How many per-class fee plans for the active session are published
   113→    // (have a current version) vs still drafts/not-configured.
   114→    const structureSession = {
   115→      session: CURRENT_ACADEMIC_YEAR,
   116→      total: feeData.feeStructures.length,
   117→      published: feeData.feeStructures.filter((st) =>
   118→        feeData.versions.some((v) => v.structureId === st.id && v.status === 'current'),
   119→      ).length,
   120→    }
   121→
   122→    // Use P&L data expense breakdown for category analysis.
   123→    const expenseBreakdown = financeStats.expenseBreakdown
   124→    // Replace the salaries line with the canonical payroll-derived figure.
   125→    const payrollDerivedExpense = expenseBreakdown.map((e) =>
   126→      e.name === 'Salaries' ? { ...e, value: annualizedPayroll, label: 'Staff & Payroll' } : e
   127→    )
   128→
   129→    // ── Cash Position ──────────────────────────────────────────────
   130→    const cashAvailable = balanceSheet.find((b) => b.account === 'Cash & Bank Balance')?.amount ?? financeStats.cashOnHand
   131→    const monthlyOperatingExpense = totalExpenses / 12
   132→    const reserveCoverage = monthlyOperatingExpense > 0 ? Math.round((cashAvailable / monthlyOperatingExpense) * 10) / 10 : 0
   133→
   134→    // ── Cash Flow ───────────────────────────────────────────────────
   135→    const operatingNet = cashflow.filter((c) => c.activity === 'operating').reduce((s, c) => s + c.inflow - c.outflow, 0)
   136→    const investingNet = cashflow.filter((c) => c.activity === 'investing').reduce((s, c) => s + c.inflow - c.outflow, 0)
   137→    const financingNet = cashflow.filter((c) => c.activity === 'financing').reduce((s, c) => s + c.inflow - c.outflow, 0)
   138→    const netCashChange = operatingNet + investingNet + financingNet
   139→    const openingCash = cashAvailable - netCashChange
   140→    const closingCash = cashAvailable
   141→
   142→    // ── Balance Sheet ──────────────────────────────────────────────
   143→    const currentAssets = balanceSheet.filter((b) => b.type === 'asset' && b.category === 'Current Assets').reduce((s, b) => s + b.amount, 0)
   144→    const fixedAssets = balanceSheet.filter((b) => b.type === 'asset' && b.category === 'Fixed Assets').reduce((s, b) => s + b.amount, 0)
   145→    const totalAssets = currentAssets + fixedAssets
   146→    const currentLiabilities = balanceSheet.filter((b) => b.type === 'liability' && b.category === 'Current Liabilities').reduce((s, b) => s + b.amount, 0)
   147→    const longTermLiabilities = balanceSheet.filter((b) => b.type === 'liability' && b.category === 'Long-term Liabilities').reduce((s, b) => s + b.amount, 0)
   148→    const totalLiabilities = currentLiabilities + longTermLiabilities
   149→    const totalEquity = balanceSheet.filter((b) => b.type === 'equity').reduce((s, b) => s + b.amount, 0)
   150→    const netWorth = totalAssets - totalLiabilities
   151→
   152→    // ── Receivables ────────────────────────────────────────────────
   153→    const feeReceivables = feeOutstanding
   154→    const otherReceivables = balanceSheet.find((b) => b.account === 'Fees Receivable')?.amount ?? 0
   155→    const totalReceivables = feeReceivables + otherReceivables
   156→    const receivableStudentCount = feeData.analytics.pendingCount
   157→
   158→    // ── Payables ────────────────────────────────────────────────────
   159→    const payrollPayable = balanceSheet.find((b) => b.account === 'Salary Payable')?.amount ?? monthlyPayroll
   160→    const vendorPayables = balanceSheet.find((b) => b.account === 'Vendor Payables')?.amount ?? 0
   161→    const totalPayables = payrollPayable + vendorPayables + longTermLiabilities / 12 // monthly loan portion
   162→
   163→    // ── Budget vs Actual ────────────────────────────────────────────
   164→    const budgetData = financeStats.budgetVsActual.map((b) => {
   165→      // Override Salaries actual with payroll-derived figure
   166→      if (b.category === 'Salaries') {
   167→        return { ...b, actual: annualizedPayroll, category: 'Staff' }
   168→      }
   169→      return b
   170→    })
   171→    const totalBudget = budgetData.reduce((s, b) => s + b.budget, 0)
   172→    const totalActual = budgetData.reduce((s, b) => s + b.actual, 0)
   173→    const totalVariance = totalBudget - totalActual
   174→    const budgetUtilization = totalBudget > 0 ? Math.round((totalActual / totalBudget) * 1000) / 10 : 0
   175→
   176→    // ── Quarterly ──────────────────────────────────────────────────
   177→    const quarterly = financeStats.quarterlyRevExp.map((q) => ({
   178→      ...q,
   179→      surplus: q.revenue - q.expense,
   180→    }))
   181→    const quarterlySurplus = quarterly.map((q) => ({ quarter: q.quarter, surplus: q.surplus }))
   182→    const bestQuarter = quarterly.reduce((best, q) => q.surplus > best.surplus ? q : best, quarterly[0])
   183→    const lowestQuarter = quarterly.reduce((low, q) => q.surplus < low.surplus ? q : low, quarterly[0])
   184→
   185→    // ── Financial Health ───────────────────────────────────────────
   186→    const currentRatio = currentLiabilities > 0 ? Math.round((currentAssets / currentLiabilities) * 100) / 100 : 0
   187→    const debtToEquity = totalEquity > 0 ? Math.round((totalLiabilities / totalEquity) * 100) / 100 : 0
   188→    const operatingEfficiency = totalRevenue > 0 ? Math.round((totalExpenses / totalRevenue) * 1000) / 10 : 0
   189→    const collectionRate = feeCollectionRate
   190→
   191→    const healthMetrics = [
   192→      { label: 'Current Ratio', value: `${currentRatio.toFixed(2)}×`, status: currentRatio >= 2 ? 'Healthy' : currentRatio >= 1 ? 'Watch' : 'Attention', severity: currentRatio >= 2 ? 'healthy' : currentRatio >= 1 ? 'watch' : 'attention' },
   193→      { label: 'Debt to Equity', value: `${debtToEquity.toFixed(2)}×`, status: debtToEquity <= 0.5 ? 'Healthy' : debtToEquity <= 1 ? 'Watch' : 'Attention', severity: debtToEquity <= 0.5 ? 'healthy' : debtToEquity <= 1 ? 'watch' : 'attention' },
   194→      { label: 'Surplus Margin', value: `${surplusMargin}%`, status: surplusMargin >= 20 ? 'Healthy' : surplusMargin >= 10 ? 'Watch' : 'Attention', severity: surplusMargin >= 20 ? 'healthy' : surplusMargin >= 10 ? 'watch' : 'attention' },
   195→      { label: 'Operating Efficiency', value: `${operatingEfficiency}%`, status: operatingEfficiency <= 60 ? 'Healthy' : operatingEfficiency <= 75 ? 'Watch' : 'Attention', severity: operatingEfficiency <= 60 ? 'healthy' : operatingEfficiency <= 75 ? 'watch' : 'attention' },
   196→      { label: 'Reserve Coverage', value: `${reserveCoverage} months`, status: reserveCoverage >= 3 ? 'Healthy' : reserveCoverage >= 1 ? 'Watch' : 'Attention', severity: reserveCoverage >= 3 ? 'healthy' : reserveCoverage >= 1 ? 'watch' : 'attention' },
   197→      { label: 'Collection Rate', value: `${collectionRate}%`, status: collectionRate >= 85 ? 'Healthy' : collectionRate >= 70 ? 'Watch' : 'Attention', severity: collectionRate >= 85 ? 'healthy' : collectionRate >= 70 ? 'watch' : 'attention' },
   198→    ]
   199→    const overallHealth = healthMetrics.filter((m) => m.severity === 'attention').length > 0 ? 'Attention'
   200→      : healthMetrics.filter((m) => m.severity === 'watch').length > 0 ? 'Watch'
   201→      : 'Healthy'
   202→
   203→    // ── Alerts ────────────────────────────────────────────────────
   204→    const alerts: Array<{ id: string; title: string; description: string; severity: 'critical' | 'warning' | 'info'; action?: string; /** Where the action button lands: a finance tab ('overview'|'statements'|'reports') or an AppShell module key ('fees'|'salary'). */ actionModule?: string }> = []
   205→    if (feeOutstanding > 1000000) {
   206→      alerts.push({ id: 'fee-outstanding', title: 'Outstanding Fees', description: `${formatINR(feeOutstanding, true)} pending from ${receivableStudentCount} students`, severity: 'warning', action: 'View Pending Dues', actionModule: 'fees' })
   207→    }
   208→    const techBudget = budgetData.find((b) => b.category === 'Tech')
   209→    if (techBudget && techBudget.actual > techBudget.budget) {
   210→      alerts.push({ id: 'tech-overrun', title: 'Technology Budget Exceeded', description: `${formatINR(techBudget.actual - techBudget.budget, true)} over budget`, severity: 'critical', action: 'View Budget', actionModule: 'reports' })
   211→    }
   212→    if (salaryData.analytics.pendingAdjustments > 0) {
   213→      alerts.push({ id: 'payroll-pending', title: 'Salary Changes Awaiting Approval', description: `${salaryData.analytics.pendingAdjustments} salary change${salaryData.analytics.pendingAdjustments > 1 ? 's' : ''} awaiting employee approval`, severity: 'warning', action: 'View Payroll', actionModule: 'salary' })
   214→    }
   215→    if (collectionRate < 85) {
   216→      alerts.push({ id: 'collection-low', title: 'Fee Collection Below Target', description: `${collectionRate}% collected — target 85%`, severity: 'warning', action: 'View Fee Management', actionModule: 'fees' })
   217→    }
   218→    if (reserveCoverage < 3) {
   219→      alerts.push({ id: 'reserve-low', title: 'Cash Reserve Below Target', description: `${reserveCoverage} months coverage — target 3+ months`, severity: 'info' })
   220→    }
   221→
   222→    // ── Recent Activity ────────────────────────────────────────────
   223→    const recentActivity: Array<{ id: string; date: string; type: 'income' | 'expense' | 'payroll' | 'adjustment'; description: string; amount: number; status: string }> = [
   224→      ...feeData.analytics.recentCollections.slice(0, 3).map((t) => ({
   225→        id: t.id, date: t.date, type: 'income' as const,
   226→        description: `Fee payment from ${t.studentName}`,
   227→        amount: t.amount, status: t.status,
   228→      })),
   229→      ...salaryData.audit.slice(0, 2).map((a) => ({
   230→        id: a.id, date: a.timestamp.split('T')[0], type: a.action.includes('payment') ? 'payroll' as const : 'adjustment' as const,
   231→        description: a.detail,
   232→        amount: 0, status: 'Recorded',
   233→      })),
   234→      // A couple of expense entries from P&L
   235→      { id: 'exp-1', date: '2025-11-28', type: 'expense' as const, description: 'Vendor payment — Lab equipment supplier', amount: 840000, status: 'Paid' },
   236→      { id: 'exp-2', date: '2025-11-25', type: 'expense' as const, description: 'Utility bill — Electricity', amount: 280000, status: 'Paid' },
   237→    ].sort((a, b) => b.date.localeCompare(a.date)).slice(0, 6)
   238→
   239→    // ── Upcoming Obligations ───────────────────────────────────────
   240→    // The payroll line is REAL: when unpaid payroll exists (Employee
   241→    // Accounts ledger) it shows the actual unpaid portion; otherwise the
   242→    // forward-looking monthly payroll. Mock vendor/utility lines follow.
   243→    const upcomingObligations = [
   244→      payrollOutstanding > 0
   245→        ? { id: 'obl-1', title: 'Payroll — unpaid portion', amount: payrollOutstanding, due: 'End of month', severity: 'warning' as const }
   246→        : { id: 'obl-1', title: 'Payroll', amount: monthlyPayroll, due: 'End of month', severity: 'info' as const },
   247→      { id: 'obl-2', title: 'Utilities', amount: 420000, due: '25th of month', severity: 'info' as const },
   248→      { id: 'obl-3', title: 'Vendor Payments', amount: 860000, due: '28th of month', severity: 'info' as const },
   249→      { id: 'obl-4', title: 'Loan Repayment', amount: 400000, due: '5th next month', severity: 'info' as const },
   250→    ]
   251→
   252→    // ── Monthly trend (revenue vs expense) ─────────────────────────
   253→    const monthlyTrend = monthlyRevenue.map((m) => ({
   254→      month: m.month,
   255→      revenue: m.revenue,
   256→      expense: m.expense,
   257→      surplus: m.revenue - m.expense,
   258→    }))
   259→
   260→    return {
   261→      period: FINANCE_PERIODS.find((p) => p.id === periodId) ?? FINANCE_PERIODS[0],
   262→
   263→      // Top summary
   264→      totalRevenue,
   265→      totalExpenses,
   266→      netSurplus,
   267→      surplusMargin,
   268→      cashAvailable,
   269→      monthlyOperatingExpense,
   270→      reserveCoverage,
   271→
   272→      // Revenue breakdown
   273→      feeRevenue,
   274→      feeExpected,
   275→      feeOutstanding,
   276→      feeCollectionRate,
   277→      otherRevenue,
   278→      otherIncomeItems,
   279→
   280→      // Expenses
   281→      expenseBreakdown: payrollDerivedExpense,
   282→      monthlyPayroll,
   283→      annualizedPayroll,
   284→
   285→      // Live payroll ledger (Employee Accounts parity)
   286→      payrollPaidSession,
   287→      payrollPendingReceipts,
   288→      payrollOutstanding,
   289→
   290→      // Fee-plan session health (Fee Structures parity)
   291→      structureSession,
   292→
   293→      // Cash flow
   294→      operatingNet,
   295→      investingNet,
   296→      financingNet,
   297→      netCashChange,
   298→      openingCash,
   299→      closingCash,
   300→
   301→      // Balance sheet
   302→      currentAssets,
   303→      fixedAssets,
   304→      totalAssets,
   305→      currentLiabilities,
   306→      longTermLiabilities,
   307→      totalLiabilities,
   308→      totalEquity,
   309→      netWorth,
   310→
   311→      // Receivables / Payables
   312→      feeReceivables,
   313→      otherReceivables,
   314→      totalReceivables,
   315→      receivableStudentCount,
   316→      payrollPayable,
   317→      vendorPayables,
   318→      totalPayables,
   319→      upcomingObligations,
   320→
   321→      // Budget
   322→      budgetData,
   323→      totalBudget,
   324→      totalActual,
   325→      totalVariance,
   326→      budgetUtilization,
   327→
   328→      // Quarterly
   329→      quarterly,
   330→      quarterlySurplus,
   331→      bestQuarter,
   332→      lowestQuarter,
   333→
   334→      // Financial health
   335→      currentRatio,
   336→      debtToEquity,
   337→      operatingEfficiency,
   338→      collectionRate,
   339→      healthMetrics,
   340→      overallHealth,
   341→
   342→      // Alerts & activity
   343→      alerts,
   344→      recentActivity,
   345→
   346→      // Charts
   347→      monthlyTrend,
   348→      monthlyRevenue,
   349→
   350→      // Raw data
   351→      pnlData,
   352→      balanceSheet,
   353→      cashflow,
   354→    }
   355→  }, [feeData, salaryData, periodId])
   356→}

   360→export function formatINRCompact(n: number): string {
   361→  if (n >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`
   362→  if (n >= 100000) return `₹${(n / 100000).toFixed(2)} L`
   363→  if (n >= 1000) return `₹${(n / 1000).toFixed(1)}K`
   364→  return `₹${n}`
   365→}

   369→// ─── Unified "Needs Attention" feed ──────────────────────────────────
   370→// One actionable list assembled from the LIVE operational stores — the
   371→// same items the Fee Management and Salary & Payroll overviews surface,
   372→// deduplicated and ordered by severity (critical → warning → info).
   373→// Consumed by the Finance shell (Overview tab badge) and the Overview's
   374→// Needs Attention panel.
   375→
   376→export interface FinanceAttentionItem {
   377→  id: string
   378→  severity: 'critical' | 'warning' | 'info'
   379→  title: string
   380→  description: string
   381→  cta: string
   382→  /** Where the CTA lands: an AppShell module key, the central finance
   383→   * settings tab, or a finance dashboard tab. */
   384→  module?: 'fees' | 'salary' | 'statements' | 'reports' | 'finance-settings'
   385→}
   386→
   387→export function useFinanceAttention(): FinanceAttentionItem[] {
   388→  const feeData = useFeeData(CURRENT_ACADEMIC_YEAR)
   389→  const salaryData = useSalaryData()
   390→  // Payment infrastructure state (central Finance Settings) — the command
   391→  // centre should surface configuration problems, not just balances.
   392→  const paymentModes = useFeeStore((s) => s.paymentModes)
   393→  const gatewayConfig = useFeeStore((s) => s.gatewayConfig)
   394→  const upiQrConfigs = useFeeStore((s) => s.upiQrConfigs)
   395→  const bankAccounts = useFeeStore((s) => s.bankAccounts)
   396→
   397→  return useMemo(() => {
   398→    const items: FinanceAttentionItem[] = []
   399→    const { analytics } = feeData
   400→
   401→    // 1 — Payroll still unpaid for the current month (money the school owes its staff).
   402→    const payrollBalance = salaryData.rows.reduce((s, r) => s + r.balance, 0)
   403→    const unpaidStaff = salaryData.rows.filter((r) => r.balance > 0).length
   404→    if (payrollBalance > 0) {
   405→      items.push({
   406→        id: 'payroll-unpaid',
   407→        severity: 'critical',
   408→        title: 'Payroll unpaid',
   409→        description: `${formatINR(payrollBalance, true)} still unpaid for ${salaryData.monthLabel} · ${unpaidStaff} staff`,
   410→        cta: 'Open Payroll',
   411→        module: 'salary',
   412→      })
   413→    }
   414→
   415→    // 2 — Salary change requests waiting on the Principal.
   416→    if (salaryData.pendingChangeRequests.length > 0) {
   417→      items.push({
   418→        id: 'salary-changes',
   419→        severity: 'warning',
   420→        title: 'Salary changes awaiting approval',
   421→        description: `${salaryData.pendingChangeRequests.length} request${salaryData.pendingChangeRequests.length > 1 ? 's' : ''} pending — applies to next payroll once approved`,
   422→        cta: 'Review',
   423→        module: 'salary',
   424→      })
   425→    }
   426→
   427→    // 3 — Cash/cheque handed out, receipt not confirmed yet.
   428→    if (salaryData.currentMonth.pending.count > 0) {
   429→      items.push({
   430→        id: 'salary-receipts',
   431→        severity: 'warning',
   432→        title: 'Salary receipts pending',
   433→        description: `${salaryData.currentMonth.pending.count} payment${salaryData.currentMonth.pending.count > 1 ? 's' : ''} (${formatINR(salaryData.currentMonth.pending.amount, true)}) awaiting confirmation`,
   434→        cta: 'Confirm',
   435→        module: 'salary',
   436→      })
   437→    }
   438→
   439→    // 4 — Online fee payments sitting under verification.
   440→    if (analytics.pendingVerification > 0) {
   441→      items.push({
   442→        id: 'fee-verification',
   443→        severity: 'warning',
   444→        title: 'Fee payments to verify',
   445→        description: `${analytics.pendingVerification} payment${analytics.pendingVerification > 1 ? 's' : ''} under verification — receipts finalize after this`,
   446→        cta: 'Verify',
   447→        module: 'fees',
   448→      })
   449→    }
   450→
   451→    // 5 — Parent cash-deposit requests needing Principal acceptance.
   452→    if (analytics.pendingCashRequests > 0) {
   453→      items.push({
   454→        id: 'fee-cash-requests',
   455→        severity: 'warning',
   456→        title: 'Cash requests need acceptance',
   457→        description: `${analytics.pendingCashRequests} cash collection request${analytics.pendingCashRequests > 1 ? 's' : ''} waiting for your acceptance`,
   458→        cta: 'Review',
   459→        module: 'fees',
   460→      })
   461→    }
   462→
   463→    // 6 — Overdue student accounts (collection worklist).
   464→    if (analytics.overdueCount > 0) {
   465→      items.push({
   466→        id: 'fee-overdue',
   467→        severity: 'warning',
   468→        title: 'Overdue student accounts',
   469→        description: `${analytics.overdueCount} account${analytics.overdueCount > 1 ? 's' : ''} past due · ${formatINR(analytics.totalOutstanding, true)} outstanding overall`,
   470→        cta: 'Open Accounts',
   471→        module: 'fees',
   472→      })
   473→    }
   474→
   475→    // 7 — Class fee plans not yet published for the active session.
   476→    const total = feeData.feeStructures.length
   477→    const published = feeData.feeStructures.filter((st) =>
   478→      feeData.versions.some((v) => v.structureId === st.id && v.status === 'current'),
   479→    ).length
   480→    if (total > 0 && published < total) {
   481→      items.push({
   482→        id: 'fee-plans',
   483→        severity: 'info',
   484→        title: 'Fee plans not published',
   485→        description: `${total - published} of ${total} classes not configured for ${CURRENT_ACADEMIC_YEAR}`,
   486→        cta: 'Open Fee Structures',
   487→        module: 'fees',
   488→      })
   489→    }
   490→
   491→    // 8 — Additional collections drafted but never published — work stuck
   492→    //     BEFORE it can collect money (APPS-IA-1 drafts are invisible to
   493→    //     students until published; the Principal is the only one who can
   494→    //     unblock them).
   495→    const draftCharges = feeData.additionalCharges.filter((c) => c.status === 'Draft')
   496→    if (draftCharges.length > 0) {
   497→      items.push({
   498→        id: 'collection-drafts',
   499→        severity: 'info',
   500→        title: 'Draft collections not published',
   501→        description: `${draftCharges.length} draft collection${draftCharges.length > 1 ? 's' : ''} waiting — students can\u2019t pay until published`,
   502→        cta: 'Open Collections',
   503→        module: 'fees',
   504→      })
   505→    }
   506→
   507→    // 9 — Reserves below the 3-month operating target.
   508→    const cashAvailable = balanceSheet.find((b) => b.account === 'Cash & Bank Balance')?.amount ?? 0
   509→    const monthlyOperatingExpense = financeStats.monthlyRevenue.reduce((s, m) => s + m.expense, 0) / 12
   510→    const reserveCoverage = monthlyOperatingExpense > 0 ? Math.round((cashAvailable / monthlyOperatingExpense) * 10) / 10 : 0
   511→    if (reserveCoverage < 3) {
   512→      items.push({
   513→        id: 'reserve-low',
   514→        severity: 'info',
   515→        title: 'Reserves below target',
   516→        description: `${reserveCoverage} months of costs in bank — target is 3+ months`,
   517→        cta: 'View Statements',
   518→        module: 'statements',
   519→      })
   520→    }
   521→
   522→    // 9-10 — PAYMENT INFRASTRUCTURE (central Finance Settings parity).
   523→    // Availability mirrors the settings logic: UPI needs an active QR or a
   524→    // connected gateway; Card/Net Banking need a gateway; Bank Transfer
   525→    // needs an active bank account.
   526→    const gatewayLive = !!gatewayConfig && (gatewayConfig.status === 'connected' || gatewayConfig.status === 'test_mode')
   527→
   528→    if (gatewayConfig?.status === 'test_mode') {
   529→      items.push({
   530→        id: 'gateway-test-mode',
   531→        severity: 'warning',
   532→        title: 'Payment gateway in test mode',
   533→        description: 'Card, net-banking and gateway UPI payments only run test transactions — switch to live before opening to parents.',
   534→        cta: 'Open Settings',
   535→        module: 'finance-settings',
   536→      })
   537→    }
   538→
   539→    const unconfigured: string[] = []
   540→    for (const m of paymentModes) {
   541→      if (!m.active) continue
   542→      if (m.id === 'Cash' || m.id === 'Cheque') continue
   543→      if (m.id === 'Bank Transfer' && bankAccounts.some((b) => b.status === 'active')) continue
   544→      if (m.id === 'UPI' && (upiQrConfigs.some((c) => c.status === 'active') || gatewayLive)) continue
   545→      if ((m.id === 'Card' || m.id === 'Net Banking') && gatewayLive) continue
   546→      unconfigured.push(m.label)
   547→    }
   548→    if (unconfigured.length > 0) {
   549→      items.push({
   550→        id: 'payments-unconfigured',
   551→        severity: 'warning',
   552→        title: 'Payment methods need configuration',
   553→        description: `${unconfigured.join(', ')} enabled but not yet usable by parents — finish the setup in Finance Settings.`,
   554→        cta: 'Open Settings',
   555→        module: 'finance-settings',
   556→      })
   557→    }
   558→
   559→    const order: Record<FinanceAttentionItem['severity'], number> = { critical: 0, warning: 1, info: 2 }
   560→    return items.sort((a, b) => order[a.severity] - order[b.severity])
   561→  }, [feeData, salaryData, paymentModes, gatewayConfig, upiQrConfigs, bankAccounts])
   562→}
