   228→// ─── ClassSubjectAssignment (Spec §19/§20) ─────────────────────────────
   229→// Separates the SUBJECT (canonical, school-scoped) from its ASSIGNMENT to
   230→// a specific class (+ optional stream). One canonical "Hindi" subject can
   231→// be assigned to many classes. Removing an assignment does NOT delete the
   232→// canonical subject — historical records remain valid (Spec §16/§25).
   233→model ClassSubjectAssignment {
   234→  id           String    @id @default(cuid())
   235→  schoolId     String
   236→  school       School    @relation(fields: [schoolId], references: [id], onDelete: Cascade)
   237→  classId      String
   238→  class        Class     @relation(fields: [classId], references: [id], onDelete: Cascade)
   239→  subjectId    String
   240→  subject      Subject   @relation(fields: [subjectId], references: [id], onDelete: Cascade)
   241→  // Spec §34: distinguishes core/default subjects from additional/optional ones.
   242→  isCore       Boolean   @default(true)
   243→  // Spec §18/§35: an inactive assignment is excluded from new examinations
   244→  // but historical records remain valid. Archived subjects are not selectable.
   245→  isActive     Boolean   @default(true)
   246→  // Spec §33: some subjects are examinable, others are not (e.g. PE, Music).
   247→  examinable   Boolean   @default(true)
   248→  // Spec §33: deterministic ordering for display.
   249→  displayOrder Int       @default(0)
   250→  createdAt    DateTime  @default(now())
   251→  updatedAt    DateTime? @updatedAt
   252→
   253→  // Spec §12/§19: one assignment per (class, subject) — prevents duplicates.
   254→  @@unique([classId, subjectId])
   255→}

   972→// ─── Phase 9 — Enterprise Fee Architecture (DB-backed) ───────────────
   973→//
   974→// The fee-store has been in-memory + Zustand-persist (localStorage) since
   975→// takeover. These new models make the fee system truly DB-backed and
   976→// multi-tenant-safe, and let the webhook route auto-reconcile instead of
   977→// just logging. All new models are ADDITIVE — the existing Fee + Payment
   978→// models remain untouched so the legacy /api/fees route keeps working.
   979→//
   980→// Architectural mapping (store type → DB row):
   981→//   - school-settings-store.fees.feeHeads[]  → MasterFeeHead
   982→//   - fee-store.FEE_STRUCTURES[]             → FeeStructure (+ FeeHead rows)
   983→//   - fee-store.FeeStructureVersion          → FeeStructureVersion
   984→//   - fee-store.SEED_TRANSACTIONS[]          → FeeTransaction (immutable)
   985→//   - fee-store.settlements[]                → Settlement
   986→//   - fee-store.reconciliations[]            → Reconciliation
   987→//   - fee-store.webhookEvents[]              → WebhookEvent
   988→//
   989→// Settlement status lifecycle:
   990→//   pending → settled → reconciled
   991→//                     ↘ exception → (manual review) → reconciled
   992→//   failure → reversed
   993→
   994→/// School-wide master fee-head catalogue — single source of truth for
   995→/// head names, default frequencies, default amounts, categories.
   996→/// Picked from when building per-class fee structures.
   997→model MasterFeeHead {
   998→  id          String   @id @default(cuid())
   999→  schoolId    String
  1000→  school      School   @relation(fields: [schoolId], references: [id], onDelete: Cascade)
  1001→  name        String
  1002→  category    String   @default("Other") // Tuition, Admission, Annual, Transport, Lab, Library, Exam, Activity, Board, Other
  1003→  frequency   String   @default("Monthly") // Annual, Half-Yearly, Quarterly, Monthly, Per Term, One-Time
  1004→  amount      Float    @default(0)
  1005→  mandatory   Boolean  @default(true)
  1006→  active      Boolean  @default(true)
  1007→  description String?
  1008→  sortOrder   Int      @default(0)
  1009→  createdAt   DateTime @default(now())
  1010→  updatedAt   DateTime @updatedAt
  1011→
  1012→  /// Per-structure line items bound to this catalogue entry (read model).
  1013→  structures FeeHead[]
  1014→
  1015→  @@unique([schoolId, name])
  1016→}
