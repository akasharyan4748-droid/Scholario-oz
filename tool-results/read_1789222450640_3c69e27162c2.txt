   444→/** Does this applicability rule include the given student? The ONE matcher
   445→ *  every entry-fee decision goes through (admission engine + previews). */
   446→export function entryFeeApplies(
   447→  audience: EntryFeeAudience,
   448→  input: { gender?: string; classId?: string; className?: string },
   449→): boolean {
   450→  const wantsGender = audience.scope === 'gender' || audience.scope === 'classes_gender'
   451→  const wantsClasses = audience.scope === 'classes' || audience.scope === 'classes_gender'
   452→  if (wantsGender) {
   453→    const g = (input.gender ?? '').toLowerCase()
   454→    const isBoy = g === 'male' || g === 'boy'
   455→    const isGirl = g === 'female' || g === 'girl'
   456→    if (audience.gender === 'boys' && !isBoy) return false
   457→    if (audience.gender === 'girls' && !isGirl) return false
   458→  }
   459→  if (wantsClasses) {
   460→    const ids = audience.classIds ?? []
   461→    if (ids.length === 0) return false
   462→    if (input.classId && ids.includes(input.classId)) return true
   463→    // Fall back to class-name matching (students without a classId).
   464→    if (input.className) {
   465→      const named = ACADEMIC_CLASSES.filter((c) => ids.includes(c.id)).map((c) => c.name)
   466→      if (named.includes(input.className)) return true
   467→    }
   468→    return false
   469→  }
   470→  return true
   471→}

   473→/**
   474→ * One-time admission fee for a student under the current policy.
   475→ * Amount comes from policy.admissionAmount; eligibility from the
   476→ * 'admission' rule's applicability. ₹0 when disabled / not applicable.
   477→ */
   478→export function admissionFeeFor(
   479→  policy: EntryFeePolicy,
   480→  input: { gender?: string; classId?: string; className?: string },
   481→): number {
   482→  if (!policy.enabled) return 0
   483→  const rule = policy.rules.find((r) => r.id === 'admission')
   484→  if (!rule) return 0
   485→  return entryFeeApplies(rule.applies, input) ? policy.admissionAmount : 0
   486→}
