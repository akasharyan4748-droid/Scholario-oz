     2→     2→ * lesson-planner — server-side engine for the Automatic Curriculum
     3→     3→ * Execution Lesson Planner.
     4→     4→ *
     5→     5→ * ARCHITECTURE (v3):

    79→    79→/** SchoolEvent(type=HOLIDAY) rows expanded into a set of YYYY-MM-DD days. */
    80→    80→export async function loadHolidaySet(schoolId: string, fromDay: string, toDay: string): Promise<Set<string>> {
    81→    81→  const events = await db.schoolEvent.findMany({
    82→    82→    where: {
    83→    83→      schoolId,
    84→    84→      type: 'HOLIDAY',
    85→    85→      startDate: { lte: dayToDate(addDays(toDay, 1)) },
    86→    86→      OR: [{ endDate: null }, { endDate: { gte: dayToDate(addDays(fromDay, -1)) } }],
    87→    87→    },
    88→    88→    select: { startDate: true, endDate: true },
    89→    89→  })
    90→    90→  const set = new Set<string>()
    91→    91→  for (const ev of events) {
    92→    92→    let cur = dateToDay(ev.startDate)
    93→    93→    const end = ev.endDate ? dateToDay(ev.endDate) : cur
    94→    94→    let guard = 0
    95→    95→    while (cur <= end && guard < MAX_HOLIDAY_SPAN_DAYS) {
    96→    96→      set.add(cur)
    97→    97→      cur = addDays(cur, 1)
    98→    98→      guard++
    99→    99→    }
   100→   100→  }
   101→   101→  return set
   102→   102→}

   135→   135→export interface WalkTopic {
   136→   136→  id: string
   137→   137→  estimatedMinutes: number
   138→   138→}
   139→   139→export interface WalkSlot {
   140→   140→  topicId: string
   141→   141→  /** school day the topic starts on */
   142→   142→  day: string
   143→   143→  /** school day the topic's allocated minutes run out on */
   144→   144→  lastDay: string
   145→   145→}
   146→   146→
   147→   147→/**
   148→   148→ * Greedy contiguous walk: topics consume capacity in course order; a day
   149→   149→ * only offers capacity when it is a working day, not a holiday, and the
   150→   150→ * timetable gives the subject periods that weekday (or pacing falls back
   151→   151→ * to minutesPerPeriod on every working day). Several short topics may
   152→   152→ * share one day; a long topic spans several days (its slot keeps the
   153→   153→ * FIRST day).
   154→   154→ */
   155→   155→export function walkSchedule(
   156→   156→  pending: WalkTopic[],
   157→   157→  anchorDay: string,
   158→   158→  holidays: Set<string>,
   159→   159→  pacing: Pacing,
   160→   160→  minutesPerPeriod: number,
   161→   161→  /** >0 = anchor day's REMAINING capacity · 0 = fresh anchor · -1 = anchor exhausted */
   162→   162→  initialCarry = 0,
   163→   163→): WalkSlot[] {
   164→   164→  const capacityFor = (day: string): number => {
   165→   165→    if (holidays.has(day)) return 0
   166→   166→    const wd = weekdayOf(day)
   167→   167→    if (!SCHOOL_WEEKDAYS.has(wd)) return 0
   168→   168→    if (pacing.source === 'FALLBACK') return minutesPerPeriod
   169→   169→    return (pacing.perWeekday[WEEKDAY_NAMES[wd]] ?? 0) * minutesPerPeriod
   170→   170→  }
   171→   171→  const out: WalkSlot[] = []
   172→   172→  // Cursor semantics (capacityFor returns a day's FULL capacity, so the
   173→   173→  // advance loop must never re-enter a day whose capacity is spent):
   174→   174→  //   initialCarry > 0 → consume the anchor's remaining capacity first
   175→   175→  //   initialCarry < 0 → anchor day already exhausted → start after it
   176→   176→  //   initialCarry = 0 → fresh anchor → first advance lands ON the anchor
   177→   177→  let cursor: string
   178→   178→  let carry: number
   179→   179→  if (initialCarry > 0) {
   180→   180→    cursor = anchorDay
   181→   181→    carry = initialCarry
   182→   182→  } else if (initialCarry < 0) {
   183→   183→    cursor = anchorDay
   184→   184→    carry = 0
   185→   185→  } else {
   186→   186→    cursor = addDays(anchorDay, -1)
   187→   187→    carry = 0
   188→   188→  }
   189→   189→  let guard = 0
   190→   190→  for (const t of pending) {
   191→   191→    let remaining = Math.max(1, t.estimatedMinutes)
   192→   192→    let firstDay: string | null = null
   193→   193→    let lastDay: string | null = null
   194→   194→    while (remaining > 0) {
   195→   195→      if (guard++ > 50_000) return out
   196→   196→      if (carry <= 0) {
   197→   197→        // The current day is exhausted — never return to it. Advance to
   198→   198→        // the next day, then skip forward to the first day with capacity.
   199→   199→        let next = addDays(cursor, 1)
   200→   200→        let inner = 0
   201→   201→        while (capacityFor(next) <= 0 && inner < WALK_HORIZON_DAYS) {
   202→   202→          next = addDays(next, 1)
   203→   203→          inner++
   204→   204→        }
   205→   205→        cursor = next
   206→   206→        carry = capacityFor(cursor)
   207→   207→        if (carry <= 0) break // horizon exhausted
   208→   208→      }
   209→   209→      const used = Math.min(carry, remaining)
   210→   210→      carry -= used
   211→   211→      remaining -= used
   212→   212→      if (!firstDay) firstDay = cursor
   213→   213→      lastDay = cursor
   214→   214→    }
   215→   215→    out.push({ topicId: t.id, day: firstDay ?? cursor, lastDay: lastDay ?? firstDay ?? cursor })
   216→   216→  }
   217→   217→  return out
   218→   218→}

   245→   245→export interface PlannedPending {
   246→   246→  curriculumTopicId: string
   247→   247→  day: string
   248→   248→  lastDay: string
   249→   249→  sequence: number
   250→   250→  minutes: number

   252→   252→
   253→   253→/**
   254→   254→ * Re-plan every PENDING slot:
   255→   255→ *  - anchor = today when any pending slot has fallen behind (self-heal),
   256→   256→ *    else the earliest future pending date (deterministic when the
   257→   257→ *    calendar is unchanged, shifts when holidays/timetable change).
   258→   258→ *  - capacity already consumed on the anchor day by COMPLETED slots is
   259→   259→ *    subtracted so completing a lesson never double-books the day.
   260→   260→ */
   261→   261→export function computePendingPlan(topics: TopicRow[], slots: SlotRow[], holidays: Set<string>, pacing: Pacing, minutesPerPeriod: number): PlannedPending[] {
   262→   262→  const today = todayStr()
   263→   263→  const completedIds = new Set(slots.filter((s) => s.status === 'COMPLETED').map((s) => s.curriculumTopicId))
   264→   264→  const topicById = new Map(topics.map((t) => [t.id, t]))
   265→   265→  const pendingSlots = slots.filter((s) => s.status !== 'COMPLETED' && topicById.has(s.curriculumTopicId))
   266→   266→
   267→   267→  const futureDays = pendingSlots
   268→   268→    .map((s) => dateToDay(s.plannedDate))
   269→   269→    .filter((d) => d >= today)
   270→   270→    .sort()
   271→   271→  const behind = pendingSlots.some((s) => dateToDay(s.plannedDate) < today)
   272→   272→  // A topic completed TODAY means the stored forward plan is stale — the
   273→   273→  // remaining topics must be re-laid from today (early completion pulls
   274→   274→  // the future forward). Otherwise keep the earliest future date as the
   275→   275→  // anchor so day-over-day dates stay stable unless inputs change.
   276→   276→  const lastTaughtDay = slots.reduce<string | null>((acc, s) => {
   277→   277→    if (s.status !== 'COMPLETED' || !s.execution) return acc
   278→   278→    const d = dateToDay(s.execution.taughtOn)
   279→   279→    return !acc || d > acc ? d : acc
   280→   280→  }, null)
   281→   281→  const completedToday = Boolean(lastTaughtDay && lastTaughtDay >= today)
   282→   282→  const anchor = behind || completedToday || futureDays.length === 0 ? today : futureDays[0]
   283→   283→
   284→   284→  let initialCarry = 0
   285→   285→  {
   286→   286→    const capacityFor = (day: string): number => {
   287→   287→      if (holidays.has(day)) return 0
   288→   288→      const wd = weekdayOf(day)
   289→   289→      if (!SCHOOL_WEEKDAYS.has(wd)) return 0
   290→   290→      if (pacing.source === 'FALLBACK') return minutesPerPeriod
   291→   291→      return (pacing.perWeekday[WEEKDAY_NAMES[wd]] ?? 0) * minutesPerPeriod
   292→   292→    }
   293→   293→    if (anchor === today) {
   294→   294→      // capacity actually used today = minutes of topics TAUGHT today
   295→   295→      // (whatever their original planned date was). A exhausted day is
   296→   296→      // signalled with -1 so the walk starts tomorrow.
   297→   297→      const consumed = slots
   298→   298→        .filter((s) => s.status === 'COMPLETED' && s.execution && dateToDay(s.execution.taughtOn) === anchor)
   299→   299→        .reduce((a, s) => a + s.minutesAllocated, 0)
   300→   300→      const cap = capacityFor(anchor)
   301→   301→      initialCarry = consumed <= 0 ? 0 : cap - consumed > 0 ? cap - consumed : -1
   302→   302→    }
   303→   303→  }
   304→   304→
   305→   305→  const pendingTopics = topics.filter((t) => !completedIds.has(t.id))
   306→   306→  const walk = walkSchedule(
   307→   307→    pendingTopics.map((t) => ({ id: t.id, estimatedMinutes: t.estimatedMinutes })),
   308→   308→    anchor,
   309→   309→    holidays,
   310→   310→    pacing,
   311→   311→    minutesPerPeriod,
   312→   312→    initialCarry,
   313→   313→  )
   314→   314→  const sequenceByTopic = new Map(topics.map((t, i) => [t.id, i + 1]))
   315→   315→  return walk.map((w) => ({
   316→   316→    curriculumTopicId: w.topicId,
   317→   317→    day: w.day,
   318→   318→    lastDay: w.lastDay,
   319→   319→    sequence: sequenceByTopic.get(w.topicId) ?? 1,
   320→   320→    minutes: topicById.get(w.topicId)?.estimatedMinutes ?? minutesPerPeriod,
   321→   321→  }))
   322→   322→}
   323→   323→
   324→   324→/**
   325→   325→ * The ORIGINAL course plan from the curriculum start date — used to judge
   326→   326→ * whether the class is on pace (expectedByNow) and by the seed to create
   327→   327→ * honest completed history.
   328→   328→ */
   329→   329→export function computeCoursePlanFromStart(topics: TopicRow[], startDate: Date, holidays: Set<string>, pacing: Pacing, minutesPerPeriod: number): WalkSlot[] {
   330→   330→  return walkSchedule(
   331→   331→    topics.map((t) => ({ id: t.id, estimatedMinutes: t.estimatedMinutes })),
   332→   332→    dateToDay(startDate),
   333→   333→    holidays,
   334→   334→    pacing,
   335→   335→    minutesPerPeriod,
   336→   336→  )
   337→   337→}

   339→   339→// ─── sync (persists the computed plan for PENDING slots) ──────────────
   340→   340→
   341→   341→async function syncSchedule(
   342→   342→  schoolId: string,
   343→   343→  classId: string,
   344→   344→  subjectId: string,
   345→   345→  curriculumId: string,
   346→   346→  topics: TopicRow[],
   347→   347→  slots: SlotRow[],
   348→   348→  holidays: Set<string>,
   349→   349→  pacing: Pacing,
   350→   350→  minutesPerPeriod: number,
   351→   351→): Promise<SlotRow[]> {
   352→   352→  const plan = computePendingPlan(topics, slots, holidays, pacing, minutesPerPeriod)
   353→   353→  const planByTopic = new Map(plan.map((p) => [p.curriculumTopicId, p]))
   354→   354→  const topicById = new Map(topics.map((t) => [t.id, t]))
   355→   355→  const slotByTopic = new Map(slots.map((s) => [s.curriculumTopicId, s]))
   356→   356→
   357→   357→  const kept: SlotRow[] = []
   358→   358→  for (const s of slots) {
   359→   359→    if (s.status === 'COMPLETED' || !topicById.has(s.curriculumTopicId)) {
   360→   360→      if (s.status === 'COMPLETED') kept.push(s)
   361→   361→      else await db.scheduledCurriculumTopic.delete({ where: { id: s.id } }).catch(() => {})
   362→   362→      continue
   363→   363→    }
   364→   364→    const p = planByTopic.get(s.curriculumTopicId)
   365→   365→    if (!p) continue
   366→   366→    const wantDay = dayToDate(p.day)
   367→   367→    if (dateToDay(s.plannedDate) !== p.day || s.minutesAllocated !== p.minutes || s.sequence !== p.sequence) {
   368→   368→      const updated = await db.scheduledCurriculumTopic.update({
   369→   369→        where: { id: s.id },
   370→   370→        data: { plannedDate: wantDay, minutesAllocated: p.minutes, sequence: p.sequence, updatedAt: new Date() },
   371→   371→        include: { execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } } },
   372→   372→      })
   373→   373→      kept.push(updated as SlotRow)
   374→   374→    } else {
   375→   375→      kept.push(s)
   376→   376→    }
   377→   377→  }
   378→   378→  for (const p of plan) {
   379→   379→    if (slotByTopic.has(p.curriculumTopicId)) continue
   380→   380→    const created = await db.scheduledCurriculumTopic.create({
   381→   381→      data: {
   382→   382→        schoolId,
   383→   383→        classId,
   384→   384→        subjectId,
   385→   385→        curriculumTopicId: p.curriculumTopicId,
   386→   386→        sequence: p.sequence,
   387→   387→        plannedDate: dayToDate(p.day),
   388→   388→        minutesAllocated: p.minutes,
   389→   389→        status: 'PENDING',
   390→   390→      },
   391→   391→      include: { execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } } },
   392→   392→    })
   393→   393→    kept.push(created as SlotRow)
   394→   394→  }
   395→   395→  return kept
   396→   396→}

   441→   441→function deriveStatus(slot: SlotRow, today: string): LessonDisplayStatus {
   442→   442→  if (slot.status === 'COMPLETED') return 'COMPLETED'
   443→   443→  if (slot.status === 'NEEDS_RESCHEDULING') return 'NEEDS_RESCHEDULING'
   444→   444→  const day = dateToDay(slot.plannedDate)
   445→   445→  if (day === today) return slot.execution?.startedAt && !slot.execution.completedAt ? 'IN_PROGRESS' : 'TODAY'
   446→   446→  return 'UPCOMING'
   447→   447→}

   469→   469→function buildStats(topics: TopicRow[], slots: SlotRow[], pendingPlan: PlannedPending[], originalPlan: WalkSlot[], startDate: Date, today: string): CurriculumStatsDTO {
   470→   470→  const completed = slots.filter((s) => s.status === 'COMPLETED')
   471→   471→  const minutesTotal = topics.reduce((a, t) => a + t.estimatedMinutes, 0)
   472→   472→  const minutesCompleted = completed.reduce((a, s) => a + s.minutesAllocated, 0)
   473→   473→  const expectedByNow = originalPlan.filter((p) => p.lastDay < today).length
   474→   474→  const behindBy = Math.max(0, expectedByNow - completed.length)
   475→   475→  const projected = pendingPlan.length ? pendingPlan[pendingPlan.length - 1].lastDay : null
   476→   476→  return {
   477→   477→    totalTopics: topics.length,
   478→   478→    completedTopics: completed.length,
   479→   479→    percentComplete: topics.length ? Math.round((completed.length / topics.length) * 100) : 0,
   480→   480→    minutesTotal,
   481→   481→    minutesCompleted,
   482→   482→    projectedEndDate: projected,
   483→   483→    onTrack: dateToDay(startDate) >= today ? null : behindBy === 0,
   484→   484→    behindBy,
   485→   485→  }
   486→   486→}

   488→   488→function buildUnits(topics: TopicRow[], slots: SlotRow[], today: string): CurriculumUnitDTO[] {
   489→   489→  const slotByTopic = new Map(slots.map((s) => [s.curriculumTopicId, s]))
   490→   490→  const byUnit = new Map<number, { unitName: string; topics: TopicRow[] }>()
   491→   491→  for (const t of topics) {
   492→   492→    const g = byUnit.get(t.unitOrder) ?? { unitName: t.unitName, topics: [] }
   493→   493→    g.topics.push(t)
   494→   494→    byUnit.set(t.unitOrder, g)
   495→   495→  }
   496→   496→  const units: CurriculumUnitDTO[] = []
   497→   497→  for (const [unitOrder, g] of [...byUnit.entries()].sort((a, b) => a[0] - b[0])) {
   498→   498→    const dtos = g.topics
   499→   499→      .map((t) => {
   500→   500→        const slot = slotByTopic.get(t.id)
   501→   501→        return slot ? toTopicDTO(slot, t, today) : null
   502→   502→      })
   503→   503→      .filter((d): d is ScheduledTopicDTO => d !== null)
   504→   504→      .sort((a, b) => a.sequence - b.sequence)
   505→   505→    const done = dtos.filter((d) => d.status === 'COMPLETED').length
   506→   506→    units.push({
   507→   507→      unitOrder,
   508→   508→      unitName: g.unitName,
   509→   509→      totalTopics: g.topics.length,
   510→   510→      completedTopics: done,
   511→   511→      percent: g.topics.length ? Math.round((done / g.topics.length) * 100) : 0,
   512→   512→      topics: dtos,
   513→   513→    })
   514→   514→  }
   515→   515→  return units
   516→   516→}

   539→   539→  return { assignments, selected }

   542→   542→/** Assignments with hasCurriculum + pendingToday (read-only, no writes). */
   543→   543→async function assignmentsWithPending(ctx: TeacherHubContext): Promise<LessonAssignmentDTO[]> {
   544→   544→  const base = await teachingAssignments(ctx)
   545→   545→  if (!base.length) return base
   546→   546→  const today = todayStr()
   547→   547→  const classIds = [...new Set(base.map((a) => a.classId))]
   548→   548→
   549→   549→  const curricula = await db.schoolCurriculum.findMany({
   550→   550→    where: { schoolId: ctx.schoolId, classId: { in: classIds }, status: 'ACTIVE' },
   551→   551→    select: { id: true, classId: true, subjectId: true, startDate: true, minutesPerPeriod: true, session: true },
   552→   552→    orderBy: { session: 'desc' },
   553→   553→  })
   554→   554→  const curriculumByScope = new Map<string, (typeof curricula)[number]>()
   555→   555→  for (const c of curricula) {
   556→   556→    const k = `${c.classId}|${c.subjectId}`
   557→   557→    if (!curriculumByScope.has(k)) curriculumByScope.set(k, c) // latest session wins
   558→   558→  }
   559→   559→
   560→   560→  const curriculumIds = [...curriculumByScope.values()].map((c) => c.id)
   561→   561→  const [topicRows, slotRows, ttRows, holidays] = await Promise.all([
   562→   562→    curriculumIds.length
   563→   563→      ? db.schoolCurriculumTopic.findMany({
   564→   564→          where: { curriculumId: { in: curriculumIds } },
   565→   565→          select: { id: true, curriculumId: true, unitOrder: true, unitName: true, topicOrder: true, topicName: true, description: true, estimatedMinutes: true },
   566→   566→        })
   567→   567→      : [],
   568→   568→    curriculumIds.length
   569→   569→      ? db.scheduledCurriculumTopic.findMany({
   570→   570→          where: { classId: { in: classIds } },
   571→   571→          select: {
   572→   572→            id: true, curriculumTopicId: true, sequence: true, plannedDate: true, minutesAllocated: true,
   573→   573→            status: true, rescheduledFrom: true, flagReason: true, classId: true, subjectId: true,
   574→   574→            execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } },
   575→   575→          },
   576→   576→        })
   577→   577→      : [],
   578→   578→    db.timetable.findMany({ where: { classId: { in: classIds } }, select: { classId: true, subjectId: true, day: true } }),
   579→   579→    loadHolidaySet(ctx.schoolId, addDays(today, -30), addDays(today, WALK_HORIZON_DAYS)),
   580→   580→  ])
   581→   581→
   582→   582→  for (const a of base) {
   583→   583→    const curriculum = curriculumByScope.get(`${a.classId}|${a.subjectId}`)
   584→   584→    a.hasCurriculum = Boolean(curriculum)
   585→   585→    if (!curriculum) continue
   586→   586→    const topics = topicRows.filter((t) => t.curriculumId === curriculum.id)
   587→   587→    if (!topics.length) continue
   588→   588→    const slots = slotRows.filter((s) => s.classId === a.classId && s.subjectId === a.subjectId)
   589→   589→    const perWeekday: Record<string, number> = {}
   590→   590→    let periods = 0
   591→   591→    for (const r of ttRows) {
   592→   592→      if (r.classId !== a.classId || r.subjectId !== a.subjectId) continue
   593→   593→      perWeekday[r.day] = (perWeekday[r.day] ?? 0) + 1
   594→   594→      periods++
   595→   595→    }
   596→   596→    const pacing: Pacing =
   597→   597→      periods > 0
   598→   598→        ? { source: 'TIMETABLE', periodsPerWeek: periods, minutesPerWeek: periods * curriculum.minutesPerPeriod, perWeekday }
   599→   599→        : { source: 'FALLBACK', periodsPerWeek: null, minutesPerWeek: curriculum.minutesPerPeriod * SCHOOL_WEEKDAYS.size, perWeekday }
   600→   600→    const plan = computePendingPlan(topics, slots, holidays, pacing, curriculum.minutesPerPeriod)
   601→   601→    a.pendingToday = plan.filter((p) => p.day === today).length
   602→   602→  }
   603→   603→  return base
   604→   604→}

   606→   606→async function buildSelectedScope(ctx: TeacherHubContext, scope: LessonAssignmentDTO): Promise<SelectedScopeDTO> {
   607→   607→  const today = todayStr()
   608→   608→  const selected: SelectedScopeDTO = {
   609→   609→    classId: scope.classId,
   610→   610→    classLabel: scope.classLabel,
   611→   611→    subjectId: scope.subjectId,
   612→   612→    subjectName: scope.subjectName,
   613→   613→    curriculum: null,
   614→   614→    unavailable: null,
   615→   615→  }
   616→   616→
   617→   617→  const curriculum = await db.schoolCurriculum.findFirst({
   618→   618→    where: { schoolId: ctx.schoolId, classId: scope.classId, subjectId: scope.subjectId, status: 'ACTIVE' },
   619→   619→    orderBy: { session: 'desc' },
   620→   620→    include: {
   621→   621→      topics: { orderBy: [{ unitOrder: 'asc' }, { topicOrder: 'asc' }] },
   622→   622→    },
   623→   623→  })
   624→   624→  if (!curriculum || !curriculum.topics.length) {
   625→   625→    selected.unavailable = {
   626→   626→      reason: `No curriculum has been imported for ${scope.classLabel} · ${scope.subjectName} yet. The school office needs to import the board syllabus for this subject before lesson planning can start.`,
   627→   627→    }
   628→   628→    return selected
   629→   629→  }
   630→   630→
   631→   631→  const [holidays, pacing, existingSlots] = await Promise.all([
   632→   632→    loadHolidaySet(ctx.schoolId, addDays(today, -30), addDays(today, WALK_HORIZON_DAYS)),
   633→   633→    loadPacing(scope.classId, scope.subjectId, curriculum.minutesPerPeriod),
   634→   634→    db.scheduledCurriculumTopic.findMany({
   635→   635→      where: { classId: scope.classId, subjectId: scope.subjectId },
   636→   636→      select: {
   637→   637→        id: true, curriculumTopicId: true, sequence: true, plannedDate: true, minutesAllocated: true,
   638→   638→        status: true, rescheduledFrom: true, flagReason: true,
   639→   639→        execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } },
   640→   640→      },
   641→   641→    }),
   642→   642→  ])
   643→   643→
   644→   644→  const topics: TopicRow[] = curriculum.topics
   645→   645→  const slots = await syncSchedule(ctx.schoolId, scope.classId, scope.subjectId, curriculum.id, topics, existingSlots, holidays, pacing, curriculum.minutesPerPeriod)
   646→   646→  const pendingPlan = computePendingPlan(topics, slots, holidays, pacing, curriculum.minutesPerPeriod)
   647→   647→  const originalPlan = computeCoursePlanFromStart(topics, curriculum.startDate, holidays, pacing, curriculum.minutesPerPeriod)
   648→   648→
   649→   649→  const topicById = new Map(topics.map((t) => [t.id, t]))
   650→   650→  const all: ScheduledTopicDTO[] = []
   651→   651→  for (const s of slots) {
   652→   652→    const t = topicById.get(s.curriculumTopicId)
   653→   653→    if (t) all.push(toTopicDTO(s, t, today))
   654→   654→  }
   655→   655→  all.sort((a, b) => a.sequence - b.sequence)
   656→   656→
   657→   657→  const pendingToday = all.filter((d) => d.status !== 'COMPLETED' && d.plannedDate === today)
   658→   658→  const upcoming = all.filter((d) => d.status !== 'COMPLETED' && d.plannedDate > today).slice(0, UPCOMING_TAKE)
   659→   659→  const flagged = all.filter((d) => d.status === 'NEEDS_RESCHEDULING')
   660→   660→  const recent = all
   661→   661→    .filter((d) => d.status === 'COMPLETED')
   662→   662→    .sort((a, b) => (a.completedOn ?? '') < (b.completedOn ?? '') ? 1 : -1)
   663→   663→    .slice(0, RECENT_TAKE)
   664→   664→
   665→   665→  const stats = buildStats(topics, slots, pendingPlan, originalPlan, curriculum.startDate, today)
   666→   666→  const pacingDto: PacingInfo = {
   667→   667→    source: pacing.source,
   668→   668→    periodsPerWeek: pacing.periodsPerWeek,
   669→   669→    minutesPerWeek: pacing.minutesPerWeek,
   670→   670→  }
   671→   671→
   672→   672→  const dto: CurriculumDTO = {
   673→   673→    id: curriculum.id,
   674→   674→    board: curriculum.board,
   675→   675→    session: curriculum.session,
   676→   676→    versionLabel: curriculum.versionLabel,
   677→   677→    startDate: dateToDay(curriculum.startDate),
   678→   678→    stats,
   679→   679→    pacing: pacingDto,
   680→   680→    today: pendingToday[0] ?? null,
   681→   681→    todayAlso: pendingToday.slice(1),
   682→   682→    needsRescheduling: flagged,
   683→   683→    upcoming,
   684→   684→    recent,
   685→   685→    units: buildUnits(topics, slots, today),
   686→   686→  }
   687→   687→  selected.curriculum = dto
   688→   688→  return selected
   689→   689→}

   706→   706→/**
   707→   707→ * Apply a teaching intent to a scheduled topic (start / complete / undo /
   708→   708→ * flag / unflag). Re-validates scope on every call, audits every mutation,
   709→   709→ * and returns the refreshed payload for the slot's class+subject.
   710→   710→ */
   711→   711→export async function applyLessonIntent(
   712→   712→  user: AuthUser,
   713→   713→  ctx: TeacherHubContext,
   714→   714→  scheduledTopicId: string,
   715→   715→  intent: LessonIntent,
   716→   716→  reason?: string | null,
   717→   717→): Promise<LessonPlannerPayload> {
   718→   718→  const slot = await db.scheduledCurriculumTopic.findUnique({
   719→   719→    where: { id: scheduledTopicId },
   720→   720→    include: { execution: true },
   721→   721→  })
   722→   722→  if (!slot || slot.schoolId !== ctx.schoolId) throw new Error('NOT_FOUND')
   723→   723→  await assertLessonScope(ctx, slot.classId, slot.subjectId)
   724→   724→
   725→   725→  const today = todayStr()
   726→   726→
   727→   727→  if (intent === 'start') {
   728→   728→    if (slot.status === 'COMPLETED') throw new Error('ALREADY_COMPLETED')
   729→   729→    if (dateToDay(slot.plannedDate) > today) throw new Error('NOT_PLANNED_TODAY')
   730→   730→    if (!slot.execution?.startedAt) {
   731→   731→      await db.lessonExecution.upsert({
   732→   732→        where: { scheduledTopicId: slot.id },
   733→   733→        create: {
   734→   734→          schoolId: ctx.schoolId,
   735→   735→          scheduledTopicId: slot.id,
   736→   736→          teacherId: ctx.teacherId,
   737→   737→          classId: slot.classId,
   738→   738→          subjectId: slot.subjectId,
   739→   739→          taughtOn: dayToDate(today),
   740→   740→          startedAt: new Date(),
   741→   741→        },
   742→   742→        update: { startedAt: new Date(), taughtOn: dayToDate(today) },
   743→   743→      })
   744→   744→      await auditTeacherAction(user, ctx.schoolId, 'LESSON_STARTED', `Started lesson: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   745→   745→    }
   746→   746→  } else if (intent === 'complete') {
   747→   747→    if (slot.status === 'COMPLETED') {
   748→   748→      // idempotent — fall through to payload refresh
   749→   749→    } else {
   750→   750→      if (dateToDay(slot.plannedDate) > today) throw new Error('NOT_PLANNED_TODAY')
   751→   751→      await db.lessonExecution.upsert({
   752→   752→        where: { scheduledTopicId: slot.id },
   753→   753→        create: {
   754→   754→          schoolId: ctx.schoolId,
   755→   755→          scheduledTopicId: slot.id,
   756→   756→          teacherId: ctx.teacherId,
   757→   757→          classId: slot.classId,
   758→   758→          subjectId: slot.subjectId,
   759→   759→          taughtOn: dayToDate(today),
   760→   760→          minutesTaught: slot.minutesAllocated,
   761→   761→          startedAt: slot.execution?.startedAt ?? new Date(),
   762→   762→          completedAt: new Date(),
   763→   763→        },
   764→   764→        update: { completedAt: new Date(), taughtOn: dayToDate(today), minutesTaught: slot.minutesAllocated },
   765→   765→      })
   766→   766→      await db.scheduledCurriculumTopic.update({
   767→   767→        where: { id: slot.id },
   768→   768→        data: { status: 'COMPLETED', updatedAt: new Date() },
   769→   769→      })
   770→   770→      await auditTeacherAction(user, ctx.schoolId, 'LESSON_COMPLETED', `Completed lesson: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   771→   771→    }
   772→   772→  } else if (intent === 'undo') {
   773→   773→    if (slot.status !== 'COMPLETED') throw new Error('NOT_COMPLETED')
   774→   774→    if (slot.execution) await db.lessonExecution.delete({ where: { id: slot.execution.id } })
   775→   775→    await db.scheduledCurriculumTopic.update({
   776→   776→      where: { id: slot.id },
   777→   777→      data: { status: 'PENDING', updatedAt: new Date() },
   778→   778→    })
   779→   779→    await auditTeacherAction(user, ctx.schoolId, 'LESSON_UNDONE', `Reopened lesson: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   780→   780→  } else if (intent === 'flag') {
   781→   781→    if (slot.status === 'COMPLETED') throw new Error('ALREADY_COMPLETED')
   782→   782→    await db.scheduledCurriculumTopic.update({
   783→   783→      where: { id: slot.id },
   784→   784→      data: {
   785→   785→        status: 'NEEDS_RESCHEDULING',
   786→   786→        rescheduledFrom: slot.rescheduledFrom ?? slot.plannedDate,
   787→   787→        flagReason: reason?.slice(0, 200) ?? null,
   788→   788→        updatedAt: new Date(),
   789→   789→      },
   790→   790→    })
   791→   791→    await auditTeacherAction(user, ctx.schoolId, 'LESSON_FLAGGED', `Flagged for rescheduling: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   792→   792→  } else if (intent === 'unflag') {
   793→   793→    if (slot.status === 'NEEDS_RESCHEDULING') {
   794→   794→      await db.scheduledCurriculumTopic.update({
   795→   795→        where: { id: slot.id },
   796→   796→        data: { status: 'PENDING', rescheduledFrom: null, flagReason: null, updatedAt: new Date() },
   797→   797→      })
   798→   798→      await auditTeacherAction(user, ctx.schoolId, 'LESSON_FLAG_CLEARED', `Cleared reschedule flag: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   799→   799→    }
   800→   800→  }
   801→   801→
   802→   802→  return buildLessonPlannerPayload(ctx, slot.classId, slot.subjectId)
   803→   803→}
