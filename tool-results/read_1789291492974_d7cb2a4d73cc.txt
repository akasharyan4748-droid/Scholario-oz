     3→/**
     4→ * fee-receipt-a5 — THE canonical school fee receipt (PAY-REWORK-1).
     5→ *
     6→ * Physical formats (ReceiptSettings.paperSize):
     7→ *   • A5 LANDSCAPE — 210 × 148 mm — ONE student per page, TWO copies
     8→ *     side-by-side (STUDENT COPY left · SCHOOL COPY right) separated by a
     9→ *     subtle tear line.
    10→ *   • A4 PORTRAIT — 210 × 297 mm — TWO students per page, each student's
    11→ *     complete dual-copy receipt occupying one A5-landscape area (4 copies
    12→ *     on the sheet; a student's two copies are never split across pages).
    13→ *   (SaaS-STAGE-1: the legacy 80mm thermal renderer was consolidated into
    14→ *   this canonical design; '80mm' persisted settings migrate to 'A5'.)
    15→ *
    16→ * Layout rules (final UI/UX spec §1): the copy fills the printable area
    17→ * NATURALLY — sections distribute leftover height evenly between them
    18→ * (justify-content: space-between), so there are no giant blank gaps, the
    19→ * signatures always sit at the foot, long fee lists fit via density tiers
    20→ * and short fee lists never pool space into one hole. Fonts are NOT the
    21→ * fix — the geometry is.
    22→ *
    23→ * Data rules (spec §15):
    24→ *   - EVERY value is derived from the canonical payment record + school
    25→ *     profile + student record. Nothing is hardcoded.
    26→ *   - Fields that do not exist for a payment are simply not rendered
    27→ *     (e.g. gateway ref on a cash payment, UPI id on a card payment).
    28→ *   - Balance Dues Remaining comes from the SAME account computation the
    29→ *     Student Accounts / Transactions surfaces use — the receipt can never
    30→ *     contradict the ledger.
    31→ *
    32→ * Lifecycle rules (spec §11):
    33→ *   - 'Under Verification' → the sheet carries an honest PENDING
    34→ *     VERIFICATION notice and is NOT presented as an official receipt.
    35→ *   - 'Failed' → NOT-A-COMPLETED-PAYMENT notice, never a receipt.
    36→ *   - 'Success' → official receipt, identical from every authorised role.
    37→ *
    38→ * Print engine (spec §18): standalone HTML document with
    39→ *   @page { size: A5 landscape | A4 portrait; margin: 0 }
    40→ * No CDN dependencies (no Tailwind/Google Fonts/Lucide) — the printable
    41→ * markup carries its own inline CSS. The on-screen preview is the same
    42→ * layout rendered as DOM and scaled to fit its container.
    43→ *
    44→ * Deliberately NOT here: printer toolbars, browser instructions, grey
    45→ * backdrops (spec §19). Actions live in the calling UI.
    46→ */

    58→// ─── Public types ─────────────────────────────────────────────────────

    60→export interface ReceiptLine {
    61→  particulars: string
    62→  period?: string
    63→  amount: number
    64→}

    66→export interface ReceiptContext {
    67→  /** Balance remaining AFTER this payment (canonical account outstanding). */
    68→  balanceDue?: number | null
    69→  /** Discount / concession applied to the obligation (rendered only when > 0). */
    70→  discount?: number
    71→  /** Extra fee rows (defaults to the payment's own fee head). */
    72→  lines?: ReceiptLine[]
    73→}

    75→// ─── Data assembly (shared by preview + print/download) ──────────────
    76→
    77→/** Build the receipt's fee rows from the canonical payment. One payment =
    78→ *  one obligation in the ledger model, so the default is a single row;
    79→ *  `lines` overrides when a caller aggregates several heads. */
    80→export function buildReceiptLines(t: FeeTransaction, ctx?: ReceiptContext): ReceiptLine[] {
    81→  if (ctx?.lines?.length) return ctx.lines
    82→  return [{ particulars: t.feeHead, amount: t.amount }]
    83→}
    84→
    85→function resolveStudent(t: FeeTransaction) {
    86→  return useStudentsStore.getState().students.find((s) => s.id === t.studentId)
    87→    ?? useStudentsStore.getState().students.find((s) => s.admissionNo === t.admissionNo)
    88→    ?? null
    89→}
    90→
    91→function resolveBalanceDue(t: FeeTransaction, ctx?: ReceiptContext): number | null {
    92→  if (ctx && ctx.balanceDue !== undefined) return ctx.balanceDue
    93→  return getStudentBalanceDue(t.studentId)
    94→}
    95→
    96→function paymentRef(t: FeeTransaction): string | null {
    97→  return t.gatewayPaymentId ?? t.utr ?? t.referenceNo ?? null
    98→}
    99→
   100→function modeDetail(t: FeeTransaction): string | null {
   101→  if (t.meta?.upiId) return t.meta.upiId
   102→  if (t.meta?.chequeNumber) return `Cheque ${t.meta.chequeNumber}`
   103→  if (t.meta?.cardLast4) return `Card ****${t.meta.cardLast4}`
   104→  if (t.meta?.bankName) return t.meta.bankName
   105→  return null
   106→}
   107→
   108→function dateLabel(iso: string): string {
   109→  const d = new Date(iso)
   110→  if (Number.isNaN(d.getTime())) return iso
   111→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' }).toUpperCase()
   112→}
   113→
   114→function shortSession(t: FeeTransaction): string {
   115→  // "2025-2026" → "25-26" for the Period column; already-short values pass through.
   116→  const ay = t.academicYear
   117→  const m = ay.match(/(\d{4})\D(\d{4})/)
   118→  if (m) return `${m[1].slice(2)}-${m[2].slice(2)}`
   119→  return ay
   120→}

   129→function copyData(t: FeeTransaction, settings: ReceiptSettings, designation: CopyData['designation']): CopyData {
   130→  if (designation === 'SCHOOL COPY') {

   132→      designation,
   133→      notes: [
   134→        'Office audit record — for accounts ledger reconciliation.',
   135→        `${settings.footerMessage}`,

   137→    }

   139→  return {
   140→    designation,
   141→    notes: [
   142→      'Fees once paid are non-refundable. Please retain this receipt.',
   143→      `${settings.footerMessage}`,
   144→    ],
   145→  }
   146→}

   148→// ─── Preview (DOM, scaled) ────────────────────────────────────────────

   150→export function FeeReceiptA5Preview({
   151→  transaction: t,
   152→  settings,
   153→  ctx,
   154→  onPrint,
   155→  onDownload,
   156→  onClose,
   157→}: {
   158→  transaction: FeeTransaction
   159→  settings: ReceiptSettings
   160→  ctx?: ReceiptContext
   161→  onPrint?: () => void
   162→  onDownload?: () => void
   163→  onClose?: () => void
   164→}) {
   165→  const wrapRef = useRef<HTMLDivElement>(null)
   166→  const [scale, setScale] = useState(0.8)
   167→
   168→  // Fit the 210mm × 148mm (≈794×561px) sheet into BOTH the available width
   169→  // and a comfortable fraction of the viewport height — responsive,
   170→  // proportional scaling; never crop (spec §33).
   171→  useEffect(() => {
   172→    const el = wrapRef.current
   173→    if (!el) return
   174→    const update = () => {
   175→      const w = el.clientWidth
   176→      const h = Math.max(300, window.innerHeight * 0.58)
   177→      setScale(Math.min(1, Math.max(0.28, Math.min((w - 8) / 794, h / 561))))
   178→    }
   179→    update()
   180→    const ro = new ResizeObserver(update)
   181→    ro.observe(el)
   182→    return () => ro.disconnect()
   183→  }, [])
   184→
   185→  return (
   186→    <div className="flex flex-col items-center w-full">
   187→      {(onPrint || onDownload || onClose) && (
   188→        <div className="flex items-center justify-between w-full mb-2.5">
   189→          <p className="text-[11px] font-semibold text-muted-foreground">Receipt</p>
   190→          <div className="flex items-center gap-1">
   191→            {onPrint && (
   192→              <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1" onClick={onPrint}>
   193→                <Printer className="h-3 w-3" /> Print
   194→              </Button>
   195→            )}
   196→            {onDownload && (
   197→              <Button size="sm" variant="outline" className="h-7 text-[10px] gap-1" onClick={onDownload}>
   198→                <Download className="h-3 w-3" /> Download
   199→              </Button>
   200→            )}
   201→            {onClose && (
   202→              <Button size="sm" variant="ghost" className="h-7 w-7 p-0" onClick={onClose} aria-label="Close receipt preview">
   203→                <X className="h-3.5 w-3.5" />
   204→              </Button>
   205→            )}
   206→          </div>
   207→        </div>
   208→      )}
   209→
   210→      <div ref={wrapRef} className="w-full overflow-hidden" data-testid="a5-receipt-preview">
   211→        {/* Outer box reserves the SCALED size so surrounding layout flows
   212→            correctly; the inner node scales the exact mm-sized sheet. */}
   213→        <div style={{ width: 794 * scale, height: 561 * scale, margin: '0 auto' }}>
   214→          <div style={{ transform: `scale(${scale})`, transformOrigin: 'top left', width: 794, height: 561 }}>
   215→            <A5Sheet transaction={t} settings={settings} ctx={ctx} />
   216→          </div>
   217→        </div>
   218→      </div>
   219→    </div>
   220→  )
   221→}

   225→export function A5Sheet({ transaction: t, settings, ctx }: { transaction: FeeTransaction; settings: ReceiptSettings; ctx?: ReceiptContext }) {
   226→  const student = resolveStudent(t)
   227→  const balanceDue = resolveBalanceDue(t, ctx)
   228→  const lines = buildReceiptLines(t, ctx)
   229→  return (
   230→    <div
   231→      data-testid="a5-receipt-sheet"
   232→      className="bg-white text-slate-950 grid"
   233→      style={{
   234→        width: '210mm',
   235→        height: '148mm',
   236→        padding: '4mm 5mm',
   237→        gridTemplateColumns: '1fr 8px 1fr',
   238→        gridTemplateRows: '1fr',
   239→        gap: '3mm',
   240→        fontFamily: 'ui-sans-serif, system-ui, -apple-system, sans-serif',
   241→        fontSize: '9px',
   242→        lineHeight: 1.25,
   243→      }}
   244→    >
   245→      <ReceiptCopy transaction={t} settings={settings} ctx={{ ...ctx, balanceDue }} student={student} lines={lines} data={copyData(t, settings, 'STUDENT COPY')} />
   246→      <TearLine />
   247→      <ReceiptCopy transaction={t} settings={settings} ctx={{ ...ctx, balanceDue }} student={student} lines={lines} data={copyData(t, settings, 'SCHOOL COPY')} />
   248→    </div>
   249→  )
   250→}

   252→function TearLine() {
   253→  return (
   254→    <div className="flex flex-col items-center justify-center select-none" style={{ borderLeft: '1px dashed #94a3b8' }} aria-hidden>
   255→      <span
   256→        className="whitespace-nowrap font-bold text-slate-400"
   257→        style={{ writingMode: 'vertical-rl', fontSize: '6px', letterSpacing: '0.18em', fontFamily: 'ui-monospace, monospace' }}
   258→      >
   259→        ✂ CUT HERE
   260→      </span>
   261→    </div>
   262→  )
   263→}
   264→
   265→/** Vertical-fit tier (spec: long lists must still fit safely, short lists
   266→ *  must not pool leftover space into one giant gap). The copy lays its
   267→ *  sections out with `justify-content: space-between` so ANY remaining
   268→ *  height distributes EVENLY between sections instead of collecting above
   269→ *  the signatures; the density tiers below shrink the fee table when a
   270→ *  receipt aggregates many heads so it never overflows the 148 mm sheet. */
   271→function fitTier(lineCount: number): 0 | 1 | 2 {
   272→  if (lineCount > 12) return 2
   273→  if (lineCount > 6) return 1
   274→  return 0
   275→}

   277→const TIER = {
   278→  0: { table: 9, cellPad: '3.5px 5px', det: 9.5, detPad: '5px 7px', notes: 7.5, sigPad: 24 },
   279→  1: { table: 8, cellPad: '2.5px 4px', det: 9, detPad: '4px 6px', notes: 7, sigPad: 14 },
   280→  2: { table: 7.5, cellPad: '1.5px 4px', det: 8.5, detPad: '3px 6px', notes: 6.5, sigPad: 10 },
   281→} as const

   283→function ReceiptCopy({
   284→  transaction: t,
   285→  settings,
   286→  ctx,
   287→  student,
   288→  lines,
   289→  data,
   290→}: {
   291→  transaction: FeeTransaction
   292→  settings: ReceiptSettings
   293→  ctx: ReceiptContext
   294→  student: ReturnType<typeof resolveStudent>
   295→  lines: ReceiptLine[]
   296→  data: CopyData
   297→}) {
   298→  const ref = paymentRef(t)
   299→  const md = modeDetail(t)
   300→  const official = t.status === 'Success'
   301→  const balance = ctx.balanceDue
   302→  const tier = TIER[fitTier(lines.length)]
   303→
   304→  return (
   305→    <div
   306→      className="flex flex-col"
   307→      style={{
   308→        border: '1.2px solid #0f172a',
   309→        borderRadius: '3px',
   310→        padding: '7px 9px',
   311→        gap: '4px',
   312→        // THE blank-space fix: sections keep natural heights and any
   313→        // leftover space spreads EVENLY between them — the sheet fills
   314→        // naturally, signatures sit at the bottom, and a short fee list
   315→        // never produces one huge hole above the footer.
   316→        justifyContent: 'space-between',
   317→        background: '#fff',
   318→        minHeight: 0,
   319→        overflow: 'hidden',
   320→      }}
   321→    >
   322→      {/* Header */}
   323→      <div style={{ textAlign: 'center', borderBottom: '1.2px solid #0f172a', paddingBottom: '4px' }}>
   324→        <div className="flex items-center justify-between">
   325→          <div className="text-left font-semibold text-slate-600" style={{ fontSize: '7.5px', lineHeight: 1.25 }}>
   326→            <p>{school.affiliation}</p>
   327→            <p style={{ marginTop: '1px' }}>Code: {school.code}</p>
   328→          </div>
   329→          <div>
   330→            <p className="font-black uppercase" style={{ fontSize: '14px', letterSpacing: '0.04em', lineHeight: 1.1 }}>{school.name}</p>
   331→            <p className="font-medium text-slate-600" style={{ fontSize: '8px', marginTop: '1.5px' }}>
   332→              {school.address} • Ph: {school.phone}
   333→            </p>
   334→          </div>
   335→          <div className="text-right">
   336→            <span
   337→              className="inline-block font-extrabold uppercase text-white"
   338→              style={{ background: '#0f172a', fontSize: '7px', padding: '2px 5px', borderRadius: '2px', letterSpacing: '0.08em' }}
   339→            >
   340→              {data.designation}
   341→            </span>
   342→          </div>
   343→        </div>
   344→      </div>
   345→
   346→      {/* Honest lifecycle state (spec §11) — never present a pending/rejected
   347→          payment as an official receipt. */}
   348→      {!official && (
   349→        <div
   350→          className="font-bold"
   351→          style={{
   352→            fontSize: '8px',
   353→            padding: '3px 6px',
   354→            borderRadius: '2px',
   355→            border: '0.8px solid',
   356→            ...(t.status === 'Failed'
   357→              ? { background: '#fef2f2', borderColor: '#fca5a5', color: '#b91c1c' }
   358→              : { background: '#fffbeb', borderColor: '#fcd34d', color: '#b45309' }),
   359→          }}
   360→        >
   361→          {t.status === 'Failed'
   362→            ? 'PAYMENT NOT COMPLETED — this is not a receipt. No money was recorded.'
   363→            : 'PENDING VERIFICATION — becomes an official receipt once the school office verifies this payment.'}
   364→        </div>
   365→      )}
   366→
   367→      {/* Details */}
   368→      <div
   369→        className="grid font-medium"
   370→        style={{ gridTemplateColumns: '1fr 1fr', background: '#f8fafc', padding: tier.detPad, borderRadius: '2px', border: '0.8px solid #e2e8f0', fontSize: tier.det, columnGap: '10px', rowGap: '1.5px' }}
   371→      >

   376→        </div>
   377→        <div className="text-right">
   378→          <p><span className="text-slate-500 font-normal">Date:</span> <strong>{dateLabel(t.date)}</strong></p>
   379→          <p><span className="text-slate-500 font-normal">Class:</span> <strong>{t.className}{student?.section ? ` - ${student.section}` : ''}</strong></p>
   380→          {(student?.fatherName || student?.guardianName) && (

   434→        <span>Mode: <strong>{t.mode.toUpperCase()}</strong>{md ? ` (${md})` : ''}</span>
   435→        {ref && <span>Txn Ref: <strong className="font-mono">{ref}</strong></span>}
   436→        <span>{data.designation === 'SCHOOL COPY' ? 'Verified By:' : 'Received By:'} <strong>{official ? (t.verifiedBy ?? t.collectedBy) : '—'}</strong></span>
   437→      </div>
   438→
   439→      {/* Notes */}
   440→      <div className="text-slate-500" style={{ fontSize: tier.notes, borderTop: '0.8px solid #e2e8f0', paddingTop: '2.5px', lineHeight: 1.4 }}>
   441→        {data.notes.map((n, i) => (
   442→          <p key={i}>{i + 1}. {n}</p>

   449→      {/* Signatures — the LAST flex child: with space-between they always sit

   451→      <div className="grid text-center" style={{ gridTemplateColumns: '1fr 1fr', columnGap: '20px', fontSize: '8px', paddingTop: '2px' }}>
   452→        <div>
   453→          <p style={{ borderTop: '1px solid #94a3b8', paddingTop: `${tier.sigPad}px`, color: '#475569', fontWeight: 500 }}>Parent / Depositor Sign</p>
   454→        </div>
   455→        {settings.showAuthorizedSignature ? (

   462→      </div>
   463→    </div>

   469→/** Escape user-derived values for safe interpolation into the HTML doc. */
   470→function esc(v: string | number | null | undefined): string {
   471→  if (v == null) return ''
   472→  return String(v)
   473→    .replace(/&/g, '&amp;')
   474→    .replace(/</g, '&lt;')
   475→    .replace(/>/g, '&gt;')
   476→    .replace(/"/g, '&quot;')
   477→}

   699→/** Print ONE payment's receipt. A5 landscape → one student per page;
   700→ *  A4 portrait → the sheet reserves one A5-landscape area for this
   701→ *  student (a second student would fill the lower half when bulk-printed). */
   702→export function printReceiptA5(t: FeeTransaction, settings: ReceiptSettings, ctx?: ReceiptContext) {
   703→  const html = generateReceiptsA5HTML([{ transaction: t, ctx }], settings, { autoPrint: true })
   704→  const w = window.open('', '_blank', 'width=860,height=640')
   705→  if (!w) {
   706→    return false
   707→  }
   708→  w.document.write(html)
   709→  w.document.close()
   710→  return true
   711→}

   713→/** Bulk print — paginated by the paper-size setting: A5 = 1 student per

   720→  if (!w) return false

   723→  return true
