     4→ * TourConfigScreen — the SESSION SETUP screen for the built-in
     5→ * "Educational Tour — Parent Consent Form" (TOUR-1 §2).
     6→ *
     7→ * Replaces the retired form builder: the template's layout is FIXED and
     8→ * cannot be redesigned. The Principal (or an authorized Teacher, whose
     9→ * draft then passes through the existing Principal approval workflow)
    10→ * edits ONLY the information that genuinely changes per session/tour:
    11→ *
    12→ *   tour title · destination · tour dates · duration · tour fee · payment

    14→ *   accompanying staff · short instructions · eligible classes · gender
    15→ *   eligibility · last date to apply
    16→ *
    17→ * The pipeline is explicit and simple: DRAFT → PREVIEW → PUBLISH.
    18→ * A live, properly-scaled A4 preview of the official printed form renders
    19→ * beside the configuration on desktop (and behind a Preview toggle on
    20→ * smaller screens). No developer language, no form-design vocabulary.
    21→ */

    77→function draftOf(app?: SchoolApplication): ConfigDraft {
    78→  const t = APPLICATION_TEMPLATES.educational_tour
    79→  if (!app) {
    80→    return {
    81→      title: '',
    82→      destination: '',
    83→      eventDate: '',
    84→      tourEndDate: '',
    85→      durationDays: '',
    86→      deadline: '',
    87→      circularNo: '',
    88→      circularDate: '',
    89→      inChargeTeacherId: '',
    90→      inChargeName: '',
    91→      accompanyingStaff: '',
    92→      tourInstructions: '',
    93→      targetClassIds: [],
    94→      genderEligibility: 'All',
    95→      paymentAmount: t.defaultAmount,
    96→      paymentAvailability: 'Both',
    97→    }
    98→  }
    99→  return {
   100→    title: app.title,
   101→    destination: app.destination ?? '',
   102→    eventDate: app.eventDate ?? '',
   103→    tourEndDate: app.tourEndDate ?? '',
   104→    durationDays: app.durationDays ?? '',
   105→    deadline: app.deadline,
   106→    circularNo: app.circularNo ?? '',
   107→    circularDate: app.circularDate ?? '',
   108→    inChargeTeacherId: app.inChargeTeacherId ?? '',
   109→    inChargeName: app.inChargeName ?? '',
   110→    accompanyingStaff: app.accompanyingStaff ?? '',
   111→    tourInstructions: app.tourInstructions ?? '',
   112→    targetClassIds: [...app.targetClassIds],
   113→    genderEligibility: app.genderEligibility ?? 'All',
   114→    paymentAmount: app.payment.mode === 'None' ? 0 : app.payment.amount,
   115→    paymentAvailability: app.paymentAvailability ?? 'Both',
   116→  }
   117→}

   531→          <Group group="eligibility" icon={<Users className="h-3.5 w-3.5" />} title="Eligibility">

   628→            <p className="text-[10px] text-muted-foreground">

   630→            </p>

   633→            <p className="rounded-lg border border-sky-200 bg-sky-50 dark:border-sky-500/30 dark:bg-sky-500/10 px-3 py-2 text-[11px] text-sky-700 dark:text-sky-400">

   636→                : 'Approved by the Principal — you can publish it now.'}
   637→            </p>

   639→        </div>
   640→
   641→        {/* ── Right: live A4 preview ── */}
   642→        <div className={cn('lg:col-span-5', mobileView === 'form' && 'hidden md:block')}>
   643→          <div className="lg:sticky lg:top-4">
   644→            <div className="rounded-xl border border-border bg-muted/30 overflow-hidden">
   645→              <div className="flex items-center justify-between px-3 py-2 border-b border-border">
   646→                <p className="text-[11px] font-semibold flex items-center gap-1.5">
   647→                  <Eye className="h-3.5 w-3.5 text-muted-foreground" /> Live A4 preview
   648→                </p>
   649→                <Badge variant="outline" className="text-[9px] h-4 px-1.5">Official school form</Badge>
   650→              </div>
   651→              <div ref={previewRef} className="max-h-[70vh] overflow-auto p-2">
   652→                <div style={{ zoom: previewZoom, width: 'fit-content', margin: '0 auto' }}>
   653→                  <TourFormDocument app={previewApp} />
   654→                </div>
   655→              </div>
   656→              <p className="px-3 py-1.5 text-[9.5px] text-muted-foreground border-t border-border">
   657→                Exactly what prints on A4 paper — students, parents and the office see this document.
   658→              </p>
   659→            </div>
   660→          </div>
   661→        </div>
   662→      </div>
   663→
   664→      {/* Publish confirmation */}
   665→      <AlertDialog open={publishOpen} onOpenChange={setPublishOpen}>
   666→        <AlertDialogContent className="sm:max-w-md">
   667→          <AlertDialogHeader>
   668→            <AlertDialogTitle className="text-base">Publish this tour for students?</AlertDialogTitle>
   669→            <AlertDialogDescription className="text-xs leading-relaxed">
   670→              {draft.destination ? <span>&ldquo;{draft.title}&rdquo; to {draft.destination} </span> : <span>&ldquo;{draft.title}&rdquo; </span>}
   671→              for {draft.targetClassIds.length} class{draft.targetClassIds.length === 1 ? '' : 'es'}.
   672→              {draft.paymentAmount > 0 ? ` A tour fee of ${formatINR(draft.paymentAmount)} will be collected as an application charge.` : ' No fee is charged.'}
   673→              {' '}Eligible students receive a notification and can apply until {draft.deadline}. You can take it down anytime — every submission and payment stays on record.
   674→            </AlertDialogDescription>
   675→          </AlertDialogHeader>
   676→          <AlertDialogFooter>
   677→            <AlertDialogCancel className="h-8 text-xs">Keep as draft</AlertDialogCancel>
   678→            <AlertDialogAction className="h-8 text-xs" onClick={(e) => { e.preventDefault(); publish() }}>
   679→              <CheckCircle2 className="h-3.5 w-3.5" /> Publish
   680→            </AlertDialogAction>
   681→          </AlertDialogFooter>
   682→        </AlertDialogContent>
   683→      </AlertDialog>
   684→    </div>
   685→  )
   686→}
