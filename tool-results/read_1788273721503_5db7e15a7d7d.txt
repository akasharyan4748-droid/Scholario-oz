   172→/** Students scoped by a charge — explicit studentIds win, else the roster of

   174→function scopedStudentsFor(charge: AdditionalCharge, activeStudents: StudentRecord[]): StudentRecord[] {
   175→  const ids = charge.studentIds
   176→  if (ids && ids.length > 0) return activeStudents.filter((s) => ids.includes(s.id))
   177→  return activeStudents.filter((s) => charge.applicableClassIds.includes(s.classId))

   256→export function FeesAdditionalCharges({ data, onCollect }: {
   257→  data: ReturnType<typeof useFeeData>
   258→  /** Opens the Payments page's Collect Fee wizard (records a payment
   259→   *  against a charge through the canonical flow). */
   260→  onCollect?: () => void
   261→}) {
   262→  const { additionalCharges, transactions } = data
   263→  const students = useStudentsStore((s) => s.students)
   264→  const applications = useApplicationsStore((s) => s.applications)
   265→
   266→  const [detailId, setDetailId] = useState<string | null>(null)
   267→  const [wizard, setWizard] = useState<{ open: boolean; editing?: AdditionalCharge } | null>(null)
   268→  const [deleteTarget, setDeleteTarget] = useState<AdditionalCharge | null>(null)
   269→
   270→  const activeStudents = useMemo(() => students.filter((s) => s.status === 'Active'), [students])
   271→
   272→  // Reverse lookup — the form linked to a charge (no drift-prone field):
   273→  // the SchoolApplication whose payment.chargeId === charge.id.
   274→  const appByChargeId = useMemo(() => {
   275→    const map = new Map<string, SchoolApplication>()
   276→    for (const app of applications) {
   277→      const cid = app.payment.chargeId
   278→      if (cid && !map.has(cid)) map.set(cid, app)
   279→    }
   280→    return map
   281→  }, [applications])
   282→
   283→  // Live roll-up per charge from bound transactions.
   284→  const facts = useMemo(() => {
   285→    const map = new Map<string, ChargeFacts>()
   286→    for (const c of additionalCharges) {
   287→      const scoped = scopedStudentsFor(c, activeStudents)
   288→      const txns = transactions.filter((t) => t.additionalChargeId === c.id)
   289→      const collected = txns
   290→        .filter((t) => t.status === 'Success' || t.status === 'Under Verification')
   291→        .reduce((sum, t) => sum + t.amount, 0)
   292→      const custom = c.allowCustomAmount === true
   293→      const expected = custom ? (c.targetAmount ?? 0) : scoped.length * c.amount
   294→      const pct = expected > 0 ? Math.min(100, Math.round((collected / expected) * 100)) : null
   295→      map.set(c.id, { students: scoped.length, collected, expected, pct, txns })
   296→    }
   297→    return map
   298→  }, [additionalCharges, transactions, activeStudents])
   299→
   300→  // Actionable surface first: drafts + open collections, then closed, then
   301→  // archived/cancelled (permanent record tail).
   302→  // "Application payments" filter (spec §4): only collections linked to an
   303→  // Applications & Forms form — never standalone collections.
   304→  const [scopeFilter, setScopeFilter] = useState<'all' | 'application'>('all')
   305→  const orderedCharges = useMemo(() => {
   306→    const rank = (c: AdditionalCharge) =>
   307→      c.status === 'Draft' ? 0 : c.status === 'Active' ? 1 : c.status === 'Closed' ? 2 : 3
   308→    return [...additionalCharges]
   309→      .filter((c) => (scopeFilter === 'application' ? appByChargeId.has(c.id) : true))
   310→      .sort((a, b) => rank(a) - rank(b) || a.dueDate.localeCompare(b.dueDate))
   311→  }, [additionalCharges, scopeFilter, appByChargeId])
   312→
   313→  const detailCharge = detailId ? additionalCharges.find((c) => c.id === detailId) ?? null : null
   314→
   315→  // ── Store actions (row + drawer + wizard all funnel through these) ──
   316→  const publishCharge = (c: AdditionalCharge) => {
   317→    const r = useFeeStore.getState().publishAdditionalCharge(c.id, ACTOR)
   318→    if (r.success) {
   319→      toast.success('Collection published', {
   320→        description: `"${c.name}" is now an obligation for its ${scopedStudentsFor(c, activeStudents).length} scoped student(s).`,
   321→      })
   322→    } else {
   323→      toast.error('Could not publish', { description: r.error })
   324→    }
   325→  }
   326→  const closeCharge = (c: AdditionalCharge) => {
   327→    const r = useFeeStore.getState().closeAdditionalCharge(c.id, ACTOR)
   328→    if (r.success) toast.success('Collection closed', { description: 'Payment history stays on record.' })
   329→    else toast.error('Could not close', { description: r.error })
   330→  }
   331→  const archiveCharge = (c: AdditionalCharge) => {
   332→    const r = useFeeStore.getState().archiveAdditionalCharge(c.id, ACTOR)
   333→    if (r.success) toast.info('Archived', { description: 'Historical record — payments remain readable.' })
   334→    else toast.error('Could not archive', { description: r.error })
   335→  }
   336→  const deleteCharge = (c: AdditionalCharge) => {
   337→    // UI-level guard: a form (any state) explicitly links to this draft →
   338→    // deleting would orphan the form's payment path.
   339→    const linkedApp = appByChargeId.get(c.id)
   340→    if (linkedApp) {
   341→      toast.error('Cannot delete this draft', {
   342→        description: `The form "${linkedApp.title}" is linked to it. Delete that form draft first (Applications & Forms).`,
   343→      })
   344→      return
   345→    }
   346→    const r = useFeeStore.getState().deleteAdditionalCharge(c.id, ACTOR)
   347→    if (r.success) {
   348→      toast.success('Draft deleted', { description: `"${c.name}" was removed permanently.` })
   349→      if (detailId === c.id) setDetailId(null)
   350→    } else {
   351→      toast.error('Could not delete', { description: r.error })
   352→    }
   353→  }
   354→
   355→  return (
   356→    <>
   357→      <Panel
   358→        title={
   359→          <span className="inline-flex items-center gap-2">
   360→            Additional Collections
   361→            <Badge variant="outline" className="text-[9px] h-4 px-1.5">
   362→              {orderedCharges.filter((c) => c.status === 'Active' || c.status === 'Draft').length}
   363→            </Badge>
   364→          </span>
   365→        }
   366→        subtitle="Money collections outside the regular fee structure — tours, events, workshops and special charges."
   367→        action={
   368→          <div className="inline-flex items-center gap-1.5">
   369→            <div className="inline-flex h-7 items-center rounded-full bg-muted/70 p-0.5" role="group" aria-label="Collection scope">
   370→              {([['all', 'All collections'], ['application', 'Application payments']] as const).map(([v, label]) => (
   371→                <button
   372→                  key={v}
   373→                  type="button"
   374→                  onClick={() => setScopeFilter(v)}
   375→                  aria-pressed={scopeFilter === v}
   376→                  className={cn(
   377→                    'px-2.5 h-6 rounded-full text-[10.5px] font-medium whitespace-nowrap transition-all',
   378→                    scopeFilter === v ? 'bg-white dark:bg-white/10 shadow-sm text-foreground' : 'text-muted-foreground hover:text-foreground',
   379→                  )}
   380→                >
   381→                  {label}
   382→                </button>
   383→              ))}
   384→            </div>
   385→            <Button
   386→              variant="outline"
   387→              size="sm"
   388→              className="h-7 text-[11px] gap-1 shrink-0"
   389→              onClick={() => setWizard({ open: true })}
   390→              aria-label="Create a new collection"
   391→            >
   392→              <Plus className="h-3 w-3" /> New Collection
   393→            </Button>
   394→          </div>
   395→        }
   396→        bodyClassName="p-0"
   397→      >
   398→        {orderedCharges.length > 0 ? (
   399→          <div className="divide-y divide-border">
   400→            {orderedCharges.map((c) => {
   401→              const f = facts.get(c.id) ?? EMPTY_FACTS
   402→              const app = appByChargeId.get(c.id)
   403→              return (
   404→                <motion.div key={c.id} initial={{ opacity: 0, y: 4 }} animate={{ opacity: 1, y: 0 }}>
   405→                  <CollectionRow
   406→                    charge={c}
   407→                    facts={f}
   408→                    app={app}
   409→                    hasLinkedForm={appByChargeId.has(c.id)}
   410→                    onOpen={() => setDetailId(c.id)}
   411→                    onEdit={() => setWizard({ open: true, editing: c })}
   412→                    onPublish={() => publishCharge(c)}
   413→                    onClose={() => closeCharge(c)}
   414→                    onArchive={() => archiveCharge(c)}
   415→                    onDelete={() => setDeleteTarget(c)}
   416→                    onCollect={onCollect}
   417→                  />
   418→                </motion.div>
   419→              )
   420→            })}
   421→          </div>
   422→        ) : (
   423→          <FeeEmptyState
   424→            icon={<CircleDollarSign className="h-6 w-6" />}
   425→            title={scopeFilter === 'application' ? 'No application payments yet' : 'No additional collections yet'}
   426→            description={scopeFilter === 'application'
   427→              ? 'Collections linked to a published form appear here once a paid form goes live.'
   428→              : 'Create a collection for a tour, event, workshop, activity or any special school charge.'}
   429→            action={scopeFilter === 'all' ? (
   430→              <Button variant="outline" size="sm" className="h-7 text-[11px] gap-1" onClick={() => setWizard({ open: true })}>
   431→                <Plus className="h-3 w-3" /> New Collection
   432→              </Button>
   433→            ) : undefined}
   434→          />
   435→        )}
   436→      </Panel>
   437→
   438→      {/* Creation / edit wizard */}
   439→      {wizard?.open && (
   440→        <CollectionWizardDialog
   441→          editing={wizard.editing}
   442→          activeStudents={activeStudents}
   443→          onClose={() => setWizard(null)}
   444→        />
   445→      )}
   446→
   447→      {/* Delete-draft confirmation (§17 — honest, non-technical language) */}
   448→      {deleteTarget && (
   449→        <AlertDialog open onOpenChange={(o) => { if (!o) setDeleteTarget(null) }}>
   450→          <AlertDialogContent className="max-w-sm">
   451→            <AlertDialogHeader>
   452→              <AlertDialogTitle>Delete draft collection?</AlertDialogTitle>
   453→              <AlertDialogDescription>
   454→                {appByChargeId.has(deleteTarget.id)
   455→                  ? `The form "${appByChargeId.get(deleteTarget.id)!.title}" is linked to this collection — delete that form draft first.`
   456→                  : `"${deleteTarget.name}" has not been published and has no payments on record. It will be permanently removed.`}
   457→              </AlertDialogDescription>
   458→            </AlertDialogHeader>
   459→            <AlertDialogFooter>
   460→              <AlertDialogCancel>Cancel</AlertDialogCancel>
   461→              <AlertDialogAction
   462→                className="bg-rose-600 hover:bg-rose-700 text-white"
   463→                disabled={appByChargeId.has(deleteTarget.id)}
   464→                onClick={() => { deleteCharge(deleteTarget); setDeleteTarget(null) }}
   465→              >
   466→                Delete Draft
   467→              </AlertDialogAction>
   468→            </AlertDialogFooter>
   469→          </AlertDialogContent>
   470→        </AlertDialog>
   471→      )}
   472→
   473→      {/* Collection status drawer */}
   474→      {detailCharge && (
   475→        <CollectionStatusSheet
   476→          charge={detailCharge}
   477→          facts={facts.get(detailCharge.id) ?? EMPTY_FACTS}
   478→          app={appByChargeId.get(detailCharge.id)}
   479→          activeStudents={activeStudents}
   480→          onClose={() => setDetailId(null)}
   481→          onEdit={() => { setDetailId(null); setWizard({ open: true, editing: detailCharge }) }}
   482→          onPublish={() => publishCharge(detailCharge)}
   483→          onCloseCollection={() => closeCharge(detailCharge)}
   484→          onArchive={() => archiveCharge(detailCharge)}
   485→          onDelete={() => setDeleteTarget(detailCharge)}
   486→          onCollect={onCollect}
   487→        />
   488→      )}
   489→    </>
   490→  )
   491→}

   495→function CollectionRow({
   496→  charge, facts, app, hasLinkedForm, onOpen, onEdit, onPublish, onClose, onArchive, onDelete, onCollect,
   497→}: {
   498→  charge: AdditionalCharge
   499→  facts: ChargeFacts
   500→  app?: SchoolApplication
   501→  hasLinkedForm: boolean
   502→  onOpen: () => void
   503→  onEdit: () => void
   504→  onPublish: () => void
   505→  onClose: () => void
   506→  onArchive: () => void
   507→  onDelete: () => void
   508→  onCollect?: () => void
   509→}) {

   658→function CollectionWizardDialog({
   659→  editing, activeStudents, onClose,
   660→}: {
   661→  editing?: AdditionalCharge
   662→  activeStudents: StudentRecord[]
   663→  onClose: () => void
   664→}) {
   665→  const createAdditionalCharge = useFeeStore((s) => s.createAdditionalCharge)
   666→  const updateAdditionalCharge = useFeeStore((s) => s.updateAdditionalCharge)
   667→
   668→  const [w, setW] = useState<WizardState>(() => editing ? wizardFromCharge(editing) : freshWizard())
   669→  const patch = (p: Partial<WizardState>) => setW((f) => ({ ...f, ...p }))
   670→
   671→  const isDraft = editing?.status === 'Draft' || !editing
   672→  const isEditActive = editing?.status === 'Active'
   673→  // Money lock: amount frozen once payments exist (the store enforces too).
   674→  const boundPayments = useFeeStore((s) => (editing ? s.transactions.filter((t) => t.additionalChargeId === editing.id).length : 0))
   675→  const amountLocked = isEditActive && boundPayments > 0
   676→
   677→  const toggleClass = (id: string) => {
   678→    const next = w.applicableClassIds.includes(id)
   679→      ? w.applicableClassIds.filter((c) => c !== id)
   680→      : [...w.applicableClassIds, id]
   681→    patch({ applicableClassIds: next })
   682→  }
   683→  const toggleStudent = (id: string) => {
   684→    const next = w.studentIds.includes(id)
   685→      ? w.studentIds.filter((s) => s !== id)
   686→      : [...w.studentIds, id]
   687→    patch({ studentIds: next })
   688→  }
   689→
   690→  const candidates = useMemo(
   691→    () => wizardCandidateStudents({ applicableClassIds: w.applicableClassIds }, activeStudents),
   692→    [w.applicableClassIds, activeStudents],
   693→  )
   694→  const scopedCount = w.studentIds.length
   695→    ? w.studentIds.length
   696→    : activeStudents.filter((s) => w.applicableClassIds.includes(s.classId)).length
   697→
   698→  const handleSave = (publish: boolean) => {
   699→    if (!w.name.trim()) { toast.error('Give this collection a name.'); return }
   700→    if (!w.customAmount && !(w.amount > 0)) { toast.error('Enter the per-student amount.'); return }
   701→    if (w.customAmount && w.targetAmount > 0 && !(w.amount >= 0)) { toast.error('The suggested amount cannot be negative.'); return }
   702→    if (!publish && !w.applicableClassIds.length && !w.studentIds.length) {
   703→      // Draft without scope is allowed (Principal still picking targets).
   704→    } else if (publish && !w.applicableClassIds.length && !w.studentIds.length) {
   705→      toast.error('Select at least one class (or specific students) before publishing.')
   706→      return
   707→    }
   708→    if (!w.dueDate) { toast.error('Set a payment due date.'); return }
   709→
   710→    if (editing) {
   711→      const r = updateAdditionalCharge(editing.id, {
   712→        name: w.name,
   713→        category: w.category,
   714→        description: w.description,
   715→        amount: w.amount,
   716→        applicableClassIds: w.applicableClassIds,
   717→        studentIds: w.studentIds,
   718→        dueDate: w.dueDate,
   719→        startDate: w.startDate,
   720→        mandatory: w.mandatory,
   721→        allowCustomAmount: w.customAmount,
   722→        targetAmount: w.targetAmount,
   723→        instructions: w.instructions,
   724→      }, ACTOR)
   725→      if (!r.success) { toast.error('Could not save changes', { description: r.error }); return }
   726→      if (publish && editing.status === 'Draft') {
   727→        const p = useFeeStore.getState().publishAdditionalCharge(editing.id, ACTOR)
   728→        if (!p.success) { toast.error('Saved — but publishing failed', { description: p.error }); onClose(); return }
   729→        toast.success('Collection published', { description: `"${w.name}" is now collecting.` })
   730→      } else {
   731→        toast.success('Changes saved')
   732→      }
   733→      onClose()
   734→      return
   735→    }
   736→    const r = createAdditionalCharge({
   737→      name: w.name,
   738→      category: w.category,
   739→      amount: w.amount,
   740→      academicYear: CURRENT_YEAR,
   741→      applicableClassIds: w.applicableClassIds,
   742→      studentIds: w.studentIds.length ? w.studentIds : undefined,
   743→      dueDate: w.dueDate,
   744→      startDate: w.startDate || undefined,
   745→      mandatory: w.mandatory,
   746→      description: w.description || undefined,
   747→      instructions: w.instructions || undefined,
   748→      allowCustomAmount: w.customAmount || undefined,
   749→      targetAmount: w.customAmount && w.targetAmount > 0 ? w.targetAmount : undefined,
   750→      status: publish ? 'Active' : 'Draft',
   751→      actor: ACTOR,
   752→    })
   753→    if (!r.success || !r.charge) { toast.error('Could not create the collection', { description: r.error }); return }
   754→    toast.success(publish ? 'Collection published' : 'Draft collection created', {
   755→      description: publish
   756→        ? `"${w.name}" is now an obligation for its scoped students.`
   757→        : `"${w.name}" is saved as a draft — publish it when you are ready.`,
   758→    })
   759→    onClose()
   760→  }

   769→              ? isEditActive
   770→                ? 'Safe details only — name, amount and terms are locked while the collection is live.'
   771→                : 'Draft collections are fully editable and can be deleted until published.'
   772→              : 'A special school charge — tour, event, workshop, fund or material. No form is required.'}
   773→          </DialogDescription>
   774→        </DialogHeader>
   775→
   776→        <div className="space-y-4">
   777→          {/* Identity */}
   778→          <div className="grid gap-3">
   779→            <div className="grid gap-1.5">
   780→              <Label htmlFor="coll-name" className="text-xs">Collection name</Label>
   781→              <Input
   782→                id="coll-name"
   783→                className="h-8 text-xs"
   784→                placeholder="e.g. Educational Tour — Jaipur"
   785→                value={w.name}
   786→                disabled={isEditActive}
   787→                onChange={(e) => patch({ name: e.target.value })}

  1053→
  1054→  const paidCount = scoped.filter((s) => {

  1059→

  1077→  if (charge.status === 'Closed') {

  1086→  if (charge.status === 'Archived') {

  1092→

  1111→              <CategoryIcon category={charge.category} className="h-4 w-4" />

  1114→              <div className="flex items-center gap-1.5 min-w-0">

  1116→                <StatusChip status={charge.status} />
  1117→              </div>
  1118→              <SheetDescription className="mt-0.5 truncate text-[10px] text-muted-foreground">
