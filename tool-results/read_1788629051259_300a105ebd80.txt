     1→/**
     2→ * Messaging store — Zustand store for the Messages & Inbox module.
     3→ *
     4→ * One connected messaging system:
     5→ *   Conversations → Messages → Send/Reply → Star/Archive/Draft
     6→ *   Groups → Group Members → Group Conversation → Send to Group
     7→ *
     8→ * Recipient data comes from canonical Teachers + Students (for parents).
     9→ * No fake "online" status — we use role/department labels instead.
    10→ * No fake "read receipts" or "typing" indicators.
    11→ *
    12→ * Folders: Inbox · Starred · Sent · Groups · Drafts · Archive
    13→ * Labels: Staff · Parents · Groups · Urgent (all functional filters)
    14→ *
    15→ * State mutations:
    16→ *   - sendMessage (creates/replies to conversation; group replies use a random member name)
    17→ *   - markRead (clears unread on open)
    18→ *   - starConversation / unstarConversation
    19→ *   - archiveConversation / unarchiveConversation
    20→ *   - saveDraft / deleteDraft / sendDraft
    21→ *   - markUrgent
    22→ *   - composeNew (recipient picker)
    23→ *   - createGroup / addMember / removeMember (group management)
    24→ */

   126→/** Resolve a single member ref into a display object. Returns null if not found. */
   127→export function resolveMemberRef(ref: string): MemberDisplay | null {
   128→  if (ref.startsWith('t:')) {
   129→    const id = ref.slice(2)
   130→    const t = teachers.find((x) => x.id === id)
   131→    if (!t || t.archived) return null
   132→    return {
   133→      ref,
   134→      type: 'teacher',
   135→      name: t.name,
   136→      avatar: t.avatar,
   137→      role: `${t.designation} · ${t.department}`,
   138→    }
   139→  }
   140→  if (ref.startsWith('p:')) {
   141→    const sid = ref.slice(2)
   142→    const s = useStudentsStore.getState().students.find((x) => x.id === sid)
   143→    if (!s || s.status !== 'Active') return null
   144→    const avatar = s.fatherName.split(' ').map((n) => n[0]).slice(0, 2).join('') || 'P'
   145→    return {
   146→      ref,
   147→      type: 'parent',
   148→      name: s.fatherName,
   149→      avatar,
   150→      role: `Parent · ${s.name} (${s.className}-${s.section})`,
   151→    }
   152→  }
   153→  return null
   154→}
   155→
   156→/** Resolve a list of member refs into display objects (skips missing). */
   157→export function resolveMemberRefs(refs: string[]): MemberDisplay[] {
   158→  return refs.map(resolveMemberRef).filter((x): x is MemberDisplay => x !== null)
   159→}

   186→function formatTimeAgo(iso: string): string {
   187→  const diff = Date.now() - new Date(iso).getTime()
   188→  const min = Math.floor(diff / 60000)
   189→  if (min < 1) return 'just now'
   190→  if (min < 60) return `${min} min ago`
   191→  const hr = Math.floor(min / 60)
   192→  if (hr < 24) return `${hr} hr ago`
   193→  const day = Math.floor(hr / 24)
   194→  if (day === 1) return 'yesterday'
   195→  if (day < 7) return `${day} days ago`
   196→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short' })
   197→}

   199→function formatMessageTime(iso: string): string {
   200→  return new Date(iso).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: true })
   201→}

   374→export const useMessagingStore = create<MessagingState>((set, get) => ({
   375→  conversations: SEED_CONVERSATIONS,
   376→  messages: SEED_MESSAGES,
   377→  drafts: SEED_DRAFTS,
   378→  groups: SEED_GROUPS,
   379→  activeConversationId: 'C01',
   380→  activeFolder: 'inbox',
   381→  activeLabel: null,
   382→  searchQuery: '',
   383→
   384→  setActiveFolder: (folder) => set({ activeFolder: folder, activeLabel: null }),
   385→  setActiveLabel: (label) => set({ activeLabel: label }),
   386→  setSearchQuery: (query) => set({ searchQuery: query }),
   387→
   388→  openConversation: (id) => {
   389→    const state = get()
   390→    const convo = state.conversations.find((c) => c.id === id)
   391→    if (!convo) return
   392→    set({
   393→      activeConversationId: id,
   394→      conversations: state.conversations.map((c) => c.id === id ? { ...c, unread: 0 } : c),
   395→    })
   396→  },
   397→
   398→  sendMessage: (conversationId, text) => {
   399→    const state = get()
   400→    if (!text.trim()) return
   401→    const newMsg: Message = {
   402→      id: `M${Date.now()}`,
   403→      conversationId,
   404→      sender: 'me',
   405→      text: text.trim(),
   406→      timestamp: new Date().toISOString(),
   407→      status: 'sent',
   408→    }
   409→    const existingMessages = state.messages[conversationId] ?? []
   410→    set({
   411→      messages: {
   412→        ...state.messages,
   413→        [conversationId]: [...existingMessages, newMsg],
   414→      },
   415→      conversations: state.conversations.map((c) =>
   416→        c.id === conversationId

   726→  getFilteredConversations: () => {
   727→    const state = get()
   728→    let result = state.conversations
   729→
   730→    // Folder filter
   731→    switch (state.activeFolder) {
   732→      case 'inbox':
   733→        result = result.filter((c) => !c.archived)
   734→        break
   735→      case 'starred':
   736→        result = result.filter((c) => c.starred && !c.archived)
   737→        break
   738→      case 'sent':
   739→        // Conversations where the last message was sent by me
   740→        result = result.filter((c) => {
   741→          const msgs = state.messages[c.id] ?? []
   742→          return msgs.length > 0 && msgs[msgs.length - 1].sender === 'me' && !c.archived
   743→        })
   744→        break
   745→      case 'groups':
   746→        result = result.filter((c) => c.type === 'group' && !c.archived)
   747→        break
   748→      case 'drafts':
   749→        // Conversations that have an associated draft
   750→        const draftConvIds = new Set(state.drafts.filter((d) => d.conversationId).map((d) => d.conversationId!))
   751→        result = result.filter((c) => draftConvIds.has(c.id) && !c.archived)
   752→        break
   753→      case 'archive':
   754→        result = result.filter((c) => c.archived)
   755→        break
   756→    }
   757→
   758→    // Label filter
   759→    if (state.activeLabel) {
   760→      switch (state.activeLabel) {
   761→        case 'Staff': result = result.filter((c) => c.type === 'staff'); break
   762→        case 'Parents': result = result.filter((c) => c.type === 'parent'); break
   763→        case 'Groups': result = result.filter((c) => c.type === 'group'); break
   764→        case 'Urgent': result = result.filter((c) => c.urgent); break
   765→      }
   766→    }
   767→
   768→    // Search filter
   769→    if (state.searchQuery.trim()) {
   770→      const q = state.searchQuery.toLowerCase()
   771→      result = result.filter((c) => {
   772→        if (c.name.toLowerCase().includes(q)) return true
   773→        if (c.lastMessage.toLowerCase().includes(q)) return true
   774→        // Search in message content
   775→        const msgs = state.messages[c.id] ?? []
   776→        return msgs.some((m) => m.text.toLowerCase().includes(q))
   777→      })
   778→    }
   779→
   780→    // Sort: starred first, then by latest activity
   781→    return result.sort((a, b) => {
   782→      if (a.starred && !b.starred) return -1
   783→      if (!a.starred && b.starred) return 1
   784→      return new Date(b.lastTimestamp).getTime() - new Date(a.lastTimestamp).getTime()
   785→    })
   786→  },
   787→
   788→  getUnreadCount: () => {
   789→    return get().conversations.filter((c) => !c.archived).reduce((sum, c) => sum + c.unread, 0)
   790→  },
