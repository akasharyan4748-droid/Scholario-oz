    30→    30→function todayISO(): string {
    31→    31→  const d = new Date()
    32→    32→  const y = d.getFullYear()
    33→    33→  const m = String(d.getMonth() + 1).padStart(2, '0')
    34→    34→  const day = String(d.getDate()).padStart(2, '0')
    35→    35→  return `${y}-${m}-${day}`
    36→    36→}
    37→    37→
    38→    38→function daysFromNow(days: number, hour = 19, minute = 0): string {
    39→    39→  const d = new Date()
    40→    40→  d.setDate(d.getDate() + days)
    41→    41→  d.setHours(hour, minute, 0, 0)
    42→    42→  return d.toISOString()
    43→    43→}
    44→    44→
    45→    45→function dateOnlyShift(days: number): string {
    46→    46→  const d = new Date()
    47→    47→  d.setDate(d.getDate() + days)
    48→    48→  const y = d.getFullYear()
    49→    49→  const m = String(d.getMonth() + 1).padStart(2, '0')
    50→    50→  const day = String(d.getDate()).padStart(2, '0')
    51→    51→  return `${y}-${m}-${day}`
    52→    52→}

   659→   659→export function dueStatsOf(cards: LearningCard[], at: Date = new Date()): DueStats {
   660→   660→  let due = 0
   661→   661→  let fresh = 0
   662→   662→  let learning = 0
   663→   663→  let reviewing = 0
   664→   664→  let mastered = 0
   665→   665→  let newCards = 0
   666→   666→  for (const c of cards) {
   667→   667→    if (c.status === 'mastered') mastered += 1
   668→   668→    else if (c.status === 'learning') learning += 1
   669→   669→    else if (c.status === 'reviewing') reviewing += 1
   670→   670→    else newCards += 1
   671→   671→    const isDue = c.status !== 'mastered' && (c.dueAt === null || new Date(c.dueAt).getTime() <= at.getTime())
   672→   672→    if (isDue) due += 1
   673→   673→    if (c.dueAt === null && c.status === 'new') fresh += 1
   674→   674→  }
   675→   675→  return { due, fresh, learning, reviewing, mastered, newCards, total: cards.length }
   676→   676→}

   678→   678→/** The review queue: due cards, new-first-later ordering (learning first). */
   679→   679→export function reviewQueueOf(cards: LearningCard[], at: Date = new Date()): LearningCard[] {
   680→   680→  return cards
   681→   681→    .filter((c) => c.status !== 'mastered' && (c.dueAt === null || new Date(c.dueAt).getTime() <= at.getTime()))
   682→   682→    .sort((a, b) => {
   683→   683→      const rank = (c: LearningCard) => (c.status === 'learning' ? 0 : c.status === 'reviewing' ? 1 : 2)
   684→   684→      const ra = rank(a)
   685→   685→      const rb = rank(b)
   686→   686→      if (ra !== rb) return ra - rb
   687→   687→      const da = a.dueAt ? new Date(a.dueAt).getTime() : Number.MAX_SAFE_INTEGER
   688→   688→      const db = b.dueAt ? new Date(b.dueAt).getTime() : Number.MAX_SAFE_INTEGER
   689→   689→      return da - db
   690→   690→    })
   691→   691→}

   695→   695→  resourcesTotal: number
   696→   696→  resourcesCompleted: number
   697→   697→  resourcesInProgress: number
   698→   698→  cardsTotal: number
   699→   699→  cardsMastered: number
   700→   700→  /** Derived 0–100: completed resources + mastered cards, honest blend. */
   701→   701→  masteryPct: number
   702→   702→}
   703→   703→
   704→   704→export function subjectStatsOf(
   705→   705→  resources: LearningResource[],
   706→   706→  progress: Record<string, ResourceProgress>,
   707→   707→  cards: LearningCard[],
   708→   708→): SubjectStat[] {
   709→   709→  const subjects = Array.from(
   710→   710→    new Set([...resources.map((r) => r.subject), ...cards.map((c) => c.subject)]),
   711→   711→  ).filter(Boolean)
   712→   712→  return subjects.map((subject) => {
   713→   713→    const res = resources.filter((r) => r.subject === subject)
   714→   714→    const completed = res.filter((r) => (progress[r.id]?.pct ?? 0) >= 100).length
   715→   715→    const inProgress = res.filter((r) => {
   716→   716→      const p = progress[r.id]?.pct ?? 0
   717→   717→      return p > 0 && p < 100
   718→   718→    }).length
   719→   719→    const cardsForSubject = cards.filter((c) => c.subject === subject)
   720→   720→    const mastered = cardsForSubject.filter((c) => c.status === 'mastered').length
   721→   721→    // Blend: resources count toward half, mastered cards the other half —
   722→   722→    // only over what actually exists (a subject with no cards leans on
   723→   723→    // resources and vice-versa).
   724→   724→    const resPart = res.length > 0 ? completed / res.length : null
   725→   725→    const cardPart = cardsForSubject.length > 0 ? mastered / cardsForSubject.length : null
   726→   726→    const parts = [resPart, cardPart].filter((p): p is number => p !== null)
   727→   727→    const masteryPct = parts.length > 0 ? Math.round((parts.reduce((a, b) => a + b, 0) / parts.length) * 100) : 0
   728→   728→    return {
   729→   729→      subject,
   730→   730→      resourcesTotal: res.length,
   731→   731→      resourcesCompleted: completed,
   732→   732→      resourcesInProgress: inProgress,
   733→   733→      cardsTotal: cardsForSubject.length,
   734→   734→      cardsMastered: mastered,
   735→   735→      masteryPct,
   736→   736→    }
   737→   737→  })
   738→   738→}

   761→   761→/** ISO weekday key list of the last 7 days (oldest first), like ['Mon', …]. */
   762→   762→export function last7DayKeys(at: Date = new Date()): Array<{ key: string; label: string }> {
   763→   763→  const out: Array<{ key: string; label: string }> = []
   764→   764→  for (let i = 6; i >= 0; i--) {
   765→   765→    const d = new Date(at)
   766→   766→    d.setDate(d.getDate() - i)
   767→   767→    out.push({ key: dateKeyOf(d.toISOString()), label: d.toLocaleDateString('en-IN', { weekday: 'short' }) })
   768→   768→  }
   769→   769→  return out

   782→   782→export interface StreakInfo {
   783→   783→  current: number
   784→   784→  longest: number
   785→   785→  /** true when at least one qualifying session happened TODAY. */
   786→   786→  activeToday: boolean
   787→   787→}
   788→   788→
   789→   789→/**
   790→   790→ * A study streak day = at least one recorded study session that day
   791→   791→ * (spec §35 — opening the page never counts).
   792→   792→ */
   793→   793→export function streakOf(sessions: StudySession[], at: Date = new Date()): StreakInfo {
   794→   794→  const days = new Set(sessions.map((s) => dateKeyOf(s.endedAt)))
   795→   795→  if (days.size === 0) return { current: 0, longest: 0, activeToday: false }
   796→   796→
   797→   797→  // Longest run over the recorded history.
   798→   798→  const sorted = Array.from(days).sort()
   799→   799→  let longest = 1
   800→   800→  let run = 1
   801→   801→  for (let i = 1; i < sorted.length; i++) {
   802→   802→    const prev = new Date(sorted[i - 1] + 'T00:00:00')
   803→   803→    const curr = new Date(sorted[i] + 'T00:00:00')
   804→   804→    const diff = Math.round((curr.getTime() - prev.getTime()) / 86400000)
   805→   805→    run = diff === 1 ? run + 1 : 1
   806→   806→    longest = Math.max(longest, run)
   807→   807→  }
   808→   808→
   809→   809→  // Current run: counts back from today (or yesterday if today is a
   810→   810→  // rest day so far — the streak isn't "broken" until the day ends).
   811→   811→  const todayKey = dateKeyOf(at.toISOString())
   812→   812→  const yKey = new Date(at)
   813→   813→  yKey.setDate(yKey.getDate() - 1)
   814→   814→  const yesterdayKey = dateKeyOf(yKey.toISOString())
   815→   815→  let cursor: Date
   816→   816→  if (days.has(todayKey)) cursor = new Date(todayKey + 'T00:00:00')
   817→   817→  else if (days.has(yesterdayKey)) cursor = new Date(yesterdayKey + 'T00:00:00')
   818→   818→  else return { current: 0, longest, activeToday: false }
   819→   819→  let current = 0
   820→   820→  while (days.has(dateKeyOf(cursor.toISOString()))) {
   821→   821→    current += 1
   822→   822→    cursor.setDate(cursor.getDate() - 1)
   823→   823→  }
   824→   824→  return { current, longest, activeToday: days.has(todayKey) }
   825→   825→}
   826→   826→
   827→   827→/**
   828→   828→ * RECOMMENDED — derived only from real signals (spec §14):
   829→   829→ *   1. unfinished resources in the student's weakest subjects
   830→   830→ *      (from the canonical results store — lowest latest %),
   831→   831→ *   2. other unfinished / in-progress resources,
   832→   832→ *   3. bookmarked-but-not-started.
   833→   833→ * `weakSubjects` is passed in by the caller (results integration,
   834→   834→ * permission-safe: only the student's own subject percentages).
   835→   835→ */
   836→   836→export function recommendedOf(
   837→   837→  resources: LearningResource[],
   838→   838→  progress: Record<string, ResourceProgress>,
   839→   839→  bookmarks: string[],
   840→   840→  weakSubjects: string[],
   841→   841→  limit = 3,
   842→   842→): LearningResource[] {
   843→   843→  const weakRank = new Map(weakSubjects.map((s, i) => [s, i]))
   844→   844→  const score = (r: LearningResource): number => {
   845→   845→    const pct = progress[r.id]?.pct ?? 0
   846→   846→    let s = 0
   847→   847→    if (weakRank.has(r.subject)) s -= 100 - (weakRank.get(r.subject) ?? 0) * 10
   848→   848→    if (pct > 0 && pct < 100) s -= 20 // in progress → finish it
   849→   849→    if (bookmarks.includes(r.id)) s -= 5
   850→   850→    if (pct >= 100) s += 1000 // completed → last
   851→   851→    return s
   852→   852→  }
   853→   853→  return [...resources].sort((a, b) => score(a) - score(b)).slice(0, limit)
   854→   854→}

   856→   856→/** Goal progress (current value + target) derived from real state. */
   857→   857→export interface GoalProgress {
   858→   858→  goal: StudyGoal
   859→   859→  current: number
   860→   860→  pct: number
   861→   861→  unit: string
   862→   862→}
   863→   863→
   864→   864→export function goalProgressOf(goal: StudyGoal, ctx: { sessions: StudySession[]; tasks: PlannerTask[]; cards: LearningCard[] }): GoalProgress {
   865→   865→  switch (goal.kind) {
   866→   866→    case 'weekly-minutes': {
   867→   867→      const current = weeklyMinutes(ctx.sessions)
   868→   868→      return { goal, current, pct: Math.min(100, Math.round((current / goal.target) * 100)), unit: 'min' }
   869→   869→    }
   870→   870→    case 'weekly-tasks': {
   871→   871→      const current = weeklyCompletedTasks(ctx.tasks)
   872→   872→      return { goal, current, pct: Math.min(100, Math.round((current / goal.target) * 100)), unit: 'tasks' }
   873→   873→    }
   874→   874→    case 'deck-master': {
   875→   875→      const cards = ctx.cards.filter((c) => c.subject === goal.subject)
   876→   876→      const mastered = cards.filter((c) => c.status === 'mastered').length
   877→   877→      const pct = cards.length > 0 ? Math.round((mastered / cards.length) * 100) : 0
   878→   878→      return { goal, current: mastered, pct: goal.target <= 100 ? pct : Math.min(100, mastered), unit: 'cards' }
   879→   879→    }
   880→   880→    case 'subject-resources':
   881→   881→    default: {
   882→   882→      // Kept for future subject-completion goals.
   883→   883→      return { goal, current: 0, pct: 0, unit: '' }
   884→   884→    }
   885→   885→  }
   886→   886→}
