     3→/**
     4→ * premium-charts — Unified premium chart visualization system for SCHOLARIO.
     5→ *
     6→ * One reusable chart system used across ALL modules:
     7→ *   - DonutChart        : animated categorical distribution (gradient segments, hover pop-out,
     8→ *                         bidirectional legend sync, enhanced center content, floating tooltip)
     9→ *   - PieChart          : full pie variant (no center hole), shares DonutChart's data model
    10→ *   - RadialProgress    : single-value progress ring (gradient stroke, animated counter,
    11→ *                         optional tick marks, completion glow, optional dual-ring)
    12→ *   - AreaTrendChart    : smooth Catmull-Rom bezier area/line trend with hover dots + tooltip
    13→ *   - GroupedBarChart   : grouped bar comparison (quarterly, class-wise)
    14→ *   - HorizontalBarChart: category breakdown bars
    15→ *   - ProgressBar       : compact linear progress utility
    16→ *
    17→ * Design principles:
    18→ *   - SVG geometry with strokeDasharray/strokeDashoffset for arcs
    19→ *   - Framer Motion for smooth, staggered entrance + hover transitions
    20→ *   - Bidirectional hover: hovering a segment highlights its legend row and vice-versa
    21→ *   - Rounded stroke caps + subtle drop-shadow glow on hover
    22→ *   - Theme-aware (uses Tailwind CSS variables, works in dark/light)
    23→ *   - Responsive (scales within container via ResizeObserver)
    24→ *   - Edge cases: empty data, zero values, zero total, single 100% slice, many tiny slices
    25→ *     (auto-grouped into "Other"), negative values (clamped to 0)
    26→ *
    27→ * This is a VISUAL-LAYER ONLY upgrade. Component props and exports are stable so all
    28→ * module wrappers (fees-charts, finance-charts, library/inventory/transport) keep working
    29→ * without changes — only the rendering is richer.
    30→ */

   153→// ─── Helpers ─────────────────────────────────────────────────────────
   154→
   155→const TWO_PI = Math.PI * 2
   156→
   157→function polarToCart(cx: number, cy: number, r: number, angleRad: number) {
   158→  return { x: cx + r * Math.cos(angleRad), y: cy + r * Math.sin(angleRad) }
   159→}
   160→
   161→// Midpoint angle of a segment given its cumulative start fraction and pct
   162→function segmentMidAngle(startFrac: number, pct: number) {
   163→  // segments are drawn starting from top (12 o'clock) going clockwise;
   164→  // we rotate the whole svg by -90deg via the viewBox, so start angle in svg-space
   165→  // is: startFrac * 2π (clockwise). Mid angle = startFrac*2π + (pct/2)*2π
   166→  return startFrac * TWO_PI + (pct / 2) * TWO_PI
   167→}

   179→function DonutOrPie({
   180→  data,
   181→  size = 160,
   182→  thickness = 18,
   183→  centerLabel,
   184→  centerValue,
   185→  centerSub,
   186→  formatValue = (n) => n.toLocaleString('en-IN'),
   187→  showLegend = true,
   188→  showPercentInLegend = true,
   189→  gapDegrees = CHART_TOKENS.gapDegrees,
   190→  innerRadius = 0.62,
   191→  className,
   192→}: DonutChartProps & { innerRadius: number }) {
   193→  const [hover, setHover] = useState<number | null>(null)
   194→  const [mouse, setMouse] = useState<{ x: number; y: number } | null>(null)
   195→  const uid = useId().replace(/[:]/g, '')
   196→  const containerRef = useRef<HTMLDivElement>(null)
   197→  const svgRef = useRef<SVGSVGElement>(null)
   198→  const [actualSize, setActualSize] = useState(size)
   199→
   200→  // Responsive: compute donut size from ACTUAL available space.
   201→  // When legend is shown, the layout is: [donut] [gap] [legend flex-1].
   202→  // The donut must fit in ~33% of the container width so the legend has room
   203→  // for name + percentage + value (which can be long currency strings).
   204→  // When no legend, the donut can use the full width.
   205→  // We also clamp to the container HEIGHT (if available) so the donut never
   206→  // exceeds the vertical space and gets clipped at the bottom.
   207→  useEffect(() => {
   208→    if (!containerRef.current) return
   209→    const observer = new ResizeObserver((entries) => {
   210→      const entry = entries[0]
   211→      const width = entry?.contentRect.width ?? size
   212→      const height = entry?.contentRect.height ?? size
   213→      // With legend: donut gets ~33% of width (leaves room for legend + gap).
   214→      // Without legend: donut can be up to `size`.
   215→      const widthBased = showLegend
   216→        ? Math.min(size, Math.max(96, Math.floor(width * 0.33)))
   217→        : Math.min(size, Math.max(96, width - 24))
   218→      // Also clamp by height (donut must fit vertically too)
   219→      const heightBased = Math.max(80, Math.floor(height - 8))
   220→      const maxDonut = Math.min(widthBased, heightBased)
   221→      setActualSize(maxDonut)
   222→    })
   223→    observer.observe(containerRef.current)
   224→    return () => observer.disconnect()
   225→  }, [size, showLegend])
   226→
   227→  // Group small segments (<5%) into "Other" so the chart stays readable
   228→  const total = data.reduce((s, d) => s + Math.max(0, d.value), 0)
   229→  const groupedData = useMemo(() => {
   230→    if (total === 0 || data.length === 0) return data
   231→    const threshold = 0.05
   232→    const major = data.filter((d) => d.value / total >= threshold)
   233→    const minor = data.filter((d) => d.value / total < threshold)
   234→    if (minor.length <= 1) return data
   235→    return [
   236→      ...major,
   237→      { name: 'Other', value: minor.reduce((s, d) => s + d.value, 0), color: 'oklch(0.62 0.015 250)' },
   238→    ]
   239→  }, [data, total])
   240→
   241→  const r = (actualSize - thickness) / 2
   242→  const cx = actualSize / 2
   243→  const cy = actualSize / 2
   244→  const circumference = 2 * Math.PI * r
   245→  const gapFraction = gapDegrees / 360
   246→  const gapLength = circumference * gapFraction
   247→
   248→  // Compute segment geometry
   249→  const segments = useMemo(() => {
   250→    if (total === 0) return []
   251→    let acc = 0
   252→    return groupedData.map((d, i) => {
   253→      const pct = Math.max(0, d.value) / total
   254→      const arcLength = circumference * pct
   255→      const segLen = groupedData.length > 1 ? Math.max(0, arcLength - gapLength) : arcLength
   256→      const dashArray = `${segLen} ${circumference - segLen}`
   257→      const dashOffset = -acc
   258→      const midAngle = segmentMidAngle(acc / circumference, pct)
   259→      // pop-out direction in viewBox coords (svg is rotated -90deg, so apply that to the angle)
   260→      const popAngle = midAngle - Math.PI / 2
   261→      const pop = polarToCart(0, 0, CHART_TOKENS.hoverLift, popAngle)
   262→      acc += arcLength
   263→      return { ...d, pct, dashArray, dashOffset, index: i, startFrac: (acc - arcLength) / circumference, pop }
   264→    })
   265→  }, [groupedData, total, circumference, gapLength])
   266→
   267→  const isPie = innerRadius === 0
   268→
   269→  // Empty state
   270→  if (data.length === 0 || total === 0) {
   271→    return (
   272→      <div ref={containerRef} className={cn('flex flex-col items-center justify-center py-8', className)}>
   273→        <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted/30 text-muted-foreground/40">
   274→          <svg width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
   275→            <circle cx="12" cy="12" r="9" />
   276→          </svg>
   277→        </div>
   278→        <p className="text-xs text-muted-foreground mt-2">No data</p>
   279→      </div>
   280→    )
   281→  }
   282→
   283→  const hovered = hover !== null ? groupedData[hover] : null
   284→  const hoveredPct = hovered ? (hovered.value / total) * 100 : 0
   285→
   286→  return (
   287→    <div ref={containerRef} className={cn('flex flex-col sm:flex-row items-center sm:gap-4 gap-3 fee-mode-mix', className)}>
   288→      {/* Scoped CSS: neutralize the global `* { outline-ring/50 }` rule on
   289→          SVG elements (the blue outline that appears on focus). This is NOT
   290→          a global hack — it's scoped to this donut component only. Keyboard
   291→          accessibility is preserved via the legend buttons which have their
   292→          own focus-visible:ring. */}
   293→      <style dangerouslySetInnerHTML={{ __html: `
   294→        .fee-mode-mix .recharts-wrapper,
   295→        .fee-mode-mix .recharts-surface,
   296→        .fee-mode-mix .recharts-surface * {
   297→          outline: none !important;
   298→        }
   299→        .fee-mode-mix .recharts-sector {
   300→          outline: none !important;
   301→        }
   302→      `}} />
   303→      {/* Donut / Pie SVG */}
   304→      <div
   305→        className="relative shrink-0"
   306→        style={{ width: actualSize, height: actualSize }}
   307→        onMouseMove={(e) => {
   308→          const rect = svgRef.current?.getBoundingClientRect()
   309→          if (rect) setMouse({ x: e.clientX - rect.left, y: e.clientY - rect.top })
   310→        }}
   311→        onMouseLeave={() => { setHover(null); setMouse(null) }}
   312→      >
   313→        <svg
   314→          ref={svgRef}
   315→          viewBox={`0 0 ${actualSize} ${actualSize}`}
   316→          className="w-full h-full -rotate-90 overflow-visible"
   317→        >
   318→          <defs>
   319→            {/* Per-segment linear gradients for depth */}
   320→            {segments.map((s) => (
   321→              <linearGradient
   322→                key={s.name}
   323→                id={`${uid}-seg-${s.index}`}
   324→                x1="0" y1="0" x2="1" y2="1"
   325→                gradientUnits="userSpaceOnUse"
   326→              >
   327→                <stop offset="0%" stopColor={s.color} stopOpacity="0.95" />
   328→                <stop offset="100%" stopColor={s.color} stopOpacity="0.75" />
   329→              </linearGradient>
   330→            ))}
   331→            {/* Subtle background ring gradient */}
   332→            <radialGradient id={`${uid}-bg`} cx="0.5" cy="0.5" r="0.5">
   333→              <stop offset="60%" stopColor="currentColor" stopOpacity={CHART_TOKENS.bgRingOpacity} />
   334→              <stop offset="100%" stopColor="currentColor" stopOpacity={CHART_TOKENS.bgRingOpacity * 0.5} />
   335→            </radialGradient>
   336→          </defs>
   337→
   338→          {/* Background ring (donut only) */}
   339→          {!isPie && (
   340→            <circle
   341→              cx={cx} cy={cy} r={r}
   342→              fill="none"
   343→              stroke="currentColor"
   344→              className="text-muted/40"
   345→              strokeWidth={thickness}
   346→              opacity={0.5}
   347→            />
   348→          )}
   349→
   350→          {/* Inner hole shadow (donut only) — subtle depth ring just inside segments */}
   351→          {!isPie && (
   352→            <circle
   353→              cx={cx} cy={cy} r={r - thickness / 2 - 1.5}
   354→              fill="none"
   355→              stroke="currentColor"
   356→              className="text-foreground/5"
   357→              strokeWidth={1}
   358→            />
   359→          )}
   360→
   361→          {/* Segments */}
   362→          {segments.map((s) => {
   363→            const isHovered = hover === s.index
   364→            const dim = hover !== null && !isHovered
   365→            return (
   366→              <motion.circle
   367→                key={s.name}
   368→                cx={cx} cy={cy} r={r}
   369→                fill="none"
   370→                stroke={`url(#${uid}-seg-${s.index})`}
   371→                strokeWidth={isHovered ? thickness + CHART_TOKENS.hoverStrokeBoost : thickness}
   372→                strokeDasharray={s.dashArray}
   373→                strokeDashoffset={s.dashOffset}
   374→                strokeLinecap="round"
   375→                className="cursor-pointer transition-[stroke-width] duration-200"
   376→                onMouseEnter={() => setHover(s.index)}
   377→                initial={{ strokeDashoffset: circumference, opacity: 0, x: 0, y: 0 }}
   378→                animate={{
   379→                  strokeDashoffset: s.dashOffset,
   380→                  opacity: dim ? 0.35 : 1,
   381→                  x: isHovered ? s.pop.x : 0,
   382→                  y: isHovered ? s.pop.y : 0,
   383→                }}
   384→                transition={{
   385→                  strokeDashoffset: { duration: CHART_TOKENS.entranceDuration, ease: CHART_TOKENS.easeOutExpo, delay: s.index * CHART_TOKENS.staggerStep },
   386→                  opacity: { duration: 0.3, delay: s.index * CHART_TOKENS.staggerStep },
   387→                  x: { duration: CHART_TOKENS.settleDuration, ease: CHART_TOKENS.easeOutExpo },
   388→                  y: { duration: CHART_TOKENS.settleDuration, ease: CHART_TOKENS.easeOutExpo },
   389→                }}
   390→                style={{
   391→                  filter: isHovered ? `drop-shadow(0 0 8px ${s.color}66)` : 'none',
   392→                }}
   393→              />
   394→            )
   395→          })}
   396→
   397→          {/* Hover segment outer highlight ring (donut only) */}
   398→          {hover !== null && !isPie && (
   399→            <circle
   400→              cx={cx} cy={cy} r={r + thickness / 2 + 2}
   401→              fill="none"
   402→              stroke={groupedData[hover].color}
   403→              strokeWidth={1}
   404→              strokeDasharray="2 4"
   405→              opacity={0.5}
   406→              className="pointer-events-none"
   407→            />
   408→          )}
   409→        </svg>
   410→
   411→        {/* Center content (donut only) */}
   412→        {!isPie && (
   413→          <div className="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
   414→            <AnimatePresence mode="wait">
   415→              {hover !== null && hovered ? (
   416→                <motion.div
   417→                  key={`hover-${hover}`}
   418→                  initial={{ opacity: 0, scale: 0.7, y: 4 }}
   419→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   420→                  exit={{ opacity: 0, scale: 0.7, y: -4 }}
   421→                  transition={{ duration: 0.18, ease: CHART_TOKENS.easeOutExpo }}
   422→                  className="text-center"
   423→                >
   424→                  <div className="flex items-center justify-center gap-1 mb-0.5">
   425→                    <span
   426→                      className="h-1.5 w-1.5 rounded-full"
   427→                      style={{ background: hovered.color, boxShadow: `0 0 4px ${hovered.color}` }}
   428→                    />
   429→                    <p className="text-[9px] text-muted-foreground font-medium uppercase tracking-wider truncate max-w-[72px]">
   430→                      {hovered.name}
   431→                    </p>
   432→                  </div>
   433→                  <p className="text-sm font-bold tabular-nums leading-tight">{formatValue(hovered.value)}</p>
   434→                  <p className="text-[10px] font-semibold tabular-nums mt-0.5" style={{ color: hovered.color }}>
   435→                    {hoveredPct.toFixed(1)}%
   436→                  </p>
   437→                </motion.div>
   438→              ) : (
   439→                <motion.div
   440→                  key="default"
   441→                  initial={{ opacity: 0, scale: 0.8 }}
   442→                  animate={{ opacity: 1, scale: 1 }}
   443→                  exit={{ opacity: 0, scale: 0.8 }}
   444→                  transition={{ duration: 0.18 }}
   445→                  className="text-center"
   446→                >
   447→                  {centerValue && (
   448→                    <p className="text-base font-bold tabular-nums leading-tight">{centerValue}</p>
   449→                  )}
   450→                  {centerLabel && (
   451→                    <p className="text-[9px] text-muted-foreground uppercase tracking-wider mt-0.5">{centerLabel}</p>
   452→                  )}
   453→                  {centerSub && (
   454→                    <p className="text-[9px] text-muted-foreground/70 mt-0.5">{centerSub}</p>
   455→                  )}
   456→                </motion.div>
   457→              )}
   458→            </AnimatePresence>
   459→          </div>
   460→        )}
   461→
   462→        {/* Floating tooltip for pie (no center to show info) */}
   463→        {isPie && hover !== null && mouse && (
   464→          <motion.div
   465→            initial={{ opacity: 0, y: 4 }}
   466→            animate={{ opacity: 1, y: 0 }}
   467→            className="absolute z-20 pointer-events-none rounded-lg border border-border bg-popover/95 backdrop-blur-md shadow-xl px-2.5 py-1.5 text-[10px] min-w-[110px]"
   468→            style={{
   469→              left: Math.min(Math.max(mouse.x + 8, 8), actualSize - 110),
   470→              top: Math.min(Math.max(mouse.y - 8, 8), actualSize - 40),
   471→            }}
   472→          >
   473→            <div className="flex items-center gap-1.5">
   474→              <span className="h-2 w-2 rounded-full" style={{ background: hovered!.color }} />
   475→              <span className="font-semibold truncate">{hovered!.name}</span>
   476→            </div>
   477→            <div className="flex items-center justify-between mt-0.5 tabular-nums">
   478→              <span className="text-muted-foreground">Value</span>
   479→              <span className="font-bold">{formatValue(hovered!.value)}</span>
   480→            </div>
   481→            <div className="flex items-center justify-between tabular-nums">
   482→              <span className="text-muted-foreground">Share</span>
   483→              <span className="font-bold" style={{ color: hovered!.color }}>{hoveredPct.toFixed(1)}%</span>
   484→            </div>
   485→          </motion.div>
   486→        )}
   487→      </div>
   488→
   489→      {/* Legend */}
   490→      {showLegend && (
   491→        <div className="flex-1 min-w-0 w-full sm:w-auto space-y-1">
   492→          {groupedData.map((d, i) => {
   493→            const pct = total > 0 ? (d.value / total) * 100 : 0
   494→            const isHovered = hover === i
   495→            return (
   496→              <button
   497→                key={d.name}
   498→                onMouseEnter={() => setHover(i)}
   499→                onMouseLeave={() => setHover(null)}
   500→                className={cn(
   501→                  'w-full flex items-center gap-2 px-1.5 py-1 rounded-md transition-colors text-left',
   502→                  isHovered && 'bg-muted/60',
   503→                )}
   504→              >
   505→                <span
   506→                  className="h-2 w-2 rounded-full shrink-0 transition-all"
   507→                  style={{
   508→                    background: d.color,
   509→                    transform: isHovered ? 'scale(1.3)' : 'scale(1)',
   510→                    boxShadow: isHovered ? `0 0 5px ${d.color}80` : 'none',
   511→                  }}
   512→                />
   513→                <span className="text-[11px] font-medium truncate flex-1 text-foreground min-w-0">{d.name}</span>
   514→                {showPercentInLegend && (
   515→                  <span className="text-[11px] tabular-nums font-bold text-foreground shrink-0">{pct.toFixed(0)}%</span>
   516→                )}
   517→                <span className="text-[11px] tabular-nums text-muted-foreground whitespace-nowrap shrink-0 text-right">
   518→                  {formatValue(d.value)}

   520→              </button>
   521→            )
   522→          })}
   523→        </div>
   524→      )}

   531→export function RadialProgress({
   532→  value,
   533→  max = 100,
   534→  size = 120,
   535→  thickness = 10,
   536→  color = 'oklch(0.55 0.14 162)',
   537→  trackColor,
   538→  label,
   539→  suffix = '%',
   540→  formatValue = (n) => `${Math.round(n)}${suffix}`,
   541→  showTicks = false,
   542→  glow = false,
   543→  className,
   544→}: RadialProgressProps) {
   545→  const [display, setDisplay] = useState(0)
   546→  const r = (size - thickness) / 2
   547→  const cx = size / 2
   548→  const cy = size / 2
   549→  const circumference = 2 * Math.PI * r
   550→  const pct = Math.min(1, Math.max(0, value / max))
   551→  const dashOffset = circumference * (1 - pct)
   552→  const uid = useId().replace(/[:]/g, '')
   553→  const isComplete = pct >= 0.999
   554→
   555→  // Animated counter
   556→  useEffect(() => {
   557→    let raf: number
   558→    const start = performance.now()
   559→    const from = 0
   560→    const to = value
   561→    const dur = 900
   562→    const tick = (t: number) => {
   563→      const e = Math.min(1, (t - start) / dur)
   564→      // easeOutExpo
   565→      const eased = e === 1 ? 1 : 1 - Math.pow(2, -10 * e)
   566→      setDisplay(from + (to - from) * eased)
   567→      if (e < 1) raf = requestAnimationFrame(tick)
   568→    }
   569→    raf = requestAnimationFrame(tick)
   570→    return () => cancelAnimationFrame(raf)
   571→  }, [value])
   572→
   573→  // Tick marks (24 ticks around the ring)
   574→  const ticks = useMemo(() => {
   575→    if (!showTicks) return []
   576→    const count = 24
   577→    return Array.from({ length: count }, (_, i) => {
   578→      const a = (i / count) * TWO_PI - Math.PI / 2
   579→      const inner = r - thickness / 2 - 2
   580→      const outer = r + thickness / 2 + 2
   581→      return {
   582→        x1: cx + Math.cos(a) * inner,
   583→        y1: cy + Math.sin(a) * inner,
   584→        x2: cx + Math.cos(a) * outer,
   585→        y2: cy + Math.sin(a) * outer,
   586→        major: i % 6 === 0,
   587→      }
   588→    })
   589→  }, [showTicks, r, thickness, cx, cy])
   590→
   591→  return (
   592→    <div className={cn('relative', className)} style={{ width: size, height: size }}>
   593→      <svg viewBox={`0 0 ${size} ${size}`} className="w-full h-full -rotate-90 overflow-visible">
   594→        <defs>
   595→          <linearGradient id={`${uid}-rp`} x1="0" y1="0" x2="1" y2="1" gradientUnits="userSpaceOnUse">
   596→            <stop offset="0%" stopColor={color} stopOpacity="0.85" />
   597→            <stop offset="100%" stopColor={color} stopOpacity="1" />
   598→          </linearGradient>
   599→        </defs>
   600→
   601→        {/* Tick marks */}
   602→        {ticks.map((t, i) => (
   603→          <line
   604→            key={i}
   605→            x1={t.x1} y1={t.y1} x2={t.x2} y2={t.y2}
   606→            stroke="currentColor"
   607→            className={t.major ? 'text-muted-foreground/40' : 'text-muted-foreground/15'}
   608→            strokeWidth={t.major ? 1.2 : 0.8}
   609→          />
   610→        ))}
   611→
   612→        {/* Track */}
   613→        <circle
   614→          cx={cx} cy={cy} r={r}
   615→          fill="none"
   616→          stroke={trackColor ?? 'currentColor'}
   617→          className={trackColor ? '' : 'text-muted/25'}
   618→          strokeWidth={thickness}
   619→        />
   620→
   621→        {/* Progress arc */}
   622→        <motion.circle
   623→          cx={cx} cy={cy} r={r}
   624→          fill="none"
   625→          stroke={`url(#${uid}-rp)`}
   626→          strokeWidth={thickness}
   627→          strokeLinecap="round"
   628→          strokeDasharray={circumference}
   629→          initial={{ strokeDashoffset: circumference }}
   630→          animate={{ strokeDashoffset: dashOffset }}
   631→          transition={{ duration: CHART_TOKENS.entranceDuration, ease: CHART_TOKENS.easeOutExpo }}
   632→          style={{
   633→            filter: glow && isComplete ? `drop-shadow(0 0 6px ${color}99)` : 'none',
   634→          }}
   635→        />
   636→
   637→        {/* Completion dot at arc end */}
   638→        {isComplete && (
   639→          <circle
   640→            cx={cx} cy={cy - r} r={thickness / 2 + 1}
   641→            fill={color}
   642→            opacity={0.9}
   643→            className="pointer-events-none"
   644→          />
   645→        )}
   646→      </svg>
   647→
   648→      <div className="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
   649→        <motion.p
   650→          initial={{ opacity: 0, scale: 0.8 }}
   651→          animate={{ opacity: 1, scale: 1 }}
   652→          transition={{ delay: 0.25, duration: 0.2 }}
   653→          className="text-lg font-bold tabular-nums leading-tight"
   654→        >
   655→          {formatValue(display)}
   656→        </motion.p>
   657→        {label && (
   658→          <motion.p
   659→            initial={{ opacity: 0 }}
   660→            animate={{ opacity: 1 }}
   661→            transition={{ delay: 0.4 }}
   662→            className="text-[9px] text-muted-foreground uppercase tracking-wider mt-0.5"
   663→          >
   664→            {label}
   665→          </motion.p>
   666→        )}
   667→      </div>
   668→    </div>
   669→  )
   670→}

   674→// Format a Y-axis value compactly (₹1Cr, ₹2L, 50%, etc.)
   675→// Sign-aware: valMin can dip below zero (5% range padding), and negatives
   676→// previously fell through every ≥ branch and rendered as raw "-4320".
   677→function formatYAxisValue(n: number): string {
   678→  const sign = n < 0 ? '-' : ''
   679→  const v = Math.abs(n)
   680→  if (v >= 10000000) return `${sign}₹${(v / 10000000).toFixed(0)}Cr`
   681→  if (v >= 100000) return `${sign}₹${(v / 100000).toFixed(0)}L`
   682→  if (v >= 1000) return `${sign}₹${(v / 1000).toFixed(0)}K`
   683→  return `${sign}${Math.round(v)}`
   684→}
   685→
   686→export function AreaTrendChart({
   687→  data,
   688→  height = 140,
   689→  formatValue = (n) => n.toLocaleString('en-IN'),
   690→  primaryColor = 'oklch(0.55 0.14 162)',
   691→  secondaryColor = 'oklch(0.62 0.2 25)',
   692→  primaryLabel,
   693→  secondaryLabel,
   694→  labelKey = 'label',
   695→  primaryKey = 'primary',
   696→  secondaryKey = 'secondary',
   697→  className,
   698→  showArea = false,
   699→  strokeWidth = 2,
   700→  secondaryStrokeWidth = 2,
   701→}: AreaTrendChartProps) {
   702→  const [hover, setHover] = useState<number | null>(null)
   703→  const uid = useId().replace(/[:]/g, '')
   704→  const w = 100
   705→  const h = height
   706→  // Larger vertical padding so the curve never clips at the top/bottom.
   707→  // With preserveAspectRatio="none", the X is stretched but Y is 1:1,
   708→  // so pad is in actual pixels relative to the height.
   709→  const padX = 6
   710→  const padY = 18
   711→
   712→  const maxVal = Math.max(...data.flatMap((d) => [d[primaryKey] ?? 0, d[secondaryKey] ?? 0]), 1)
   713→  const minVal = Math.min(...data.flatMap((d) => [d[primaryKey] ?? 0, d[secondaryKey] ?? 0]), 0)
   714→  // Add 10% headroom to the range so the curve doesn't touch the edges
   715→  const valRange = (maxVal - minVal) * 1.1 || 1
   716→  const valMin = minVal - (maxVal - minVal) * 0.05
   717→
   718→  // Build a MONOTONE cubic interpolation path (no overshoot).
   719→  // Catmull-Rom can overshoot when data points vary dramatically, creating
   720→  // artificial peaks/valleys below the real data range. Monotone cubic
   721→  // interpolation guarantees the curve passes through every data point
   722→  // WITHOUT overshooting — data accuracy > decorative smoothness.
   723→  // This is the same approach used by D3's curveMonotoneX.
   724→  const monotonePath = (pts: { x: number; y: number }[]) => {
   725→    if (pts.length < 2) return ''
   726→    if (pts.length === 2) return `M ${pts[0].x.toFixed(2)},${pts[0].y.toFixed(2)} L ${pts[1].x.toFixed(2)},${pts[1].y.toFixed(2)}`
   727→
   728→    // Compute slopes (tangents) for each point — monotone-clamped
   729→    const n = pts.length
   730→    const dx: number[] = new Array(n - 1)
   731→    const dy: number[] = new Array(n - 1)
   732→    const m: number[] = new Array(n) // slope at each point
   733→
   734→    for (let i = 0; i < n - 1; i++) {
   735→      dx[i] = pts[i + 1].x - pts[i].x
   736→      dy[i] = pts[i + 1].y - pts[i].y
   737→    }
   738→
   739→    // Slopes at endpoints
   740→    m[0] = dy[0] / dx[0]
   741→    m[n - 1] = dy[n - 2] / dx[n - 2]
   742→
   743→    // Interior slopes — monotone: take the harmonic mean of the two adjacent
   744→    // secant slopes, but only if they have the same sign (otherwise slope=0).
   745→    for (let i = 1; i < n - 1; i++) {
   746→      const s1 = dy[i - 1] / dx[i - 1]
   747→      const s2 = dy[i] / dx[i]
   748→      if (s1 * s2 <= 0) {
   749→        m[i] = 0 // sign change → flat tangent (prevents overshoot)
   750→      } else {
   751→        m[i] = (s1 + s2) / 2
   752→      }
   753→    }
   754→
   755→    // Monotone slope clamping: ensure the control points don't overshoot
   756→    for (let i = 0; i < n - 1; i++) {
   757→      const s = dy[i] / dx[i]
   758→      if (Math.abs(m[i]) > 3 * Math.abs(s)) m[i] = 3 * s * Math.sign(m[i])
   759→      if (Math.abs(m[i + 1]) > 3 * Math.abs(s)) m[i + 1] = 3 * s * Math.sign(m[i + 1])
   760→    }
   761→
   762→    // Build the cubic bezier path using the monotone slopes
   763→    let path = `M ${pts[0].x.toFixed(2)},${pts[0].y.toFixed(2)}`
   764→    for (let i = 0; i < n - 1; i++) {
   765→      const cp1x = pts[i].x + dx[i] / 3
   766→      const cp1y = pts[i].y + m[i] * dx[i] / 3
   767→      const cp2x = pts[i + 1].x - dx[i] / 3
   768→      const cp2y = pts[i + 1].y - m[i + 1] * dx[i] / 3
   769→      path += ` C ${cp1x.toFixed(2)},${cp1y.toFixed(2)} ${cp2x.toFixed(2)},${cp2y.toFixed(2)} ${pts[i + 1].x.toFixed(2)},${pts[i + 1].y.toFixed(2)}`
   770→    }
   771→    return path
   772→  }
   773→
   774→  const primaryPoints = data.map((d, i) => ({
   775→    x: padX + (i / Math.max(1, data.length - 1)) * (w - 2 * padX),
   776→    // Parentheses matter: `d[k] ?? 0 - valMin` parses as `d[k] ?? (0 - valMin)`,
   777→    // so valMin was never subtracted from real values.
   778→    y: padY + (h - 2 * padY) - (((d[primaryKey] ?? 0) - valMin) / valRange) * (h - 2 * padY),
   779→  }))
   780→  const secondaryPoints = data.map((d, i) => ({
   781→    x: padX + (i / Math.max(1, data.length - 1)) * (w - 2 * padX),
   782→    y: padY + (h - 2 * padY) - (((d[secondaryKey] ?? 0) - valMin) / valRange) * (h - 2 * padY),
   783→  }))
   784→
   785→  const primaryPath = useMemo(() => monotonePath(primaryPoints), [primaryPoints])
   786→  const secondaryPath = useMemo(() => monotonePath(secondaryPoints), [secondaryPoints])
   787→  const primaryArea = useMemo(() => primaryPath ? `${primaryPath} L ${primaryPoints[primaryPoints.length - 1].x.toFixed(2)},${h - padY} L ${primaryPoints[0].x.toFixed(2)},${h - padY} Z` : '', [primaryPath, primaryPoints, h, padY])
   788→  const secondaryArea = useMemo(() => secondaryPath ? `${secondaryPath} L ${secondaryPoints[secondaryPoints.length - 1].x.toFixed(2)},${h - padY} L ${secondaryPoints[0].x.toFixed(2)},${h - padY} Z` : '', [secondaryPath, secondaryPoints, h, padY])
   789→
   790→  return (
   791→    <div className="relative w-full" style={{ height }}>
   792→      <svg viewBox={`0 0 ${w} ${h}`} preserveAspectRatio="none" className="absolute inset-0 w-full h-full overflow-visible">
   793→        <defs>
   794→          <linearGradient id={`${uid}-p`} x1="0" y1="0" x2="0" y2="1">
   795→            <stop offset="0%" stopColor={primaryColor} stopOpacity="0.22" />
   796→            <stop offset="60%" stopColor={primaryColor} stopOpacity="0.06" />
   797→            <stop offset="100%" stopColor={primaryColor} stopOpacity="0" />
   798→          </linearGradient>
   799→          {data.some((d) => d[secondaryKey] !== undefined) && (
   800→            <linearGradient id={`${uid}-s`} x1="0" y1="0" x2="0" y2="1">
   801→              <stop offset="0%" stopColor={secondaryColor} stopOpacity="0.14" />
   802→              <stop offset="100%" stopColor={secondaryColor} stopOpacity="0" />
   803→            </linearGradient>
   804→          )}
   805→        </defs>
   806→        {/* Subtle horizontal gridlines only — no vertical grid, no heavy box */}
   807→        {[0.25, 0.5, 0.75].map((frac) => (
   808→          <line key={frac} x1={0} y1={padY + frac * (h - 2 * padY)} x2={w} y2={padY + frac * (h - 2 * padY)}
   809→            stroke="currentColor" strokeOpacity={0.08} strokeDasharray="2 6" vectorEffect="non-scaling-stroke" />
   810→        ))}
   811→        {/* Secondary area + line (drawn first) */}
   812→        {data.some((d) => d[secondaryKey] !== undefined) && (
   813→          <>
   814→            {showArea && (
   815→              <motion.path d={secondaryArea} fill={`url(#${uid}-s)`}
   816→                initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.6, delay: 0.1 }} />
   817→            )}
   818→            {/* Plain path inside an opacity-fading group — NEVER animate
   819→                pathLength here. A completed pathLength animation leaves
   820→                pathLength="1" + stroke-dasharray on the element, and with
   821→                vector-effect:non-scaling-stroke Chromium computes the dash
   822→                in screen pixels, rendering the line as fragments. */}
   823→            <motion.g initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.8, delay: 0.1, ease: 'easeOut' }}>
   824→              <path d={secondaryPath} fill="none" stroke={secondaryColor} strokeWidth={secondaryStrokeWidth}
   825→                strokeLinecap="round" strokeLinejoin="round" vectorEffect="non-scaling-stroke" />
   826→            </motion.g>
   827→          </>
   828→        )}
   829→        {/* Primary area + line */}
   830→        {showArea && (
   831→          <motion.path d={primaryArea} fill={`url(#${uid}-p)`}
   832→            initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.6, delay: 0.2 }} />
   833→        )}
   834→        {/* Primary line — same rule: opacity fade, no pathLength (see above). */}
   835→        <motion.g initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.9, ease: CHART_TOKENS.easeInOut }}>
   836→          <path d={primaryPath} fill="none" stroke={primaryColor} strokeWidth={strokeWidth}
   837→            strokeLinecap="round" strokeLinejoin="round" vectorEffect="non-scaling-stroke" />
   838→        </motion.g>
   839→        {/* Hover interaction zones + dots */}
   840→        {primaryPoints.map((p, i) => (
   841→          <g key={i}>
   842→            <rect x={p.x - 5} y={0} width={10} height={h} fill="transparent"
   843→              onMouseEnter={() => setHover(i)} onMouseLeave={() => setHover(null)} />
   844→            <circle cx={p.x} cy={p.y} r={hover === i ? 4 : 3} fill={primaryColor} stroke="white" strokeWidth={1.5}
   845→              style={{ opacity: hover === i ? 1 : 0 }} className="transition-all" vectorEffect="non-scaling-stroke" />
   846→            {data[i].secondary !== undefined && (
   847→              <circle cx={p.x} cy={secondaryPoints[i].y} r={hover === i ? 3.5 : 2.5} fill={secondaryColor} stroke="white" strokeWidth={1.2}
   848→                style={{ opacity: hover === i ? 1 : 0 }} className="transition-all" vectorEffect="non-scaling-stroke" />
   849→            )}
   850→            {hover === i && (
   851→              <line x1={p.x} y1={Math.min(p.y, secondaryPoints[i]?.y ?? h)} x2={p.x} y2={h - padY}
   852→                stroke={primaryColor} strokeWidth={1} strokeDasharray="3 3" strokeOpacity={0.3} vectorEffect="non-scaling-stroke" />
   853→            )}
   854→          </g>
   855→        ))}
   856→      </svg>
   857→      {/* Y-axis labels (left) — subtle, formatted */}
   858→      <div className="absolute left-0 top-0 bottom-4 flex flex-col justify-between py-1 text-[8px] text-muted-foreground/50 pointer-events-none tabular-nums leading-none w-12">
   859→        <span>{formatYAxisValue(maxVal)}</span>
   860→        <span>{formatYAxisValue(valMin + valRange * 0.5)}</span>
   861→        <span>{formatYAxisValue(valMin)}</span>
   862→      </div>
   863→      {/* Legend above the chart (only if multi-series) */}
   864→      {data.some((d) => d[secondaryKey] !== undefined) && (
   865→        <div className="absolute top-0 right-2 flex items-center gap-3 text-[10px] text-muted-foreground pointer-events-none">
   866→          <span className="flex items-center gap-1.5">

   868→            {primaryLabel ?? 'Primary'}

   900→            </div>
   901→          )}

   903→      )}

   905→  )
