   187→export type FeeHeadCategory =

   191→/**
   192→ * FINANCIAL CATEGORY of a payment/transaction (Core vs Additional vs Exam).
   193→ *
   194→ *   CORE        — money against the standard annual fee structure
   195→ *                 (recurring heads: Tuition, Transport, Library…).
   196→ *   EXAMINATION — money against per-examination charges (the exam fee
   197→ *                 schedule + board/exam heads).
   198→ *   ADDITIONAL  — money against an event-based Additional Charge
   199→ *                 (Educational Tour, Workshop, Competition…) — tracked
   200→ *                 SEPARATELY from core fees so the two are never mixed.
   201→ *
   202→ * Optional on FeeTransaction for backward compatibility: legacy
   203→ * transactions (recorded before this field existed) derive their category
   204→ * via `txnCategory()` (feeHead containing "exam" → EXAMINATION, else CORE).
   205→ */
   206→export type TransactionCategory = 'CORE' | 'EXAMINATION' | 'ADDITIONAL'
   207→
   208→/** Resolve the financial category of any transaction (new or legacy). */
   209→export function txnCategory(
   210→  t: Pick<FeeTransaction, 'category' | 'additionalChargeId' | 'feeHead'>,
   211→): TransactionCategory {
   212→  if (t.additionalChargeId) return 'ADDITIONAL'
   213→  if (t.category) return t.category
   214→  if (/exam/i.test(t.feeHead ?? '')) return 'EXAMINATION'
   215→  return 'CORE'
   216→}

   224→/**
   225→ * Collection lifecycle (FIN-COLLECTION → APPS-IA-1 generalization):
   226→ *   Draft    — being composed; NOT an obligation yet (excluded from student
   227→ *              accounts, the collect-payment wizard and student fees).

   233→ *   Archived — historical read-only record; payment history stays

   236→ *              stay on record (audit trail).

   961→/**
   962→ * Returns the reason a structure cannot be edited in place, or null when
   963→ * editing is allowed. Rules (PART 7/8/9):
   964→ *   • draft / not-configured structures → editable freely
   965→ *   • historical (non-current) session → read-only, always
   966→ *   • published CURRENT-session structure → locked unless a temporary
   967→ *     editing window is open for it
   968→ */
   969→export function structureLockReason(
   970→  state: Pick<FeeState, 'versions' | 'structureEditWindow'>,
   971→  struct: Pick<FeeStructureConfig, 'id' | 'academicYear' | 'className'>,
   972→): string | null {
   973→  const isPublished = state.versions.some((v) => v.structureId === struct.id && v.status === 'current')
   974→  if (!isPublished) return null // draft — editable
   975→  const year = struct.academicYear ?? CURRENT_ACADEMIC_YEAR
   976→  if (year !== CURRENT_ACADEMIC_YEAR) {
   977→    return 'Historical session structures are read-only — the record stays historically intact.'
   978→  }
   979→  const w = state.structureEditWindow
   980→  const live = w.structureId === struct.id && !!w.expiresAt && Date.now() < w.expiresAt
   981→  if (live) return null
   982→  return `"${struct.className}" is locked — it is active for students. Request a temporary edit window to make exceptional changes.`
   983→}

   995→// ─── Helper: find the FeeStructureConfig that applies to a student ──
   996→// FEE-PER-CLASS (Phase 5): tries an EXACT classId match first (using the
   997→// student's classId when available, derived from className via the
   998→// ACADEMIC_CLASSES catalogue). Falls back to className exact match,
   999→// then classLevel substring matching. This priority order ensures:
  1000→//   1. A student in Class 9 (C12) finds FS04 (classId='C12') directly —
  1001→//      no substring walk.
  1002→//   2. A student in Class 11-PCB (C14-PCB) finds the future Class 11
  1003→//      structure via applicableClassIds=['C14-PCM','C14-PCB'] — handles
  1004→//      the stream-class ambiguity that className-only matching couldn't.
  1005→//   3. A student whose className has no structure (e.g. "Class 4") falls
  1006→//      back to the level (Primary → FS02).
  1007→//
  1008→// The signature accepts an optional classId (Phase 5 addition). Callers
  1009→// that pass only className continue to work — backward compatible.
  1010→export function findStructureForStudent(className: string, classId?: string): FeeStructureConfig | undefined {
  1011→  // STRUCT-SESSION/REV — resolve against the LIVE store state (the pool the
  1012→  // publish pipeline updates), falling back to the seed constant only when
  1013→  // the store is not yet hydrated. Previously this read the static
  1014→  // FEE_STRUCTURES seed directly, so a newly published version never
  1015→  // reached student accounts (PART 21 violation).
  1016→  let pool: FeeStructureConfig[] = FEE_STRUCTURES
  1017→  try {
  1018→    const live = useFeeStore.getState().feeStructures
  1019→    if (Array.isArray(live) && live.length > 0) pool = live
  1020→  } catch {
  1021→    /* store not initialised yet — seed fallback is correct at init time */
  1022→  }
  1023→  // 1. classId exact match (preferred path — Phase 5)
  1024→  if (classId) {
  1025→    const byApplicable = pool.find((f) => f.applicableClassIds?.includes(classId))
  1026→    if (byApplicable) return byApplicable
  1027→    const byClassId = pool.find((f) => f.classId === classId)
  1028→    if (byClassId) return byClassId
  1029→  }
  1030→  // 2. className exact match (legacy path — pre-Phase-5 primary lookup)
  1031→  const byName = pool.find((f) => f.className === className)
  1032→  if (byName) return byName
  1033→  // 3. classLevel fallback (last resort — keeps backward compat with
  1034→  //    pre-FEE-PER-CLASS seed which used range names like "Class 9–10")
  1035→  return pool.find((f) => f.classLevel === studentClassLevel(className))
  1036→}
  1037→
  1038→// ─── Helper: count affected students for a structure ───────────────
  1039→// FEE-PER-CLASS (Phase 5): tries an EXACT classId/applicableClassIds
  1040→// match first (so a Class 9 structure reports only Class 9 students,
  1041→// and a Class 12 structure reports BOTH PCM and PCB stream students).
  1042→// Falls back to className exact match, then classLevel substring
  1043→// matching when no student has an exact className match (e.g. a custom
  1044→// structure with className="Custom").
  1045→//
  1046→// Mirrors the matching logic in `findStructureForStudent` so the
  1047→// changeLog's `affectedStudents` field reflects the same students
  1048→// whose fee accounts would be re-derived when the structure changes.
  1049→function countStudentsForStructure(struct: {
  1050→  className: string
  1051→  classLevel: string
  1052→  classId?: string
  1053→  applicableClassIds?: string[]
  1054→}): number {
  1055→  const students = useStudentsStore.getState().students.filter((s) => s.status === 'Active')
  1056→  // 1. classId / applicableClassIds exact match (Phase 5 — preferred
  1057→  //    path when the structure is bound to academic class ids).
  1058→  if (struct.applicableClassIds && struct.applicableClassIds.length > 0) {
  1059→    const matched = students.filter((s) => {
  1060→      const sid = deriveStudentClassId(s.className)
  1061→      return sid != null && struct.applicableClassIds!.includes(sid)
  1062→    })
  1063→    if (matched.length > 0) return matched.length
  1064→  }
  1065→  if (struct.classId) {
  1066→    const matched = students.filter((s) => deriveStudentClassId(s.className) === struct.classId)
  1067→    if (matched.length > 0) return matched.length
  1068→  }
  1069→  // 2. className exact match (legacy path)
  1070→  if (struct.className) {
  1071→    const exact = students.filter((s) => s.className === struct.className)
  1072→    if (exact.length > 0) return exact.length
  1073→  }
  1074→  // 3. classLevel fallback (last resort)
  1075→  return students.filter((s) => studentClassLevel(s.className) === struct.classLevel).length
  1076→}

  1078→// PHASE 5 — derive a student's AcademicClassDef.id from their className.
  1079→// Looks up the canonical class catalogue to find the matching id (e.g.
  1080→// "Class 9" → "C12", "Pre-Nursery" → "C01"). Returns undefined if no
  1081→// academic class matches (e.g. custom draft className "Class 9 — Copy").
  1082→// Used by countStudentsForStructure's classId path. Cached at module
  1083→// scope so the lookup happens once per className.
  1084→const _studentClassNameToClassId = new Map<string, string | undefined>()
  1085→export function deriveStudentClassId(className: string): string | undefined {
  1086→  if (_studentClassNameToClassId.has(className)) {
  1087→    return _studentClassNameToClassId.get(className)
  1088→  }
  1089→  const def = ACADEMIC_CLASSES.find((c) => c.name === className)
  1090→  _studentClassNameToClassId.set(className, def?.id)
  1091→  return def?.id
  1092→}

  1459→// Compute the diff between two head arrays for the changeLog `changes` field.
  1460→function diffHeads(oldHeads: FeeHead[], newHeads: FeeHead[]): { headName: string; oldValue: number; newValue: number }[] {
  1461→  const changes: { headName: string; oldValue: number; newValue: number }[] = []
  1462→  const seen = new Set<string>()
  1463→  for (const h of newHeads) {
  1464→    seen.add(h.name)
  1465→    const old = oldHeads.find((o) => o.name === h.name)
  1466→    if (!old) {
  1467→      changes.push({ headName: h.name, oldValue: 0, newValue: h.amount })
  1468→    } else if (old.amount !== h.amount) {
  1469→      changes.push({ headName: h.name, oldValue: old.amount, newValue: h.amount })
  1470→    }
  1471→  }
  1472→  for (const h of oldHeads) {
  1473→    if (!seen.has(h.name)) {
  1474→      changes.push({ headName: h.name, oldValue: h.amount, newValue: 0 })
  1475→    }
  1476→  }
  1477→  return changes
  1478→}
