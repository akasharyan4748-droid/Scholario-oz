     3→/**
     4→ * Class Attendance — the CT baseline + ST prefill workflow (spec §11 of
     5→ * the Teacher Workspace cleanup brief).
     6→ *
     7→ * WORKFLOW:
     8→ *   Class Teacher marks attendance  →  the class's daily baseline
     9→ *   ↓ (other teachers open their assigned class)
    10→ *   students are ALREADY marked (prefilled from the baseline)
    11→ *   ↓
    12→ *   the subject teacher adjusts only exceptions and explicitly submits
    13→ *   their OWN subject-session record (kept separate from the baseline).
    14→ *
    15→ * Data layer discipline mirrors use-lesson-planner: one aggregate fetch
    16→ * (`cache: 'no-store', credentials: 'same-origin'`), a 401 that routes
    17→ * through the shared signOut(), a single-flight submit lock, the POST
    18→ * response REPLACES the view (no re-GET), and a friendly error
    19→ * vocabulary for the server's error codes. Simply viewing prefilled
    20→ * statuses never creates a submission — only the explicit Save does.

    51→// A dead server session cannot be retried — reset auth ONCE, land on login.
    52→let sessionExpiredInFlight = false
    53→
    54→function handleExpiredSession(): void {
    55→  if (sessionExpiredInFlight) return
    56→  sessionExpiredInFlight = true
    57→  void signOut().finally(() => {
    58→    window.setTimeout(() => {
    59→      sessionExpiredInFlight = false
    60→    }, 2000)
    61→  })
    62→}
    63→
    64→async function caRequest<T>(url: string, init?: RequestInit): Promise<T> {
    65→  const r = await fetch(url, {
    66→    ...init,
    67→    cache: 'no-store',
    68→    credentials: 'same-origin',
    69→    headers: init?.body
    70→      ? { 'Content-Type': 'application/json', ...(init?.headers ?? {}) }
    71→      : init?.headers,
    72→  })
    73→  if (r.status === 401) {
    74→    handleExpiredSession()
    75→    throw new Error('UNAUTHORIZED')
    76→  }
    77→  let json: unknown = null
    78→  try {
    79→    json = await r.json()
    80→  } catch {
    81→    /* non-JSON error body — fall through to the generic message */
    82→  }
    83→  const envelope = json as { ok?: unknown; error?: unknown; data?: T } | null
    84→  if (!r.ok || !envelope || envelope.ok !== true) {
    85→    const message =
    86→      envelope && typeof envelope.error === 'string' && envelope.error
    87→        ? envelope.error
    88→        : `Request failed (${r.status})`
    89→    throw new Error(message)
    90→  }
    91→  return envelope.data as T
    92→}

   180→  const draft: Record<string, AttendanceMark> = {}

   212→function useClassAttendance(): ClassAttendanceState {
   213→  const [scopesPayload, setScopesPayload] = useState<ClassAttendanceScopesPayload | null>(null)
   214→  const [scopesError, setScopesError] = useState<string | null>(null)
   215→  const [scopeSel, setScopeSel] = useState<{ classId: string; subjectId: string | null }>({ classId: '', subjectId: null })
   216→  const [date, setDate] = useState('')
   217→  const [view, setView] = useState<ClassAttendanceViewDTO | null>(null)
   218→  const [viewLoading, setViewLoading] = useState(false)
   219→  const [viewError, setViewError] = useState<string | null>(null)
   220→  const [draft, setDraft] = useState<Record<string, AttendanceMark>>({})
   221→  const [submitting, setSubmitting] = useState(false)
   222→  const [scopesTick, setScopesTick] = useState(0)
   223→  const [viewTick, setViewTick] = useState(0)
   224→
   225→  const mounted = useRef(true)
   226→  const submittingRef = useRef(false)
   227→
   228→  useEffect(() => {
   229→    mounted.current = true
   230→    return () => {
   231→      mounted.current = false
   232→    }
   233→  }, [])
   234→
   235→  const applyView = useCallback((v: ClassAttendanceViewDTO) => {
   236→    setView(v)
   237→    setDraft(buildDraft(v))
   238→  }, [])
   239→
   240→  // ① scopes — fetched once (and on retry); picks the default scope + date.
   241→  useEffect(() => {
   242→    let cancelled = false
   243→    caRequest<ClassAttendanceScopesPayload>('/api/teacher/class-attendance')
   244→      .then((d) => {
   245→        if (cancelled || !mounted.current) return
   246→        setScopesPayload(d)
   247→        setScopesError(null)
   248→        if (d.scopes.length) {
   249→          setScopeSel((cur) =>
   250→            cur.classId
   251→              ? cur
   252→              : { classId: d.scopes[0].classId, subjectId: d.scopes[0].subjects[0]?.subjectId ?? null },
   253→          )
   254→          setDate((cur) => cur || d.today)
   255→        }
   256→      })
   257→      .catch((e: unknown) => {
   258→        if (cancelled || !mounted.current) return
   259→        setScopesError(e instanceof Error && e.message === 'UNAUTHORIZED' ? 'UNAUTHORIZED' : friendlyError(e))
   260→      })
   261→    return () => {
   262→      cancelled = true
   263→    }
   264→  }, [scopesTick])
   265→
   266→  // ② the marking view — re-fetched on every scope / date change.
   267→  useEffect(() => {
   268→    if (!scopeSel.classId || !date) return
   269→    let cancelled = false
   270→    setViewLoading(true)
   271→    setViewError(null)
   272→    const params = new URLSearchParams({ classId: scopeSel.classId, date })
   273→    if (scopeSel.subjectId) params.set('subjectId', scopeSel.subjectId)
   274→    caRequest<ClassAttendanceViewDTO>(`/api/teacher/class-attendance?${params.toString()}`)
   275→      .then((d) => {
   276→        if (!cancelled && mounted.current) applyView(d)
   277→      })
   278→      .catch((e: unknown) => {
   279→        if (!cancelled && mounted.current) setViewError(friendlyError(e))
   280→      })
   281→      .finally(() => {
   282→        if (!cancelled && mounted.current) setViewLoading(false)
   283→      })
   284→    return () => {
   285→      cancelled = true
   286→    }
   287→  }, [scopeSel, date, viewTick, applyView])
   288→
   289→  const selectClass = useCallback((classId: string) => {
   290→    setScopeSel((cur) => {
   291→      if (cur.classId === classId) return cur
   292→      const scope = (view?.scopes ?? scopesPayload?.scopes ?? []).find((s) => s.classId === classId)
   293→      return { classId, subjectId: scope?.subjects[0]?.subjectId ?? null }
   294→    })
   295→  }, [view, scopesPayload])
   296→
   297→  const selectSubject = useCallback((subjectId: string) => {
   298→    setScopeSel((cur) => (cur.subjectId === subjectId ? cur : { ...cur, subjectId }))
   299→  }, [])
   300→
   301→  const setStatus = useCallback((studentId: string, status: AttendanceMark) => {
   302→    setDraft((prev) => (prev[studentId] === status ? prev : { ...prev, [studentId]: status }))
   303→  }, [])
   304→
   305→  const markAllPresent = useCallback((studentIds: string[]) => {
   306→    setDraft((prev) => {
   307→      const next = { ...prev }
   308→      let changed = 0
   309→      for (const id of studentIds) {
   310→        if (next[id] !== 'present') changed++
   311→        next[id] = 'present'
   312→      }
   313→      return changed ? next : prev
   314→    })
   315→  }, [])
   316→
   317→  const reloadView = useCallback(() => setViewTick((t) => t + 1), [])
   318→  const retryScopes = useCallback(() => setScopesTick((t) => t + 1), [])
   319→
   320→  // ③ the explicit submission — the ONLY write path. Single-flight; the
   321→  //    POST response replaces the view (persistence without a re-GET).
   322→  const submit = useCallback(async () => {
   323→    if (!view || submittingRef.current || !view.roster.length) return
   324→    submittingRef.current = true
   325→    setSubmitting(true)
   326→    const entries = view.roster.map((s) => ({
   327→      studentId: s.studentId,
   328→      status: draft[s.studentId] ?? 'present',
   329→    }))
   330→    try {
   331→      const data = await caRequest<ClassAttendanceViewDTO>('/api/teacher/class-attendance', {
   332→        method: 'POST',
   333→        body: JSON.stringify({
   334→          classId: view.classId,
   335→          subjectId: view.subjectId,
   336→          date: view.date,
   337→          entries,
   338→        }),
   339→      })
   340→      submittingRef.current = false
   341→      if (mounted.current) applyView(data)
   342→      const present = entries.filter((e) => e.status === 'present').length
   343→      if (data.mode === 'CT_DAILY') {
   344→        toast.success('Class baseline saved', {
   345→          description: `${data.classLabel} · ${fmtDay(data.date)} · ${present} of ${entries.length} present`,
   346→        })
   347→      } else if (data.isClassTeacher) {
   348→        toast.success(`${data.subjectName} attendance saved`, {
   349→          description: `${present} of ${entries.length} present · the class baseline for ${data.classLabel} was updated too`,
   350→        })
   351→      } else {
   352→        toast.success(`${data.subjectName} attendance saved`, {
   353→          description: `${data.classLabel} · ${fmtDay(data.date)} · ${present} of ${entries.length} present`,
   354→        })
   355→      }
   356→    } catch (e: unknown) {
   357→      submittingRef.current = false
   358→      const code = e instanceof Error ? e.message : ''
   359→      toast.error(friendlyError(e))
   360→      if (code === 'ROSTER_MISMATCH') reloadView()
   361→    } finally {
   362→      if (mounted.current) setSubmitting(false)
   363→    }
   364→  }, [view, draft, applyView, reloadView])
   365→
   366→  return {
   367→    scopes: view?.scopes ?? scopesPayload?.scopes ?? [],
   368→    scopesLoaded: scopesPayload != null,
   369→    scopesErrorMessage: scopesError,
   370→    today: view?.today ?? scopesPayload?.today ?? date,
   371→    scopeSel,
   372→    date,
   373→    view,
   374→    viewLoading,
   375→    viewError,
   376→    draft,
   377→    submitting,
   378→    selectClass,
   379→    selectSubject,
   380→    setDate,
   381→    setStatus,
   382→    markAllPresent,
   383→    reloadView,
   384→    retryScopes,
   385→    submit,
   386→  }
   387→}

   391→export function AttendanceModule() {
   392→  const reduce = useReducedMotion()
   393→  const [search, setSearch] = useState('')
   394→  const ca = useClassAttendance()
   395→  const { view, scopes, today, date, scopeSel, draft, submitting, viewLoading, viewError } = ca

   397→  // ── derived state (the whole UI reads from these) ──

   399→  const currentScope = useMemo(
   400→    () => scopes.find((s) => s.classId === scopeSel.classId) ?? null,
   401→    [scopes, scopeSel.classId],
   402→  )
   403→  const mode = view?.mode ?? (scopeSel.subjectId ? 'SUBJECT' : 'CT_DAILY')
   404→
   405→  const counts = useMemo(() => {
   406→    const c = { present: 0, absent: 0, late: 0, leave: 0 }
   407→    for (const s of roster) c[draft[s.studentId] ?? 'present']++
   408→    return c
   409→  }, [roster, draft])
   410→
   411→  const total = roster.length
   412→  const rate = total > 0 ? ((counts.present / total) * 100).toFixed(1) : '0.0'
   413→
   414→  const filtered = useMemo(() => {
   415→    const q = search.trim().toLowerCase()
   416→    if (!q) return roster
   417→    return roster.filter(
   418→      (s) => s.name.toLowerCase().includes(q) || (s.rollNo ?? '').toLowerCase().includes(q),
   419→    )
   420→  }, [roster, search])
   421→
   422→  // A change = any draft status differs from what is persisted for that
   423→  // student (own session → CT baseline → the implicit 'present' default).
   424→  const hasChanges = useMemo(
   425→    () => roster.some((s) => (draft[s.studentId] ?? 'present') !== persistedStatus(s)),
   426→    [roster, draft],
   427→  )
   428→  // "Persisted" only counts when EVERY roster student has a row — a day

   432→    view && total > 0 && (mode === 'SUBJECT'
   433→      ? view.meta.session.count >= total
   434→      : view.meta.baseline.count >= total),
   435→  )

   437→
   438→  const submittedNote =
   439→    view && mode === 'SUBJECT' && view.meta.session.submitted && view.meta.session.submittedAt
   440→      ? `Submitted ${fmtTime(view.meta.session.submittedAt)}`
   441→      : view && mode === 'CT_DAILY' && view.meta.baseline.marked
   442→        ? 'Baseline saved'
   443→        : null
   444→
   445→  const saveLabel = submitting
   446→    ? 'Saving…'
   447→    : clean
   448→      ? mode === 'SUBJECT' ? 'Submitted' : 'Saved'
   449→      : fullyPersisted
   450→        ? mode === 'SUBJECT' ? 'Update Submission' : 'Update Attendance'
   451→        : 'Save Attendance'
   452→  const saveDisabled = submitting || clean || !total
   453→
   454→  // Prefill semantics — calm, one line, never per-row shouting.
   455→  const showPrefillNote =
   456→    view != null && mode === 'SUBJECT' && !view.meta.session.submitted && view.meta.baseline.marked
   457→  const showClassTeacherNote =
   458→    view != null && (mode === 'CT_DAILY' || (mode === 'SUBJECT' && view.isClassTeacher && !view.meta.baseline.marked))
   459→
   460→  const handleBulkPresent = () => {
   461→    const ids = filtered.map((s) => s.studentId)
   462→    if (!ids.length) return
   463→    ca.markAllPresent(ids)
   464→    toast.success(`Marked ${ids.length} student${ids.length === 1 ? '' : 's'} present`, {
   465→      description: 'Review and save the attendance.',
   466→    })
   467→  }
   468→
   469→  // ── early states ──
   470→  if (!ca.scopesLoaded && ca.scopesErrorMessage != null) {
   471→    return (
   472→      <PageTransition className="space-y-5">
   473→        <GlassCard className="p-4" hover={false}>
   474→          <HubSectionError message={ca.scopesErrorMessage} onRetry={ca.retryScopes} />
   475→        </GlassCard>
   476→      </PageTransition>
   477→    )
   478→  }
   479→
   480→  if (!ca.scopesLoaded) {
   481→    return (
   482→      <PageTransition>
   483→        <HubModuleSkeleton />
   484→      </PageTransition>
   485→    )
   486→  }
   487→
   488→  if (!scopes.length) {
   489→    return (
   490→      <PageTransition className="space-y-5">
   491→        <GlassCard hover={false} className="overflow-hidden">
   492→          <HubEmptyState
   493→            icon={Users}
   494→            title="No classes assigned"
   495→            hint="Attendance follows your class-subject assignments and class-teacher duties. Once the school office assigns you a class, its roster appears here automatically."
   496→          />
   497→        </GlassCard>
   498→      </PageTransition>
   499→    )
   500→  }
   501→
   502→  return (
   503→    <PageTransition className="space-y-5">
   504→      <ModuleToolbar
   505→        className="flex-col items-stretch sm:flex-row sm:items-center"
   506→        context={`Mark daily attendance · ${fmtDay(date || today)}`}
   507→        action={
   508→          <div className="flex flex-wrap items-center gap-2">
   509→            <Select value={scopeSel.classId} onValueChange={ca.selectClass}>
   510→              <SelectTrigger className="h-9 w-[128px]" aria-label="Select class">
   511→                <SelectValue placeholder="Class" />
   512→              </SelectTrigger>
   513→              <SelectContent>
   514→                {scopes.map((s) => (
   515→                  <SelectItem key={s.classId} value={s.classId}>
   516→                    {s.classLabel}
   517→                  </SelectItem>
   518→                ))}
   519→              </SelectContent>
   520→            </Select>
   521→
   522→            {mode === 'SUBJECT' && currentScope ? (
   523→              <Select value={scopeSel.subjectId ?? undefined} onValueChange={ca.selectSubject}>
   524→                <SelectTrigger className="h-9 w-[136px]" aria-label="Select subject">
   525→                  <SelectValue placeholder="Subject" />
   526→                </SelectTrigger>
   527→                <SelectContent>
