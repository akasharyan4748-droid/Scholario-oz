     6→     6→ * Flow (mirrored by the compact 4-step stepper):
     7→     7→ *   1. Document type — compact 7-type picker grid, always visible, with a

     9→     9→ *      pre-selected by default so the Live Preview pane renders a REAL

    12→    12→ *      Marksheets add exam → class; Fee Receipts add the fee transaction.
    13→    13→ *   3. Details — template & style, default pre-selected.
    14→    14→ *   4. Preview & issue — live paper preview + Generate (store + toast).
    15→    15→ *
    16→    16→ * The preview pane wraps the real previews.tsx documents on a neutral desk
    17→    17→ * with a paper drop shadow — it must read as a printed document, not a web
    18→    18→ * dashboard.
    19→    19→ */

    52→    52→export function GenerateTab() {
    53→    53→  // Bonafide pre-selected — the preview pane shows a real document from
    54→    54→  // the first paint; the picker grid keeps the selection visible.
    55→    55→  const [docType, setDocType] = useState<DocType | null>('Bonafide')
    56→    56→  const [studentId, setStudentId] = useState<string>('')
    57→    57→  const [examId, setExamId] = useState<string>('')
    58→    58→  const [classId, setClassId] = useState<string>('')
    59→    59→  const [txnId, setTxnId] = useState<string>('')
    60→    60→  const [templateId, setTemplateId] = useState<string>('')
    61→    61→  const [purpose, setPurpose] = useState<string>('')
    62→    62→  const [genSearch, setGenSearch] = useState<string>('')
    63→    63→
    64→    64→  const students = useStudentsStore((s) => s.students)
    65→    65→  const exams = useMockExamsStore((s) => s.exams)
    66→    66→  const transactions = useFeeStore((s) => s.transactions)
    67→    67→  const templates = useCertificatesStore((s) => s.templates)
    68→    68→  const generateDocument = useCertificatesStore((s) => s.generateDocument)
    69→    69→
    70→    70→  // ─── Derived data ──────────────────────────────────────────────────
    71→    71→  const meta = docType ? DOC_TYPE_BY_LABEL[docType] : null
    72→    72→
    73→    73→  const filteredStudents = useMemo(() => {
    74→    74→    if (!genSearch.trim()) return students.slice(0, 40)
    75→    75→    const q = genSearch.toLowerCase().trim()
    76→    76→    return students
    77→    77→      .filter((s) =>
    78→    78→        s.name.toLowerCase().includes(q) ||
    79→    79→        s.admissionNo.toLowerCase().includes(q) ||
    80→    80→        s.className.toLowerCase().includes(q),
    81→    81→      )
    82→    82→      .slice(0, 40)
    83→    83→  }, [students, genSearch])
    84→    84→
    85→    85→  // For marksheet — only show students in the selected class
    86→    86→  const exam = exams.find((e) => e.id === examId)
    87→    87→  const examClasses = exam?.classes ?? []
    88→    88→  const examSubjects = exam ? exam.subjects.filter((s) => s.classId === classId) : []
    89→    89→  const marks = useMockMarksStore((s) => s.marks)
    90→    90→  const initMarks = useMockMarksStore((s) => s.initMarks)
    91→    91→
    92→    92→  // Auto-init marks for the selected exam when a class is picked (so the
    93→    93→  // marksheet preview has data even before the user opens the exam in the
    94→    94→  // Examinations module).
    95→    95→  useEffect(() => {
    96→    96→    if (!exam || !classId) return
    97→    97→    const hasMarks = marks.some((m) => m.examId === exam.id && m.classId === classId)
    98→    98→    if (hasMarks) return
    99→    99→    const classStudents = students
   100→   100→      .filter((s) => s.classId === classId)
   101→   101→      .map((s) => ({ id: s.id, name: s.name, rollNo: s.rollNo, classId: s.classId, className: s.className }))
   102→   102→    if (classStudents.length > 0) {
   103→   103→      initMarks(exam, classStudents)
   104→   104→    }
   105→   105→  }, [exam, classId, students, marks, initMarks])
   106→   106→
   107→   107→  // Filter students by class for marksheet selection
   108→   108→  const classStudents = useMemo(() => {
   109→   109→    if (docType !== 'Marksheet') return students
   110→   110→    return students.filter((s) => s.classId === classId)
   111→   111→  }, [students, classId, docType])
   112→   112→
   113→   113→  // ─── Template list for the active doc type ────────────────────────
   114→   114→  const docTemplates = useMemo(
   115→   115→    () => (docType ? templates.filter((t) => t.docType === docType && t.active) : []),
   116→   116→    [templates, docType],
   117→   117→  )
   118→   118→
   119→   119→  // Pick a default template when doc type changes
   120→   120→  useEffect(() => {
   121→   121→    if (!docType) return
   122→   122→    const def = docTemplates.find((t) => t.isDefault) ?? docTemplates[0]
   123→   123→    setTemplateId(def?.id ?? '')
   124→   124→  }, [docType, docTemplates])
   125→   125→
   126→   126→  // ─── Reset sub-state when doc type changes ────────────────────────
   127→   127→  useEffect(() => {
   128→   128→    setStudentId('')
   129→   129→    setExamId('')
   130→   130→    setClassId('')
   131→   131→    setTxnId('')
   132→   132→    setPurpose('')
   133→   133→    setGenSearch('')
   134→   134→  }, [docType])
   135→   135→
   136→   136→  // Pre-select the first eligible student so the Live Preview renders a
   137→   137→  // real document immediately (the picker shows the active check; the
   138→   138→  // operator can switch at any time before issuing).
   139→   139→  useEffect(() => {
   140→   140→    if (!docType) return
   141→   141→    if (studentId && students.some((s) => s.id === studentId)) return
   142→   142→    const pool = docType === 'Marksheet' ? classStudents : students
   143→   143→    if (pool.length > 0) setStudentId(pool[0].id)
   144→   144→  }, [docType, studentId, students, classStudents])
   145→   145→
   146→   146→  // ─── Selected student + transaction + marksheet data ──────────────
   147→   147→  const selectedStudent = students.find((s) => s.id === studentId) as StudentRecord | undefined
   148→   148→  const selectedTxn = transactions.find((t) => t.id === txnId) as FeeTransaction | undefined
   149→   149→  const selectedTemplate = templates.find((t) => t.id === templateId) as DocumentTemplate | undefined
   150→   150→
   151→   151→  // Compute marksheet data on-the-fly (cheap — at most ~10 subjects).
   152→   152→  const marksheetData: MarksheetData | undefined =
   153→   153→    docType === 'Marksheet' && selectedStudent && exam && classId
   154→   154→      ? computeMarksheet(exam, classId, selectedStudent, marks, examSubjects)
   155→   155→      : undefined
   156→   156→
   157→   157→  // ─── Can generate? ────────────────────────────────────────────────
   158→   158→  const canGenerate = (() => {
   159→   159→    if (!docType || !templateId) return false
   160→   160→    if (!selectedStudent) return false
   161→   161→    if (docType === 'Marksheet' && (!examId || !classId || !marksheetData)) return false
   162→   162→    if (docType === 'Fee Receipt' && !selectedTxn) return false
   163→   163→    return true
   164→   164→  })()
   165→   165→
   166→   166→  // ─── Stepper state (4 real flow steps) ────────────────────────────
   167→   167→  const studentStepDone = !!selectedStudent
   168→   168→    && !(meta?.needsExam && !(examId && classId && marksheetData))
   169→   169→    && !(meta?.needsFeeTxn && !selectedTxn)
   170→   170→  const stepIndex = !docType ? 0 : !studentStepDone ? 1 : !selectedTemplate ? 2 : 3
   171→   171→  const steps = ['Document type', 'Student', 'Details', 'Preview & issue'].map((label, i) => ({
   172→   172→    label,
   173→   173→    done: i < stepIndex,
   174→   174→    current: i === stepIndex,
   175→   175→  }))
   176→   176→
   177→   177→  // ─── Generate ─────────────────────────────────────────────────────
   178→   178→  function handleGenerate() {
   179→   179→    if (!docType || !selectedStudent || !selectedTemplate) return
   180→   180→    let data: Record<string, any> = {}
   181→   181→    let classInfo = selectedStudent.className
   182→   182→    if (docType === 'Marksheet' && marksheetData && exam) {
   183→   183→      data = { examId: exam.id, examName: exam.name, classId, marksheet: marksheetData }
   184→   184→      classInfo = `${exam.name} · ${selectedStudent.className}`
   185→   185→    } else if (docType === 'Fee Receipt' && selectedTxn) {
   186→   186→      data = {
   187→   187→        transactionId: selectedTxn.id,
   188→   188→        receiptNo: selectedTxn.receiptNo,
   189→   189→        amount: selectedTxn.amount,
   190→   190→        mode: selectedTxn.mode,
   191→   191→        purpose: selectedTxn.purpose,
   192→   192→      }
   193→   193→      classInfo = selectedTxn.className
   194→   194→    } else {
   195→   195→      data = { purpose: purpose || '—' }
   196→   196→    }
   197→   197→    const doc = generateDocument({
   198→   198→      docType,
   199→   199→      templateId,
   200→   200→      student: {
   201→   201→        id: selectedStudent.id,
   202→   202→        name: selectedStudent.name,
   203→   203→        admissionNo: selectedStudent.admissionNo,
   204→   204→        class: classInfo,
   205→   205→      },
   206→   206→      generatedBy: school.principal,
   207→   207→      data,
   208→   208→    })
   209→   209→    toast.success(`${docType} generated`, {
   210→   210→      description: `${doc.docNumber} · ${selectedStudent.name}`,
   211→   211→    })
   212→   212→  }
   213→   213→
   214→   214→  // ─── Render ────────────────────────────────────────────────────────
   215→   215→  return (
   216→   216→    <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   217→   217→      {/* LEFT — workflow */}
   218→   218→      <div className="lg:col-span-5 space-y-4">
   219→   219→        {/* Compact step indicator — 4 real steps; completed get a check,
   220→   220→            the current step is emphasised. */}
   221→   221→        <FlowStepper steps={steps} />
   222→   222→
   223→   223→        {/* Step 1 — Doc type picker (always visible so switching types is
   224→   224→            one click; the selected card carries ring + check + tint). */}
   225→   225→        <CertPanel
   226→   226→          title="Document type"
   227→   227→          subtitle={docType ? DOC_TYPE_BY_LABEL[docType].description : 'Pick the document to issue'}
   228→   228→        >
   229→   229→          <div className="grid grid-cols-2 gap-2">
   230→   230→            {DOC_TYPES.map((d) => (
   231→   231→              <DocTypeCard
   232→   232→                key={d.label}
   233→   233→                meta={d}
   234→   234→                selected={docType === d.label}
   235→   235→                onClick={() => setDocType(d.label)}
   236→   236→              />
   237→   237→            ))}
   238→   238→          </div>
   239→   239→        </CertPanel>
   240→   240→
   241→   241→        {docType && (
   242→   242→          <CertPanel
   243→   243→            title="Student"
   244→   244→            subtitle={
   245→   245→              meta?.needsExam
   246→   246→                ? 'Examination → Class → Student'
   247→   247→                : meta?.needsFeeTxn
   248→   248→                  ? 'Student → Fee transaction'
   249→   249→                  : 'Search and pick the student'
   250→   250→            }
   251→   251→          >
   252→   252→            {meta?.needsExam ? (
   253→   253→              <div className="space-y-3">
   254→   254→                <FieldLabel label="Examination" />
   255→   255→                <Select value={examId} onValueChange={setExamId}>
   256→   256→                  <SelectTrigger className="h-9 text-xs">
   257→   257→                    <SelectValue placeholder="Choose examination" />
   258→   258→                  </SelectTrigger>
   259→   259→                  <SelectContent>
   260→   260→                    {exams.map((e) => (
   261→   261→                      <SelectItem key={e.id} value={e.id}>
   262→   262→                        {e.name} · {e.session} · {e.status}
   263→   263→                      </SelectItem>
   264→   264→                    ))}
   265→   265→                  </SelectContent>
   266→   266→                </Select>
   267→   267→
   268→   268→                <FieldLabel label="Class" />
   269→   269→                <Select value={classId} onValueChange={setClassId} disabled={!examId}>
   270→   270→                  <SelectTrigger className="h-9 text-xs">
   271→   271→                    <SelectValue placeholder={examId ? 'Choose class' : 'Pick examination first'} />
   272→   272→                  </SelectTrigger>
   273→   273→                  <SelectContent>
   274→   274→                    {examClasses.map((c) => (
   275→   275→                      <SelectItem key={c.classId} value={c.classId}>
   276→   276→                        {c.className}{c.stream ? ` · ${c.stream}` : ''}
   277→   277→                      </SelectItem>
   278→   278→                    ))}
   279→   279→                  </SelectContent>
   280→   280→                </Select>
   281→   281→
   282→   282→                <FieldLabel label="Student" />
   283→   283→                <StudentPicker
   284→   284→                  students={classStudents}
   285→   285→                  value={studentId}
   286→   286→                  onChange={setStudentId}
   287→   287→                  search={genSearch}
   288→   288→                  onSearch={setGenSearch}
   289→   289→                />
   290→   290→              </div>
   291→   291→            ) : (
   292→   292→              <div className="space-y-3">
   293→   293→                <FieldLabel label="Student" />
   294→   294→                <StudentPicker
   295→   295→                  students={filteredStudents}
   296→   296→                  value={studentId}
   297→   297→                  onChange={setStudentId}
   298→   298→                  search={genSearch}
   299→   299→                  onSearch={setGenSearch}
   300→   300→                />
   301→   301→
   302→   302→                {meta?.needsFeeTxn && selectedStudent && (
   303→   303→                  <>
   304→   304→                    <FieldLabel label="Fee transaction" />
   305→   305→                    <Select
   306→   306→                      value={txnId}
   307→   307→                      onValueChange={setTxnId}
   308→   308→                    >
   309→   309→                      <SelectTrigger className="h-9 text-xs">
   310→   310→                        <SelectValue placeholder="Choose fee payment" />
   311→   311→                      </SelectTrigger>
   312→   312→                      <SelectContent>
   313→   313→                        {transactions
   314→   314→                          .filter((t) => t.studentId === selectedStudent.id)
   315→   315→                          .slice(0, 25)
   316→   316→                          .map((t) => (
   317→   317→                            <SelectItem key={t.id} value={t.id}>
   318→   318→                              {t.receiptNo} · {formatDate(t.date)} · {formatINR(t.amount)} · {t.mode}
   319→   319→                            </SelectItem>
   320→   320→                          ))}
   321→   321→                        {transactions.filter((t) => t.studentId === selectedStudent.id).length === 0 && (
   322→   322→                          <SelectItem value="__none" disabled>No transactions found for this student</SelectItem>
   323→   323→                        )}
   324→   324→                      </SelectContent>
   325→   325→                    </Select>
   326→   326→                  </>
   327→   327→                )}
   328→   328→
   329→   329→                {docType === 'Bonafide' && selectedStudent && (
   330→   330→                  <>
   331→   331→                    <FieldLabel label="Purpose (optional)" />
   332→   332→                    <Input
   333→   333→                      value={purpose}
   334→   334→                      onChange={(e) => setPurpose(e.target.value)}
   335→   335→                      placeholder="e.g. Bank account opening, passport, scholarship"
   336→   336→                      className="h-9 text-xs"
   337→   337→                    />
   338→   338→                  </>
   339→   339→                )}
   340→   340→              </div>
   341→   341→            )}
   342→   342→          </CertPanel>
   343→   343→        )}
   344→   344→
   345→   345→        {docType && docTemplates.length > 0 && (
   346→   346→          <CertPanel
   347→   347→            title="Template & style"
   348→   348→            action={selectedTemplate && (
   349→   349→              <span className="text-[10px] text-muted-foreground">
   350→   350→                Style: <strong className="text-foreground">{selectedTemplate.style}</strong>
   351→   351→              </span>
   352→   352→            )}
   353→   353→          >
   354→   354→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   355→   355→              {docTemplates.map((t) => {
   356→   356→                const active = t.id === templateId
   357→   357→                return (
   358→   358→                  <button
   359→   359→                    key={t.id}
   360→   360→                    onClick={() => setTemplateId(t.id)}
   361→   361→                    className={cn(
   362→   362→                      'relative flex flex-col items-start gap-1.5 rounded-lg border p-2 text-left transition-all hover:-translate-y-0.5',
   363→   363→                      active
   364→   364→                        ? 'border-emerald-500/40 bg-emerald-500/[0.05] ring-2 ring-emerald-500/30'
   365→   365→                        : 'border-border hover:border-foreground/40',
   366→   366→                    )}
   367→   367→                  >
   368→   368→                    <div className="flex items-center gap-1.5 w-full">
   369→   369→                      <span
   370→   370→                        className="h-4 w-4 rounded shrink-0"
   371→   371→                        style={{ background: t.accentColor }}
   372→   372→                      />
   373→   373→                      <span className="text-[10px] font-semibold leading-tight truncate">{t.style}</span>
   374→   374→                      {t.isDefault && (
   375→   375→                        <span className="ml-auto text-[8px] px-1 py-0.5 rounded bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 font-semibold">
   376→   376→                          DEFAULT
   377→   377→                        </span>
   378→   378→                      )}
   379→   379→                    </div>
   380→   380→                    <p className="text-[9px] text-muted-foreground leading-tight line-clamp-2">{t.name}</p>
   381→   381→                    {active && (
   382→   382→                      <span className="absolute -top-1.5 -right-1.5 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-600 text-white">
   383→   383→                        <Check className="h-2.5 w-2.5" />
   384→   384→                      </span>
   385→   385→                    )}
   386→   386→                  </button>
   387→   387→                )
   388→   388→              })}
   389→   389→            </div>
   390→   390→          </CertPanel>
   391→   391→        )}
   392→   392→
   393→   393→        {docType && (
   394→   394→          <div className="flex items-center gap-2">
   395→   395→            <Button
   396→   396→              size="sm"
   397→   397→              disabled={!canGenerate}
   398→   398→              onClick={handleGenerate}
   399→   399→              className="flex-1 h-9 text-xs gap-1.5 bg-emerald-600 hover:bg-emerald-700 text-white"
   400→   400→            >
   401→   401→              <Sparkles className="h-3.5 w-3.5" /> Generate {docType}
   402→   402→            </Button>
   403→   403→            <Button
   404→   404→              size="sm"
   405→   405→              variant="outline"
   406→   406→              disabled={!canGenerate}
   407→   407→              onClick={() => window.print()}
   408→   408→              className="h-9 text-xs gap-1 no-print"
   409→   409→            >
   410→   410→              <Printer className="h-3.5 w-3.5" /> Print
   411→   411→            </Button>
   412→   412→          </div>
   413→   413→        )}
   414→   414→      </div>
   415→   415→
   416→   416→      {/* RIGHT — live preview on a paper desk */}
   417→   417→      <div className="lg:col-span-7">
   418→   418→        <CertPanel
   419→   419→          title="Live preview"
   420→   420→          subtitle={docType ? `${docType}${selectedStudent ? ` · ${selectedStudent.name}` : ''}` : 'Pick a document type'}
   421→   421→          className="h-full"
   422→   422→          bodyClassName="p-2 sm:p-4 bg-muted/40"
   423→   423→        >
   424→   424→          <PreviewArea
   425→   425→            docType={docType}
   426→   426→            student={selectedStudent}
   427→   427→            template={selectedTemplate}
   428→   428→            txn={selectedTxn}
   429→   429→            marksheetData={marksheetData}
   430→   430→            purpose={purpose}
   431→   431→          />
   432→   432→        </CertPanel>
   433→   433→      </div>
   434→   434→    </div>
   435→   435→  )
   436→   436→}

   440→   440→/** Compact horizontal 4-step indicator. Completed steps show a check;
   441→   441→ *  the current step is emphasised (dark pill). Wraps on narrow screens. */
   442→   442→function FlowStepper({ steps }: { steps: { label: string; done: boolean; current: boolean }[] }) {
   443→   443→  return (
   444→   444→    <div className="flex items-center gap-1 flex-wrap" aria-label="Generation progress">
   445→   445→      {steps.map((s, i) => (
   446→   446→        <div key={s.label} className="flex items-center gap-1">
   447→   447→          {i > 0 && <ChevronRight className="h-3 w-3 text-muted-foreground/40" aria-hidden />}

   668→   668→import type { ExamDTO, ExamSubjectConfigDTO, ExamMarkDTO } from '@/lib/exams/types'
   669→   669→
   670→   670→function computeMarksheet(
   671→   671→  exam: ExamDTO,
   672→   672→  classId: string,
   673→   673→  student: StudentRecord,
   674→   674→  marks: ExamMarkDTO[],
   675→   675→  examSubjects: ExamSubjectConfigDTO[],
   676→   676→): MarksheetData {
   677→   677→  // Try real exam marks first
   678→   678→  if (examSubjects.length > 0) {
   679→   679→    const rows = examSubjects.map((subj) => {
   680→   680→      const mark = marks.find(
   681→   681→        (m) =>
   682→   682→          m.examId === exam.id &&
   683→   683→          m.classId === classId &&
   684→   684→          m.subjectId === subj.subjectId &&
   685→   685→          m.studentId === student.id,
   686→   686→      )
   687→   687→      const obtained = mark?.marksObtained ?? 0
   688→   688→      return {
   689→   689→        subject: subj.subjectName,
   690→   690→        max: subj.maxMarks,
   691→   691→        pass: subj.passMarks,
   692→   692→        obtained,
   693→   693→        isAbsent: mark?.status === 'ABSENT',
   694→   694→      }
   695→   695→    })
   696→   696→    const totalMax = rows.reduce((s, r) => s + r.max, 0)
   697→   697→    const totalObtained = rows.reduce((s, r) => s + (r.isAbsent ? 0 : r.obtained), 0)
   698→   698→    const pct = totalMax > 0 ? (totalObtained / totalMax) * 100 : 0
   699→   699→    const passed = rows.every((r) => r.isAbsent || r.obtained >= r.pass)
   700→   700→    return {
   701→   701→      examName: exam.name,
   702→   702→      className: student.className,
   703→   703→      section: student.section,
   704→   704→      session: exam.session ?? school.academicYear,
   705→   705→      rows,
   706→   706→      totalMax,
   707→   707→      totalObtained,
   708→   708→      percentage: pct,
   709→   709→      grade: gradeFor(pct),
   710→   710→      result: passed ? 'PASS' : 'FAIL',
   711→   711→      remarks: passed ? 'Conduct: Excellent. Regularity: Satisfactory.' : 'Re-examination recommended.',
   712→   712→    }
   713→   713→  }
   714→   714→  // Fallback — derive from student.academics.subjects (max 100)
   715→   715→  const rows: MarksheetRow[] = student.academics.subjects.map((subj) => ({
   716→   716→    subject: subj.name,
   717→   717→    max: 100,
   718→   718→    pass: 33,
   719→   719→    obtained: Math.round((subj.percent / 100) * 100),
   720→   720→  }))
   721→   721→  const totalMax = rows.length * 100
   722→   722→  const totalObtained = rows.reduce((s, r) => s + r.obtained, 0)
   723→   723→  const pct = totalMax > 0 ? (totalObtained / totalMax) * 100 : 0
   724→   724→  return {
   725→   725→    examName: 'Academic Performance Report',
   726→   726→    className: student.className,
   727→   727→    section: student.section,
   728→   728→    session: school.academicYear,
   729→   729→    rows,
   730→   730→    totalMax,
   731→   731→    totalObtained,
   732→   732→    percentage: pct,
   733→   733→    grade: gradeFor(pct),
   734→   734→    result: pct >= 33 ? 'PASS' : 'FAIL',
   735→   735→    remarks: 'Based on cumulative academic record.',
   736→   736→  }
   737→   737→}
   738→   738→
   739→   739→function gradeFor(pct: number): string {
   740→   740→  if (pct >= 90) return 'A1'
   741→   741→  if (pct >= 80) return 'A2'
   742→   742→  if (pct >= 70) return 'B1'
   743→   743→  if (pct >= 60) return 'B2'
   744→   744→  if (pct >= 50) return 'C1'
   745→   745→  if (pct >= 33) return 'C2'
   746→   746→  return 'E'
   747→   747→}
