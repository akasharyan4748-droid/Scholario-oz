     2→ * lesson-planner — server-side engine for the Automatic Curriculum
     3→ * Execution Lesson Planner.
     4→ *
     5→ * ARCHITECTURE (v3):
     6→ *   1. Authorization: an ACTIVE ClassSubjectAssignment (teacherId = the
     7→ *      signed-in teacher) is the ONLY source of planner access. Class
     8→ *      teachers get nothing here unless they also teach the subject.
     9→ *   2. Scheduling: the engine derives each topic's planned school day from

    11→ *      skipping Sundays and SchoolEvent rows of type HOLIDAY. Classes
    12→ *      without timetable rows for the subject fall back to

    20→ *      flags write LessonExecution rows — the layer that survives every

    79→/** SchoolEvent(type=HOLIDAY) rows expanded into a set of YYYY-MM-DD days. */
    80→export async function loadHolidaySet(schoolId: string, fromDay: string, toDay: string): Promise<Set<string>> {
    81→  const events = await db.schoolEvent.findMany({
    82→    where: {
    83→      schoolId,
    84→      type: 'HOLIDAY',
    85→      startDate: { lte: dayToDate(addDays(toDay, 1)) },
    86→      OR: [{ endDate: null }, { endDate: { gte: dayToDate(addDays(fromDay, -1)) } }],
    87→    },
    88→    select: { startDate: true, endDate: true },
    89→  })
    90→  const set = new Set<string>()
    91→  for (const ev of events) {
    92→    let cur = dateToDay(ev.startDate)
    93→    const end = ev.endDate ? dateToDay(ev.endDate) : cur
    94→    let guard = 0
    95→    while (cur <= end && guard < MAX_HOLIDAY_SPAN_DAYS) {
    96→      set.add(cur)
    97→      cur = addDays(cur, 1)
    98→      guard++
    99→    }
   100→  }
   101→  return set
   102→}

   113→export async function loadPacing(classId: string, subjectId: string, minutesPerPeriod: number): Promise<Pacing> {
   114→  const rows = await db.timetable.findMany({ where: { classId, subjectId }, select: { day: true } })
   115→  const perWeekday: Record<string, number> = {}
   116→  for (const r of rows) perWeekday[r.day] = (perWeekday[r.day] ?? 0) + 1
   117→  if (rows.length > 0) {
   118→    return {
   119→      source: 'TIMETABLE',
   120→      periodsPerWeek: rows.length,
   121→      minutesPerWeek: rows.length * minutesPerPeriod,
   122→      perWeekday,
   123→    }
   124→  }
   125→  return {
   126→    source: 'FALLBACK',
   127→    periodsPerWeek: null,
   128→    minutesPerWeek: minutesPerPeriod * SCHOOL_WEEKDAYS.size,
   129→    perWeekday,
   130→  }
   131→}

   139→export interface WalkSlot {
   140→  topicId: string
   141→  /** school day the topic starts on */
   142→  day: string
   143→  /** school day the topic's allocated minutes run out on */
   144→  lastDay: string
   145→}

   147→/**
   148→ * Greedy contiguous walk: topics consume capacity in course order; a day
   149→ * only offers capacity when it is a working day, not a holiday, and the
   150→ * timetable gives the subject periods that weekday (or pacing falls back
   151→ * to minutesPerPeriod on every working day). Several short topics may
   152→ * share one day; a long topic spans several days (its slot keeps the
   153→ * FIRST day).
   154→ */
   155→export function walkSchedule(
   156→  pending: WalkTopic[],
   157→  anchorDay: string,
   158→  holidays: Set<string>,
   159→  pacing: Pacing,
   160→  minutesPerPeriod: number,
   161→  /** >0 = anchor day's REMAINING capacity · 0 = fresh anchor · -1 = anchor exhausted */
   162→  initialCarry = 0,
   163→): WalkSlot[] {
   164→  const capacityFor = (day: string): number => {
   165→    if (holidays.has(day)) return 0
   166→    const wd = weekdayOf(day)
   167→    if (!SCHOOL_WEEKDAYS.has(wd)) return 0
   168→    if (pacing.source === 'FALLBACK') return minutesPerPeriod
   169→    return (pacing.perWeekday[WEEKDAY_NAMES[wd]] ?? 0) * minutesPerPeriod
   170→  }
   171→  const out: WalkSlot[] = []
   172→  // Cursor semantics (capacityFor returns a day's FULL capacity, so the
   173→  // advance loop must never re-enter a day whose capacity is spent):
   174→  //   initialCarry > 0 → consume the anchor's remaining capacity first
   175→  //   initialCarry < 0 → anchor day already exhausted → start after it
   176→  //   initialCarry = 0 → fresh anchor → first advance lands ON the anchor
   177→  let cursor: string
   178→  let carry: number
   179→  if (initialCarry > 0) {
   180→    cursor = anchorDay
   181→    carry = initialCarry
   182→  } else if (initialCarry < 0) {
   183→    cursor = anchorDay
   184→    carry = 0
   185→  } else {
   186→    cursor = addDays(anchorDay, -1)
   187→    carry = 0
   188→  }
   189→  let guard = 0
   190→  for (const t of pending) {
   191→    let remaining = Math.max(1, t.estimatedMinutes)
   192→    let firstDay: string | null = null
   193→    let lastDay: string | null = null
   194→    while (remaining > 0) {
   195→      if (guard++ > 50_000) return out
   196→      if (carry <= 0) {
   197→        // The current day is exhausted — never return to it. Advance to
   198→        // the next day, then skip forward to the first day with capacity.
   199→        let next = addDays(cursor, 1)
   200→        let inner = 0
   201→        while (capacityFor(next) <= 0 && inner < WALK_HORIZON_DAYS) {
   202→          next = addDays(next, 1)
   203→          inner++
   204→        }
   205→        cursor = next
   206→        carry = capacityFor(cursor)
   207→        if (carry <= 0) break // horizon exhausted
   208→      }
   209→      const used = Math.min(carry, remaining)
   210→      carry -= used
   211→      remaining -= used
   212→      if (!firstDay) firstDay = cursor
   213→      lastDay = cursor
   214→    }
   215→    out.push({ topicId: t.id, day: firstDay ?? cursor, lastDay: lastDay ?? firstDay ?? cursor })
   216→  }
   217→  return out
   218→}

   245→export interface PlannedPending {

   247→  day: string

   253→/**
   254→ * Re-plan every PENDING slot:
   255→ *  - anchor = today when any pending slot has fallen behind (self-heal),
   256→ *    else the earliest future pending date (deterministic when the
   257→ *    calendar is unchanged, shifts when holidays/timetable change).
   258→ *  - capacity already consumed on the anchor day by COMPLETED slots is
   259→ *    subtracted so completing a lesson never double-books the day.
   260→ */
   261→export function computePendingPlan(topics: TopicRow[], slots: SlotRow[], holidays: Set<string>, pacing: Pacing, minutesPerPeriod: number): PlannedPending[] {
   262→  const today = todayStr()
   263→  const completedIds = new Set(slots.filter((s) => s.status === 'COMPLETED').map((s) => s.curriculumTopicId))
   264→  const topicById = new Map(topics.map((t) => [t.id, t]))
   265→  const pendingSlots = slots.filter((s) => s.status !== 'COMPLETED' && topicById.has(s.curriculumTopicId))
   266→
   267→  const futureDays = pendingSlots
   268→    .map((s) => dateToDay(s.plannedDate))
   269→    .filter((d) => d >= today)
   270→    .sort()
   271→  const behind = pendingSlots.some((s) => dateToDay(s.plannedDate) < today)
   272→  // A topic completed TODAY means the stored forward plan is stale — the
   273→  // remaining topics must be re-laid from today (early completion pulls
   274→  // the future forward). Otherwise keep the earliest future date as the
   275→  // anchor so day-over-day dates stay stable unless inputs change.
   276→  const lastTaughtDay = slots.reduce<string | null>((acc, s) => {
   277→    if (s.status !== 'COMPLETED' || !s.execution) return acc
   278→    const d = dateToDay(s.execution.taughtOn)
   279→    return !acc || d > acc ? d : acc
   280→  }, null)
   281→  const completedToday = Boolean(lastTaughtDay && lastTaughtDay >= today)
   282→  const anchor = behind || completedToday || futureDays.length === 0 ? today : futureDays[0]
   283→
   284→  let initialCarry = 0
   285→  {
   286→    const capacityFor = (day: string): number => {
   287→      if (holidays.has(day)) return 0
   288→      const wd = weekdayOf(day)
   289→      if (!SCHOOL_WEEKDAYS.has(wd)) return 0
   290→      if (pacing.source === 'FALLBACK') return minutesPerPeriod
   291→      return (pacing.perWeekday[WEEKDAY_NAMES[wd]] ?? 0) * minutesPerPeriod
   292→    }
   293→    if (anchor === today) {
   294→      // capacity actually used today = minutes of topics TAUGHT today
   295→      // (whatever their original planned date was). A exhausted day is
   296→      // signalled with -1 so the walk starts tomorrow.
   297→      const consumed = slots
   298→        .filter((s) => s.status === 'COMPLETED' && s.execution && dateToDay(s.execution.taughtOn) === anchor)
   299→        .reduce((a, s) => a + s.minutesAllocated, 0)
   300→      const cap = capacityFor(anchor)
   301→      initialCarry = consumed <= 0 ? 0 : cap - consumed > 0 ? cap - consumed : -1
   302→    }
   303→  }
   304→
   305→  const pendingTopics = topics.filter((t) => !completedIds.has(t.id))
   306→  const walk = walkSchedule(
   307→    pendingTopics.map((t) => ({ id: t.id, estimatedMinutes: t.estimatedMinutes })),
   308→    anchor,
   309→    holidays,
   310→    pacing,
   311→    minutesPerPeriod,
   312→    initialCarry,
   313→  )
   314→  const sequenceByTopic = new Map(topics.map((t, i) => [t.id, i + 1]))
   315→  return walk.map((w) => ({
   316→    curriculumTopicId: w.topicId,
   317→    day: w.day,
   318→    lastDay: w.lastDay,
   319→    sequence: sequenceByTopic.get(w.topicId) ?? 1,
   320→    minutes: topicById.get(w.topicId)?.estimatedMinutes ?? minutesPerPeriod,
   321→  }))
   322→}

   339→// ─── sync (persists the computed plan for PENDING slots) ──────────────
   340→
   341→async function syncSchedule(
   342→  schoolId: string,
   343→  classId: string,
   344→  subjectId: string,
   345→  curriculumId: string,
   346→  topics: TopicRow[],
   347→  slots: SlotRow[],
   348→  holidays: Set<string>,
   349→  pacing: Pacing,
   350→  minutesPerPeriod: number,
   351→): Promise<SlotRow[]> {
   352→  const plan = computePendingPlan(topics, slots, holidays, pacing, minutesPerPeriod)
   353→  const planByTopic = new Map(plan.map((p) => [p.curriculumTopicId, p]))
   354→  const topicById = new Map(topics.map((t) => [t.id, t]))
   355→  const slotByTopic = new Map(slots.map((s) => [s.curriculumTopicId, s]))
   356→
   357→  const kept: SlotRow[] = []
   358→  for (const s of slots) {
   359→    if (s.status === 'COMPLETED' || !topicById.has(s.curriculumTopicId)) {
   360→      if (s.status === 'COMPLETED') kept.push(s)
   361→      else await db.scheduledCurriculumTopic.delete({ where: { id: s.id } }).catch(() => {})
   362→      continue
   363→    }
   364→    const p = planByTopic.get(s.curriculumTopicId)
   365→    if (!p) continue
   366→    const wantDay = dayToDate(p.day)
   367→    if (dateToDay(s.plannedDate) !== p.day || s.minutesAllocated !== p.minutes || s.sequence !== p.sequence) {
   368→      const updated = await db.scheduledCurriculumTopic.update({
   369→        where: { id: s.id },
   370→        data: { plannedDate: wantDay, minutesAllocated: p.minutes, sequence: p.sequence, updatedAt: new Date() },
   371→        include: { execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } } },
   372→      })
   373→      kept.push(updated as SlotRow)
   374→    } else {
   375→      kept.push(s)
   376→    }
   377→  }
   378→  for (const p of plan) {
   379→    if (slotByTopic.has(p.curriculumTopicId)) continue
   380→    const created = await db.scheduledCurriculumTopic.create({
   381→      data: {
   382→        schoolId,
   383→        classId,
   384→        subjectId,
   385→        curriculumTopicId: p.curriculumTopicId,
   386→        sequence: p.sequence,
   387→        plannedDate: dayToDate(p.day),
   388→        minutesAllocated: p.minutes,
   389→        status: 'PENDING',
   390→      },
   391→      include: { execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } } },
   392→    })
   393→    kept.push(created as SlotRow)
   394→  }
   395→  return kept
   396→}

   441→function deriveStatus(slot: SlotRow, today: string): LessonDisplayStatus {
   442→  if (slot.status === 'COMPLETED') return 'COMPLETED'
   443→  if (slot.status === 'NEEDS_RESCHEDULING') return 'NEEDS_RESCHEDULING'
   444→  const day = dateToDay(slot.plannedDate)
   445→  if (day === today) return slot.execution?.startedAt && !slot.execution.completedAt ? 'IN_PROGRESS' : 'TODAY'
   446→  return 'UPCOMING'
   447→}

   469→function buildStats(topics: TopicRow[], slots: SlotRow[], pendingPlan: PlannedPending[], originalPlan: WalkSlot[], startDate: Date, today: string): CurriculumStatsDTO {
   470→  const completed = slots.filter((s) => s.status === 'COMPLETED')
   471→  const minutesTotal = topics.reduce((a, t) => a + t.estimatedMinutes, 0)
   472→  const minutesCompleted = completed.reduce((a, s) => a + s.minutesAllocated, 0)
   473→  const expectedByNow = originalPlan.filter((p) => p.lastDay < today).length
   474→  const behindBy = Math.max(0, expectedByNow - completed.length)
   475→  const projected = pendingPlan.length ? pendingPlan[pendingPlan.length - 1].lastDay : null
   476→  return {
   477→    totalTopics: topics.length,
   478→    completedTopics: completed.length,
   479→    percentComplete: topics.length ? Math.round((completed.length / topics.length) * 100) : 0,
   480→    minutesTotal,
   481→    minutesCompleted,
   482→    projectedEndDate: projected,
   483→    onTrack: dateToDay(startDate) >= today ? null : behindBy === 0,
   484→    behindBy,
   485→  }
   486→}

   488→function buildUnits(topics: TopicRow[], slots: SlotRow[], today: string): CurriculumUnitDTO[] {
   489→  const slotByTopic = new Map(slots.map((s) => [s.curriculumTopicId, s]))
   490→  const byUnit = new Map<number, { unitName: string; topics: TopicRow[] }>()
   491→  for (const t of topics) {
   492→    const g = byUnit.get(t.unitOrder) ?? { unitName: t.unitName, topics: [] }
   493→    g.topics.push(t)
   494→    byUnit.set(t.unitOrder, g)
   495→  }
   496→  const units: CurriculumUnitDTO[] = []
   497→  for (const [unitOrder, g] of [...byUnit.entries()].sort((a, b) => a[0] - b[0])) {
   498→    const dtos = g.topics
   499→      .map((t) => {
   500→        const slot = slotByTopic.get(t.id)
   501→        return slot ? toTopicDTO(slot, t, today) : null
   502→      })
   503→      .filter((d): d is ScheduledTopicDTO => d !== null)
   504→      .sort((a, b) => a.sequence - b.sequence)
   505→    const done = dtos.filter((d) => d.status === 'COMPLETED').length
   506→    units.push({
   507→      unitOrder,
   508→      unitName: g.unitName,
   509→      totalTopics: g.topics.length,
   510→      completedTopics: done,
   511→      percent: g.topics.length ? Math.round((done / g.topics.length) * 100) : 0,
   512→      topics: dtos,
   513→    })
   514→  }
   515→  return units
   516→}

   522→
   523→/**
   524→ * The whole planner workspace in one server-resolved call. `classId` +
   525→ * `subjectId` are optional (defaults to the first assignment); both are
   526→ * re-validated against the teacher's ACTIVE assignments when supplied.
   527→ */
   528→export async function buildLessonPlannerPayload(ctx: TeacherHubContext, classId?: string | null, subjectId?: string | null): Promise<LessonPlannerPayload> {
   529→  const assignments = await assignmentsWithPending(ctx)
   530→  if (!assignments.length) return { assignments, selected: null }
   531→
   532→  let scope = assignments.find((a) => a.classId === classId && a.subjectId === subjectId) ?? null
   533→  if (!scope && classId && subjectId) {
   534→    await assertLessonScope(ctx, classId, subjectId) // throws NOT_FOUND / FORBIDDEN
   535→  }
   536→  if (!scope) scope = assignments[0]
   537→
   538→  const selected = await buildSelectedScope(ctx, scope)
   539→  return { assignments, selected }
   540→}

   542→/** Assignments with hasCurriculum + pendingToday (read-only, no writes). */
   543→async function assignmentsWithPending(ctx: TeacherHubContext): Promise<LessonAssignmentDTO[]> {
   544→  const base = await teachingAssignments(ctx)
   545→  if (!base.length) return base
   546→  const today = todayStr()
   547→  const classIds = [...new Set(base.map((a) => a.classId))]
   548→
   549→  const curricula = await db.schoolCurriculum.findMany({
   550→    where: { schoolId: ctx.schoolId, classId: { in: classIds }, status: 'ACTIVE' },
   551→    select: { id: true, classId: true, subjectId: true, startDate: true, minutesPerPeriod: true, session: true },
   552→    orderBy: { session: 'desc' },
   553→  })
   554→  const curriculumByScope = new Map<string, (typeof curricula)[number]>()
   555→  for (const c of curricula) {
   556→    const k = `${c.classId}|${c.subjectId}`
   557→    if (!curriculumByScope.has(k)) curriculumByScope.set(k, c) // latest session wins
   558→  }
   559→
   560→  const curriculumIds = [...curriculumByScope.values()].map((c) => c.id)
   561→  const [topicRows, slotRows, ttRows, holidays] = await Promise.all([
   562→    curriculumIds.length
   563→      ? db.schoolCurriculumTopic.findMany({
   564→          where: { curriculumId: { in: curriculumIds } },
   565→          select: { id: true, curriculumId: true, unitOrder: true, unitName: true, topicOrder: true, topicName: true, description: true, estimatedMinutes: true },
   566→        })
   567→      : [],
   568→    curriculumIds.length
   569→      ? db.scheduledCurriculumTopic.findMany({
   570→          where: { classId: { in: classIds } },
   571→          select: {
   572→            id: true, curriculumTopicId: true, sequence: true, plannedDate: true, minutesAllocated: true,
   573→            status: true, rescheduledFrom: true, flagReason: true, classId: true, subjectId: true,
   574→            execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } },
   575→          },
   576→        })
   577→      : [],
   578→    db.timetable.findMany({ where: { classId: { in: classIds } }, select: { classId: true, subjectId: true, day: true } }),
   579→    loadHolidaySet(ctx.schoolId, addDays(today, -30), addDays(today, WALK_HORIZON_DAYS)),
   580→  ])
   581→
   582→  for (const a of base) {
   583→    const curriculum = curriculumByScope.get(`${a.classId}|${a.subjectId}`)
   584→    a.hasCurriculum = Boolean(curriculum)
   585→    if (!curriculum) continue
   586→    const topics = topicRows.filter((t) => t.curriculumId === curriculum.id)
   587→    if (!topics.length) continue
   588→    const slots = slotRows.filter((s) => s.classId === a.classId && s.subjectId === a.subjectId)
   589→    const perWeekday: Record<string, number> = {}
   590→    let periods = 0
   591→    for (const r of ttRows) {
   592→      if (r.classId !== a.classId || r.subjectId !== a.subjectId) continue
   593→      perWeekday[r.day] = (perWeekday[r.day] ?? 0) + 1
   594→      periods++
   595→    }
   596→    const pacing: Pacing =
   597→      periods > 0
   598→        ? { source: 'TIMETABLE', periodsPerWeek: periods, minutesPerWeek: periods * curriculum.minutesPerPeriod, perWeekday }
   599→        : { source: 'FALLBACK', periodsPerWeek: null, minutesPerWeek: curriculum.minutesPerPeriod * SCHOOL_WEEKDAYS.size, perWeekday }
   600→    const plan = computePendingPlan(topics, slots, holidays, pacing, curriculum.minutesPerPeriod)
   601→    a.pendingToday = plan.filter((p) => p.day === today).length
   602→  }
   603→  return base
   604→}

   606→async function buildSelectedScope(ctx: TeacherHubContext, scope: LessonAssignmentDTO): Promise<SelectedScopeDTO> {
   607→  const today = todayStr()
   608→  const selected: SelectedScopeDTO = {
   609→    classId: scope.classId,
   610→    classLabel: scope.classLabel,
   611→    subjectId: scope.subjectId,
   612→    subjectName: scope.subjectName,
   613→    curriculum: null,
   614→    unavailable: null,
   615→  }
   616→
   617→  const curriculum = await db.schoolCurriculum.findFirst({
   618→    where: { schoolId: ctx.schoolId, classId: scope.classId, subjectId: scope.subjectId, status: 'ACTIVE' },
   619→    orderBy: { session: 'desc' },
   620→    include: {
   621→      topics: { orderBy: [{ unitOrder: 'asc' }, { topicOrder: 'asc' }] },
   622→    },
   623→  })
   624→  if (!curriculum || !curriculum.topics.length) {
   625→    selected.unavailable = {
   626→      reason: `No curriculum has been imported for ${scope.classLabel} · ${scope.subjectName} yet. The school office needs to import the board syllabus for this subject before lesson planning can start.`,
   627→    }
   628→    return selected
   629→  }
   630→
   631→  const [holidays, pacing, existingSlots] = await Promise.all([
   632→    loadHolidaySet(ctx.schoolId, addDays(today, -30), addDays(today, WALK_HORIZON_DAYS)),
   633→    loadPacing(scope.classId, scope.subjectId, curriculum.minutesPerPeriod),
   634→    db.scheduledCurriculumTopic.findMany({
   635→      where: { classId: scope.classId, subjectId: scope.subjectId },
   636→      select: {
   637→        id: true, curriculumTopicId: true, sequence: true, plannedDate: true, minutesAllocated: true,
   638→        status: true, rescheduledFrom: true, flagReason: true,
   639→        execution: { select: { startedAt: true, completedAt: true, taughtOn: true, teacherId: true } },
   640→      },
   641→    }),
   642→  ])
   643→
   644→  const topics: TopicRow[] = curriculum.topics
   645→  const slots = await syncSchedule(ctx.schoolId, scope.classId, scope.subjectId, curriculum.id, topics, existingSlots, holidays, pacing, curriculum.minutesPerPeriod)
   646→  const pendingPlan = computePendingPlan(topics, slots, holidays, pacing, curriculum.minutesPerPeriod)
   647→  const originalPlan = computeCoursePlanFromStart(topics, curriculum.startDate, holidays, pacing, curriculum.minutesPerPeriod)
   648→
   649→  const topicById = new Map(topics.map((t) => [t.id, t]))
   650→  const all: ScheduledTopicDTO[] = []
   651→  for (const s of slots) {
   652→    const t = topicById.get(s.curriculumTopicId)
   653→    if (t) all.push(toTopicDTO(s, t, today))
   654→  }
   655→  all.sort((a, b) => a.sequence - b.sequence)
   656→
   657→  const pendingToday = all.filter((d) => d.status !== 'COMPLETED' && d.plannedDate === today)
   658→  const upcoming = all.filter((d) => d.status !== 'COMPLETED' && d.plannedDate > today).slice(0, UPCOMING_TAKE)
   659→  const flagged = all.filter((d) => d.status === 'NEEDS_RESCHEDULING')
   660→  const recent = all
   661→    .filter((d) => d.status === 'COMPLETED')
   662→    .sort((a, b) => (a.completedOn ?? '') < (b.completedOn ?? '') ? 1 : -1)
   663→    .slice(0, RECENT_TAKE)
   664→
   665→  const stats = buildStats(topics, slots, pendingPlan, originalPlan, curriculum.startDate, today)
   666→  const pacingDto: PacingInfo = {
   667→    source: pacing.source,
   668→    periodsPerWeek: pacing.periodsPerWeek,
   669→    minutesPerWeek: pacing.minutesPerWeek,
   670→  }
   671→
   672→  const dto: CurriculumDTO = {
   673→    id: curriculum.id,
   674→    board: curriculum.board,
   675→    session: curriculum.session,
   676→    versionLabel: curriculum.versionLabel,
   677→    startDate: dateToDay(curriculum.startDate),
   678→    stats,
   679→    pacing: pacingDto,
   680→    today: pendingToday[0] ?? null,
   681→    todayAlso: pendingToday.slice(1),
   682→    needsRescheduling: flagged,
   683→    upcoming,
   684→    recent,
   685→    units: buildUnits(topics, slots, today),
   686→  }
   687→  selected.curriculum = dto
   688→  return selected
   689→}

   706→/**
   707→ * Apply a teaching intent to a scheduled topic (start / complete / undo /
   708→ * flag / unflag). Re-validates scope on every call, audits every mutation,
   709→ * and returns the refreshed payload for the slot's class+subject.
   710→ */
   711→export async function applyLessonIntent(
   712→  user: AuthUser,
   713→  ctx: TeacherHubContext,
   714→  scheduledTopicId: string,
   715→  intent: LessonIntent,
   716→  reason?: string | null,
   717→): Promise<LessonPlannerPayload> {
   718→  const slot = await db.scheduledCurriculumTopic.findUnique({
   719→    where: { id: scheduledTopicId },
   720→    include: { execution: true },
   721→  })
   722→  if (!slot || slot.schoolId !== ctx.schoolId) throw new Error('NOT_FOUND')
   723→  await assertLessonScope(ctx, slot.classId, slot.subjectId)
   724→
   725→  const today = todayStr()
   726→
   727→  if (intent === 'start') {
   728→    if (slot.status === 'COMPLETED') throw new Error('ALREADY_COMPLETED')
   729→    if (dateToDay(slot.plannedDate) > today) throw new Error('NOT_PLANNED_TODAY')
   730→    if (!slot.execution?.startedAt) {
   731→      await db.lessonExecution.upsert({
   732→        where: { scheduledTopicId: slot.id },
   733→        create: {
   734→          schoolId: ctx.schoolId,
   735→          scheduledTopicId: slot.id,
   736→          teacherId: ctx.teacherId,
   737→          classId: slot.classId,
   738→          subjectId: slot.subjectId,
   739→          taughtOn: dayToDate(today),
   740→          startedAt: new Date(),
   741→        },
   742→        update: { startedAt: new Date(), taughtOn: dayToDate(today) },
   743→      })
   744→      await auditTeacherAction(user, ctx.schoolId, 'LESSON_STARTED', `Started lesson: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   745→    }
   746→  } else if (intent === 'complete') {
   747→    if (slot.status === 'COMPLETED') {
   748→      // idempotent — fall through to payload refresh
   749→    } else {
   750→      if (dateToDay(slot.plannedDate) > today) throw new Error('NOT_PLANNED_TODAY')
   751→      await db.lessonExecution.upsert({
   752→        where: { scheduledTopicId: slot.id },
   753→        create: {
   754→          schoolId: ctx.schoolId,
   755→          scheduledTopicId: slot.id,
   756→          teacherId: ctx.teacherId,
   757→          classId: slot.classId,
   758→          subjectId: slot.subjectId,
   759→          taughtOn: dayToDate(today),
   760→          minutesTaught: slot.minutesAllocated,
   761→          startedAt: slot.execution?.startedAt ?? new Date(),
   762→          completedAt: new Date(),
   763→        },
   764→        update: { completedAt: new Date(), taughtOn: dayToDate(today), minutesTaught: slot.minutesAllocated },
   765→      })
   766→      await db.scheduledCurriculumTopic.update({
   767→        where: { id: slot.id },
   768→        data: { status: 'COMPLETED', updatedAt: new Date() },
   769→      })
   770→      await auditTeacherAction(user, ctx.schoolId, 'LESSON_COMPLETED', `Completed lesson: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   771→    }
   772→  } else if (intent === 'undo') {
   773→    if (slot.status !== 'COMPLETED') throw new Error('NOT_COMPLETED')
   774→    if (slot.execution) await db.lessonExecution.delete({ where: { id: slot.execution.id } })
   775→    await db.scheduledCurriculumTopic.update({
   776→      where: { id: slot.id },
   777→      data: { status: 'PENDING', updatedAt: new Date() },
   778→    })
   779→    await auditTeacherAction(user, ctx.schoolId, 'LESSON_UNDONE', `Reopened lesson: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   780→  } else if (intent === 'flag') {
   781→    if (slot.status === 'COMPLETED') throw new Error('ALREADY_COMPLETED')
   782→    await db.scheduledCurriculumTopic.update({
   783→      where: { id: slot.id },
   784→      data: {
   785→        status: 'NEEDS_RESCHEDULING',
   786→        rescheduledFrom: slot.rescheduledFrom ?? slot.plannedDate,
   787→        flagReason: reason?.slice(0, 200) ?? null,
   788→        updatedAt: new Date(),
   789→      },
   790→    })
   791→    await auditTeacherAction(user, ctx.schoolId, 'LESSON_FLAGGED', `Flagged for rescheduling: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   792→  } else if (intent === 'unflag') {
   793→    if (slot.status === 'NEEDS_RESCHEDULING') {
   794→      await db.scheduledCurriculumTopic.update({
   795→        where: { id: slot.id },
   796→        data: { status: 'PENDING', rescheduledFrom: null, flagReason: null, updatedAt: new Date() },
   797→      })
   798→      await auditTeacherAction(user, ctx.schoolId, 'LESSON_FLAG_CLEARED', `Cleared reschedule flag: ${slot.id} (${slot.classId}/${slot.subjectId})`)
   799→    }
   800→  }
   801→
   802→  return buildLessonPlannerPayload(ctx, slot.classId, slot.subjectId)
   803→}
