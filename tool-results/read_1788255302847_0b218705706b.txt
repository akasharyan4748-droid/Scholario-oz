   331→function TCRow({ label, value, serif }: { label: string; value: string; serif?: boolean }) {
   332→  return (
   333→    <tr>
   334→      <td className={cn('py-1 pr-2 text-slate-500 w-1/3', serif && 'font-serif')}>{label}</td>
   335→      <td className={cn('py-1 text-slate-900 font-medium', serif && 'font-serif')}>{value}</td>

   340→// Quick "in words" for date of birth (very compact).
   341→function birthWords(dob: string): string {
   342→  const d = new Date(dob)
   343→  if (Number.isNaN(d.getTime())) return '—'
   344→  const months = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December']
   345→  return `${ordinal(d.getDate())} ${months[d.getMonth()]} ${d.getFullYear()}`
   346→}
   347→function ordinal(n: number): string {
   348→  const s = ['th', 'st', 'nd', 'rd']
   349→  const v = n % 100
   350→  return n + (s[(v - 20) % 10] || s[v] || s[0])
   351→}

   461→              <td className="px-2 py-1.5 border border-slate-300 text-center text-slate-900">{data.grade}</td>
   462→              <td className="px-2 py-1.5 border border-slate-300 text-center text-slate-900">{data.result}</td>
   463→            </tr>
   464→          </tfoot>
   465→        </table>
   466→
   467→        {/* Footer */}
   468→        <div className="mt-3 grid grid-cols-3 gap-2 text-[10px]">
   469→          <div className="border border-slate-200 rounded px-2 py-1">
   470→            <p className="text-slate-500 text-[8px] uppercase">Percentage</p>
   471→            <p className="font-bold text-slate-900">{data.percentage.toFixed(2)}%</p>
   472→          </div>
   473→          <div className="border border-slate-200 rounded px-2 py-1">
   474→            <p className="text-slate-500 text-[8px] uppercase">Division</p>
   475→            <p className="font-bold text-slate-900">{divisionFor(data.percentage)}</p>
   476→          </div>
   477→          <div className="border border-slate-200 rounded px-2 py-1">
   478→            <p className="text-slate-500 text-[8px] uppercase">Rank</p>
   479→            <p className="font-bold text-slate-900">{data.rank ?? '—'}</p>
   480→          </div>
   481→        </div>
   482→
   483→        <div className="mt-4 flex items-end justify-between">
   484→          <div className="text-[9px] text-slate-600 max-w-[60%]">
   485→            <p className="font-semibold text-slate-800">Remarks:</p>
   486→            <p className="italic">{data.remarks ?? 'Conduct: Excellent. Regularity: Satisfactory.'}</p>
   487→          </div>
   488→          <div className="text-center">
   489→            <div className="h-0.5 w-32 bg-slate-400 mb-1" />
   490→            <p className="text-[9px] text-slate-800 font-semibold">{school.principal}</p>
   491→            <p className="text-[8px] text-slate-500">Principal</p>
   492→          </div>
   493→        </div>
   494→      </div>
   495→    </div>
   496→  )
   497→}
   498→
   499→function gradeFor(pct: number): string {
   500→  if (pct >= 90) return 'A1'
   501→  if (pct >= 80) return 'A2'
   502→  if (pct >= 70) return 'B1'
   503→  if (pct >= 60) return 'B2'
   504→  if (pct >= 50) return 'C1'
   505→  if (pct >= 33) return 'C2'
   506→  return 'E'
   507→}
   508→function divisionFor(pct: number): string {
   509→  if (pct >= 60) return 'First'
   510→  if (pct >= 45) return 'Second'
   511→  if (pct >= 33) return 'Third'
   512→  return 'Fail'
   513→}
