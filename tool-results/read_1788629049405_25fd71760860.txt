     4→ * GroupsPanel — Group management UI for the Messages & Inbox module.
     5→ *
     6→ * Surfaces when the "Groups" folder is active (replaces the conversation
     7→ * list). Each row shows the group's avatar, name, type pill, member count

    17→ * (search teachers + parents from canonical data — NO duplicate data).
    18→ *
    19→ * Manage Members dialog: list current members with remove buttons + add
    20→ * a single member at a time via the SearchableSelect.
    21→ *
    22→ * SCHOLARIO visual language preserved: rounded-xl cards, soft tinted
    23→ * type pills, emerald → teal gradient on primary CTAs, violet → purple
    24→ * gradient on group avatars (consistent with the existing group colour).

    96→function getMemberPool(): PoolMember[] {
    97→  const students = useStudentsStore.getState().students
    98→  const activeStudents = students.filter((s) => s.status === 'Active')
    99→
   100→  const teachersPool: PoolMember[] = teachers
   101→    .filter((t) => !t.archived)
   102→    .map((t) => ({
   103→      ref: `t:${t.id}`,
   104→      name: t.name,
   105→      avatar: t.avatar,
   106→      role: `${t.designation} · ${t.department}`,
   107→      type: 'teacher' as const,
   108→    }))
   109→
   110→  const parentsPool: PoolMember[] = activeStudents.map((s) => ({
   111→    ref: `p:${s.id}`,
   112→    name: s.fatherName,
   113→    avatar: s.fatherName.split(' ').map((n) => n[0]).slice(0, 2).join(''),
   114→    role: `Parent · ${s.name} (${s.className}-${s.section})`,
   115→    type: 'parent' as const,
   116→  }))
   117→
   118→  return [...teachersPool, ...parentsPool]
   119→}

   131→  const groups = useMessagingStore((s) => s.groups)
   132→  const conversations = useMessagingStore((s) => s.conversations)
   133→  const openConversation = useMessagingStore((s) => s.openConversation)
   134→  const searchQuery = useMessagingStore((s) => s.searchQuery)
   135→  const setSearchQuery = useMessagingStore((s) => s.setSearchQuery)
   136→  const activeConversationId = useMessagingStore((s) => s.activeConversationId)
   137→  const deleteGroup = useMessagingStore((s) => s.deleteGroup)
   138→
   139→  const [createOpen, setCreateOpen] = useState(false)
   140→  const [manageGroup, setManageGroup] = useState<Group | null>(null)
   141→
   142→  // Filter groups by search query (name or type)
   143→  const filtered = useMemo(() => {
   144→    const q = searchQuery.trim().toLowerCase()
   145→    if (!q) return groups
   146→    return groups.filter(

   152→    openConversation(g.conversationId)

   154→
   155→  const handleDelete = (g: Group) => {

   157→    toast.success('Group deleted', { description: g.name })
   158→  }
   159→
   160→  return (
   161→    <div className="flex flex-col border-r border-border bg-card min-w-0">
   162→      {/* Header — search + Create Group */}
   163→      <div className="p-3 border-b border-border space-y-2">
   164→        <div className="relative">
   165→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   166→          <input
   167→            value={searchQuery}
   168→            onChange={(e) => setSearchQuery(e.target.value)}
   169→            placeholder="Search groups…"
   170→            className="w-full h-8 pl-8 pr-8 text-xs rounded-md border border-border bg-card focus:outline-none focus:ring-2 focus:ring-primary/30"
   171→          />
   172→          {searchQuery && (
   173→            <button
   174→              onClick={() => setSearchQuery('')}
   175→              className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   176→            >
   177→              <X className="h-3.5 w-3.5" />
   178→            </button>
   179→          )}
   180→        </div>
   181→        <button
   182→          onClick={() => setCreateOpen(true)}
   183→          className="w-full inline-flex items-center justify-center gap-1.5 rounded-md bg-gradient-to-r from-emerald-600 to-teal-600 hover:from-emerald-700 hover:to-teal-700 px-3 py-1.5 text-xs font-semibold text-white shadow-sm transition-colors"
   184→        >
   185→          <Plus className="h-3.5 w-3.5" /> Create Group
   186→        </button>
   187→      </div>
   188→
   189→      {/* Groups list */}
   190→      <div className="flex-1 overflow-y-auto">
   191→        {filtered.length > 0 ? (
   192→          filtered.map((g) => {
   193→            const convo = conversations.find((c) => c.id === g.conversationId)
   194→            const memberCount = g.memberRefs.length
   195→            const isActive = activeConversationId === g.conversationId
   196→            const lastActivity = convo ? formatTimeAgo(convo.lastTimestamp) : '—'
   197→            const lastMessage = convo?.lastMessage ?? 'No messages yet'
   198→            const unread = convo?.unread ?? 0
   199→            return (
   200→              <GroupRow
   201→                key={g.id}
   202→                group={g}
   203→                isActive={isActive}
   204→                memberCount={memberCount}
   205→                lastActivity={lastActivity}
   206→                lastMessage={lastMessage}
   207→                unread={unread}
   208→                onOpenChat={() => handleOpenChat(g)}
   209→                onManage={() => setManageGroup(g)}
   210→                onCompose={() => onCompose(g.name)}
   211→                onDelete={() => handleDelete(g)}
   212→              />
   213→            )
   214→          })
   215→        ) : (
   216→          <div className="flex flex-col items-center justify-center py-16 px-4 text-center">
   217→            <Users className="h-8 w-8 text-muted-foreground/40 mb-2" />

   219→              {searchQuery ? 'No groups match your search' : 'No groups yet'}
   220→            </p>
   221→            <p className="text-[10px] text-muted-foreground/70 mt-0.5 mb-3">
   222→              {searchQuery ? 'Try a different search term.' : 'Create a group to start a group conversation.'}
   223→            </p>
   224→            {!searchQuery && (
   225→              <button
   226→                onClick={() => setCreateOpen(true)}
   227→                className="inline-flex items-center gap-1.5 rounded-md bg-emerald-600 hover:bg-emerald-700 px-3 py-1.5 text-xs font-semibold text-white"
   228→              >
   229→                <Plus className="h-3 w-3" /> Create Group
   230→              </button>
   231→            )}
   232→          </div>
   233→        )}
   234→      </div>
   235→
   236→      {/* Create Group dialog */}
   237→      <CreateGroupDialog open={createOpen} onOpenChange={setCreateOpen} />
   238→
   239→      {/* Manage Members dialog */}
   240→      <ManageMembersDialog
   241→        group={manageGroup}
   242→        onOpenChange={(open) => { if (!open) setManageGroup(null) }}
   243→      />
   244→    </div>

   376→function CreateGroupDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (open: boolean) => void }) {
   377→  const createGroup = useMessagingStore((s) => s.createGroup)
   378→  const [name, setName] = useState('')
   379→  const [type, setType] = useState<GroupType>('Class Group')
   380→  // Smart picker state
   381→  const [className, setClassName] = useState<string>('') // e.g. "Class 10"
   382→  const [section, setSection] = useState<string>('')      // e.g. "A"
   383→  const [department, setDepartment] = useState<string>('') // e.g. "Science"
   384→  const [allStaff, setAllStaff] = useState(false)
   385→  // Members
   386→  const [selectedRefs, setSelectedRefs] = useState<string[]>([])
   387→  const [memberSearch, setMemberSearch] = useState('')
   388→
   389→  const pool = useMemo(() => getMemberPool(), [])
   390→
   391→  // Class + section + department options (deduped)
   392→  const classOptions = useMemo(() => {
   393→    const seen = new Set<string>()
   394→    const out: { id: string; label: string }[] = []
   395→    for (const c of ACADEMIC_CLASSES) {
   396→      if (!seen.has(c.name)) {
   397→        seen.add(c.name)
   398→        out.push({ id: c.name, label: c.name })
   399→      }
   400→    }
   401→    return out
   402→  }, [])
   403→  const sectionOptions = useMemo(() => {
   404→    const cls = ACADEMIC_CLASSES.find((c) => c.name === className)
   405→    return (cls?.sections ?? []).map((s) => ({ id: s, label: s }))
   406→  }, [className])
   407→  const departmentOptions = useMemo(() => {
   408→    const seen = new Set<string>()
   409→    const out: { id: string; label: string }[] = []
   410→    for (const t of teachers) {
   411→      if (t.archived) continue
   412→      if (!seen.has(t.department)) {
   413→        seen.add(t.department)
   414→        out.push({ id: t.department, label: t.department })
   415→      }
   416→    }
   417→    return out
   418→  }, [])
   419→
   420→  // Smart auto-fill: when the user picks a class+section / class / department,
   421→  // pre-fill the suggested members and a suggested name. The user can still
   422→  // tweak both before submitting.
   423→  const smartFill = useMemo(() => {
   424→    if (type === 'Class Group' || type === 'Parents Group') {
   425→      if (!className || !section) return null
   426→      const refs = getParentsOfClassSection(className, section)
   427→      const suggestedName = `${className}-${section} Parents`
   428→      return { refs, suggestedName }
   429→    }
   430→    if (type === 'Teachers Group') {
   431→      if (!className) return null
   432→      // Pull teachers across all sections of this class
   433→      const cls = ACADEMIC_CLASSES.find((c) => c.name === className)
   434→      const sections = cls?.sections ?? ['A']
   435→      const refs = Array.from(new Set(sections.flatMap((s) => getTeachersOfClass(`${className}-${s}`))))
   436→      const suggestedName = `${className} Teachers`
   437→      return { refs, suggestedName }
   438→    }
   439→    if (type === 'Department Group') {
   440→      if (!department) return null
   441→      const refs = getTeachersOfDepartment(department)
   442→      const suggestedName = `${department} Department`
   443→      return { refs, suggestedName }
   444→    }
   445→    if (type === 'Staff Group') {
   446→      const refs = getAllStaffRefs()
   447→      const suggestedName = 'All Staff'
   448→      return { refs, suggestedName }
   449→    }
   450→    return null
   451→  }, [type, className, section, department])
   452→
   453→  // Reset on dialog open + when type changes (clear smart picker state)
   454→  useEffect(() => {
   455→    if (open) {
   456→      setName('')
   457→      setType('Class Group')
   458→      setClassName('')
   459→      setSection('')
   460→      setDepartment('')
   461→      setAllStaff(false)
   462→      setSelectedRefs([])
   463→      setMemberSearch('')
   464→    }
   465→  }, [open])
   466→
   467→  // When smart-fill produces a suggestion, sync the name (if user hasn't typed
   468→  // anything yet) and merge the suggested refs into the selection (without
   469→  // losing any the user has explicitly added).
   470→  useEffect(() => {
   471→    if (!open) return
   472→    if (smartFill) {
   473→      setName((prev) => prev || smartFill.suggestedName)
   474→      setSelectedRefs((prev) => Array.from(new Set([...prev, ...smartFill.refs])))
   475→    }
   476→  }, [smartFill, open])
   477→
   478→  // Filter the pool by member search
   479→  const filteredPool = useMemo(() => {
   480→    const q = memberSearch.trim().toLowerCase()
   481→    if (!q) return pool
   482→    return pool.filter((p) => p.name.toLowerCase().includes(q) || p.role.toLowerCase().includes(q))
   483→  }, [pool, memberSearch])
   484→
   485→  const selectedSet = useMemo(() => new Set(selectedRefs), [selectedRefs])
   486→  const selectedMembers = useMemo(
   487→    () => selectedRefs.map(resolveMemberRef).filter((x): x is MemberDisplay => x !== null),
   488→    [selectedRefs],
   489→  )
   490→
   491→  const canSubmit = name.trim().length > 0 && selectedRefs.length > 0
   492→
   493→  const toggleMember = (ref: string) => {
   494→    setSelectedRefs((prev) => prev.includes(ref) ? prev.filter((r) => r !== ref) : [...prev, ref])
   495→  }
   496→
   497→  const clearSmart = () => {
   498→    setSelectedRefs([])
   499→    setName('')
   500→  }
   501→
   502→  const handleSubmit = () => {
   503→    if (!canSubmit) {
   504→      if (!name.trim()) toast.error('Group name is required')
   505→      else if (selectedRefs.length === 0) toast.error('Add at least one member')
   506→      return
   507→    }
   508→    createGroup({ name: name.trim(), type, memberRefs: selectedRefs })
   509→    toast.success('Group created', {
   510→      description: `${name.trim()} · ${selectedRefs.length} member${selectedRefs.length === 1 ? '' : 's'}`,
   511→    })

   783→function ManageMembersDialog({
   784→  group, onOpenChange,
   785→}: {
   786→  group: Group | null
   787→  onOpenChange: (open: boolean) => void
   788→}) {
   789→  const addMember = useMessagingStore((s) => s.addMember)
   790→  const removeMember = useMessagingStore((s) => s.removeMember)
   791→  const groups = useMessagingStore((s) => s.groups)
   792→  const [addRef, setAddRef] = useState('')
   793→
   794→  // Always read the latest group state from the store so member add/remove
   795→  // updates show without remounting.
   796→  const liveGroup = group ? groups.find((g) => g.id === group.id) : null
   797→  const members = useMemo(
   798→    () => liveGroup ? resolveMemberRefs(liveGroup.memberRefs) : [],
   799→    [liveGroup],
   800→  )
   801→
   802→  const pool = useMemo(() => getMemberPool(), [])
   803→  const memberSet = useMemo(() => new Set(liveGroup?.memberRefs ?? []), [liveGroup])
   804→  const addable = useMemo(
   805→    () => pool
   806→      .filter((p) => !memberSet.has(p.ref))
   807→      .map((p) => ({ id: p.ref, label: p.name, avatar: p.avatar, meta: p.role })),
   808→    [pool, memberSet],
   809→  )
   810→
   811→  useEffect(() => {
   812→    setAddRef('')
   813→  }, [group?.id])
   814→
   815→  const handleAdd = () => {
   816→    if (!liveGroup || !addRef) return
   817→    const result = addMember(liveGroup.id, addRef)
   818→    if (!result.success) {
   819→      toast.error(result.error || 'Could not add member')
   820→      return
   821→    }
   822→    const display = resolveMemberRef(addRef)
   823→    toast.success('Member added', {
   824→      description: display ? `${display.name} added to ${liveGroup.name}` : 'Member added',
   825→    })
   826→    setAddRef('')
   827→  }
   828→
   829→  const handleRemove = (ref: string) => {
   830→    if (!liveGroup) return
   831→    const display = resolveMemberRef(ref)
   832→    removeMember(liveGroup.id, ref)
   833→    toast.success('Member removed', {
   834→      description: display ? `${display.name} removed from ${liveGroup.name}` : 'Member removed',
   835→    })
   836→  }
   837→
   838→  if (!group) return null
   839→
   840→  return (
   841→    <Dialog open={!!group} onOpenChange={onOpenChange}>
   842→      <DialogContent className="sm:max-w-md max-h-[85vh] overflow-y-auto">
   843→        <DialogHeader>
   844→          <DialogTitle className="flex items-center gap-2 text-base">
   845→            <div className="flex h-7 w-7 items-center justify-center rounded-full bg-gradient-to-br from-violet-500 to-purple-600 text-white text-[10px] font-semibold">
   846→              {avatarFromName(group.name)}
   847→            </div>
   848→            {group.name}
   849→          </DialogTitle>
   850→          <DialogDescription className="text-xs flex items-center gap-1.5">
   851→            <span className={cn(
   852→              'inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[9px] font-semibold',
   853→              TYPE_PILL[group.type],
   854→            )}>
   855→              {TYPE_ICON[group.type]}
   856→              {group.type}
   857→            </span>
   858→            <span className="inline-flex items-center gap-0.5 text-[10px] text-muted-foreground tabular-nums">
   859→              <Users className="h-2.5 w-2.5" />
   860→              {members.length} member{members.length === 1 ? '' : 's'}
   861→            </span>
   862→          </DialogDescription>
   863→        </DialogHeader>
   864→
   865→        <div className="space-y-3">
   866→          {/* Add member */}
   867→          <div className="space-y-1">
   868→            <Label className="text-[11px] flex items-center gap-1.5">
   869→              <UserPlus className="h-3 w-3" /> Add a Member
   870→            </Label>
   871→            <div className="flex items-end gap-1.5">
   872→              <div className="flex-1 min-w-0">
   873→                <SearchableSelect
   874→                  selectedId={addRef}
   875→                  onSelect={setAddRef}
   876→                  placeholder="Search teachers or parents…"
   877→                  options={addable}
   878→                  pickerId="grp-add-member"
   879→                  popoverWidth="w-64"
   880→                />
   881→              </div>
   882→              <Button
   883→                size="sm"
   884→                onClick={handleAdd}
   885→                disabled={!addRef}
   886→                className="h-9 gap-1 bg-gradient-to-r from-emerald-600 to-teal-600 hover:from-emerald-700 hover:to-teal-700 text-white"
   887→              >
   888→                <Plus className="h-3.5 w-3.5" /> Add
   889→              </Button>
   890→            </div>
   891→            {addable.length === 0 && (
   892→              <p className="text-[10px] text-muted-foreground mt-1">
   893→                Everyone is already a member of this group.
   894→              </p>
   895→            )}
   896→          </div>
   897→
   898→          {/* Current members */}
   899→          <div className="space-y-1">
   900→            <Label className="text-[11px] flex items-center justify-between">
   901→              <span>Current Members ({members.length})</span>
   902→            </Label>
   903→            <div className="max-h-64 overflow-y-auto rounded-md border border-border divide-y divide-border/30">
   904→              {members.length > 0 ? (
   905→                members.map((m) => (
   906→                  <div key={m.ref} className="px-2.5 py-1.5 flex items-center gap-2 hover:bg-muted/30">
   907→                    <div className={cn(
   908→                      'h-7 w-7 shrink-0 rounded-full text-white text-[9px] font-bold flex items-center justify-center',
   909→                      m.type === 'teacher'
   910→                        ? 'bg-gradient-to-br from-emerald-500 to-teal-600'
   911→                        : 'bg-gradient-to-br from-amber-500 to-orange-600',
   912→                    )}>
   913→                      {m.avatar}
   914→                    </div>
   915→                    <div className="min-w-0 flex-1">
   916→                      <p className="text-[11px] font-medium truncate">{m.name}</p>
   917→                      <p className="text-[9px] text-muted-foreground truncate">{m.role}</p>
   918→                    </div>
   919→                    <button
   920→                      onClick={() => handleRemove(m.ref)}
   921→                      className="p-1 text-muted-foreground hover:text-rose-500 rounded hover:bg-muted"
   922→                      title="Remove from group"
   923→                    >
   924→                      <Trash2 className="h-3 w-3" />
   925→                    </button>
   926→                  </div>
   927→                ))
   928→              ) : (
   929→                <p className="px-3 py-4 text-[10px] text-muted-foreground text-center">
   930→                  No members yet. Add some above.
   931→                </p>
   932→              )}
   933→            </div>
   934→          </div>
   935→        </div>
   936→
   937→        <DialogFooter>
   938→          <Button variant="outline" onClick={() => onOpenChange(false)}>Done</Button>
   939→        </DialogFooter>
   940→      </DialogContent>
   941→    </Dialog>
   942→  )
   943→}
