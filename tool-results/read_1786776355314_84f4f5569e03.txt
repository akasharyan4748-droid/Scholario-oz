    72→export function ExamWorkspace({ examId, onBack, onMutated }: Props) {
    73→  const [tab, setTab] = useState<Tab>('overview')
    74→  const { exam, loading, error, reload } = useExam(examId)
    75→
    76→  const handleReload = () => {
    77→    reload()
    78→    onMutated()
    79→  }
    80→
    81→  return (
    82→    <div className="flex flex-col h-full -mt-4 -mx-4 sm:-mx-6">
    83→      {/* Top header — full width */}
    84→      <div className="border-b border-border bg-card/80 backdrop-blur sticky top-0 z-20">
    85→        <div className="px-4 sm:px-6 py-3">
    86→          <div className="flex items-center justify-between gap-3 flex-wrap">
    87→            <div className="flex items-center gap-3 min-w-0">
    88→              <Button variant="ghost" size="sm" className="h-8 px-2 text-xs gap-1" onClick={onBack}>
    89→                <ArrowLeft className="h-3.5 w-3.5" /> Examinations
    90→              </Button>
    91→              <div className="h-5 w-px bg-border" />
    92→              <div className="min-w-0">
    93→                <h1 className="text-base font-semibold truncate">{exam?.name ?? 'Loading…'}</h1>
    94→                <p className="text-[10px] text-muted-foreground mt-0.5">
    95→                  {exam ? `${exam.type} · ${exam.session} · ${exam.classes.length} classes · ${exam.subjects.length} subjects` : ''}
    96→                </p>
    97→              </div>
    98→            </div>
    99→            {exam && (
   100→              <div className="flex items-center gap-2 shrink-0">
   101→                <StatusPill status={exam.status} />
   102→                <ResultStatusPill status={exam.resultStatus} />
   103→              </div>
   104→            )}
   105→          </div>
   106→        </div>
   107→        {/* Section navigation */}
   108→        <div className="px-4 sm:px-6 pb-3 overflow-x-auto">
   109→          <SegmentedTabs tabs={TABS} value={tab} onValueChange={(v) => setTab(v as Tab)} />
   110→        </div>
   111→      </div>
   112→
   113→      {/* Main content — full available width */}
   114→      <div className="flex-1 overflow-y-auto p-4 sm:p-6">
   115→        {loading ? (
   116→          <InlineLoading label="Loading examination…" />
   117→        ) : error ? (
   118→          <div className="rounded-lg border border-rose-500/30 bg-rose-500/5 p-3">
   119→            <p className="text-xs text-rose-700">{error}</p>
   120→          </div>
   121→        ) : !exam ? null : (
   122→          <AnimatePresence mode="wait">
   123→            <motion.div
   124→              key={tab}
   125→              initial={{ opacity: 0, y: 4 }}
   126→              animate={{ opacity: 1, y: 0 }}
   127→              exit={{ opacity: 0, y: -4 }}
   128→              transition={{ duration: 0.15 }}
   129→            >
   130→              {tab === 'overview' && <OverviewSection exam={exam} onReload={handleReload} onNavigate={setTab} />}
   131→              {tab === 'schedule' && <ScheduleSection exam={exam} onReload={handleReload} />}
   132→              {tab === 'marks' && <MarksSection exam={exam} onReload={handleReload} />}
   133→              {tab === 'import' && <CsvImportSection examId={exam.id} exam={exam} onReload={handleReload} />}
   134→              {tab === 'results' && <ResultsSection exam={exam} onReload={handleReload} />}
   135→              {tab === 'outcomes' && <OutcomesSection examId={exam.id} exam={exam} onReload={handleReload} />}
   136→              {tab === 'seating' && <SeatingSection examId={exam.id} exam={exam} onReload={handleReload} />}
   137→              {tab === 'attendance' && <AttendanceSection examId={exam.id} exam={exam} onReload={handleReload} />}
   138→              {tab === 'grace' && <GraceSection examId={exam.id} exam={exam} onReload={handleReload} />}
   139→              {tab === 'audit' && <AuditSection examId={exam.id} />}
   140→            </motion.div>
   141→          </AnimatePresence>
   142→        )}
   143→      </div>
   144→    </div>
   145→  )
   146→}

   172→function OverviewSection({ exam, onReload, onNavigate }: { exam: any; onReload: () => void; onNavigate: (t: Tab) => void }) {
   173→  const { update } = useUpdateExam()
   174→  const gate = useRoleGate()
   175→  const [editing, setEditing] = useState(false)
   176→  const [name, setName] = useState(exam.name)
   177→  const [status, setStatus] = useState(exam.status)
   178→  const [startDate, setStartDate] = useState(exam.startDate ?? '')
   179→  const [endDate, setEndDate] = useState(exam.endDate ?? '')
   180→
   181→  const handleSave = async () => {
   182→    try {
   183→      await update(exam.id, { name, status, startDate, endDate })
   184→      toast.success('Exam updated')
   185→      setEditing(false)
   186→      onReload()
   187→    } catch (e: any) {
   188→      toast.error('Failed to update exam', { description: e.message })
   189→    }
   190→  }
   191→
   192→  // Compute real readiness indicators
   193→  const readiness = [
   194→    { label: 'Classes configured', done: exam.classes.length > 0, navigate: 'overview' as Tab },
   195→    { label: 'Subjects configured', done: exam.subjects.length > 0, navigate: 'overview' as Tab },
   196→    { label: 'Schedule published', done: exam.schedule.length > 0, navigate: 'schedule' as Tab },
   197→    { label: 'Marks entry started', done: exam.markSummary.entered > 0, navigate: 'marks' as Tab },
   198→    { label: 'Marks verified', done: exam.markSummary.verified > 0, navigate: 'marks' as Tab },
   199→    { label: 'Marks locked', done: exam.markSummary.locked > 0, navigate: 'marks' as Tab },
   200→    { label: 'Results declared', done: exam.resultStatus === 'Result Declared', navigate: 'results' as Tab },
   201→  ]
   202→
   203→  const entered = exam.markSummary.entered
   204→  const total = exam.markSummary.total
   205→  const pct = exam.markSummary.pct
   206→
   207→  return (
   208→    <div className="space-y-4">
   209→      {/* KPI grid */}
   210→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   211→        <Kpi label="Classes" value={exam.classes.length} sub={`${exam.classes.reduce((s: number, c: any) => s + c.studentCount, 0)} students`} />
   212→        <Kpi label="Subjects" value={exam.subjects.length} sub={`${exam.schedule.length} scheduled`} />
   213→        <Kpi label="Marks Entry" value={`${entered}/${total}`} sub={`${pct}% entered`} progress={pct} />
   214→        <Kpi label="Schedule" value={exam.schedule.length} sub={`${exam.schedule.length} sessions`} />
   215→      </div>
   216→
   217→      {/* Exam readiness */}
   218→      <div className="rounded-xl border border-border bg-card p-4">
   219→        <h3 className="text-sm font-semibold mb-3">Exam Readiness</h3>
   220→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2">
   221→          {readiness.map((r) => (
   222→            <button
   223→              key={r.label}
   224→              onClick={() => onNavigate(r.navigate)}
   225→              className="flex items-center gap-2 p-2 rounded-lg border border-border hover:bg-muted/40 transition-colors text-left"
   226→            >
   227→              <span className={cn(
   228→                'flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-[10px] font-bold',
   229→                r.done ? 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-300' : 'bg-muted text-muted-foreground'
   230→              )}>
   231→                {r.done ? '✓' : '—'}
   232→              </span>
   233→              <span className="text-xs flex-1">{r.label}</span>
   234→              {!r.done && <span className="text-[9px] text-muted-foreground">pending</span>}
   235→            </button>
   236→          ))}
   237→        </div>
   238→      </div>
   239→
   240→      {/* Edit / details */}
   241→      {gate.canEdit ? (
   242→        <div className="rounded-xl border border-border bg-card p-4">
   243→          <div className="flex items-center justify-between mb-3">
   244→            <h3 className="text-sm font-semibold">Examination Details</h3>
   245→            {!editing ? (
   246→              <Button size="sm" variant="ghost" className="h-7 text-xs gap-1" onClick={() => setEditing(true)}>
   247→                <Pencil className="h-3 w-3" /> Edit
   248→              </Button>
   249→            ) : (
   250→              <div className="flex gap-2">
   251→                <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => setEditing(false)}>Cancel</Button>
   252→                <Button size="sm" variant="default" className="h-7 text-xs gap-1" onClick={handleSave}>
   253→                  <Save className="h-3 w-3" /> Save
   254→                </Button>
   255→              </div>
   256→            )}
   257→          </div>
   258→          {!editing ? (
   259→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   260→              <DetailField label="Name" value={exam.name} />
   261→              <DetailField label="Status" value={exam.status} />
   262→              <DetailField label="Start" value={exam.startDate ?? '—'} />
   263→              <DetailField label="End" value={exam.endDate ?? '—'} />
   264→            </div>
   265→          ) : (
   266→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   267→              <div>
   268→                <Label className="text-[10px]">Name</Label>
   269→                <Input value={name} onChange={(e) => setName(e.target.value)} className="h-8 text-xs" />
   270→              </div>
   271→              <div>
   272→                <Label className="text-[10px]">Status</Label>
   273→                <Select value={status} onValueChange={setStatus}>
   274→                  <SelectTrigger size="sm" className="text-xs"><SelectValue /></SelectTrigger>
   275→                  <SelectContent>
   276→                    <SelectItem value="Draft">Draft</SelectItem>
   277→                    <SelectItem value="Scheduled">Scheduled</SelectItem>
   278→                    <SelectItem value="Ongoing">Ongoing</SelectItem>
   279→                    <SelectItem value="Completed">Completed</SelectItem>
   280→                    <SelectItem value="Cancelled">Cancelled</SelectItem>
   281→                  </SelectContent>
   282→                </Select>
   283→              </div>
   284→              <div>
   285→                <Label className="text-[10px]">Start Date</Label>
   286→                <DatePicker value={startDate} onChange={setStartDate} />
   287→              </div>
   288→              <div>
   289→                <Label className="text-[10px]">End Date</Label>
   290→                <DatePicker value={endDate} onChange={setEndDate} />
   291→              </div>
   292→            </div>
   293→          )}
   294→        </div>
   295→      ) : null}
   296→    </div>
   297→  )
   298→}
   299→
   300→function Kpi({ label, value, sub, progress }: { label: string; value: any; sub?: string; progress?: number }) {
   301→  return (
   302→    <div className="rounded-xl border border-border bg-card p-3">
   303→      <p className="text-[10px] uppercase font-semibold text-muted-foreground">{label}</p>
   304→      <p className="font-display text-2xl font-bold tabular-nums">{value}</p>
   305→      {sub && <p className="text-[10px] text-muted-foreground mt-0.5">{sub}</p>}
   306→      {progress !== undefined && (
   307→        <div className="mt-2 h-1 rounded-full bg-muted/60 overflow-hidden">
   308→          <div className="h-full bg-primary" style={{ width: `${progress}%` }} />
   309→        </div>
   310→      )}
   311→    </div>
   312→  )
   313→}
   314→
   315→function DetailField({ label, value }: { label: string; value: string }) {
   316→  return (
   317→    <div>
   318→      <p className="text-[9px] uppercase font-semibold text-muted-foreground">{label}</p>
   319→      <p className="text-xs font-medium truncate">{value}</p>
   320→    </div>
   321→  )
   322→}

   326→function ScheduleSection({ exam, onReload }: { exam: any; onReload: () => void }) {
   327→  const { add } = useAddScheduleItem()
   328→  const { remove } = useDeleteScheduleItem()
   329→  const { update: updateItem } = useUpdateScheduleItemV2()
   330→  const { teachers } = useTeachers(exam.id)
   331→  const { assign: assignInvigilator } = useAssignInvigilator()
   332→  const gate = useRoleGate()
   333→  const [classId, setClassId] = useState(exam.classes[0]?.classId ?? '')
   334→  const [subjectId, setSubjectId] = useState('')
   335→  const [date, setDate] = useState('')
   336→  const [startTime, setStartTime] = useState('09:00')
   337→  const [endTime, setEndTime] = useState('10:00')
   338→  const [room, setRoom] = useState('')
   339→  const [invigilator, setInvigilator] = useState('')
   340→  const [editingId, setEditingId] = useState<string | null>(null)
   341→  const [editRoom, setEditRoom] = useState('')
   342→  const [editDate, setEditDate] = useState('')
   343→  const [editStart, setEditStart] = useState('')
   344→  const [editEnd, setEditEnd] = useState('')
   345→
   346→  const handleAdd = async () => {
   347→    if (!classId || !subjectId || !date) {
   348→      toast.error('Class, subject, and date are required')
   349→      return
   350→    }
   351→    try {
   352→      await add(exam.id, { classId, subjectId, date, startTime, endTime, room, invigilatorName: invigilator })
   353→      toast.success('Schedule item added')
   354→      setSubjectId(''); setDate(''); setRoom(''); setInvigilator('')
   355→      onReload()
   356→    } catch (e: any) {
   357→      toast.error('Failed to add schedule item', { description: e.message })
   358→    }
   359→  }
   360→
   361→  const handleDelete = async (itemId: string) => {
   362→    try {
   363→      await remove(exam.id, itemId)
   364→      toast.success('Schedule item removed')
   365→      onReload()
   366→    } catch (e: any) {
   367→      toast.error('Failed to remove schedule item', { description: e.message })
   368→    }
   369→  }
   370→
   371→  const handleSaveEdit = async (item: any) => {
   372→    try {
   373→      await updateItem(exam.id, item.id, {
   374→        date: editDate || undefined,
   375→        startTime: editStart || undefined,
   376→        endTime: editEnd || undefined,
   377→        room: editRoom,
   378→      })
   379→      toast.success('Schedule item updated')
   380→      setEditingId(null)
   381→      onReload()
   382→    } catch (e: any) {
   383→      toast.error('Failed to update', { description: e.message })
   384→    }
   385→  }
   386→
   387→  const handleAssignInvigilator = async (itemId: string, teacherId: string) => {
   388→    if (!teacherId) return
   389→    try {
   390→      await assignInvigilator(exam.id, itemId, teacherId)
   391→      toast.success('Invigilator assigned')
   392→      onReload()
   393→    } catch (e: any) {
   394→      toast.error('Failed to assign invigilator', { description: e.message })
   395→    }
   396→  }
   397→
   398→  const subjectsForClass = exam.subjects.filter((s: any) => s.classId === classId)
   399→  const scheduleForClass = exam.schedule.filter((s: any) => s.classId === classId)
   400→
   401→  return (
   402→    <div className="space-y-3">
   403→      {gate.canManageSchedule && (
   404→        <div className="rounded-xl border border-border bg-card p-3">
   405→          <p className="text-[10px] uppercase font-semibold text-muted-foreground mb-2">Add Schedule Item</p>
   406→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   407→            <Select value={classId} onValueChange={(v) => { setClassId(v); setSubjectId('') }}>
   408→              <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Class" /></SelectTrigger>
   409→              <SelectContent>
   410→                {exam.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   411→              </SelectContent>
   412→            </Select>
   413→            <Select value={subjectId} onValueChange={setSubjectId}>
   414→              <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Subject" /></SelectTrigger>
   415→              <SelectContent>
   416→                {subjectsForClass.map((s: any) => <SelectItem key={s.subjectId} value={s.subjectId}>{s.subjectName}</SelectItem>)}
   417→              </SelectContent>
   418→            </Select>
   419→            <DatePicker value={date} onChange={setDate} placeholder="Date" />
   420→            <Input type="time" value={startTime} onChange={(e) => setStartTime(e.target.value)} className="h-8 text-xs" />
   421→            <Input type="time" value={endTime} onChange={(e) => setEndTime(e.target.value)} className="h-8 text-xs" />
   422→            <Input value={room} onChange={(e) => setRoom(e.target.value)} placeholder="Room" className="h-8 text-xs" />
   423→            <Input value={invigilator} onChange={(e) => setInvigilator(e.target.value)} placeholder="Invigilator" className="h-8 text-xs" />
   424→          </div>
   425→          <Button size="sm" className="mt-2 h-7 text-xs gap-1" onClick={handleAdd}>
   426→            <Plus2 /> Add to Schedule
   427→          </Button>
   428→        </div>
   429→      )}
   430→
   431→      <div className="rounded-xl border border-border bg-card overflow-hidden">
   432→        <div className="px-3 py-2 border-b border-border flex items-center gap-2">
   433→          <p className="text-[10px] uppercase font-semibold text-muted-foreground">
   434→            Schedule for {exam.classes.find((c: any) => c.classId === classId)?.className ?? '—'} ({scheduleForClass.length} items)
   435→          </p>
   436→          {teachers.length > 0 && (
   437→            <span className="text-[10px] text-muted-foreground ml-auto">{teachers.length} teachers available</span>
   438→          )}
   439→        </div>
   440→        {scheduleForClass.length === 0 ? (
   441→          <div className="p-6 text-center text-xs text-muted-foreground">No schedule items yet.</div>
   442→        ) : (
   443→          <div className="overflow-x-auto">
   444→            <table className="w-full text-xs">
   445→              <thead className="bg-muted/40">
   446→                <tr>
   447→                  <th className="text-left text-[9px] uppercase font-semibold text-muted-foreground px-3 py-2">Subject</th>
   448→                  <th className="text-left text-[9px] uppercase font-semibold text-muted-foreground px-3 py-2">Date</th>
   449→                  <th className="text-left text-[9px] uppercase font-semibold text-muted-foreground px-3 py-2">Time</th>

   553→function MarksSection({ exam, onReload }: { exam: any; onReload: () => void }) {
   554→  const { submit } = useSubmitMarks()
   555→  const { verify } = useVerifyMarks()
   556→  const { lock } = useLockMarks()
   557→  const { declare } = useDeclareResults()
   558→  const { publish } = usePublishResults()
   559→  const gate = useRoleGate()
   560→  const [classId, setClassId] = useState(exam.classes[0]?.classId ?? '')
   561→  const [subjectId, setSubjectId] = useState('')
   562→
   563→  const subjectsForClass = exam.subjects.filter((s: any) => s.classId === classId)
   564→
   565→  const handleAction = async (action: 'submit' | 'verify' | 'lock' | 'declare' | 'publish') => {
   566→    try {
   567→      const filter = { classId, ...(subjectId ? { subjectId } : {}) }
   568→      if (action === 'submit') {
   569→        const r = await submit(exam.id, filter)
   570→        toast.success(`Submitted ${r.submitted} marks`)
   571→      } else if (action === 'verify') {
   572→        const r = await verify(exam.id, filter)
   573→        toast.success(`Verified ${r.verified} marks`)
   574→      } else if (action === 'lock') {
   575→        const r = await lock(exam.id, filter)
   576→        toast.success(`Locked ${r.locked} marks`)
   577→      } else if (action === 'declare') {
   578→        await declare(exam.id)
   579→        toast.success('Results declared')
   580→      } else if (action === 'publish') {
   581→        const r = await publish(exam.id, { notifyStudents: true, notifyParents: true })
   582→        toast.success(`Results published`, { description: `${r.notificationsSent} notifications sent.` })
   583→      }
   584→      onReload()
   585→    } catch (e: any) {
   586→      toast.error(`Failed to ${action}`, { description: e.message })
   587→    }
   588→  }
   589→
   590→  return (
   591→    <div className="space-y-3">
   592→      <div className="rounded-xl border border-border bg-card p-3">
   593→        <p className="text-[10px] uppercase font-semibold text-muted-foreground mb-2">Workflow Controls</p>
   594→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   595→          <Select value={classId} onValueChange={(v) => { setClassId(v); setSubjectId('') }}>
   596→            <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="Class" /></SelectTrigger>
   597→            <SelectContent>
   598→              {exam.classes.map((c: any) => <SelectItem key={c.classId} value={c.classId}>{c.className}</SelectItem>)}
   599→            </SelectContent>
   600→          </Select>
   601→          <Select value={subjectId || 'all'} onValueChange={(v) => setSubjectId(v === 'all' ? '' : v)}>
   602→            <SelectTrigger size="sm" className="text-xs"><SelectValue placeholder="All Subjects" /></SelectTrigger>
   603→            <SelectContent>
   604→              <SelectItem value="all">All Subjects in Class</SelectItem>
   605→              {subjectsForClass.map((s: any) => <SelectItem key={s.subjectId} value={s.subjectId}>{s.subjectName}</SelectItem>)}
   606→            </SelectContent>
   607→          </Select>
   608→        </div>
   609→        <div className="flex flex-wrap gap-2 mt-3">
   610→          {gate.canSubmitMarks && (
   611→            <Button size="sm" variant="outline" className="h-7 text-xs gap-1.5" onClick={() => handleAction('submit')} disabled={exam.resultStatus === 'Result Declared'}>
   612→              Submit
   613→            </Button>
   614→          )}
   615→          {gate.canVerifyMarks && (
   616→            <Button size="sm" variant="outline" className="h-7 text-xs gap-1.5" onClick={() => handleAction('verify')} disabled={exam.resultStatus === 'Result Declared'}>
   617→              Verify
   618→            </Button>
   619→          )}
   620→          {gate.canLockMarks && (
   621→            <Button size="sm" variant="outline" className="h-7 text-xs gap-1.5" onClick={() => handleAction('lock')} disabled={exam.resultStatus === 'Result Declared'}>
   622→              Lock
   623→            </Button>
   624→          )}
   625→          {gate.canDeclareResults && (
   626→            <Button size="sm" variant="default" className="h-7 text-xs gap-1.5 bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => handleAction('declare')} disabled={exam.resultStatus !== 'Result Ready'}>
   627→              Declare
   628→            </Button>
   629→          )}
   630→          {gate.canPublishResults && (
   631→            <Button size="sm" variant="outline" className="h-7 text-xs gap-1.5" onClick={() => handleAction('publish')} disabled={exam.resultStatus !== 'Result Declared'}>
   632→              Publish & Notify
   633→            </Button>
   634→          )}
   635→        </div>
   636→      </div>
   637→
   638→      <div className="rounded-xl border border-border bg-card p-3">
   639→        <p className="text-[10px] uppercase font-semibold text-muted-foreground mb-2">Progress Summary</p>
   640→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   641→          <Stat label="Total Mark Rows" value={exam.markSummary.total} />
   642→          <Stat label="Entered" value={exam.markSummary.entered} />
   643→          <Stat label="Submitted" value={exam.markSummary.submitted} />
   644→          <Stat label="Verified" value={exam.markSummary.verified} />
   645→        </div>
   646→      </div>
   647→      <p className="text-[10px] text-muted-foreground px-1">
   648→        Tip: Use the Marks tab in the main Examinations navigation for spreadsheet entry.
   649→      </p>
   650→    </div>
   651→  )
