    28→// ─── Date helpers (local-time, stable YYYY-MM-DD keys) ────────────

    30→function todayISO(): string {
    31→  const d = new Date()
    32→  const y = d.getFullYear()
    33→  const m = String(d.getMonth() + 1).padStart(2, '0')
    34→  const day = String(d.getDate()).padStart(2, '0')
    35→  return `${y}-${m}-${day}`
    36→}
    37→
    38→function daysFromNow(days: number, hour = 19, minute = 0): string {
    39→  const d = new Date()
    40→  d.setDate(d.getDate() + days)
    41→  d.setHours(hour, minute, 0, 0)
    42→  return d.toISOString()
    43→}
    44→
    45→function dateOnlyShift(days: number): string {
    46→  const d = new Date()
    47→  d.setDate(d.getDate() + days)
    48→  const y = d.getFullYear()
    49→  const m = String(d.getMonth() + 1).padStart(2, '0')
    50→  const day = String(d.getDate()).padStart(2, '0')
    51→  return `${y}-${m}-${day}`
    52→}

   659→export function dueStatsOf(cards: LearningCard[], at: Date = new Date()): DueStats {
   660→  let due = 0
   661→  let fresh = 0
   662→  let learning = 0
   663→  let reviewing = 0
   664→  let mastered = 0
   665→  let newCards = 0
   666→  for (const c of cards) {
   667→    if (c.status === 'mastered') mastered += 1
   668→    else if (c.status === 'learning') learning += 1
   669→    else if (c.status === 'reviewing') reviewing += 1
   670→    else newCards += 1
   671→    const isDue = c.status !== 'mastered' && (c.dueAt === null || new Date(c.dueAt).getTime() <= at.getTime())
   672→    if (isDue) due += 1
   673→    if (c.dueAt === null && c.status === 'new') fresh += 1
   674→  }
   675→  return { due, fresh, learning, reviewing, mastered, newCards, total: cards.length }
   676→}

   678→/** The review queue: due cards, new-first-later ordering (learning first). */
   679→export function reviewQueueOf(cards: LearningCard[], at: Date = new Date()): LearningCard[] {
   680→  return cards
   681→    .filter((c) => c.status !== 'mastered' && (c.dueAt === null || new Date(c.dueAt).getTime() <= at.getTime()))
   682→    .sort((a, b) => {
   683→      const rank = (c: LearningCard) => (c.status === 'learning' ? 0 : c.status === 'reviewing' ? 1 : 2)
   684→      const ra = rank(a)
   685→      const rb = rank(b)
   686→      if (ra !== rb) return ra - rb
   687→      const da = a.dueAt ? new Date(a.dueAt).getTime() : Number.MAX_SAFE_INTEGER
   688→      const db = b.dueAt ? new Date(b.dueAt).getTime() : Number.MAX_SAFE_INTEGER
   689→      return da - db
   690→    })
   691→}

   693→export interface SubjectStat {
   694→  subject: string
   695→  resourcesTotal: number
   696→  resourcesCompleted: number
   697→  resourcesInProgress: number
   698→  cardsTotal: number
   699→  cardsMastered: number
   700→  /** Derived 0–100: completed resources + mastered cards, honest blend. */
   701→  masteryPct: number
   702→}
   703→
   704→export function subjectStatsOf(
   705→  resources: LearningResource[],
   706→  progress: Record<string, ResourceProgress>,
   707→  cards: LearningCard[],
   708→): SubjectStat[] {
   709→  const subjects = Array.from(
   710→    new Set([...resources.map((r) => r.subject), ...cards.map((c) => c.subject)]),
   711→  ).filter(Boolean)
   712→  return subjects.map((subject) => {
   713→    const res = resources.filter((r) => r.subject === subject)
   714→    const completed = res.filter((r) => (progress[r.id]?.pct ?? 0) >= 100).length
   715→    const inProgress = res.filter((r) => {
   716→      const p = progress[r.id]?.pct ?? 0
   717→      return p > 0 && p < 100
   718→    }).length
   719→    const cardsForSubject = cards.filter((c) => c.subject === subject)
   720→    const mastered = cardsForSubject.filter((c) => c.status === 'mastered').length
   721→    // Blend: resources count toward half, mastered cards the other half —
   722→    // only over what actually exists (a subject with no cards leans on
   723→    // resources and vice-versa).
   724→    const resPart = res.length > 0 ? completed / res.length : null
   725→    const cardPart = cardsForSubject.length > 0 ? mastered / cardsForSubject.length : null
   726→    const parts = [resPart, cardPart].filter((p): p is number => p !== null)
   727→    const masteryPct = parts.length > 0 ? Math.round((parts.reduce((a, b) => a + b, 0) / parts.length) * 100) : 0
   728→    return {
   729→      subject,
   730→      resourcesTotal: res.length,
   731→      resourcesCompleted: completed,
   732→      resourcesInProgress: inProgress,
   733→      cardsTotal: cardsForSubject.length,
   734→      cardsMastered: mastered,
   735→      masteryPct,
   736→    }
   737→  })
   738→}

   789→/**
   790→ * A study streak day = at least one recorded study session that day
   791→ * (spec §35 — opening the page never counts).
   792→ */
   793→export function streakOf(sessions: StudySession[], at: Date = new Date()): StreakInfo {
   794→  const days = new Set(sessions.map((s) => dateKeyOf(s.endedAt)))
   795→  if (days.size === 0) return { current: 0, longest: 0, activeToday: false }
   796→
   797→  // Longest run over the recorded history.
   798→  const sorted = Array.from(days).sort()
   799→  let longest = 1
   800→  let run = 1
   801→  for (let i = 1; i < sorted.length; i++) {
   802→    const prev = new Date(sorted[i - 1] + 'T00:00:00')
   803→    const curr = new Date(sorted[i] + 'T00:00:00')
   804→    const diff = Math.round((curr.getTime() - prev.getTime()) / 86400000)
   805→    run = diff === 1 ? run + 1 : 1
   806→    longest = Math.max(longest, run)
   807→  }
   808→
   809→  // Current run: counts back from today (or yesterday if today is a
   810→  // rest day so far — the streak isn't "broken" until the day ends).
   811→  const todayKey = dateKeyOf(at.toISOString())
   812→  const yKey = new Date(at)
   813→  yKey.setDate(yKey.getDate() - 1)
   814→  const yesterdayKey = dateKeyOf(yKey.toISOString())
   815→  let cursor: Date
   816→  if (days.has(todayKey)) cursor = new Date(todayKey + 'T00:00:00')
   817→  else if (days.has(yesterdayKey)) cursor = new Date(yesterdayKey + 'T00:00:00')
   818→  else return { current: 0, longest, activeToday: false }
   819→  let current = 0
   820→  while (days.has(dateKeyOf(cursor.toISOString()))) {
   821→    current += 1
   822→    cursor.setDate(cursor.getDate() - 1)
   823→  }
   824→  return { current, longest, activeToday: days.has(todayKey) }
   825→}

   827→/**
   828→ * RECOMMENDED — derived only from real signals (spec §14):
   829→ *   1. unfinished resources in the student's weakest subjects
   830→ *      (from the canonical results store — lowest latest %),
   831→ *   2. other unfinished / in-progress resources,
   832→ *   3. bookmarked-but-not-started.
   833→ * `weakSubjects` is passed in by the caller (results integration,
   834→ * permission-safe: only the student's own subject percentages).
   835→ */
   836→export function recommendedOf(
   837→  resources: LearningResource[],
   838→  progress: Record<string, ResourceProgress>,
   839→  bookmarks: string[],
   840→  weakSubjects: string[],
   841→  limit = 3,
   842→): LearningResource[] {
   843→  const weakRank = new Map(weakSubjects.map((s, i) => [s, i]))
   844→  const score = (r: LearningResource): number => {
   845→    const pct = progress[r.id]?.pct ?? 0
   846→    let s = 0
   847→    if (weakRank.has(r.subject)) s -= 100 - (weakRank.get(r.subject) ?? 0) * 10
   848→    if (pct > 0 && pct < 100) s -= 20 // in progress → finish it
   849→    if (bookmarks.includes(r.id)) s -= 5
   850→    if (pct >= 100) s += 1000 // completed → last
   851→    return s
   852→  }
   853→  return [...resources].sort((a, b) => score(a) - score(b)).slice(0, limit)
   854→}

   856→/** Goal progress (current value + target) derived from real state. */
   857→export interface GoalProgress {
   858→  goal: StudyGoal
   859→  current: number
   860→  pct: number
   861→  unit: string
   862→}
   863→
   864→export function goalProgressOf(goal: StudyGoal, ctx: { sessions: StudySession[]; tasks: PlannerTask[]; cards: LearningCard[] }): GoalProgress {
   865→  switch (goal.kind) {
   866→    case 'weekly-minutes': {
   867→      const current = weeklyMinutes(ctx.sessions)
   868→      return { goal, current, pct: Math.min(100, Math.round((current / goal.target) * 100)), unit: 'min' }
   869→    }
   870→    case 'weekly-tasks': {
   871→      const current = weeklyCompletedTasks(ctx.tasks)
   872→      return { goal, current, pct: Math.min(100, Math.round((current / goal.target) * 100)), unit: 'tasks' }
   873→    }
   874→    case 'deck-master': {
   875→      const cards = ctx.cards.filter((c) => c.subject === goal.subject)
   876→      const mastered = cards.filter((c) => c.status === 'mastered').length
   877→      const pct = cards.length > 0 ? Math.round((mastered / cards.length) * 100) : 0
   878→      return { goal, current: mastered, pct: goal.target <= 100 ? pct : Math.min(100, mastered), unit: 'cards' }
   879→    }
   880→    case 'subject-resources':
   881→    default: {
   882→      // Kept for future subject-completion goals.
   883→      return { goal, current: 0, pct: 0, unit: '' }
   884→    }
   885→  }
   886→}
