    97→function toExamDTO(e: any): ExamDTO {
    98→  const marks = e.marks ?? []
    99→  const total = marks.length
   100→  const entered = marks.filter((m: any) => m.marksObtained !== null || m.status !== 'PRESENT').length
   101→  const locked = marks.filter((m: any) => m.workflowStatus === 'LOCKED').length
   102→  const submitted = marks.filter((m: any) => m.workflowStatus === 'SUBMITTED').length
   103→  const verified = marks.filter((m: any) => m.workflowStatus === 'VERIFIED').length
   104→  return {
   105→    id: e.id,
   106→    schoolId: e.schoolId,
   107→    name: e.name,
   108→    type: e.type ?? 'Unit Test',
   109→    session: e.session ?? '2025-2026',
   110→    term: e.term ?? null,
   111→    status: e.status ?? 'Draft',
   112→    resultStatus: e.resultStatus ?? 'Not Started',
   113→    passPercentage: e.passPercentage ?? 33,
   114→    startDate: e.startDate ? new Date(e.startDate).toISOString().split('T')[0] : null,
   115→    endDate: e.endDate ? new Date(e.endDate).toISOString().split('T')[0] : null,
   116→    declaredAt: e.declaredAt ? e.declaredAt.toISOString() : null,
   117→    declaredBy: e.declaredBy,
   118→    createdBy: e.createdBy,
   119→    createdAt: e.createdAt.toISOString(),
   120→    updatedAt: e.updatedAt.toISOString(),
   121→    classes: (e.examClasses ?? []).map(toExamClassDTO),
   122→    subjects: (e.examSubjects ?? []).map(toSubjectConfigDTO),
   123→    schedule: (e.scheduleItems ?? []).map(toScheduleDTO),
   124→    markSummary: {
   125→      total,
   126→      entered,
   127→      locked,
   128→      submitted,
   129→      verified,
   130→      pct: total > 0 ? Math.round((entered / total) * 100) : 0,
   131→    },
   132→  }
   133→}

   280→export async function createExam(
   281→  schoolId: string,
   282→  user: AuthUserLike | null,
   283→  input: CreateExamInput
   284→): Promise<ExamDTO> {
   285→  if (!input.name?.trim()) throw new Error('Examination name is required')
   286→  if (input.classIds.length === 0) throw new Error('At least one class is required')
   287→
   288→  // Validate classes belong to school
   289→  const validClasses = await db.class.findMany({
   290→    where: { id: { in: input.classIds }, schoolId },
   291→    select: { id: true, name: true },
   292→  })
   293→  if (validClasses.length !== input.classIds.length) {
   294→    throw new Error('One or more classes not found in this school')
   295→  }
   296→
   297→  // Validate subjects exist
   298→  for (const classId of input.classIds) {
   299→    const subjects = input.subjectsByClass[classId] ?? []
   300→    if (subjects.length === 0) throw new Error(`No subjects selected for class ${validClasses.find((c) => c.id === classId)?.name}`)
   301→    const validSubjects = await db.subject.findMany({
   302→      where: { id: { in: subjects.map((s) => s.subjectId) }, classId },
   303→      select: { id: true },
   304→    })
   305→    if (validSubjects.length !== subjects.length) {
   306→      throw new Error('One or more subjects not found for the selected class')
   307→    }
   308→  }
   309→
   310→  const exam = await db.exam.create({
   311→    data: {
   312→      schoolId,
   313→      name: input.name.trim(),
   314→      type: input.type,
   315→      session: input.session ?? '2025-2026',
   316→      startDate: input.startDate ? new Date(input.startDate) : null,
   317→      endDate: input.endDate ? new Date(input.endDate) : null,
   318→      status: 'Draft',
   319→      resultStatus: 'Not Started',
   320→      passPercentage: input.passPercentage ?? 33,
   321→      createdBy: user?.id ?? null,
   322→      examClasses: {
   323→        create: input.classIds.map((classId) => ({ classId })),
   324→      },
   325→      examSubjects: {
   326→        create: input.classIds.flatMap((classId) =>
   327→          (input.subjectsByClass[classId] ?? []).map((s, idx) => ({
   328→            classId,
   329→            subjectId: s.subjectId,
   330→            maxMarks: s.maxMarks ?? 100,
   331→            passMarks: s.passMarks ?? 33,
   332→            theoryMarks: s.theoryMarks ?? s.maxMarks ?? 100,
   333→            practicalMarks: s.practicalMarks ?? 0,
   334→            sortOrder: idx,
   335→          }))
   336→        ),
   337→      },
   338→    },
   339→    include: EXAM_INCLUDE,
   340→  })
   341→
   342→  // Create schedule items if provided
   343→  if (input.schedule && input.schedule.length > 0) {
   344→    for (const item of input.schedule) {
   345→      await db.examScheduleItem.create({
   346→        data: {
   347→          examId: exam.id,
   348→          classId: item.classId,
   349→          subjectId: item.subjectId,
   350→          date: new Date(item.date),
   351→          startTime: item.startTime,
   352→          endTime: item.endTime,
   353→          room: item.room ?? null,
   354→          invigilatorName: item.invigilatorName ?? null,
   355→        },
   356→      })
   357→    }
   358→  }
   359→
   360→  // Auto-create empty ExamMark rows for every student in every class+subject
   361→  for (const classId of input.classIds) {
   362→    const students = await db.student.findMany({ where: { classId, schoolId }, select: { id: true } })
   363→    const subjects = input.subjectsByClass[classId] ?? []
   364→    for (const student of students) {
   365→      for (const subject of subjects) {
   366→        await db.examMark.upsert({
   367→          where: {
   368→            examId_classId_subjectId_studentId: {
   369→              examId: exam.id,
   370→              classId,
   371→              subjectId: subject.subjectId,
   372→              studentId: student.id,
   373→            },
   374→          },
   375→          create: {
   376→            examId: exam.id,
   377→            classId,
   378→            subjectId: subject.subjectId,
   379→            studentId: student.id,
   380→            marksObtained: null,
   381→            status: 'PRESENT',
   382→            workflowStatus: 'DRAFT',
   383→          },
   384→          update: {},
   385→        })
   386→      }
   387→    }
   388→  }
   389→
   390→  await audit(exam.id, user, 'EXAM_CREATED', 'EXAM', exam.id, null, { name: exam.name, type: exam.type })
   391→
   392→  const fresh = await db.exam.findUnique({ where: { id: exam.id }, include: EXAM_INCLUDE })
   393→  return toExamDTO(fresh!)
   394→}

   398→export async function updateExam(
   399→  examId: string,
   400→  schoolId: string,
   401→  user: AuthUserLike | null,
   402→  updates: { name?: string; type?: string; session?: string; startDate?: string; endDate?: string; status?: string; passPercentage?: number }
   403→): Promise<ExamDTO> {
   404→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   405→  if (!exam) throw new Error('Exam not found')
   406→  const oldSnapshot = { name: exam.name, type: exam.type, status: exam.status, resultStatus: exam.resultStatus }
   407→  const updated = await db.exam.update({
   408→    where: { id: examId },
   409→    data: {
   410→      ...(updates.name !== undefined ? { name: updates.name } : {}),
   411→      ...(updates.type !== undefined ? { type: updates.type } : {}),
   412→      ...(updates.session !== undefined ? { session: updates.session } : {}),
   413→      ...(updates.startDate !== undefined ? { startDate: updates.startDate ? new Date(updates.startDate) : null } : {}),
   414→      ...(updates.endDate !== undefined ? { endDate: updates.endDate ? new Date(updates.endDate) : null } : {}),
   415→      ...(updates.status !== undefined ? { status: updates.status } : {}),
   416→      ...(updates.passPercentage !== undefined ? { passPercentage: updates.passPercentage } : {}),
   417→    },
   418→    include: EXAM_INCLUDE,
   419→  })
   420→  await audit(examId, user, 'EXAM_UPDATED', 'EXAM', examId, oldSnapshot, {
   421→    name: updates.name,
   422→    type: updates.type,
   423→    status: updates.status,
   424→  })
   425→  return toExamDTO(updated)
   426→}

   435→// ─── Schedule management ──────────────────────────────────────────────
   436→
   437→export async function addScheduleItem(
   438→  examId: string,
   439→  schoolId: string,
   440→  user: AuthUserLike | null,
   441→  data: {
   442→    classId: string
   443→    subjectId: string
   444→    date: string
   445→    startTime: string
   446→    endTime: string
   447→    room?: string
   448→    invigilatorName?: string
   449→  }
   450→): Promise<ScheduleItemDTO> {
   451→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   452→  if (!exam) throw new Error('Exam not found')
   453→  // Conflict detection — same room+date+time OR same class+date+time OR same invigilator+date+time
   454→  const conflicts = await db.examScheduleItem.findMany({
   455→    where: {
   456→      examId,
   457→      OR: [
   458→        { classId: data.classId, date: new Date(data.date), startTime: data.startTime },
   459→        { room: data.room ?? undefined, date: new Date(data.date), startTime: data.startTime },
   460→      ],
   461→    },
   462→  })
   463→  if (conflicts.length > 0) {
   464→    throw new Error('Schedule conflict detected: same class or room already booked at this time')
   465→  }
   466→  const item = await db.examScheduleItem.create({
   467→    data: {
   468→      examId,
   469→      classId: data.classId,
   470→      subjectId: data.subjectId,
   471→      date: new Date(data.date),
   472→      startTime: data.startTime,
   473→      endTime: data.endTime,
   474→      room: data.room ?? null,
   475→      invigilatorName: data.invigilatorName ?? null,
   476→    },
   477→    include: { class: true, subject: true },
   478→  })
   479→  await audit(examId, user, 'SCHEDULE_ADDED', 'SCHEDULE', item.id, null, data)
   480→  return toScheduleDTO(item)
   481→}

   531→// ─── Marks management ─────────────────────────────────────────────────
   532→
   533→export async function setMark(
   534→  examId: string,
   535→  schoolId: string,
   536→  user: AuthUserLike | null,
   537→  input: SetMarkInput
   538→): Promise<ExamMarkDTO> {
   539→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   540→  if (!exam) throw new Error('Exam not found')
   541→  if (exam.resultStatus === 'Result Declared') {
   542→    throw new Error('Cannot modify marks after results are declared')
   543→  }
   544→
   545→  // Validate marks against max
   546→  const subjectConfig = await db.examSubjectConfig.findFirst({
   547→    where: { examId, classId: input.classId, subjectId: input.subjectId },
   548→  })
   549→  if (!subjectConfig) throw new Error('Subject not configured for this exam/class')
   550→
   551→  const max = subjectConfig.maxMarks
   552→  if (input.status === 'PRESENT' && input.marksObtained !== null) {
   553→    if (input.marksObtained < 0) throw new Error('Marks cannot be negative')
   554→    if (input.marksObtained > max) throw new Error(`Marks cannot exceed maximum (${max})`)
   555→  }
   556→  if (input.status !== 'PRESENT' && input.marksObtained !== null && input.marksObtained !== 0) {
   557→    // For ABSENT/MEDICAL/EXEMPTED we typically store null; coerce
   558→    input.marksObtained = null
   559→  }
   560→
   561→  const existing = await db.examMark.findUnique({
   562→    where: {
   563→      examId_classId_subjectId_studentId: {
   564→        examId,
   565→        classId: input.classId,
   566→        subjectId: input.subjectId,
   567→        studentId: input.studentId,
   568→      },
   569→    },
   570→  })
   571→
   572→  // If mark is already LOCKED, cannot modify without override flow
   573→  if (existing?.workflowStatus === 'LOCKED') {
   574→    throw new Error('Mark is locked. Use the unlock/reopen workflow to modify.')
   575→  }
   576→  if (existing?.workflowStatus === 'VERIFIED' && user?.role !== 'PRINCIPAL' && user?.role !== 'MANAGEMENT') {
   577→    throw new Error('Verified marks can only be modified by Principal')
   578→  }
   579→
   580→  const oldValue = existing ? {
   581→    marksObtained: existing.marksObtained,
   582→    status: existing.status,
   583→  } : null
   584→
   585→  // Preserve workflow status if already SUBMITTED/VERIFIED — but updates after submission
   586→  // remain allowed for teachers only if not yet LOCKED.
   587→  const updated = await db.examMark.upsert({
   588→    where: {
   589→      examId_classId_subjectId_studentId: {
   590→        examId,
   591→        classId: input.classId,
   592→        subjectId: input.subjectId,
   593→        studentId: input.studentId,
   594→      },
   595→    },
   596→    create: {
   597→      examId,
   598→      classId: input.classId,
   599→      subjectId: input.subjectId,
   600→      studentId: input.studentId,
   601→      marksObtained: input.status === 'PRESENT' ? input.marksObtained : null,
   602→      status: input.status,
   603→      workflowStatus: 'DRAFT',
   604→      originalMarks: input.status === 'PRESENT' ? input.marksObtained : null,
   605→      remarks: input.remarks ?? null,
   606→      enteredBy: user?.id ?? null,
   607→      enteredAt: new Date(),
   608→    },
   609→    update: {
   610→      marksObtained: input.status === 'PRESENT' ? input.marksObtained : null,
   611→      status: input.status,
   612→      remarks: input.remarks ?? existing?.remarks ?? null,
   613→      enteredBy: user?.id ?? existing?.enteredBy ?? null,
   614→      enteredAt: new Date(),
   615→      // If reverting from SUBMITTED to DRAFT (any edit), allow it
   616→      ...(existing?.workflowStatus === 'SUBMITTED' ? { workflowStatus: 'DRAFT' as const } : {}),
   617→    },
   618→    include: { student: { include: { user: { select: { name: true } } } } },
   619→  })
   620→
   621→  // Update exam resultStatus to "Marks Entry" if it was "Not Started"
   622→  if (exam.resultStatus === 'Not Started') {
   623→    await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Marks Entry' } })
   624→  }
   625→
   626→  await audit(examId, user, 'MARK_ENTERED', 'MARK', updated.id, oldValue, {
   627→    marksObtained: updated.marksObtained,
   628→    status: updated.status,
   629→  })
   630→  return toMarkDTO(updated)
   631→}

   633→export async function setMarksBatch(
   634→  examId: string,
   635→  schoolId: string,
   636→  user: AuthUserLike | null,
   637→  marks: SetMarkInput[]
   638→): Promise<{ updated: number }> {
   639→  let updated = 0
   640→  for (const m of marks) {
   641→    try {
   642→      await setMark(examId, schoolId, user, m)
   643→      updated++
   644→    } catch (err) {
   645→      // continue but skip invalid entries
   646→      console.warn('setMarksBatch skip', m, err)
   647→    }
   648→  }
   649→  return { updated }
   650→}

   652→export async function submitMarks(
   653→  examId: string,
   654→  schoolId: string,
   655→  user: AuthUserLike | null,
   656→  filter: { classId?: string; subjectId?: string }
   657→): Promise<{ submitted: number }> {
   658→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   659→  if (!exam) throw new Error('Exam not found')
   660→  if (exam.resultStatus === 'Result Declared') {
   661→    throw new Error('Cannot modify marks after results are declared')
   662→  }
   663→  const result = await db.examMark.updateMany({
   664→    where: {
   665→      examId,
   666→      ...(filter.classId ? { classId: filter.classId } : {}),
   667→      ...(filter.subjectId ? { subjectId: filter.subjectId } : {}),
   668→      workflowStatus: 'DRAFT',
   669→      // Must have either marks entered OR a non-PRESENT status (absent/medical/exempted)
   670→      OR: [
   671→        { marksObtained: { not: null } },
   672→        { status: { not: 'PRESENT' } },
   673→      ],
   674→    },
   675→    data: {
   676→      workflowStatus: 'SUBMITTED',
   677→      enteredBy: user?.id ?? null,
   678→    },
   679→  })
   680→  // Move exam to Under Verification if all marks submitted
   681→  const remainingDrafts = await db.examMark.count({ where: { examId, workflowStatus: 'DRAFT' } })
   682→  if (remainingDrafts === 0) {
   683→    await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Under Verification' } })
   684→  } else {
   685→    await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Marks Entry' } })
   686→  }
   687→  await audit(examId, user, 'MARK_SUBMITTED', 'MARK', null, null, filter)
   688→  return { submitted: result.count }
   689→}

   691→export async function verifyMarks(
   692→  examId: string,
   693→  schoolId: string,
   694→  user: AuthUserLike | null,
   695→  filter: { classId?: string; subjectId?: string }
   696→): Promise<{ verified: number }> {
   697→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   698→  if (!exam) throw new Error('Exam not found')
   699→  if (exam.resultStatus === 'Result Declared') {
   700→    throw new Error('Cannot modify marks after results are declared')
   701→  }
   702→  const result = await db.examMark.updateMany({
   703→    where: {
   704→      examId,
   705→      ...(filter.classId ? { classId: filter.classId } : {}),
   706→      ...(filter.subjectId ? { subjectId: filter.subjectId } : {}),
   707→      workflowStatus: 'SUBMITTED',
   708→    },
   709→    data: {
   710→      workflowStatus: 'VERIFIED',
   711→      verifiedBy: user?.id ?? null,
   712→      verifiedAt: new Date(),
   713→    },
   714→  })
   715→  const remainingSubmitted = await db.examMark.count({ where: { examId, workflowStatus: 'SUBMITTED' } })
   716→  if (remainingSubmitted === 0) {
   717→    const remainingDrafts = await db.examMark.count({ where: { examId, workflowStatus: 'DRAFT' } })
   718→    if (remainingDrafts === 0) {
   719→      await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Result Ready' } })
   720→    }
   721→  }
   722→  await audit(examId, user, 'MARK_VERIFIED', 'MARK', null, null, filter)
   723→  return { verified: result.count }
   724→}

   726→export async function lockMarks(
   727→  examId: string,
   728→  schoolId: string,
   729→  user: AuthUserLike | null,
   730→  filter: { classId?: string; subjectId?: string }
   731→): Promise<{ locked: number }> {
   732→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   733→  if (!exam) throw new Error('Exam not found')
   734→  if (exam.resultStatus === 'Result Declared') {
   735→    throw new Error('Cannot modify marks after results are declared')
   736→  }
   737→  // Lock everything matching filter (DRAFT/SUBMITTED/VERIFIED → LOCKED)
   738→  const result = await db.examMark.updateMany({
   739→    where: {
   740→      examId,
   741→      ...(filter.classId ? { classId: filter.classId } : {}),
   742→      ...(filter.subjectId ? { subjectId: filter.subjectId } : {}),
   743→      workflowStatus: { in: ['DRAFT', 'SUBMITTED', 'VERIFIED'] },
   744→    },
   745→    data: {
   746→      workflowStatus: 'LOCKED',
   747→      lockedBy: user?.id ?? null,
   748→    },
   749→  })
   750→  await db.exam.update({ where: { id: examId }, data: { resultStatus: 'Result Ready' } })
   751→  await audit(examId, user, 'MARK_LOCKED', 'MARK', null, null, filter)
   752→  return { locked: result.count }
   753→}

   757→export async function getResultsForClass(
   758→  examId: string,
   759→  classId: string,
   760→  schoolId: string
   761→): Promise<{ students: StudentDTO[]; subjects: ExamSubjectConfigDTO[]; marks: ExamMarkDTO[]; results: StudentResult[]; analytics: ExamAnalyticsDTO }> {
   762→  const exam = await db.exam.findFirst({
   763→    where: { id: examId, schoolId },
   764→    include: {
   765→      examSubjects: { include: { subject: true }, where: { classId } },
   766→      marks: { where: { classId }, include: { student: { include: { user: { select: { name: true } } } } } },
   767→      examClasses: { include: { class: { include: { _count: { select: { students: true } } } } } },
   768→    },
   769→  })
   770→  if (!exam) throw new Error('Exam not found')
   771→  const cls = exam.examClasses.find((ec) => ec.classId === classId)?.class
   772→  if (!cls) throw new Error('Class not part of this exam')
   773→
   774→  // Get real students of this class
   775→  const students = await db.student.findMany({
   776→    where: { classId, schoolId },
   777→    orderBy: { rollNo: 'asc' },
   778→    include: { user: { select: { name: true } } },
   779→  })
   780→
   781→  const studentDTOs: StudentDTO[] = students.map((s) => ({
   782→    id: s.id,
   783→    rollNo: s.rollNo,
   784→    admissionNo: s.admissionNo,
   785→    name: s.user?.name ?? '',
   786→    classId: s.classId,
   787→  }))
   788→
   789→  const subjectDTOs = exam.examSubjects.map(toSubjectConfigDTO)
   790→  const markDTOs = exam.marks.map(toMarkDTO)
   791→
   792→  const results = computeAllResults({
   793→    students: studentDTOs,
   794→    subjects: subjectDTOs,
   795→    marks: markDTOs,
   796→    passPercentage: exam.passPercentage,
   797→  })
   798→  for (const r of results) r.className = cls.name
   799→
   800→  const analytics = computeAnalytics(
   801→    {
   802→      students: studentDTOs,
   803→      subjects: subjectDTOs,
   804→      marks: markDTOs,
   805→      passPercentage: exam.passPercentage,
   806→    },
   807→    cls.name
   808→  )
   809→
   810→  return { students: studentDTOs, subjects: subjectDTOs, marks: markDTOs, results, analytics }
   811→}

   814→  examId: string,
   815→  schoolId: string,
   816→  user: AuthUserLike | null
   817→): Promise<{ declared: boolean }> {
   818→  const exam = await db.exam.findFirst({ where: { id: examId, schoolId } })
   819→  if (!exam) throw new Error('Exam not found')
   820→  if (exam.resultStatus !== 'Result Ready') {
   821→    throw new Error('Results can only be declared from Result Ready state')
   822→  }
   823→  await db.exam.update({
   824→    where: { id: examId },
   825→    data: {
   826→      resultStatus: 'Result Declared',
   827→      status: 'Completed',
   828→      declaredAt: new Date(),
   829→      declaredBy: user?.id ?? null,
   830→    },
   831→  })
   832→  await audit(examId, user, 'RESULT_DECLARED', 'EXAM', examId, null, null)
   833→  return { declared: true }
   834→}
