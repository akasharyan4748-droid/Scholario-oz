     3→/**
     4→ * generate-tab — the document generation workflow.
     5→ *
     6→ * Flow (mirrored by the compact 4-step stepper):
     7→ *   1. Document type — compact 7-type picker grid, always visible, with a
     8→ *      clear selected state (emerald ring + check). Bonafide is
     9→ *      pre-selected by default so the Live Preview pane renders a REAL
    10→ *      sample document on first paint instead of an empty placeholder.
    11→ *   2. Student — the first student is pre-selected (real preview data);
    12→ *      Marksheets add exam → class; Fee Receipts add the fee transaction.
    13→ *   3. Details — template & style, default pre-selected.
    14→ *   4. Preview & issue — live paper preview + Generate (store + toast).
    15→ *
    16→ * The preview pane wraps the real previews.tsx documents on a neutral desk
    17→ * with a paper drop shadow — it must read as a printed document, not a web
    18→ * dashboard.
    19→ */
    20→

    22→import {
    23→  Search, Sparkles, Check, Printer, ChevronRight,
    24→} from 'lucide-react'
    25→import { Button } from '@/components/ui/button'
    26→import { Input } from '@/components/ui/input'
    27→import {
    28→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    29→} from '@/components/ui/select'
    30→import { toast } from 'sonner'

    32→import { school } from '@/lib/mock/school'
    33→import { formatDate, formatINR } from '@/lib/format'
    34→import { useStudentsStore } from '@/lib/store/students-store'
    35→import type { StudentRecord } from '@/lib/store/students-store'
    36→import { useFeeStore } from '@/lib/store/fee-store'
    37→import type { FeeTransaction } from '@/lib/store/fee-store'
    38→import { useMockExamsStore } from '@/lib/exams/mock-exams-data'
    39→import { useMockMarksStore } from '@/lib/exams/mock-marks-data'
    40→import {
    41→  useCertificatesStore,
    42→  type DocType, type DocumentTemplate,
    43→} from '@/lib/store/certificates-store'
    44→import {
    45→  DOC_TYPES, DOC_TYPE_BY_LABEL, CertPanel,
    46→} from './cert-shared'
    47→import {
    48→  CertificatePreview, MarksheetPreview, IDCardPreview, FeeReceiptPreview,
    49→  type MarksheetData, type MarksheetRow,
    50→} from './previews'
    51→
    52→export function GenerateTab() {
    53→  // Bonafide pre-selected — the preview pane shows a real document from
    54→  // the first paint; the picker grid keeps the selection visible.
    55→  const [docType, setDocType] = useState<DocType | null>('Bonafide')
    56→  const [studentId, setStudentId] = useState<string>('')
    57→  const [examId, setExamId] = useState<string>('')
    58→  const [classId, setClassId] = useState<string>('')
    59→  const [txnId, setTxnId] = useState<string>('')
    60→  const [templateId, setTemplateId] = useState<string>('')
    61→  const [purpose, setPurpose] = useState<string>('')
    62→  const [genSearch, setGenSearch] = useState<string>('')
    63→
    64→  const students = useStudentsStore((s) => s.students)
    65→  const exams = useMockExamsStore((s) => s.exams)
    66→  const transactions = useFeeStore((s) => s.transactions)
    67→  const templates = useCertificatesStore((s) => s.templates)
    68→  const generateDocument = useCertificatesStore((s) => s.generateDocument)
    69→
    70→  // ─── Derived data ──────────────────────────────────────────────────
    71→  const meta = docType ? DOC_TYPE_BY_LABEL[docType] : null
    72→
    73→  const filteredStudents = useMemo(() => {
    74→    if (!genSearch.trim()) return students.slice(0, 40)
    75→    const q = genSearch.toLowerCase().trim()
    76→    return students
    77→      .filter((s) =>
    78→        s.name.toLowerCase().includes(q) ||
    79→        s.admissionNo.toLowerCase().includes(q) ||
    80→        s.className.toLowerCase().includes(q),
    81→      )
    82→      .slice(0, 40)
    83→  }, [students, genSearch])
    84→
    85→  // For marksheet — only show students in the selected class
    86→  const exam = exams.find((e) => e.id === examId)
    87→  const examClasses = exam?.classes ?? []
    88→  const examSubjects = exam ? exam.subjects.filter((s) => s.classId === classId) : []
    89→  const marks = useMockMarksStore((s) => s.marks)
    90→  const initMarks = useMockMarksStore((s) => s.initMarks)
    91→
    92→  // Auto-init marks for the selected exam when a class is picked (so the
    93→  // marksheet preview has data even before the user opens the exam in the
    94→  // Examinations module).
    95→  useEffect(() => {
    96→    if (!exam || !classId) return
    97→    const hasMarks = marks.some((m) => m.examId === exam.id && m.classId === classId)
    98→    if (hasMarks) return
    99→    const classStudents = students
   100→      .filter((s) => s.classId === classId)
   101→      .map((s) => ({ id: s.id, name: s.name, rollNo: s.rollNo, classId: s.classId, className: s.className }))
   102→    if (classStudents.length > 0) {
   103→      initMarks(exam, classStudents)
   104→    }
   105→  }, [exam, classId, students, marks, initMarks])
   106→
   107→  // Filter students by class for marksheet selection
   108→  const classStudents = useMemo(() => {
   109→    if (docType !== 'Marksheet') return students
   110→    return students.filter((s) => s.classId === classId)
   111→  }, [students, classId, docType])
   112→
   113→  // ─── Template list for the active doc type ────────────────────────
   114→  const docTemplates = useMemo(
   115→    () => (docType ? templates.filter((t) => t.docType === docType && t.active) : []),
   116→    [templates, docType],
   117→  )
   118→
   119→  // Pick a default template when doc type changes
   120→  useEffect(() => {
   121→    if (!docType) return
   122→    const def = docTemplates.find((t) => t.isDefault) ?? docTemplates[0]
   123→    setTemplateId(def?.id ?? '')
   124→  }, [docType, docTemplates])
   125→
   126→  // ─── Reset sub-state when doc type changes ────────────────────────
   127→  useEffect(() => {
   128→    setStudentId('')
   129→    setExamId('')
   130→    setClassId('')
   131→    setTxnId('')
   132→    setPurpose('')
   133→    setGenSearch('')
   134→  }, [docType])
   135→
   136→  // Pre-select the first eligible student so the Live Preview renders a
   137→  // real document immediately (the picker shows the active check; the
   138→  // operator can switch at any time before issuing).
   139→  useEffect(() => {
   140→    if (!docType) return
   141→    if (studentId && students.some((s) => s.id === studentId)) return
   142→    const pool = docType === 'Marksheet' ? classStudents : students
   143→    if (pool.length > 0) setStudentId(pool[0].id)
   144→  }, [docType, studentId, students, classStudents])
   145→
   146→  // ─── Selected student + transaction + marksheet data ──────────────
   147→  const selectedStudent = students.find((s) => s.id === studentId) as StudentRecord | undefined
   148→  const selectedTxn = transactions.find((t) => t.id === txnId) as FeeTransaction | undefined
   149→  const selectedTemplate = templates.find((t) => t.id === templateId) as DocumentTemplate | undefined
   150→
   151→  // Compute marksheet data on-the-fly (cheap — at most ~10 subjects).
   152→  const marksheetData: MarksheetData | undefined =
   153→    docType === 'Marksheet' && selectedStudent && exam && classId
   154→      ? computeMarksheet(exam, classId, selectedStudent, marks, examSubjects)
   155→      : undefined
   156→
   157→  // ─── Can generate? ────────────────────────────────────────────────
   158→  const canGenerate = (() => {
   159→    if (!docType || !templateId) return false
   160→    if (!selectedStudent) return false
   161→    if (docType === 'Marksheet' && (!examId || !classId || !marksheetData)) return false
   162→    if (docType === 'Fee Receipt' && !selectedTxn) return false
   163→    return true
   164→  })()
   165→
   166→  // ─── Stepper state (4 real flow steps) ────────────────────────────
   167→  const studentStepDone = !!selectedStudent
   168→    && !(meta?.needsExam && !(examId && classId && marksheetData))
   169→    && !(meta?.needsFeeTxn && !selectedTxn)
   170→  const stepIndex = !docType ? 0 : !studentStepDone ? 1 : !selectedTemplate ? 2 : 3
   171→  const steps = ['Document type', 'Student', 'Details', 'Preview & issue'].map((label, i) => ({
   172→    label,
   173→    done: i < stepIndex,
   174→    current: i === stepIndex,
   175→  }))
   176→
   177→  // ─── Generate ─────────────────────────────────────────────────────
   178→  function handleGenerate() {
   179→    if (!docType || !selectedStudent || !selectedTemplate) return
   180→    let data: Record<string, any> = {}
   181→    let classInfo = selectedStudent.className
   182→    if (docType === 'Marksheet' && marksheetData && exam) {
   183→      data = { examId: exam.id, examName: exam.name, classId, marksheet: marksheetData }
   184→      classInfo = `${exam.name} · ${selectedStudent.className}`
   185→    } else if (docType === 'Fee Receipt' && selectedTxn) {
   186→      data = {
   187→        transactionId: selectedTxn.id,
   188→        receiptNo: selectedTxn.receiptNo,
   189→        amount: selectedTxn.amount,
   190→        mode: selectedTxn.mode,
   191→        purpose: selectedTxn.purpose,
   192→      }
   193→      classInfo = selectedTxn.className
   194→    } else {
   195→      data = { purpose: purpose || '—' }
   196→    }
   197→    const doc = generateDocument({
   198→      docType,
   199→      templateId,
   200→      student: {
   201→        id: selectedStudent.id,
   202→        name: selectedStudent.name,
   203→        admissionNo: selectedStudent.admissionNo,
   204→        class: classInfo,
   205→      },
   206→      generatedBy: school.principal,
   207→      data,
   208→    })
   209→    toast.success(`${docType} generated`, {
   210→      description: `${doc.docNumber} · ${selectedStudent.name}`,
   211→    })
   212→  }
   213→
   214→  // ─── Render ────────────────────────────────────────────────────────
   215→  return (
   216→    <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   217→      {/* LEFT — workflow */}
   218→      <div className="lg:col-span-5 space-y-4">
   219→        {/* Compact step indicator — 4 real steps; completed get a check,
   220→            the current step is emphasised. */}
   221→        <FlowStepper steps={steps} />
   222→
   223→        {/* Step 1 — Doc type picker (always visible so switching types is
   224→            one click; the selected card carries ring + check + tint). */}
   225→        <CertPanel
   226→          title="Document type"
   227→          subtitle={docType ? DOC_TYPE_BY_LABEL[docType].description : 'Pick the document to issue'}
   228→        >
   229→          <div className="grid grid-cols-2 gap-2">
   230→            {DOC_TYPES.map((d) => (
   231→              <DocTypeCard
   232→                key={d.label}
   233→                meta={d}
   234→                selected={docType === d.label}
   235→                onClick={() => setDocType(d.label)}
   236→              />
   237→            ))}
   238→          </div>
   239→        </CertPanel>
   240→
   241→        {docType && (
   242→          <CertPanel
   243→            title="Student"
   244→            subtitle={
   245→              meta?.needsExam
   246→                ? 'Examination → Class → Student'
   247→                : meta?.needsFeeTxn
   248→                  ? 'Student → Fee transaction'
   249→                  : 'Search and pick the student'
   250→            }
   251→          >
   252→            {meta?.needsExam ? (
   253→              <div className="space-y-3">
   254→                <FieldLabel label="Examination" />
   255→                <Select value={examId} onValueChange={setExamId}>
   256→                  <SelectTrigger className="h-9 text-xs">
   257→                    <SelectValue placeholder="Choose examination" />
   258→                  </SelectTrigger>
   259→                  <SelectContent>
   260→                    {exams.map((e) => (
   261→                      <SelectItem key={e.id} value={e.id}>
   262→                        {e.name} · {e.session} · {e.status}
   263→                      </SelectItem>
   264→                    ))}
   265→                  </SelectContent>
   266→                </Select>
   267→
   268→                <FieldLabel label="Class" />
   269→                <Select value={classId} onValueChange={setClassId} disabled={!examId}>
   270→                  <SelectTrigger className="h-9 text-xs">
   271→                    <SelectValue placeholder={examId ? 'Choose class' : 'Pick examination first'} />
   272→                  </SelectTrigger>
   273→                  <SelectContent>
   274→                    {examClasses.map((c) => (
   275→                      <SelectItem key={c.classId} value={c.classId}>
   276→                        {c.className}{c.stream ? ` · ${c.stream}` : ''}
   277→                      </SelectItem>
   278→                    ))}
   279→                  </SelectContent>
   280→                </Select>
   281→
   282→                <FieldLabel label="Student" />
   283→                <StudentPicker
   284→                  students={classStudents}
   285→                  value={studentId}
   286→                  onChange={setStudentId}
   287→                  search={genSearch}
   288→                  onSearch={setGenSearch}
   289→                />
   290→              </div>
   291→            ) : (
   292→              <div className="space-y-3">
   293→                <FieldLabel label="Student" />
   294→                <StudentPicker
   295→                  students={filteredStudents}
   296→                  value={studentId}
   297→                  onChange={setStudentId}
   298→                  search={genSearch}
   299→                  onSearch={setGenSearch}
   300→                />
   301→
   302→                {meta?.needsFeeTxn && selectedStudent && (
   303→                  <>
   304→                    <FieldLabel label="Fee transaction" />
   305→                    <Select
   306→                      value={txnId}
   307→                      onValueChange={setTxnId}
   308→                    >
   309→                      <SelectTrigger className="h-9 text-xs">
   310→                        <SelectValue placeholder="Choose fee payment" />
   311→                      </SelectTrigger>
   312→                      <SelectContent>
   313→                        {transactions
   314→                          .filter((t) => t.studentId === selectedStudent.id)
   315→                          .slice(0, 25)
   316→                          .map((t) => (
   317→                            <SelectItem key={t.id} value={t.id}>
   318→                              {t.receiptNo} · {formatDate(t.date)} · {formatINR(t.amount)} · {t.mode}
   319→                            </SelectItem>
   320→                          ))}
   321→                        {transactions.filter((t) => t.studentId === selectedStudent.id).length === 0 && (
   322→                          <SelectItem value="__none" disabled>No transactions found for this student</SelectItem>
   323→                        )}
   324→                      </SelectContent>
   325→                    </Select>
   326→                  </>
   327→                )}
   328→
   329→                {docType === 'Bonafide' && selectedStudent && (
   330→                  <>
   331→                    <FieldLabel label="Purpose (optional)" />
   332→                    <Input
   333→                      value={purpose}
   334→                      onChange={(e) => setPurpose(e.target.value)}
   335→                      placeholder="e.g. Bank account opening, passport, scholarship"
   336→                      className="h-9 text-xs"
   337→                    />
   338→                  </>
   339→                )}
   340→              </div>
   341→            )}
   342→          </CertPanel>
   343→        )}
   344→
   345→        {docType && docTemplates.length > 0 && (
   346→          <CertPanel
   347→            title="Template & style"
   348→            action={selectedTemplate && (
   349→              <span className="text-[10px] text-muted-foreground">
   350→                Style: <strong className="text-foreground">{selectedTemplate.style}</strong>
   351→              </span>
   352→            )}
   353→          >
   354→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   355→              {docTemplates.map((t) => {
   356→                const active = t.id === templateId
   357→                return (
   358→                  <button
   359→                    key={t.id}
   360→                    onClick={() => setTemplateId(t.id)}
   361→                    className={cn(
   362→                      'relative flex flex-col items-start gap-1.5 rounded-lg border p-2 text-left transition-all hover:-translate-y-0.5',
   363→                      active
   364→                        ? 'border-emerald-500/40 bg-emerald-500/[0.05] ring-2 ring-emerald-500/30'
   365→                        : 'border-border hover:border-foreground/40',
   366→                    )}
   367→                  >
   368→                    <div className="flex items-center gap-1.5 w-full">
   369→                      <span
   370→                        className="h-4 w-4 rounded shrink-0"
   371→                        style={{ background: t.accentColor }}
   372→                      />
   373→                      <span className="text-[10px] font-semibold leading-tight truncate">{t.style}</span>
   374→                      {t.isDefault && (
   375→                        <span className="ml-auto text-[8px] px-1 py-0.5 rounded bg-emerald-500/10 text-emerald-700 dark:text-emerald-300 font-semibold">
   376→                          DEFAULT
   377→                        </span>
   378→                      )}
   379→                    </div>
   380→                    <p className="text-[9px] text-muted-foreground leading-tight line-clamp-2">{t.name}</p>
   381→                    {active && (
   382→                      <span className="absolute -top-1.5 -right-1.5 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-600 text-white">
   383→                        <Check className="h-2.5 w-2.5" />
   384→                      </span>
   385→                    )}
   386→                  </button>
   387→                )
   388→              })}
   389→            </div>
   390→          </CertPanel>
   391→        )}
   392→
   393→        {docType && (
   394→          <div className="flex items-center gap-2">
   395→            <Button
   396→              size="sm"
   397→              disabled={!canGenerate}
   398→              onClick={handleGenerate}
   399→              className="flex-1 h-9 text-xs gap-1.5 bg-emerald-600 hover:bg-emerald-700 text-white"
   400→            >
   401→              <Sparkles className="h-3.5 w-3.5" /> Generate {docType}
   402→            </Button>
   403→            <Button
   404→              size="sm"
   405→              variant="outline"
   406→              disabled={!canGenerate}
   407→              onClick={() => window.print()}
   408→              className="h-9 text-xs gap-1 no-print"
   409→            >
   410→              <Printer className="h-3.5 w-3.5" /> Print
   411→            </Button>
   412→          </div>
   413→        )}
   414→      </div>
   415→
   416→      {/* RIGHT — live preview on a paper desk */}
   417→      <div className="lg:col-span-7">
   418→        <CertPanel
   419→          title="Live preview"
   420→          subtitle={docType ? `${docType}${selectedStudent ? ` · ${selectedStudent.name}` : ''}` : 'Pick a document type'}
   421→          className="h-full"
   422→          bodyClassName="p-2 sm:p-4 bg-muted/40"
   423→        >
   424→          <PreviewArea
   425→            docType={docType}
   426→            student={selectedStudent}
   427→            template={selectedTemplate}
   428→            txn={selectedTxn}
   429→            marksheetData={marksheetData}
   430→            purpose={purpose}
   431→          />
   432→        </CertPanel>
   433→      </div>
   434→    </div>
   435→  )
   436→}

   440→/** Compact horizontal 4-step indicator. Completed steps show a check;
   441→ *  the current step is emphasised (dark pill). Wraps on narrow screens. */
   442→function FlowStepper({ steps }: { steps: { label: string; done: boolean; current: boolean }[] }) {
   443→  return (
   444→    <div className="flex items-center gap-1 flex-wrap" aria-label="Generation progress">
   445→      {steps.map((s, i) => (
   446→        <div key={s.label} className="flex items-center gap-1">
   447→          {i > 0 && <ChevronRight className="h-3 w-3 text-muted-foreground/40" aria-hidden />}
   448→          <div
   449→            className={cn(
   450→              'flex items-center gap-1.5 rounded-full px-2 py-1 transition-colors',
   451→              s.current ? 'bg-foreground text-background' : 'text-muted-foreground',
   452→            )}
   453→            aria-current={s.current ? 'step' : undefined}
   454→          >
   455→            <span
   456→              className={cn(
   457→                'flex h-4 w-4 items-center justify-center rounded-full text-[9px] font-bold tabular-nums',
   458→                s.done
   459→                  ? 'bg-emerald-500/20 text-emerald-700 dark:text-emerald-300'
   460→                  : s.current
   461→                    ? 'bg-background/20 text-background'
   462→                    : 'bg-muted text-muted-foreground',
   463→              )}
   464→            >
   465→              {s.done ? <Check className="h-2.5 w-2.5" strokeWidth={3} /> : i + 1}
   466→            </span>
   467→            <span className="text-[10px] font-semibold whitespace-nowrap">{s.label}</span>
   468→          </div>
   469→        </div>
   470→      ))}
   471→    </div>
   472→  )
   473→}

   475→/** Compact document-type picker card. Distinct lucide icon per type,
   476→ *  single emerald accent. Selected: ring-2 ring-emerald-500/40 + emerald
   477→ *  check top-right + slight bg tint. Hover: border lift. */
   478→function DocTypeCard({
   479→  meta, selected, onClick,
   480→}: {
   481→  meta: (typeof DOC_TYPES)[number]
   482→  selected: boolean
   483→  onClick: () => void
   484→}) {
   485→  const Icon = meta.icon
   486→  return (
   487→    <button
   488→      type="button"
   489→      onClick={onClick}
   490→      aria-pressed={selected}
   491→      className={cn(
   492→        'group relative flex items-start gap-2.5 rounded-lg border p-2.5 text-left transition-all',
   493→        selected
   494→          ? 'border-emerald-500/40 bg-emerald-500/[0.05] ring-2 ring-emerald-500/40'
   495→          : 'border-border hover:border-foreground/20 hover:-translate-y-0.5 hover:shadow-sm',
   496→      )}
   497→    >
   498→      <span
   499→        className={cn(
   500→          'flex h-8 w-8 shrink-0 items-center justify-center rounded-md ring-1 transition-colors',
   501→          selected
   502→            ? 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-300 ring-emerald-500/25'
   503→            : 'bg-muted/60 text-muted-foreground ring-border group-hover:text-foreground',
   504→        )}
   505→      >
   506→        <Icon className="h-4 w-4" />
   507→      </span>
   508→      <div className="min-w-0 flex-1 pr-4">
   509→        <p className="text-xs font-semibold leading-tight truncate">{meta.short}</p>
   510→        <p className="text-[9px] text-muted-foreground leading-tight mt-0.5 line-clamp-2">{meta.description}</p>
   511→      </div>
   512→      {selected && (
   513→        <span className="absolute top-1.5 right-1.5 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-600 text-white">
   514→          <Check className="h-2.5 w-2.5" strokeWidth={3} />
   515→        </span>
   516→      )}
   517→    </button>
   518→  )
   519→}

   521→function FieldLabel({ label }: { label: string }) {
   522→  return <p className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wider">{label}</p>

   525→function StudentPicker({
   526→  students, value, onChange, search, onSearch,
   527→}: {
   528→  students: StudentRecord[]
   529→  value: string
   530→  onChange: (id: string) => void
   531→  search: string
   532→  onSearch: (q: string) => void
   533→}) {
   534→  return (
   535→    <div className="space-y-2">

   668→import type { ExamDTO, ExamSubjectConfigDTO, ExamMarkDTO } from '@/lib/exams/types'
   669→
   670→function computeMarksheet(
   671→  exam: ExamDTO,
   672→  classId: string,
   673→  student: StudentRecord,
   674→  marks: ExamMarkDTO[],
   675→  examSubjects: ExamSubjectConfigDTO[],
   676→): MarksheetData {
   677→  // Try real exam marks first
   678→  if (examSubjects.length > 0) {
   679→    const rows = examSubjects.map((subj) => {
   680→      const mark = marks.find(
   681→        (m) =>
   682→          m.examId === exam.id &&
   683→          m.classId === classId &&
   684→          m.subjectId === subj.subjectId &&
   685→          m.studentId === student.id,
   686→      )
   687→      const obtained = mark?.marksObtained ?? 0
   688→      return {
   689→        subject: subj.subjectName,
   690→        max: subj.maxMarks,
   691→        pass: subj.passMarks,
   692→        obtained,
   693→        isAbsent: mark?.status === 'ABSENT',
   694→      }
   695→    })
   696→    const totalMax = rows.reduce((s, r) => s + r.max, 0)
   697→    const totalObtained = rows.reduce((s, r) => s + (r.isAbsent ? 0 : r.obtained), 0)
   698→    const pct = totalMax > 0 ? (totalObtained / totalMax) * 100 : 0
   699→    const passed = rows.every((r) => r.isAbsent || r.obtained >= r.pass)
   700→    return {
   701→      examName: exam.name,
   702→      className: student.className,
   703→      section: student.section,
   704→      session: exam.session ?? school.academicYear,
   705→      rows,
   706→      totalMax,
   707→      totalObtained,
   708→      percentage: pct,
   709→      grade: gradeFor(pct),
   710→      result: passed ? 'PASS' : 'FAIL',
   711→      remarks: passed ? 'Conduct: Excellent. Regularity: Satisfactory.' : 'Re-examination recommended.',
   712→    }
   713→  }
   714→  // Fallback — derive from student.academics.subjects (max 100)
   715→  const rows: MarksheetRow[] = student.academics.subjects.map((subj) => ({
   716→    subject: subj.name,
   717→    max: 100,
   718→    pass: 33,
   719→    obtained: Math.round((subj.percent / 100) * 100),
   720→  }))
   721→  const totalMax = rows.length * 100
   722→  const totalObtained = rows.reduce((s, r) => s + r.obtained, 0)
   723→  const pct = totalMax > 0 ? (totalObtained / totalMax) * 100 : 0
   724→  return {
   725→    examName: 'Academic Performance Report',
   726→    className: student.className,
   727→    section: student.section,
   728→    session: school.academicYear,
   729→    rows,
   730→    totalMax,
   731→    totalObtained,
   732→    percentage: pct,
   733→    grade: gradeFor(pct),
   734→    result: pct >= 33 ? 'PASS' : 'FAIL',
   735→    remarks: 'Based on cumulative academic record.',
   736→  }
   737→}
   738→
   739→function gradeFor(pct: number): string {
   740→  if (pct >= 90) return 'A1'
   741→  if (pct >= 80) return 'A2'
   742→  if (pct >= 70) return 'B1'
   743→  if (pct >= 60) return 'B2'
   744→  if (pct >= 50) return 'C1'
   745→  if (pct >= 33) return 'C2'
   746→  return 'E'
   747→}
